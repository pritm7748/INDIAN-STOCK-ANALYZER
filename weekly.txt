```javascript
// ═══════════════════════════════════════════════
//  WEEKLY CANDLE CONSTRUCTION
// ═══════════════════════════════════════════════

/**
 * Build weekly candles from daily candles
 * Handles holidays, short weeks, and edge cases
 *
 * @param {Array} dailyCandles - [{date, o, h, l, c, v}, ...] ascending
 * @returns {Array} [{weekStart, weekEnd, o, h, l, c, v, tradingDays, dailyCandles, isShortWeek}, ...]
 */
function buildWeeklyCandles(dailyCandles) {
  if (!dailyCandles || dailyCandles.length === 0) return [];

  const weeks = [];
  let current = null;

  for (const day of dailyCandles) {
    const dow = getDow(day.date);
    const needsNew = !current
      || dow === 1                                      // Monday
      || dayGap(current.weekEnd, day.date) > 3;         // >3 calendar day gap (holiday week)

    if (needsNew && current) {
      finalizeWeek(current);
      weeks.push(current);
      current = null;
    }

    if (!current) {
      current = {
        weekStart: day.date,
        weekEnd: day.date,
        o: day.o,
        h: day.h,
        l: day.l,
        c: day.c,
        v: 0,
        tradingDays: 0,
        dailyCandles: [],
      };
    }

    current.h = Math.max(current.h, day.h);
    current.l = Math.min(current.l, day.l);
    current.c = day.c;
    current.v += day.v;
    current.weekEnd = day.date;
    current.tradingDays++;
    current.dailyCandles.push(day);
  }

  if (current && current.tradingDays > 0) {
    finalizeWeek(current);
    weeks.push(current);
  }

  return weeks;
}

function finalizeWeek(week) {
  week.isShortWeek = week.tradingDays < 4;
  week.range = week.h - week.l;
  week.rangePercent = week.c > 0 ? r((week.range / week.c) * 100) : 0;
  week.body = Math.abs(week.c - week.o);
  week.bodyPercent = week.range > 0 ? r((week.body / week.range) * 100) : 0;
  week.isBullish = week.c > week.o;
  week.closePosition = week.range > 0 ? r((week.c - week.l) / week.range, 3) : 0.5;

  // weekly candle type classification
  week.candleType = classifyWeeklyCandle(week);
}

/**
 * Classify weekly candle for follow-through probability
 */
function classifyWeeklyCandle(week) {
  const bodyRatio = week.range > 0 ? week.body / week.range : 0;
  const upperWick = week.isBullish ? week.h - week.c : week.h - week.o;
  const lowerWick = week.isBullish ? week.o - week.l : week.c - week.l;
  const upperWickRatio = week.range > 0 ? upperWick / week.range : 0;
  const lowerWickRatio = week.range > 0 ? lowerWick / week.range : 0;

  if (bodyRatio > 0.75 && week.isBullish)  return 'BULLISH_MARUBOZU';
  if (bodyRatio > 0.75 && !week.isBullish) return 'BEARISH_MARUBOZU';
  if (bodyRatio < 0.10)                     return 'DOJI';
  if (week.isBullish && lowerWickRatio > 0.60 && upperWickRatio < 0.10) return 'HAMMER';
  if (!week.isBullish && upperWickRatio > 0.60 && lowerWickRatio < 0.10) return 'SHOOTING_STAR';
  if (week.isBullish && bodyRatio > 0.50)  return 'BULLISH';
  if (!week.isBullish && bodyRatio > 0.50) return 'BEARISH';
  if (upperWickRatio > 0.35 && lowerWickRatio > 0.35) return 'SPINNING_TOP';
  return week.isBullish ? 'BULLISH_WEAK' : 'BEARISH_WEAK';
}

/**
 * Follow-through probability based on previous week candle type
 */
function candleFollowThroughBias(candleType, direction) {
  // empirical probabilities from Indian market data patterns
  const biases = {
    BULLISH_MARUBOZU:  { LONG: 0.72, SHORT: 0.28 },
    BEARISH_MARUBOZU:  { LONG: 0.30, SHORT: 0.70 },
    BULLISH:           { LONG: 0.62, SHORT: 0.38 },
    BEARISH:           { LONG: 0.38, SHORT: 0.62 },
    BULLISH_WEAK:      { LONG: 0.55, SHORT: 0.45 },
    BEARISH_WEAK:      { LONG: 0.45, SHORT: 0.55 },
    DOJI:              { LONG: 0.50, SHORT: 0.50 },
    HAMMER:            { LONG: 0.65, SHORT: 0.35 },
    SHOOTING_STAR:     { LONG: 0.35, SHORT: 0.65 },
    SPINNING_TOP:      { LONG: 0.50, SHORT: 0.50 },
  };
  const entry = biases[candleType] || biases.DOJI;
  return entry[direction] || 0.5;
}

//  MULTI-WEEK BASE DETECTION

/**
 * Detect multi-week consolidation bases
 * This is THE most important enhancement — single-week NR is noise,
 * multi-week base is conviction
 */
function detectBase(weeklyCandles, maxWeeks = 12) {
  if (!weeklyCandles || weeklyCandles.length < 6) return null;

  const recent = weeklyCandles.slice(-Math.min(maxWeeks + 1, weeklyCandles.length));
  if (recent.length < 4) return null;

  const lastWeek = recent[recent.length - 1];

  // find the "base" — consecutive weeks where highs and lows stay within a range
  // work backwards from current week
  let baseHighest = lastWeek.h;
  let baseLowest = lastWeek.l;
  let baseWeeks = 1;
  let baseVolumes = [lastWeek.v];

  for (let i = recent.length - 2; i >= 0; i--) {
    const w = recent[i];
    const candidateHigh = Math.max(baseHighest, w.h);
    const candidateLow = Math.min(baseLowest, w.l);
    const candidateRange = ((candidateHigh - candidateLow) / candidateLow) * 100;

    // base stays tight if total range < 8%
    // (individual weeks are < 5%, but over multiple weeks allow up to 8%)
    if (candidateRange < 8) {
      baseHighest = candidateHigh;
      baseLowest = candidateLow;
      baseWeeks++;
      baseVolumes.push(w.v);
    } else {
      break; // range expanded — base broken
    }
  }

  const baseRange = baseHighest - baseLowest;
  const baseRangePercent = (baseRange / baseLowest) * 100;

  // volume pattern within base
  // ideal: declining volume during consolidation (drying up)
  let volumeDeclining = false;
  if (baseVolumes.length >= 3) {
    const firstHalf = baseVolumes.slice(0, Math.floor(baseVolumes.length / 2));
    const secondHalf = baseVolumes.slice(Math.floor(baseVolumes.length / 2));
    const avgFirst = firstHalf.reduce((a, b) => a + b, 0) / firstHalf.length;
    const avgSecond = secondHalf.reduce((a, b) => a + b, 0) / secondHalf.length;
    volumeDeclining = avgSecond < avgFirst * 0.85;
  }

  // NR detection within the base
  const ranges = recent.slice(-8).map(w => w.range);
  const currentRange = lastWeek.range;
  const isNR4 = ranges.length >= 4 && currentRange <= Math.min(...ranges.slice(-4));
  const isNR7 = ranges.length >= 7 && currentRange <= Math.min(...ranges.slice(-7));

  // inside week: current week entirely within previous week
  const prevWeek = recent[recent.length - 2];
  const isInsideWeek = lastWeek.h < prevWeek.h && lastWeek.l > prevWeek.l;

  // consecutive contracting ranges
  let contractingCount = 0;
  for (let i = ranges.length - 1; i > 0; i--) {
    if (ranges[i] < ranges[i - 1]) contractingCount++;
    else break;
  }

  // is this a proper base?
  const isValidBase = baseWeeks >= 3;
  const isStrongBase = baseWeeks >= 4 && baseRangePercent < 6;
  const isCoiledSpring = isStrongBase && (volumeDeclining || isNR4);
  const isPerfectSetup = isCoiledSpring && isInsideWeek;

  // base quality score (0-100)
  let baseScore = 0;
  if (baseWeeks >= 6) baseScore += 25;
  else if (baseWeeks >= 4) baseScore += 20;
  else if (baseWeeks >= 3) baseScore += 12;
  else baseScore += 5;

  if (baseRangePercent < 4) baseScore += 20;
  else if (baseRangePercent < 6) baseScore += 14;
  else if (baseRangePercent < 8) baseScore += 8;

  if (volumeDeclining) baseScore += 20;
  if (isNR7) baseScore += 15;
  else if (isNR4) baseScore += 10;
  if (isInsideWeek) baseScore += 12;
  if (contractingCount >= 3) baseScore += 8;

  return {
    exists: isValidBase,
    baseWeeks,
    baseHighest: r(baseHighest),
    baseLowest: r(baseLowest),
    baseRange: r(baseRange),
    baseRangePercent: r(baseRangePercent),
    baseMid: r((baseHighest + baseLowest) / 2),
    volumeDeclining,
    isNR4,
    isNR7,
    isInsideWeek,
    contractingCount,
    isStrongBase,
    isCoiledSpring,
    isPerfectSetup,
    baseScore: r(clamp(baseScore)),
    classification: isPerfectSetup ? 'PERFECT_COIL'
      : isCoiledSpring ? 'COILED_SPRING'
      : isStrongBase ? 'STRONG_BASE'
      : isValidBase ? 'VALID_BASE'
      : 'NO_BASE',
  };
}

//  RELATIVE STRENGTH vs INDEX
// ═══════════════════════════════════════════════

/**
 * Mansfield Relative Strength — stock performance vs Nifty
 *
 * @param {Array} stockWeekly - weekly candles for stock
 * @param {Array} niftyWeekly - weekly candles for Nifty
 * @param {number} period - lookback in weeks
 */
function relativeStrength(stockWeekly, niftyWeekly, period = 12) {
  const len = Math.min(stockWeekly.length, niftyWeekly.length);
  if (len < period + 1) return null;

  const stockCloses = stockWeekly.slice(-len).map(w => w.c);
  const niftyCloses = niftyWeekly.slice(-len).map(w => w.c);

  // RS line = stock / nifty (normalized)
  const rsLine = [];
  for (let i = 0; i < stockCloses.length; i++) {
    rsLine.push(niftyCloses[i] > 0 ? stockCloses[i] / niftyCloses[i] : 0);
  }

  // current RS vs period-ago RS
  const currentRS = rsLine[rsLine.length - 1];
  const pastRS = rsLine[rsLine.length - 1 - period];
  const rsChange = pastRS > 0 ? ((currentRS - pastRS) / pastRS) * 100 : 0;

  // RS slope (last 4 weeks)
  const recent4 = rsLine.slice(-4);
  let risingCount = 0;
  for (let i = 1; i < recent4.length; i++) {
    if (recent4[i] > recent4[i - 1]) risingCount++;
  }
  const rsRising = risingCount >= 2;

  // stock return vs nifty return over period
  const stockReturn = ((stockCloses[stockCloses.length - 1] - stockCloses[stockCloses.length - 1 - period])
    / stockCloses[stockCloses.length - 1 - period]) * 100;
  const niftyReturn = ((niftyCloses[niftyCloses.length - 1] - niftyCloses[niftyCloses.length - 1 - period])
    / niftyCloses[niftyCloses.length - 1 - period]) * 100;
  const excessReturn = stockReturn - niftyReturn;

  // RS rank classification
  let rsRank;
  if (rsChange > 10 && rsRising) rsRank = 'VERY_STRONG';
  else if (rsChange > 5 || (rsChange > 0 && rsRising)) rsRank = 'STRONG';
  else if (rsChange > -5 && rsChange <= 5) rsRank = 'NEUTRAL';
  else if (rsChange > -10) rsRank = 'WEAK';
  else rsRank = 'VERY_WEAK';

  return {
    rsChange: r(rsChange),
    rsRising,
    stockReturn: r(stockReturn),
    niftyReturn: r(niftyReturn),
    excessReturn: r(excessReturn),
    rsRank,
    outperforming: excessReturn > 0,
    // RS score (0-100)
    rsScore: r(clamp(mapRange(rsChange, -15, 20, 10, 100))),
  };
}

//  VOLUME PATTERN ANALYSIS

/**
 * Pre-breakout volume pattern — THE critical volume signal
 * Declining volume during consolidation = sellers exhausted = accumulation
 * Then volume spike on breakout = conviction
 */
function preBreakoutVolumePattern(weeklyCandles, basePeriod = 6) {
  if (!weeklyCandles || weeklyCandles.length < basePeriod + 2) return null;

  const recent = weeklyCandles.slice(-(basePeriod + 1));
  const consolidation = recent.slice(0, -1);   // weeks before current
  const currentWeek = recent[recent.length - 1];

  // volume trend during consolidation
  const consolVols = consolidation.map(w => w.v);

  // linear regression slope of volume
  const volSlope = linearSlope(consolVols);
  const avgConsolVol = consolVols.reduce((a, b) => a + b, 0) / consolVols.length;

  // is volume declining?
  const isDeclining = volSlope < 0;
  const declineRate = avgConsolVol > 0 ? (volSlope / avgConsolVol) * 100 : 0;

  // volume on breakout week vs consolidation average
  const breakoutVolRatio = avgConsolVol > 0 ? currentWeek.v / avgConsolVol : 1;

  // ideal pattern: declining consolidation volume + spike on breakout
  const idealPattern = isDeclining && breakoutVolRatio > 1.5;
  const goodPattern = isDeclining && breakoutVolRatio > 1.2;

  // up-volume vs down-volume during consolidation
  let upVol = 0, downVol = 0;
  for (const w of consolidation) {
    if (w.isBullish) upVol += w.v;
    else downVol += w.v;
  }
  const upDownRatio = downVol > 0 ? upVol / downVol : 999;

  // accumulation score (Wyckoff-inspired)
  let accumScore = 0;
  if (isDeclining) accumScore += 30;
  if (breakoutVolRatio > 2.0) accumScore += 25;
  else if (breakoutVolRatio > 1.5) accumScore += 18;
  else if (breakoutVolRatio > 1.2) accumScore += 10;
  if (upDownRatio > 1.3) accumScore += 20;
  else if (upDownRatio > 1.0) accumScore += 10;
  if (idealPattern) accumScore += 15;
  if (declineRate < -5) accumScore += 10; // strong decline in volume

  return {
    isDeclining,
    declineRate: r(declineRate),
    avgConsolidationVolume: Math.round(avgConsolVol),
    breakoutWeekVolume: currentWeek.v,
    breakoutVolRatio: r(breakoutVolRatio),
    upDownVolumeRatio: r(upDownRatio),
    idealPattern,
    goodPattern,
    accumScore: r(clamp(accumScore)),
    pattern: idealPattern ? 'IDEAL_ACCUMULATION'
      : goodPattern ? 'GOOD_ACCUMULATION'
      : isDeclining ? 'DECLINING_BUT_NO_SPIKE'
      : breakoutVolRatio > 1.5 ? 'VOLUME_SPIKE_NO_DRYUP'
      : 'WEAK_VOLUME_PATTERN',
  };
}

/**
 * Breakout day intraday volume analysis
 * Uses 5-min candles on the breakout day to check SUSTAINABILITY
 *
 * @param {Array} intradayCandles - 5-min candles of breakout day
 * @param {number} breakoutLevel - PW High (for longs) or PW Low (for shorts)
 * @param {string} direction - 'LONG' | 'SHORT'
 */
function breakoutDayIntradayAnalysis(intradayCandles, breakoutLevel, direction = 'LONG') {
  if (!intradayCandles || intradayCandles.length < 10) return null;

  const isLong = direction === 'LONG';

  // when did the stock first cross the breakout level?
  let firstCrossIdx = -1;
  for (let i = 0; i < intradayCandles.length; i++) {
    const c = intradayCandles[i];
    if (isLong ? c.h > breakoutLevel : c.l < breakoutLevel) {
      firstCrossIdx = i;
      break;
    }
  }

  if (firstCrossIdx === -1) return { crossedLevel: false };

  const totalCandles = intradayCandles.length;
  const crossTime = extractTime(intradayCandles[firstCrossIdx].time);

  // how much of the day was above the breakout level?
  let candlesAbove = 0;
  for (let i = firstCrossIdx; i < totalCandles; i++) {
    const c = intradayCandles[i];
    if (isLong ? c.c > breakoutLevel : c.c < breakoutLevel) candlesAbove++;
  }
  const holdRatio = candlesAbove / (totalCandles - firstCrossIdx);

  // did it hold above the level in the LAST HOUR (2:30-3:30)?
  const lastHourCandles = intradayCandles.filter(c => extractTime(c.time) >= '14:30');
  let lastHourAbove = 0;
  for (const c of lastHourCandles) {
    if (isLong ? c.c > breakoutLevel : c.c < breakoutLevel) lastHourAbove++;
  }
  const lastHourHoldRatio = lastHourCandles.length > 0
    ? lastHourAbove / lastHourCandles.length
    : 0;

  // volume acceleration after cross
  const beforeCross = intradayCandles.slice(0, firstCrossIdx);
  const afterCross = intradayCandles.slice(firstCrossIdx);
  const volBefore = beforeCross.length > 0
    ? beforeCross.reduce((s, c) => s + c.v, 0) / beforeCross.length
    : 0;
  const volAfter = afterCross.reduce((s, c) => s + c.v, 0) / afterCross.length;
  const volAcceleration = volBefore > 0 ? volAfter / volBefore : 1;

  // did it retest the breakout level and hold? (pullback test)
  let retestFound = false;
  let retestHeld = false;
  for (let i = firstCrossIdx + 2; i < totalCandles; i++) {
    const c = intradayCandles[i];
    const touched = isLong
      ? c.l <= breakoutLevel * 1.003 && c.l >= breakoutLevel * 0.995
      : c.h >= breakoutLevel * 0.997 && c.h <= breakoutLevel * 1.005;

    if (touched) {
      retestFound = true;
      // did subsequent candles hold?
      if (i + 2 < totalCandles) {
        const next1 = intradayCandles[i + 1];
        const next2 = intradayCandles[i + 2];
        retestHeld = isLong
          ? next1.c > breakoutLevel && next2.c > breakoutLevel
          : next1.c < breakoutLevel && next2.c < breakoutLevel;
      }
      break;
    }
  }

  // closing candle analysis
  const lastCandle = intradayCandles[totalCandles - 1];
  const closedStrong = isLong
    ? lastCandle.c > breakoutLevel * 1.003
    : lastCandle.c < breakoutLevel * 0.997;

  const quality = (holdRatio > 0.7 && lastHourHoldRatio > 0.8 && closedStrong)
    ? 'STRONG'
    : (holdRatio > 0.5 && closedStrong)
      ? 'MODERATE'
      : (closedStrong)
        ? 'WEAK'
        : 'FAILED';

  return {
    crossedLevel: true,
    firstCrossTime: crossTime,
    holdRatio: r(holdRatio, 3),
    lastHourHoldRatio: r(lastHourHoldRatio, 3),
    volAccelerationOnCross: r(volAcceleration),
    retestFound,
    retestHeld,
    closedStrong,
    quality,
    intradayScore: r(clamp(
      holdRatio * 30
      + lastHourHoldRatio * 25
      + (volAcceleration > 1.3 ? 15 : volAcceleration > 1.0 ? 8 : 0)
      + (retestHeld ? 15 : retestFound ? 5 : 0)
      + (closedStrong ? 15 : 0)
    )),
  };
}

/**
 * Where is the stock relative to its 52-week range?
 * Breakouts near 52W high behave differently from mid-range breakouts
 */
function fiftyTwoWeekContext(weeklyCandles) {
  if (!weeklyCandles || weeklyCandles.length < 40) return null;

  const last52 = weeklyCandles.slice(-52);
  const high52 = Math.max(...last52.map(w => w.h));
  const low52 = Math.min(...last52.map(w => w.l));
  const currentClose = last52[last52.length - 1].c;
  const range52 = high52 - low52;

  const position = range52 > 0 ? (currentClose - low52) / range52 : 0.5;
  const distFromHigh = ((high52 - currentClose) / high52) * 100;
  const distFromLow = ((currentClose - low52) / low52) * 100;

  // is it at/near 52-week high?
  const nearHighZone = distFromHigh < 5;   // within 5% of 52W high
  const nearLowZone = position < 0.15;

  // count new 52W highs in last 12 weeks (momentum)
  let newHighCount = 0;
  let runningHigh = Math.max(...last52.slice(0, -12).map(w => w.h));
  for (const w of last52.slice(-12)) {
    if (w.h > runningHigh) {
      newHighCount++;
      runningHigh = w.h;
    }
  }

  return {
    high52: r(high52),
    low52: r(low52),
    currentClose: r(currentClose),
    position: r(position, 3),        // 0 = at 52W low, 1 = at 52W high
    distFromHigh: r(distFromHigh),
    distFromLow: r(distFromLow),
    nearHighZone,
    nearLowZone,
    newHighsLast12Weeks: newHighCount,
    // breakout context
    isAllTimeHighBreakout: distFromHigh < 2,
    isMidRangeBreakout: position > 0.3 && position < 0.7,
    isRecoveryBreakout: position < 0.4,
    contextScore: r(clamp(
      nearHighZone ? 85 :               // near 52W high = strong momentum
      position > 0.7 ? 70 :             // upper quartile
      position > 0.5 ? 55 :             // above midpoint
      position > 0.3 ? 40 :             // below midpoint but not bottom
      25                                 // bottom quartile
    )),
  };
}


// ═══════════════════════════════════════════════
//  FALSE BREAKOUT TRACKING
// ═══════════════════════════════════════════════

/**
 * How many recent breakout attempts failed?
 * Stock with 3 consecutive false breakouts should be penalized
 */
function falseBreakoutHistory(weeklyCandles, lookback = 20) {
  if (!weeklyCandles || weeklyCandles.length < lookback + 2) return null;

  const start = Math.max(1, weeklyCandles.length - lookback);
  let falseBreakouts = 0;
  let trueBreakouts = 0;
  let consecutiveFalse = 0;
  let maxConsecutiveFalse = 0;
  let lastFalseDate = null;

  for (let i = start; i < weeklyCandles.length - 1; i++) {
    const prev = weeklyCandles[i - 1];
    const curr = weeklyCandles[i];
    const next = weeklyCandles[i + 1];

    // did this week break above prev week's high?
    if (curr.h > prev.h && curr.c > prev.h) {
      // did the breakout hold next week?
      if (next.c > prev.h) {
        trueBreakouts++;
        consecutiveFalse = 0;
      } else {
        falseBreakouts++;
        consecutiveFalse++;
        maxConsecutiveFalse = Math.max(maxConsecutiveFalse, consecutiveFalse);
        lastFalseDate = curr.weekStart;
      }
    }
  }

  const totalAttempts = falseBreakouts + trueBreakouts;
  const falseRate = totalAttempts > 0 ? (falseBreakouts / totalAttempts) * 100 : 0;

  // recency — how long ago was last false breakout?
  let weeksSinceLastFalse = null;
  if (lastFalseDate) {
    weeksSinceLastFalse = Math.round(
      dayGap(lastFalseDate, weeklyCandles[weeklyCandles.length - 1].weekEnd) / 7
    );
  }

  // penalty score (higher = MORE false breakouts = WORSE)
  let penaltyScore = 0;
  if (falseRate > 60) penaltyScore = 40;
  else if (falseRate > 40) penaltyScore = 25;
  else if (falseRate > 20) penaltyScore = 10;

  if (maxConsecutiveFalse >= 3) penaltyScore += 20;
  else if (maxConsecutiveFalse >= 2) penaltyScore += 10;

  if (weeksSinceLastFalse !== null && weeksSinceLastFalse <= 3) penaltyScore += 15;

  return {
    totalAttempts,
    falseBreakouts,
    trueBreakouts,
    falseRate: r(falseRate),
    maxConsecutiveFalse,
    weeksSinceLastFalse,
    isSerialFalseBreaker: falseRate > 50 && totalAttempts >= 3,
    penaltyScore: r(clamp(penaltyScore)),
    // inverted: higher = better (fewer false breakouts)
    reliabilityScore: r(clamp(100 - penaltyScore)),
  };
}

/**
 * Check if multiple stocks in the same sector are breaking out
 * Sector-wide breakout = sector rotation = much higher probability
 *
 * @param {Array} allCandidates - all breakout candidates from scanner
 * @param {string} targetSector - the sector we're evaluating
 */
function sectorBreakoutConfirmation(allCandidates, targetSector) {
  if (!allCandidates || !targetSector) return null;

  const sectorStocks = allCandidates.filter(c => c.sector === targetSector);
  const totalCandidates = allCandidates.length;

  const sectorRatio = totalCandidates > 0
    ? sectorStocks.length / totalCandidates
    : 0;

  return {
    sectorStockCount: sectorStocks.length,
    totalCandidates,
    sectorConcentration: r(sectorRatio * 100),
    isSectorBreakout: sectorStocks.length >= 3,
    sectorStocks: sectorStocks.map(s => s.symbol),
    sectorScore: r(clamp(
      sectorStocks.length >= 4 ? 100 :
      sectorStocks.length >= 3 ? 80 :
      sectorStocks.length >= 2 ? 55 :
      sectorStocks.length >= 1 ? 30 : 10
    )),
  };
}

//  MARKET REGIME DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect if market is in trending or choppy regime
 * Breakouts succeed in trends, fail in chop
 */
function marketRegime(niftyWeekly, lookback = 12) {
  if (!niftyWeekly || niftyWeekly.length < lookback + 4) return null;

  const recent = niftyWeekly.slice(-lookback);
  const closes = recent.map(w => w.c);

  // directional movement
  const returns = [];
  for (let i = 1; i < closes.length; i++) {
    returns.push(((closes[i] - closes[i - 1]) / closes[i - 1]) * 100);
  }

  // trend strength: absolute sum of returns vs sum of absolute returns
  // efficiency ratio (Perry Kaufman)
  const totalMove = Math.abs(closes[closes.length - 1] - closes[0]);
  const totalPath = recent.reduce((s, w) => s + w.range, 0);
  const efficiency = totalPath > 0 ? totalMove / totalPath : 0;

  // count consecutive same-direction weeks
  let maxTrendWeeks = 0, currentTrend = 0;
  for (let i = 1; i < returns.length; i++) {
    if ((returns[i] > 0 && returns[i - 1] > 0) || (returns[i] < 0 && returns[i - 1] < 0)) {
      currentTrend++;
      maxTrendWeeks = Math.max(maxTrendWeeks, currentTrend);
    } else {
      currentTrend = 0;
    }
  }

  // ATR expansion/contraction (weekly)
  const ranges = recent.map(w => w.rangePercent);
  const avgRange = ranges.reduce((a, b) => a + b, 0) / ranges.length;
  const recentRange = ranges.slice(-4).reduce((a, b) => a + b, 0) / 4;
  const rangeExpanding = recentRange > avgRange * 1.2;

  // weekly SMA slope
  const sma20 = niftyWeekly.length >= 20
    ? niftyWeekly.slice(-20).reduce((s, w) => s + w.c, 0) / 20
    : null;
  const sma20_4wAgo = niftyWeekly.length >= 24
    ? niftyWeekly.slice(-24, -4).reduce((s, w) => s + w.c, 0) / 20
    : null;
  const smaSlope = sma20 && sma20_4wAgo
    ? ((sma20 - sma20_4wAgo) / sma20_4wAgo) * 100
    : 0;

  let regime;
  if (efficiency > 0.6 && maxTrendWeeks >= 4) regime = 'STRONG_TREND';
  else if (efficiency > 0.4 || maxTrendWeeks >= 3) regime = 'TRENDING';
  else if (efficiency < 0.2 && maxTrendWeeks <= 1) regime = 'CHOPPY';
  else regime = 'MIXED';

  // breakout success probability estimate
  const breakoutEdge = {
    STRONG_TREND: 0.65,
    TRENDING: 0.58,
    MIXED: 0.50,
    CHOPPY: 0.38,
  };

  return {
    regime,
    efficiency: r(efficiency, 3),
    maxConsecutiveTrendWeeks: maxTrendWeeks,
    avgWeeklyRange: r(avgRange),
    recentRangeVsAvg: r(recentRange / avgRange),
    rangeExpanding,
    smaSlope: r(smaSlope),
    estimatedBreakoutSuccessRate: breakoutEdge[regime],
    regimeScore: r(clamp(mapRange(efficiency, 0.15, 0.65, 20, 100))),
  };
}


// ═══════════════════════════════════════════════
//  PULLBACK / RETEST DETECTION
// ═══════════════════════════════════════════════

/**
 * After breakout day (Monday), check if Tuesday/Wednesday offers
 * a pullback retest entry — often BETTER than breakout-day entry
 *
 * @param {Object} breakoutDayCandle - the day it broke above PW High
 * @param {Object} retestDayCandle - the day we're checking for retest
 * @param {number} breakoutLevel - PW High (for longs) / PW Low (for shorts)
 * @param {string} direction
 */
function detectRetest(breakoutDayCandle, retestDayCandle, breakoutLevel, direction = 'LONG') {
  if (!breakoutDayCandle || !retestDayCandle || !breakoutLevel) return null;

  const isLong = direction === 'LONG';
  const rd = retestDayCandle;

  // did it pull back toward the breakout level?
  const touchedLevel = isLong
    ? rd.l <= breakoutLevel * 1.005   // came within 0.5% of PW High from above
    : rd.h >= breakoutLevel * 0.995;

  const stayedAbove = isLong
    ? rd.l >= breakoutLevel * 0.99    // didn't fall more than 1% below PW High
    : rd.h <= breakoutLevel * 1.01;

  const closedAbove = isLong
    ? rd.c > breakoutLevel
    : rd.c < breakoutLevel;

  const isBullishRetest = isLong
    ? touchedLevel && stayedAbove && closedAbove && rd.c > rd.o
    : touchedLevel && stayedAbove && closedAbove && rd.c < rd.o;

  // did it leave a long lower wick at the retest level? (rejection wick)
  const range = rd.h - rd.l;
  const lowerWick = isLong ? rd.o - rd.l : 0;
  const upperWick = !isLong ? rd.h - rd.o : 0;
  const rejectionWick = isLong
    ? (range > 0 && lowerWick / range > 0.50)
    : (range > 0 && upperWick / range > 0.50);

  // retest entry price (slightly above the breakout level)
  const retestEntry = isLong
    ? r(breakoutLevel * 1.002)
    : r(breakoutLevel * 0.998);

  // stop loss for retest entry (tighter than initial breakout entry)
  const retestStop = isLong
    ? r(breakoutLevel * 0.985)  // 1.5% below PW High
    : r(breakoutLevel * 1.015);

  return {
    touchedLevel,
    stayedAbove,
    closedAbove,
    isBullishRetest,
    rejectionWick,
    isValidRetest: touchedLevel && stayedAbove && closedAbove,
    isPerfectRetest: isBullishRetest && rejectionWick,
    retestEntry,
    retestStop,
    retestRiskPercent: r(Math.abs(retestEntry - retestStop) / retestEntry * 100),
    // retest gives better R:R because stop is tighter
    advantage: 'Tighter stop loss → better R:R vs breakout-day entry',
  };
}

function weeklySMA(weeklyCandles, period = 20) {
  if (!weeklyCandles || weeklyCandles.length < period) return null;
  return r(weeklyCandles.slice(-period).reduce((s, w) => s + w.c, 0) / period);
}

function weeklyEMA(weeklyCandles, period = 20) {
  if (!weeklyCandles || weeklyCandles.length < period) return null;
  const closes = weeklyCandles.map(w => w.c);
  const k = 2 / (period + 1);
  let val = closes.slice(0, period).reduce((a, b) => a + b, 0) / period;
  for (let i = period; i < closes.length; i++) val = closes[i] * k + val * (1 - k);
  return r(val);
}

function weeklyTrend(weeklyCandles, period = 20) {
  if (!weeklyCandles || weeklyCandles.length < period + 4) return null;
  const sma = weeklySMA(weeklyCandles, period);
  const sma4ago = weeklySMA(weeklyCandles.slice(0, -4), period);
  const close = weeklyCandles[weeklyCandles.length - 1].c;
  const above = close > sma;
  const slope = sma4ago > 0 ? ((sma - sma4ago) / sma4ago) * 100 : 0;

  let trend;
  if (above && slope > 0.5) trend = 'STRONG_UPTREND';
  else if (above) trend = 'UPTREND';
  else if (!above && slope < -0.5) trend = 'STRONG_DOWNTREND';
  else if (!above) trend = 'DOWNTREND';
  else trend = 'SIDEWAYS';

  return { trend, sma20: sma, aboveSMA: above, smaSlope: r(slope), distFromSMA: r(((close - sma) / sma) * 100) };
}

function dailyRSI(closes, period = 14) {
  if (!closes || closes.length < period + 1) return null;
  let ag = 0, al = 0;
  for (let i = 1; i <= period; i++) {
    const d = closes[i] - closes[i - 1];
    if (d >= 0) ag += d; else al -= d;
  }
  ag /= period; al /= period;
  for (let i = period + 1; i < closes.length; i++) {
    const d = closes[i] - closes[i - 1];
    ag = (ag * (period - 1) + Math.max(d, 0)) / period;
    al = (al * (period - 1) + Math.max(-d, 0)) / period;
  }
  return al === 0 ? 100 : r(100 - 100 / (1 + ag / al));
}

function dailyATR(candles, period = 14) {
  if (!candles || candles.length < period + 1) return null;
  const trs = [];
  for (let i = 1; i < candles.length; i++) {
    trs.push(Math.max(
      candles[i].h - candles[i].l,
      Math.abs(candles[i].h - candles[i - 1].c),
      Math.abs(candles[i].l - candles[i - 1].c)
    ));
  }
  let val = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
  for (let i = period; i < trs.length; i++) val = (val * (period - 1) + trs[i]) / period;
  return r(val);
}

/**
 * ADR — Average Daily Range (for dynamic targets)
 * More useful than PW range for setting profit targets
 */
function avgDailyRange(dailyCandles, period = 20) {
  if (!dailyCandles || dailyCandles.length < period) return null;
  const ranges = dailyCandles.slice(-period).map(c => c.h - c.l);
  const adr = ranges.reduce((a, b) => a + b, 0) / period;
  const adrPercent = dailyCandles[dailyCandles.length - 1].c > 0
    ? (adr / dailyCandles[dailyCandles.length - 1].c) * 100
    : 0;
  return { adr: r(adr), adrPercent: r(adrPercent) };
}

function avgDailyVolume(dailyCandles, period = 50) {
  if (!dailyCandles || dailyCandles.length < period) return null;
  return Math.round(dailyCandles.slice(-period).reduce((s, c) => s + c.v, 0) / period);
}

function avgWeeklyVolume(weeklyCandles, period = 10) {
  if (!weeklyCandles || weeklyCandles.length < period) return null;
  return Math.round(weeklyCandles.slice(-period).reduce((s, w) => s + w.v, 0) / period);
}

function analyzeOI(oiData, currentPrice, direction = 'LONG') {
  if (!oiData) return null;

  const oiChange = oiData.oiChangePercent || 0;
  const pcr = oiData.putCallRatio || 1;

  let interpretation;
  if (direction === 'LONG') {
    if (oiChange > 5) interpretation = 'FRESH_LONGS';
    else if (oiChange > 0) interpretation = 'MILD_LONG_BUILD';
    else if (oiChange > -5) interpretation = 'SHORT_COVERING';
    else interpretation = 'LONG_UNWINDING';
  } else {
    if (oiChange > 5) interpretation = 'FRESH_SHORTS';
    else if (oiChange > 0) interpretation = 'MILD_SHORT_BUILD';
    else if (oiChange > -5) interpretation = 'LONG_UNWINDING';
    else interpretation = 'SHORT_COVERING';
  }

  let oiScore = 50;
  if (direction === 'LONG') {
    if (interpretation === 'FRESH_LONGS') oiScore += 25;
    else if (interpretation === 'SHORT_COVERING') oiScore += 10;
    else if (interpretation === 'LONG_UNWINDING') oiScore -= 20;
    if (pcr > 1.2) oiScore += 15;
    if (pcr < 0.7) oiScore -= 15;
  } else {
    if (interpretation === 'FRESH_SHORTS') oiScore += 25;
    else if (interpretation === 'LONG_UNWINDING') oiScore += 10;
    else if (interpretation === 'SHORT_COVERING') oiScore -= 20;
    if (pcr < 0.7) oiScore += 15;
    if (pcr > 1.5) oiScore -= 15;
  }

  return {
    oiChange: r(oiChange),
    interpretation,
    putCallRatio: r(pcr),
    oiScore: r(clamp(oiScore)),
    oiSupport: oiData.highestPutOIStrike || null,
    oiResistance: oiData.highestCallOIStrike || null,
    maxPainStrike: oiData.maxPainStrike || null,
    supportsBias: oiScore >= 60,
  };
}


// ═══════════════════════════════════════════════
//  BREAKOUT / BREAKDOWN DETECTION

function detectBreakout(todayCandle, pwRange, avgVol) {
  if (!todayCandle || !pwRange) return null;
  const d = todayCandle;
  const volRatio = avgVol > 0 ? d.v / avgVol : 0;
  const range = d.h - d.l;
  const cs = range > 0 ? (d.c - d.l) / range : 0;

  const breakout = {
    occurred: d.c > pwRange.pwHigh,
    intradayTouch: d.h > pwRange.pwHigh && d.c <= pwRange.pwHigh,
    margin: r(((d.c - pwRange.pwHigh) / pwRange.pwHigh) * 100),
    volumeRatio: r(volRatio),
    volumeConfirm: volRatio >= 1.3,
    closeNearHigh: r(cs, 3),
    isBullishCandle: d.c > d.o,
    isCleanBreakout: d.c > pwRange.pwHigh && d.c > d.o && cs > 0.70 && volRatio >= 1.3,
  };

  const closeNearLow = range > 0 ? (d.h - d.c) / range : 0;
  const breakdown = {
    occurred: d.c < pwRange.pwLow,
    intradayTouch: d.l < pwRange.pwLow && d.c >= pwRange.pwLow,
    margin: r(((pwRange.pwLow - d.c) / pwRange.pwLow) * 100),
    volumeRatio: r(volRatio),
    volumeConfirm: volRatio >= 1.3,
    closeNearLow: r(closeNearLow, 3),
    isBearishCandle: d.c < d.o,
    isCleanBreakdown: d.c < pwRange.pwLow && d.c < d.o && closeNearLow > 0.70 && volRatio >= 1.3,
  };

  return { breakout, breakdown };
}

function detectMondayGapBreakout(mondayCandle, pwRange, fridayCandle) {
  if (!mondayCandle || !pwRange || !fridayCandle) return null;
  if (getDow(mondayCandle.date) !== 1) return { isMonday: false };

  const gap = ((mondayCandle.o - fridayCandle.c) / fridayCandle.c) * 100;
  const gapAbovePWH = mondayCandle.o > pwRange.pwHigh;
  const gapBelowPWL = mondayCandle.o < pwRange.pwLow;

  const gapHeldUp = gapAbovePWH && mondayCandle.l >= pwRange.pwHigh;
  const gapHeldDown = gapBelowPWL && mondayCandle.h <= pwRange.pwLow;

  return {
    isMonday: true,
    gapPercent: r(gap),
    gapUpBreakout: gapAbovePWH && gap > 0.3,
    gapDownBreakdown: gapBelowPWL && gap < -0.3,
    gapHeld: gapHeldUp || gapHeldDown,
    gapFilled: (gapAbovePWH && mondayCandle.l < pwRange.pwHigh)
      || (gapBelowPWL && mondayCandle.h > pwRange.pwLow),
    isPowerBreakout: gapAbovePWH && gapHeldUp && mondayCandle.c > mondayCandle.o,
    isPowerBreakdown: gapBelowPWL && gapHeldDown && mondayCandle.c < mondayCandle.o,
  };
}

function getDow(dateStr) { return new Date(dateStr).getDay(); }

function dayGap(d1, d2) {
  return Math.round(Math.abs(new Date(d2) - new Date(d1)) / 864e5);
}

function extractTime(timeStr) {
  if (!timeStr) return '00:00';
  if (timeStr.includes('T')) return timeStr.split('T')[1].substring(0, 5);
  return timeStr.substring(0, 5);
}

function linearSlope(values) {
  const n = values.length;
  if (n < 2) return 0;
  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
  for (let i = 0; i < n; i++) {
    sumX += i; sumY += values[i]; sumXY += i * values[i]; sumX2 += i * i;
  }
  const denom = n * sumX2 - sumX * sumX;
  return denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
}

function r(val, d = 2) {
  if (val == null || isNaN(val)) return null;
  return Math.round(val * 10 ** d) / 10 ** d;
}

function clamp(val, min = 0, max = 100) {
  return Math.min(max, Math.max(min, val));
}

function mapRange(val, inMin, inMax, outMin, outMax) {
  return ((val - inMin) / (inMax - inMin)) * (outMax - outMin) + outMin;
}
```

## `weekly-scanner.js`

```javascript
const ind = require('./weekly-indicators');

/**
 * Two-phase scanner:
 *   PHASE 1 (WEEKEND):  Build watchlist from weekly structure + multi-week base + RS
 *   PHASE 2 (MON-TUE):  Confirm breakout + check intraday quality + detect retest
 */
class WeeklyBreakoutScanner {
  constructor(config = {}) {
    this.c = {
      // base filters
      maxPWRangePercent: config.maxPWRangePercent ?? 5.0,
      minWeeklyVolMultiple: config.minWeeklyVolMultiple ?? 1.0,
      weeklySMAPeriod: config.weeklySMAPeriod ?? 20,
      requireFnO: config.requireFnO ?? true,

      // enhanced filters
      minBaseWeeks: config.minBaseWeeks ?? 2,
      maxFalseBreakoutRate: config.maxFalseBreakoutRate ?? 65,
      minRSScore: config.minRSScore ?? 30,
      shortWeekPenalty: config.shortWeekPenalty ?? true,

      // confirmation
      minDailyVolMultiple: config.minDailyVolMultiple ?? 1.3,
      allowRetest: config.allowRetest ?? true,
      maxConfirmDays: config.maxConfirmDays ?? 3,  // Mon + Tue + Wed

      // shorts
      allowShorts: config.allowShorts ?? true,
      requireBearMarketForShorts: config.requireBearMarketForShorts ?? true,
    };
  }

  // ═══════════════════════════════════════
  //  PHASE 1: WEEKEND WATCHLIST
  // ═══════════════════════════════════════

  /**
   * @param {Array<StockData>} stocks
   *   [{
   *     symbol, sector, isFnO,
   *     dailyCandles: [{date,o,h,l,c,v}, ...],  // 6+ months
   *     intradayCandles: null,                    // only needed for Phase 2
   *     oiData: {...} | null
   *   }]
   *
   * @param {Object} market
   *   {
   *     niftyDailyCandles: [...],
   *     indiaVix, isExpiryWeek, majorEventThisWeek,
   *     sectorPerformance: { AUTO: 1.5, ... }
   *   }
   */
  weekendScan(stocks, market) {
    // build nifty weekly for regime + RS
    const niftyWeekly = ind.buildWeeklyCandles(market.niftyDailyCandles || []);
    const regime = ind.marketRegime(niftyWeekly, 12);
    const niftyTrend = ind.weeklyTrend(niftyWeekly, this.c.weeklySMAPeriod);

    const niftySMA50 = market.niftyDailyCandles
      ? (() => {
          const closes = market.niftyDailyCandles.map(c => c.c);
          if (closes.length < 50) return null;
          return closes.slice(-50).reduce((a, b) => a + b, 0) / 50;
        })()
      : null;
    const niftyLatest = niftyWeekly.length > 0 ? niftyWeekly[niftyWeekly.length - 1].c : null;
    const niftyAbove50SMA = niftySMA50 && niftyLatest ? niftyLatest > niftySMA50 : true;

    const breakoutWatch = [];
    const breakdownWatch = [];
    const rejected = [];

    for (const stock of stocks) {
      try {
        const evaluation = this._weekendEval(stock, niftyWeekly, niftyTrend, niftyAbove50SMA, regime);

        if (evaluation.breakoutCandidate) {
          breakoutWatch.push(evaluation);
        } else if (evaluation.breakdownCandidate) {
          breakdownWatch.push(evaluation);
        } else {
          rejected.push({ symbol: stock.symbol, reason: evaluation.rejectionReason });
        }
      } catch (err) {
        rejected.push({ symbol: stock.symbol, error: err.message });
      }
    }

    // sort by composite weekend score
    breakoutWatch.sort((a, b) => b.weekendScore - a.weekendScore);
    breakdownWatch.sort((a, b) => b.weekendScore - a.weekendScore);

    // sector confirmation pass
    const allCandidates = [...breakoutWatch, ...breakdownWatch];
    for (const c of allCandidates) {
      c.sectorConfirmation = ind.sectorBreakoutConfirmation(allCandidates, c.sector);
    }

    return {
      breakoutWatchlist: breakoutWatch,
      breakdownWatchlist: breakdownWatch,
      rejected,
      regime,
      niftyTrend,
      meta: {
        scanned: stocks.length,
        breakoutCandidates: breakoutWatch.length,
        breakdownCandidates: breakdownWatch.length,
        marketRegime: regime?.regime,
        breakoutSuccessEstimate: regime?.estimatedBreakoutSuccessRate,
      },
    };
  }

  _weekendEval(stock, niftyWeekly, niftyTrend, niftyAbove50SMA, regime) {
    const daily = stock.dailyCandles;
    if (!daily || daily.length < 100) {
      return { breakoutCandidate: false, breakdownCandidate: false, rejectionReason: 'Insufficient data (<100 days)' };
    }

    const weekly = ind.buildWeeklyCandles(daily);
    if (weekly.length < 24) {
      return { breakoutCandidate: false, breakdownCandidate: false, rejectionReason: 'Need 24+ weeks' };
    }

    // ── core weekly data ──
    const pw = ind.previousWeekRange(weekly);
    if (!pw) return { breakoutCandidate: false, breakdownCandidate: false, rejectionReason: 'No PW range' };

    const trend = ind.weeklyTrend(weekly, this.c.weeklySMAPeriod);
    const base = ind.detectBase(weekly, 12);
    const rs = ind.relativeStrength(weekly, niftyWeekly, 12);
    const volPattern = ind.preBreakoutVolumePattern(weekly, Math.min(base?.baseWeeks || 6, 8));
    const falseBO = ind.falseBreakoutHistory(weekly, 20);
    const context52w = ind.fiftyTwoWeekContext(weekly);
    const pwCandleBias = ind.candleFollowThroughBias(pw.pwCandleType, 'LONG');

    // daily indicators
    const allCloses = daily.map(c => c.c);
    const rsi = ind.dailyRSI(allCloses, 14);
    const atr = ind.dailyATR(daily, 14);
    const adrData = ind.avgDailyRange(daily, 20);
    const avgVol = ind.avgDailyVolume(daily, 50);
    const avgWeekVol = ind.avgWeeklyVolume(weekly, 10);

    // ── FILTERS ──
    const checks = {};

    // 1. PW range < 5%
    checks.tightRange = { value: pw.pwRangePercent, pass: pw.pwRangePercent < this.c.maxPWRangePercent };

    // 2. Short week penalty (3-day PW is unreliable)
    checks.notShortWeek = { value: pw.pwTradingDays, pass: !this.c.shortWeekPenalty || !pw.isShortWeek };

    // 3. F&O
    checks.fno = { value: stock.isFnO, pass: this.c.requireFnO ? stock.isFnO : true };

    // 4. Volume
    const weekVolRatio = avgWeekVol > 0 ? pw.pwVolume / avgWeekVol : 0;
    checks.volume = { value: ind.r(weekVolRatio), pass: weekVolRatio >= this.c.minWeeklyVolMultiple };

    // 5. Trend for longs
    checks.trendLong = { value: trend?.trend, pass: trend ? trend.aboveSMA : false };

    // 6. Trend for shorts
    checks.trendShort = { value: trend?.trend, pass: trend ? !trend.aboveSMA : false };

    // 7. RSI (long: 45-75, short: 25-55)
    checks.rsiLong = { value: rsi, pass: rsi != null && rsi > 45 && rsi < 75 };
    checks.rsiShort = { value: rsi, pass: rsi != null && rsi > 25 && rsi < 55 };

    // 8. False breakout history
    checks.reliability = {
      value: falseBO?.falseRate,
      pass: !falseBO || falseBO.falseRate <= this.c.maxFalseBreakoutRate,
    };

    // 9. Relative strength (minimum threshold)
    checks.rsStrength = {
      value: rs?.rsScore,
      pass: !rs || rs.rsScore >= this.c.minRSScore,
    };

    // ── determine candidacy ──
    const isBreakoutCandidate =
      checks.tightRange.pass
      && checks.fno.pass
      && checks.volume.pass
      && checks.trendLong.pass
      && checks.rsiLong.pass
      && checks.reliability.pass
      && checks.rsStrength.pass
      && (pw.closedNearHigh || pw.closedMidRange)  // not closed at low
      && checks.notShortWeek.pass;

    const isBreakdownCandidate =
      this.c.allowShorts
      && checks.tightRange.pass
      && checks.fno.pass
      && checks.volume.pass
      && checks.trendShort.pass
      && checks.rsiShort.pass
      && checks.reliability.pass
      && pw.closedNearLow
      && (!this.c.requireBearMarketForShorts || !niftyAbove50SMA);

    // rejection reason
    let rejectionReason = null;
    if (!isBreakoutCandidate && !isBreakdownCandidate) {
      const failedChecks = Object.entries(checks)
        .filter(([k, v]) => !v.pass && !k.includes('Short'))
        .map(([k]) => k);
      rejectionReason = failedChecks.length > 0
        ? `Failed: ${failedChecks.join(', ')}`
        : 'Close position not near PW High or Low';
    }

    // ── weekend score ──
    let weekendScore = 0;
    const direction = isBreakoutCandidate ? 'LONG' : isBreakdownCandidate ? 'SHORT' : null;

    if (direction) {
      weekendScore = this._weekendScore(
        pw, base, trend, rs, volPattern, falseBO, context52w,
        rsi, regime, pwCandleBias, direction
      );
    }

    // ── build levels ──
    // ADR-based targets (more dynamic than PW range)
    const adrTarget = adrData
      ? ind.r(pw.pwHigh + adrData.adr * 3)
      : ind.r(pw.pwHigh + pw.pwRange);

    return {
      symbol: stock.symbol,
      sector: stock.sector,
      isFnO: stock.isFnO,
      breakoutCandidate: isBreakoutCandidate,
      breakdownCandidate: isBreakdownCandidate,
      direction,
      rejectionReason,
      weekendScore: ind.r(weekendScore),

      levels: {
        breakoutTrigger: pw.pwHigh,
        breakdownTrigger: pw.pwLow,
        stopLossLong: base?.exists ? base.baseLowest : pw.pwLow,
        stopLossLongMid: pw.pwMid,
        stopLossShort: base?.exists ? base.baseHighest : pw.pwHigh,
        targetLong: adrTarget,
        targetLongConservative: ind.r(pw.pwHigh + pw.pwRange),
        targetShort: adrData
          ? ind.r(pw.pwLow - adrData.adr * 3)
          : ind.r(pw.pwLow - pw.pwRange),
      },

      pw,
      base,
      trend,
      relativeStrength: rs,
      volumePattern: volPattern,
      falseBreakoutHistory: falseBO,
      fiftyTwoWeek: context52w,
      regime,
      pwCandleBias,

      indicators: {
        rsi14: rsi,
        atr14: atr,
        adr: adrData,
        avgDailyVolume: avgVol,
        avgWeeklyVolume: avgWeekVol,
      },
      checks,
    };
  }

  _weekendScore(pw, base, trend, rs, volPattern, falseBO, w52, rsi, regime, candleBias, direction) {
    let s = 0;
    const isLong = direction === 'LONG';

    // 1. Multi-week base quality [0-25]
    if (base) {
      s += (base.baseScore / 100) * 25;
    }

    // 2. Trend alignment [0-15]
    if (trend) {
      const idealTrend = isLong ? 'STRONG_UPTREND' : 'STRONG_DOWNTREND';
      const goodTrend = isLong ? 'UPTREND' : 'DOWNTREND';
      if (trend.trend === idealTrend) s += 15;
      else if (trend.trend === goodTrend) s += 10;
      else s += 3;
    }

    // 3. Relative strength [0-12]
    if (rs) {
      if (isLong) s += (rs.rsScore / 100) * 12;
      else s += ((100 - rs.rsScore) / 100) * 12;
    }

    // 4. Pre-breakout volume pattern [0-12]
    if (volPattern) {
      s += (volPattern.accumScore / 100) * 12;
    }

    // 5. PW close position [0-8]
    if (isLong) s += Math.min(pw.pwClosePosition * 8, 8);
    else s += Math.min((1 - pw.pwClosePosition) * 8, 8);

    // 6. False breakout history [0-8]
    if (falseBO) {
      s += (falseBO.reliabilityScore / 100) * 8;
    }

    // 7. 52-week context [0-7]
    if (w52) {
      s += (w52.contextScore / 100) * 7;
    }

    // 8. Market regime [0-5]
    if (regime) {
      s += (regime.regimeScore / 100) * 5;
    }

    // 9. PW candle type bias [0-4]
    s += candleBias * 4;

    // 10. RSI positioning [0-4]
    if (rsi != null) {
      if (isLong && rsi >= 55 && rsi <= 68) s += 4;
      else if (isLong && rsi >= 48 && rsi <= 72) s += 2;
      else if (!isLong && rsi >= 32 && rsi <= 45) s += 4;
      else if (!isLong && rsi >= 28 && rsi <= 52) s += 2;
    }

    return s;
  }


  // ═══════════════════════════════════════
  //  PHASE 2: INTRA-WEEK CONFIRMATION
  // ═══════════════════════════════════════

  /**
   * Called on Monday, Tuesday, Wednesday with that day's data
   *
   * @param {Array} watchlist - from weekendScan
   * @param {Object} todayCandles - { 'SYMBOL': {date,o,h,l,c,v}, ... }
   * @param {Object} intradayData - { 'SYMBOL': [{time,o,h,l,c,v}, ...], ... } 5-min candles (optional)
   * @param {Object} oiDataMap - { 'SYMBOL': {...oiData}, ... }
   * @param {Object|null} yesterdayCandles - for retest detection (day after breakout)
   */
  confirmationScan(watchlist, todayCandles, intradayData = {}, oiDataMap = {}, yesterdayCandles = null) {
    const confirmed = [];
    const retestEntries = [];
    const pending = [];

    for (const candidate of watchlist) {
      const sym = candidate.symbol;
      const today = todayCandles[sym];
      if (!today) { pending.push({ symbol: sym, reason: 'No data' }); continue; }

      const isLong = candidate.direction === 'LONG';
      const avgVol = candidate.indicators.avgDailyVolume;
      const pw = candidate.pw;

      // ── CHECK FOR BREAKOUT ──
      const detection = ind.detectBreakout(today, pw, avgVol);
      if (!detection) { pending.push({ symbol: sym, reason: 'Detection error' }); continue; }

      // Monday gap check
      const fridayClose = pw.pwClose;
      const mondayGap = ind.detectMondayGapBreakout(today, pw, { c: fridayClose, date: pw.weekEnd });

      // OI
      const oi = oiDataMap[sym] ? ind.analyzeOI(oiDataMap[sym], today.c, candidate.direction) : null;

      // intraday quality (if available)
      const intradayQuality = intradayData[sym]
        ? ind.breakoutDayIntradayAnalysis(
            intradayData[sym],
            isLong ? pw.pwHigh : pw.pwLow,
            candidate.direction
          )
        : null;

      // ── BREAKOUT CONFIRMED ──
      if (isLong && detection.breakout.occurred) {
        confirmed.push({
          ...candidate,
          entryType: 'BREAKOUT',
          confirmationType: detection.breakout.isCleanBreakout ? 'CLEAN' : 'WEAK',
          todayCandle: today,
          breakoutData: detection.breakout,
          mondayGap,
          oiAnalysis: oi,
          intradayQuality,
          confirmationScore: this._confirmScore(
            detection.breakout, mondayGap, oi, intradayQuality, candidate
          ),
        });
        continue;
      }

      if (!isLong && detection.breakdown.occurred) {
        confirmed.push({
          ...candidate,
          entryType: 'BREAKDOWN',
          confirmationType: detection.breakdown.isCleanBreakdown ? 'CLEAN' : 'WEAK',
          todayCandle: today,
          breakdownData: detection.breakdown,
          mondayGap,
          oiAnalysis: oi,
          intradayQuality,
          confirmationScore: this._confirmScore(
            detection.breakdown, mondayGap, oi, intradayQuality, candidate
          ),
        });
        continue;
      }

      // ── RETEST CHECK (if yesterday was breakout day) ──
      if (this.c.allowRetest && yesterdayCandles?.[sym]) {
        const yesterday = yesterdayCandles[sym];
        const yesterdayDetection = ind.detectBreakout(yesterday, pw, avgVol);

        if (yesterdayDetection) {
          const yesterdayBroke = isLong
            ? yesterdayDetection.breakout.occurred
            : yesterdayDetection.breakdown.occurred;

          if (yesterdayBroke) {
            const retest = ind.detectRetest(
              yesterday, today,
              isLong ? pw.pwHigh : pw.pwLow,
              candidate.direction
            );

            if (retest?.isValidRetest) {
              retestEntries.push({
                ...candidate,
                entryType: 'RETEST',
                confirmationType: retest.isPerfectRetest ? 'PERFECT_RETEST' : 'VALID_RETEST',
                todayCandle: today,
                yesterdayCandle: yesterday,
                retestData: retest,
                oiAnalysis: oi,
                // retest entries get a bonus for better R:R
                confirmationScore: (candidate.weekendScore + 15)
                  * (retest.isPerfectRetest ? 1.2 : 1.0),
              });
              continue;
            }
          }
        }
      }

      // ── STILL PENDING ──
      if (detection.breakout.intradayTouch || detection.breakdown.intradayTouch) {
        pending.push({
          symbol: sym,
          reason: `Touched level intraday but closed back inside — watch tomorrow`,
          watchTomorrow: true,
        });
      } else {
        pending.push({
          symbol: sym,
          reason: 'No breakout yet',
          watchTomorrow: true,
        });
      }
    }

    // merge confirmed + retest entries, sort by score
    const allConfirmed = [...confirmed, ...retestEntries];
    allConfirmed.sort((a, b) => b.confirmationScore - a.confirmationScore);

    return {
      confirmed: allConfirmed,
      pending: pending.filter(p => p.watchTomorrow),
      meta: {
        watchlistSize: watchlist.length,
        breakoutConfirmed: confirmed.length,
        retestEntries: retestEntries.length,
        stillWatching: pending.filter(p => p.watchTomorrow).length,
      },
    };
  }

  _confirmScore(breakData, mondayGap, oi, intraday, candidate) {
    let s = candidate.weekendScore || 0;

    // breakout quality [0-20]
    const clean = breakData.isCleanBreakout || breakData.isCleanBreakdown;
    s += clean ? 20 : breakData.occurred ? 10 : 0;

    // volume [0-12]
    if (breakData.volumeConfirm) s += 12;
    else if (breakData.volumeRatio > 1.0) s += 6;

    // close strength [0-8]
    const cs = breakData.closeNearHigh || breakData.closeNearLow || 0;
    s += cs * 8;

    // Monday gap [0-12]
    if (mondayGap?.isPowerBreakout || mondayGap?.isPowerBreakdown) s += 12;
    else if (mondayGap?.gapHeld) s += 8;
    else if (mondayGap?.gapUpBreakout || mondayGap?.gapDownBreakdown) s += 5;

    // OI [0-8]
    if (oi?.supportsBias) s += 8;
    else if (oi) s += Math.max(0, (oi.oiScore - 50) / 6);

    // intraday quality [0-10]
    if (intraday?.quality === 'STRONG') s += 10;
    else if (intraday?.quality === 'MODERATE') s += 6;
    else if (intraday?.quality === 'WEAK') s += 2;

    // margin [0-5]
    const margin = Math.abs(breakData.margin || 0);
    if (margin > 1.5) s += 5;
    else if (margin > 0.5) s += 3;

    return ind.r(s);
  }
}

module.exports = { WeeklyBreakoutScanner };
```

```javascript
const ind = require('./weekly-indicators');

/**
 * Entry / exit signal engine
 * Handles: breakout entry, retest entry, daily management, Friday exit
 */
class WeeklyBreakoutSignalEngine {
  constructor(config = {}) {
    this.stopLossMode = config.stopLossMode ?? 'BASE_LOW';  // BASE_LOW | PW_LOW | PW_MID | ATR
    this.atrStopMultiple = config.atrStopMultiple ?? 2.0;
    this.targetMode = config.targetMode ?? 'ADR_MULTI';     // ADR_MULTI | RANGE_EXP | ATR | TRAIL
    this.adrTargetMultiple = config.adrTargetMultiple ?? 3;
    this.trailActivation = config.trailActivation ?? 3.0;   // % profit to start trailing
    this.trailProtection = config.trailProtection ?? 0.6;   // protect 60% of peak profit
    this.failedBreakoutDays = config.failedBreakoutDays ?? 2;
    this.maxHoldDays = config.maxHoldDays ?? 5;
  }

  // ═══════════════════════════════════════
  //  ENTRY SIGNAL
  // ═══════════════════════════════════════

  generateEntry(confirmed, positionSize = null) {
    const dir = confirmed.direction;
    const isLong = dir === 'LONG';
    const isRetest = confirmed.entryType === 'RETEST';

    // entry price
    let entryPrice;
    if (isRetest && confirmed.retestData) {
      entryPrice = confirmed.retestData.retestEntry;
    } else {
      entryPrice = confirmed.todayCandle.c;
    }

    const pw = confirmed.pw;
    const base = confirmed.base;
    const atr = confirmed.indicators?.atr14;
    const adrData = confirmed.indicators?.adr;

    // ── STOP LOSS ──
    let stopLoss;
    switch (this.stopLossMode) {
      case 'BASE_LOW':
        // use base low (for multi-week consolidation) — widest, most reliable
        if (isRetest && confirmed.retestData) {
          stopLoss = confirmed.retestData.retestStop;  // tighter for retest
        } else if (base?.exists && isLong) {
          stopLoss = base.baseLowest;
        } else if (base?.exists && !isLong) {
          stopLoss = base.baseHighest;
        } else {
          stopLoss = isLong ? pw.pwLow : pw.pwHigh;
        }
        break;
      case 'PW_LOW':
        stopLoss = isLong ? pw.pwLow : pw.pwHigh;
        break;
      case 'PW_MID':
        stopLoss = pw.pwMid;
        break;
      case 'ATR':
        if (atr) {
          stopLoss = isLong
            ? ind.r(entryPrice - atr * this.atrStopMultiple)
            : ind.r(entryPrice + atr * this.atrStopMultiple);
        } else {
          stopLoss = isLong ? pw.pwLow : pw.pwHigh;
        }
        break;
      default:
        stopLoss = isLong ? pw.pwLow : pw.pwHigh;
    }

    // cap stop at 7% max
    const stopPercent = Math.abs((entryPrice - stopLoss) / entryPrice) * 100;
    if (stopPercent > 7) {
      stopLoss = isLong ? ind.r(entryPrice * 0.93) : ind.r(entryPrice * 1.07);
    }

    // ── TARGET ──
    let target;
    switch (this.targetMode) {
      case 'ADR_MULTI':
        if (adrData) {
          target = isLong
            ? ind.r(entryPrice + adrData.adr * this.adrTargetMultiple)
            : ind.r(entryPrice - adrData.adr * this.adrTargetMultiple);
        } else {
          target = isLong
            ? ind.r(pw.pwHigh + pw.pwRange)
            : ind.r(pw.pwLow - pw.pwRange);
        }
        break;
      case 'RANGE_EXP':
        target = isLong
          ? ind.r(pw.pwHigh + pw.pwRange)
          : ind.r(pw.pwLow - pw.pwRange);
        break;
      case 'ATR':
        target = atr
          ? isLong
            ? ind.r(entryPrice + atr * 3)
            : ind.r(entryPrice - atr * 3)
          : isLong
            ? ind.r(pw.pwHigh + pw.pwRange)
            : ind.r(pw.pwLow - pw.pwRange);
        break;
      case 'TRAIL':
        target = null;
        break;
      default:
        target = isLong
          ? ind.r(pw.pwHigh + pw.pwRange)
          : ind.r(pw.pwLow - pw.pwRange);
    }

    const riskPerShare = Math.abs(entryPrice - stopLoss);
    const rewardPerShare = target ? Math.abs(target - entryPrice) : null;

    const dow = ind.getDow(confirmed.todayCandle.date);
    const isMondayGap = confirmed.mondayGap?.isPowerBreakout || confirmed.mondayGap?.isPowerBreakdown;

    return {
      symbol: confirmed.symbol,
      direction: dir,
      action: isLong ? 'BUY' : 'SELL_SHORT',
      orderType: isLong ? 'CNC' : 'NRML',
      entryType: confirmed.entryType,  // BREAKOUT | RETEST
      entryPrice: ind.r(entryPrice),
      stopLoss: ind.r(stopLoss),
      stopLossPercent: ind.r(Math.abs((entryPrice - stopLoss) / entryPrice) * 100),
      target: target ? ind.r(target) : null,
      targetPercent: target ? ind.r((Math.abs(target - entryPrice) / entryPrice) * 100) : null,
      riskRewardRatio: rewardPerShare && riskPerShare > 0 ? ind.r(rewardPerShare / riskPerShare) : null,
      riskPerShare: ind.r(riskPerShare),

      levels: {
        pwHigh: pw.pwHigh,
        pwLow: pw.pwLow,
        pwMid: pw.pwMid,
        baseHigh: base?.baseHighest || pw.pwHigh,
        baseLow: base?.baseLowest || pw.pwLow,
      },

      entryDate: confirmed.todayCandle.date,
      entryDayOfWeek: dow,
      expectedHoldDays: Math.max(1, 5 - dow),
      isMondayGapEntry: isMondayGap,
      isRetestEntry: isRetest,
      confirmationType: confirmed.confirmationType,
      confirmationScore: confirmed.confirmationScore,
      weekendScore: confirmed.weekendScore,
      sector: confirmed.sector,

      // context for exit decisions
      _context: {
        base: confirmed.base,
        relativeStrength: confirmed.relativeStrength,
        fiftyTwoWeek: confirmed.fiftyTwoWeek,
        falseBreakoutHistory: confirmed.falseBreakoutHistory,
        intradayQuality: confirmed.intradayQuality,
        oiAnalysis: confirmed.oiAnalysis,
        pwCandleType: pw.pwCandleType,
      },

      // mutable state
      _state: {
        daysInsidePW: 0,
        highSinceEntry: entryPrice,
        lowSinceEntry: entryPrice,
        trailingActive: false,
        trailingStop: null,
        daysSinceEntry: 0,
      },

      quantity: positionSize?.quantity ?? null,
      investmentAmount: positionSize ? ind.r(positionSize.quantity * entryPrice) : null,
      maxLoss: positionSize ? ind.r(positionSize.quantity * riskPerShare) : null,

      timestamp: new Date().toISOString(),
    };
  }


  // ═══════════════════════════════════════
  //  DAILY EXIT MANAGEMENT
  // ═══════════════════════════════════════

  dailyExitCheck(entry, todayCandle, dayNumber) {
    const isLong = entry.direction === 'LONG';
    const price = todayCandle.c;
    const state = entry._state;

    state.daysSinceEntry = dayNumber;
    state.highSinceEntry = Math.max(state.highSinceEntry, todayCandle.h);
    state.lowSinceEntry = Math.min(state.lowSinceEntry, todayCandle.l);

    const profit = isLong
      ? ((price - entry.entryPrice) / entry.entryPrice) * 100
      : ((entry.entryPrice - price) / entry.entryPrice) * 100;

    const peakProfit = isLong
      ? ((state.highSinceEntry - entry.entryPrice) / entry.entryPrice) * 100
      : ((entry.entryPrice - state.lowSinceEntry) / entry.entryPrice) * 100;

    const dow = ind.getDow(todayCandle.date);

    // ── STOP LOSS ──
    const slHit = isLong ? todayCandle.l <= entry.stopLoss : todayCandle.h >= entry.stopLoss;
    if (slHit) {
      return {
        action: 'EXIT_STOP_LOSS',
        exitPrice: entry.stopLoss,
        profitPercent: ind.r(-entry.stopLossPercent),
        reason: 'Stop loss breached',
        dayNumber,
      };
    }

    // ── TARGET HIT ──
    if (entry.target) {
      const tgtHit = isLong ? todayCandle.h >= entry.target : todayCandle.l <= entry.target;
      if (tgtHit) {
        return {
          action: 'EXIT_TARGET',
          exitPrice: entry.target,
          profitPercent: entry.targetPercent,
          reason: 'Target reached',
          dayNumber,
        };
      }
    }

    // ── FRIDAY EXIT ──
    if (dow === 5 || dayNumber >= this.maxHoldDays) {
      return {
        action: 'EXIT_FRIDAY',
        exitPrice: price,
        profitPercent: ind.r(profit),
        reason: dow === 5 ? 'Friday close — end of week' : `Max hold ${this.maxHoldDays} days`,
        dayNumber,
      };
    }

    // ── FAILED BREAKOUT (back inside PW for 2+ days) ──
    const insidePW = isLong
      ? price < entry.levels.pwHigh && price > entry.levels.pwLow
      : price > entry.levels.pwLow && price < entry.levels.pwHigh;

    if (insidePW) state.daysInsidePW++;
    else state.daysInsidePW = 0;

    if (state.daysInsidePW >= this.failedBreakoutDays) {
      return {
        action: 'EXIT_FAILED_BREAKOUT',
        exitPrice: price,
        profitPercent: ind.r(profit),
        reason: `Back inside PW range ${state.daysInsidePW} days — breakout failed`,
        dayNumber,
      };
    }

    // ── TRAILING STOP ──
    if (profit >= this.trailActivation) {
      state.trailingActive = true;
      const peak = isLong ? state.highSinceEntry : state.lowSinceEntry;
      const protectedMove = Math.abs(peak - entry.entryPrice) * this.trailProtection;

      let trail = isLong
        ? ind.r(entry.entryPrice + protectedMove)
        : ind.r(entry.entryPrice - protectedMove);

      // trail can't be worse than breakeven + 0.5%
      trail = isLong
        ? Math.max(trail, entry.entryPrice * 1.005)
        : Math.min(trail, entry.entryPrice * 0.995);

      state.trailingStop = trail;

      const trailHit = isLong ? price <= trail : price >= trail;
      if (trailHit) {
        const lockInProfit = isLong
          ? ((trail - entry.entryPrice) / entry.entryPrice) * 100
          : ((entry.entryPrice - trail) / entry.entryPrice) * 100;

        return {
          action: 'EXIT_TRAILING_STOP',
          exitPrice: trail,
          profitPercent: ind.r(lockInProfit),
          peakProfit: ind.r(peakProfit),
          reason: `Trailing stop at ₹${trail} after ${ind.r(peakProfit)}% peak`,
          dayNumber,
        };
      }
    }

    // ── THURSDAY: tighten for Friday ──
    if (dow === 4 && profit > 1) {
      const tightStop = isLong
        ? ind.r(Math.max(entry.entryPrice * 1.005, state.trailingStop || entry.stopLoss))
        : ind.r(Math.min(entry.entryPrice * 0.995, state.trailingStop || entry.stopLoss));

      entry.stopLoss = tightStop;

      return {
        action: 'HOLD_TIGHTEN',
        currentProfit: ind.r(profit),
        peakProfit: ind.r(peakProfit),
        stopLoss: tightStop,
        reason: 'Thursday — tightened stop. Exits tomorrow.',
        dayNumber,
        exitTomorrow: true,
      };
    }

    // ── HOLD ──
    return {
      action: 'HOLD',
      currentProfit: ind.r(profit),
      peakProfit: ind.r(peakProfit),
      stopLoss: entry.stopLoss,
      trailingStop: state.trailingStop,
      target: entry.target,
      daysInsidePW: state.daysInsidePW,
```

```javascript
const ind = require('./weekly-indicators');

class WeeklyBreakoutRiskManager {
  constructor(config = {}) {
    this.totalCapital = config.totalCapital ?? 500000;
    this.maxAllocation = config.maxAllocation ?? 0.60;
    this.maxPositions = config.maxPositions ?? 5;
    this.maxPerStock = config.maxPerStock ?? 0.15;
    this.maxRiskPerTrade = config.maxRiskPerTrade ?? 0.02;
    this.maxTotalRisk = config.maxTotalRisk ?? 0.06;
    this.maxSectorConc = config.maxSectorConc ?? 2;
  }

  preWeekGate(market) {
    const blocks = [];
    const warnings = [];

    if (market.indiaVix > 25) {
      blocks.push({ rule: 'VIX', msg: `VIX ${market.indiaVix} > 25 — skip weekly trades` });
    } else if (market.indiaVix > 20) {
      warnings.push({ rule: 'VIX', msg: `VIX ${market.indiaVix} elevated`, sizeMultiplier: 0.6 });
    }

    if (market.isExpiryWeek) {
      warnings.push({ rule: 'EXPIRY', msg: 'F&O expiry week — OI-driven whipsaws', sizeMultiplier: 0.7 });
    }

    if (market.majorEventThisWeek) {
      warnings.push({
        rule: 'EVENT', msg: `Event: ${market.majorEventThisWeek}`, sizeMultiplier: 0.5,
      });
    }

    // regime-based adjustment
    if (market.regime?.regime === 'CHOPPY') {
      warnings.push({
        rule: 'CHOPPY', msg: 'Choppy regime — breakouts less reliable', sizeMultiplier: 0.5,
      });
    }

    let sizeMultiplier = 1.0;
    for (const w of warnings) {
      if (w.sizeMultiplier) sizeMultiplier = Math.min(sizeMultiplier, w.sizeMultiplier);
    }

    return {
      allowed: blocks.length === 0,
      blocks,
      warnings,
      sizeMultiplier,
    };
  }

  calculatePosition(entryPrice, stopLoss, sector, existing = [], sizeMultiplier = 1.0) {
    if (existing.length >= this.maxPositions) {
      return { allowed: false, reason: `Max ${this.maxPositions} positions` };
    }

    const sectorCount = existing.filter(p => p.sector === sector).length;
    if (sectorCount >= this.maxSectorConc) {
      return { allowed: false, reason: `Max ${this.maxSectorConc} in ${sector}` };
    }

    const existingRisk = existing.reduce((s, p) => s + (p.maxLoss || 0), 0);
    const remainingRiskBudget = this.totalCapital * this.maxTotalRisk - existingRisk;
    if (remainingRiskBudget <= 0) {
      return { allowed: false, reason: 'Total risk budget exhausted' };
    }

    const pool = this.totalCapital * this.maxAllocation * sizeMultiplier;
    const existingExposure = existing.reduce((s, p) => s + p.quantity * p.entryPrice, 0);
    const remainingPool = pool - existingExposure;
    if (remainingPool <= 0) {
      return { allowed: false, reason: 'Capital pool exhausted' };
    }

    const riskPerShare = Math.abs(entryPrice - stopLoss);
    if (riskPerShare <= 0) return { allowed: false, reason: 'Invalid stop' };

    const riskBudget = Math.min(this.totalCapital * this.maxRiskPerTrade * sizeMultiplier, remainingRiskBudget);
    const byRisk = Math.floor(riskBudget / riskPerShare);
    const byCap = Math.floor(this.totalCapital * this.maxPerStock * sizeMultiplier / entryPrice);
    const slots = this.maxPositions - existing.length;
    const byPool = Math.floor((remainingPool / slots) / entryPrice);

    const qty = Math.max(1, Math.min(byRisk, byCap, byPool));
    const investment = ind.r(qty * entryPrice);
    const maxLoss = ind.r(qty * riskPerShare);

    const limits = { risk: byRisk, capital: byCap, pool: byPool };
    const limiting = Object.entries(limits).sort((a, b) => a[1] - b[1])[0][0].toUpperCase();

    return {
      allowed: true,
      quantity: qty,
      entryPrice: ind.r(entryPrice),
      stopLoss: ind.r(stopLoss),
      investmentAmount: investment,
      maxLoss,
      maxLossPercent: ind.r((maxLoss / this.totalCapital) * 100),
      capitalUtilization: ind.r((investment / this.totalCapital) * 100),
      limitingFactor: limiting,
      remainingSlots: slots - 1,
    };
  }

  portfolioRisk(positions) {
    if (positions.length === 0) return { positionCount: 0, riskRating: 'NONE' };

    const exposure = positions.reduce((s, p) => s + (p.quantity || 0) * (p.entryPrice || 0), 0);
    const risk = positions.reduce((s, p) => s + (p.maxLoss || 0), 0);
    const worstCase = ind.r((risk / this.totalCapital) * 100);

    const sectors = {};
    const dirs = { LONG: 0, SHORT: 0 };
    for (const p of positions) {
      sectors[p.sector || 'UNK'] = (sectors[p.sector || 'UNK'] || 0) + 1;
      dirs[p.direction || 'LONG']++;
    }

    return {
      positionCount: positions.length,
      symbols: positions.map(p => p.symbol),
      totalExposure: ind.r(exposure),
      exposurePercent: ind.r((exposure / this.totalCapital) * 100),
      totalMaxLoss: ind.r(risk),
      worstCaseDrawdown: worstCase,
      sectorBreakdown: sectors,
      directionBreakdown: dirs,
      hedged: dirs.LONG > 0 && dirs.SHORT > 0,
      riskRating: worstCase > 6 ? 'EXCESSIVE' : worstCase > 4 ? 'HIGH' : worstCase > 2 ? 'MODERATE' : 'LOW',
      withinLimits: worstCase <= this.maxTotalRisk * 100,
    };
  }
}

module.exports = { WeeklyBreakoutRiskManager };
```

```javascript
class WeeklyBreakoutScorer {
  constructor(weights = {}) {
    this.w = {
      multiWeekBase:       weights.multiWeekBase       ?? 20,
      relativeStrength:    weights.relativeStrength    ?? 12,
      trendAlignment:      weights.trendAlignment      ?? 10,

  rank(candidates, sectorData = {}) {
    const scored = candidates.map(c => {
      const f = this._factors(c);
      const total = this._total(f);
      const grade = total >= 78 ? 'A+' : total >= 68 ? 'A' : total >= 58 ? 'B+' : total >= 48 ? 'B' : total >= 38 ? 'C' : 'D';

      const weak = Object.entries(f).filter(([_, v]) => v.score < 35).map(([k]) => k);
      const strong = Object.entries(f).filter(([_, v]) => v.score >= 75).map(([k]) => k);

      let rec;
      if (grade <= 'A') {
        rec = {
          action: 'STRONG_ENTRY', confidence: 'HIGH',
          message: `${strong.length} strong factors aligned${c.base?.isPerfectSetup ? ' — PERFECT COIL SETUP' : ''}`,
          suggestedSize: grade === 'A+' ? 1.0 : 0.85,
          strengths: strong, weaknesses: weak,
        };
      } else if (grade <= 'B') {
        rec = {
          action: 'ENTER', confidence: 'MODERATE',
          message: `Decent setup. Watch: ${weak.join(', ') || 'nothing specific'}`,
          suggestedSize: grade === 'B+' ? 0.7 : 0.5,
          strengths: strong, weaknesses: weak,
        };
      } else if (grade === 'C') {
        rec = {
          action: 'MARGINAL', confidence: 'LOW',
          message: 'Take only if no better options.',
          suggestedSize: 0.3, weaknesses: weak,
        };
      } else {
        rec = { action: 'SKIP', confidence: 'NONE', suggestedSize: 0, weaknesses: weak };
      }

      return {
        ...c,
        scoring: { factors: f, totalScore: ind.r(total), grade, recommendation: rec },
      };
    });

    scored.sort((a, b) => b.scoring.totalScore - a.scoring.totalScore);
    scored.forEach((s, i) => { s.rank = i + 1; });
    return scored;
  }

  _factors(c) {
    const isLong = c.direction === 'LONG';

    const base = c.base
      ? { score: ind.r(c.base.baseScore), raw: c.base.classification }
      : { score: 30, raw: 'NO_DATA' };

    const rs = c.relativeStrength
      ? { score: ind.r(isLong ? c.relativeStrength.rsScore : 100 - c.relativeStrength.rsScore), raw: c.relativeStrength.rsRank }
      : { score: 50, raw: 'NO_DATA' };

    let trendScore = 50;
    if (c.trend) {
      const idealTrend = isLong ? 'STRONG_UPTREND' : 'STRONG_DOWNTREND';
      const goodTrend = isLong ? 'UPTREND' : 'DOWNTREND';
      if (c.trend.trend === idealTrend) trendScore = 100;
      else if (c.trend.trend === goodTrend) trendScore = 75;
      else trendScore = 25;
    }

    const vol = c.volumePattern
      ? { score: ind.r(c.volumePattern.accumScore), raw: c.volumePattern.pattern }
      : { score: 50, raw: 'NO_DATA' };

    const falseBo = c.falseBreakoutHistory
      ? { score: ind.r(c.falseBreakoutHistory.reliabilityScore), raw: c.falseBreakoutHistory.falseRate }
      : { score: 70, raw: 'NO_DATA' };

    const w52 = c.fiftyTwoWeek
      ? { score: ind.r(c.fiftyTwoWeek.contextScore), raw: c.fiftyTwoWeek.position }
      : { score: 50, raw: 'NO_DATA' };

    let closeScore = 50;
    if (c.pw) {
      closeScore = isLong
        ? ind.clamp(ind.mapRange(c.pw.pwClosePosition, 0.6, 1.0, 20, 100))
        : ind.clamp(ind.mapRange(1 - c.pw.pwClosePosition, 0.6, 1.0, 20, 100));
    }

    const sector = c.sectorConfirmation
      ? { score: ind.r(c.sectorConfirmation.sectorScore), raw: c.sectorConfirmation.sectorStockCount }
      : { score: 30, raw: 0 };

    const regime = c.regime
      ? { score: ind.r(c.regime.regimeScore), raw: c.regime.regime }
      : { score: 50, raw: 'UNKNOWN' };

    const candleBias = c.pwCandleBias != null
      ? { score: ind.r(c.pwCandleBias * 100), raw: c.pw?.pwCandleType }
      : { score: 50, raw: 'UNKNOWN' };

    const oi = c.oiAnalysis
      ? { score: ind.r(c.oiAnalysis.oiScore), raw: c.oiAnalysis.interpretation }
      : { score: 50, raw: 'NO_DATA' };

    let rsiScore = 50;
    if (c.indicators?.rsi14 != null) {
      const v = c.indicators.rsi14;
      if (isLong) {
        if (v >= 55 && v <= 68) rsiScore = 100;
        else if (v >= 48 && v <= 72) rsiScore = 65;
        else rsiScore = 30;
      } else {
        if (v >= 32 && v <= 45) rsiScore = 100;
        else if (v >= 28 && v <= 52) rsiScore = 65;
        else rsiScore = 30;
      }
    }

    return {
      multiWeekBase: base,
      relativeStrength: rs,
      trendAlignment: { score: ind.r(trendScore), raw: c.trend?.trend },
      volumePattern: vol,
      falseBreakoutPenalty: falseBo,
      fiftyTwoWeekContext: w52,
      pwClosePosition: { score: ind.r(closeScore), raw: c.pw?.pwClosePosition },
      sectorConfirmation: sector,
      marketRegime: regime,
      candleBias,
      oiConfirmation: oi,
      rsiOptimality: { score: ind.r(rsiScore), raw: c.indicators?.rsi14 },
    };
  }

  _total(factors) {
    let t = 0;
    for (const [key, f] of Object.entries(factors)) {
      t += f.score * ((this.w[key] || 0) / 100);
    }
    return t;
  }
}
```

```javascript
const ind = require('./weekly-indicators');
const { WeeklyBreakoutScanner } = require('./weekly-scanner');
const { WeeklyBreakoutSignalEngine } = require('./weekly-signals');

class WeeklyBreakoutBacktester {
  constructor(config = {}) {
    this.maxPositions = config.maxPositions ?? 4;
    this.stopLossMode = config.stopLossMode ?? 'BASE_LOW';
    this.scannerConfig = config.scannerConfig ?? {};
    this.signalConfig = config.signalConfig ?? {};
    this.lookbackWeeks = config.lookbackWeeks ?? 26;
    this.allowRetest = config.allowRetest ?? true;
    this.exitMode = config.exitMode ?? 'MANAGED'; // MANAGED | FRIDAY_ONLY
  }

  run(stocksHistory, niftyDaily, fnoList, options = {}) {
    const scanner = new WeeklyBreakoutScanner(this.scannerConfig);
    const signalEngine = new WeeklyBreakoutSignalEngine(this.signalConfig);

    const niftyWeekly = ind.buildWeeklyCandles(niftyDaily);
    const allWeekStarts = niftyWeekly.map(w => w.weekStart);

    const startIdx = options.startDate
      ? Math.max(allWeekStarts.findIndex(d => d >= options.startDate), this.lookbackWeeks)
      : this.lookbackWeeks;
    const endIdx = options.endDate
      ? allWeekStarts.findIndex(d => d > options.endDate)
      : allWeekStarts.length;

    const trades = [];
    const weeklyLog = [];

    for (let wi = startIdx; wi < endIdx; wi++) {
      const weekStart = allWeekStarts[wi];

      // prepare data up to end of previous week
      const niftyToHere = niftyDaily.filter(c => c.date < weekStart);
      const niftyWeeklyToHere = ind.buildWeeklyCandles(niftyToHere);
      const niftyTrend = ind.weeklyTrend(niftyWeeklyToHere, 20);
      const regime = ind.marketRegime(niftyWeeklyToHere, 12);

      const niftySMA50 = niftyToHere.length >= 50
        ? niftyToHere.slice(-50).reduce((s, c) => s + c.c, 0) / 50
        : null;
      const niftyLast = niftyToHere.length > 0 ? niftyToHere[niftyToHere.length - 1].c : null;

      const market = {
        niftyDailyCandles: niftyToHere,
        niftyWeeklyTrend: niftyTrend?.trend || 'SIDEWAYS',
        niftyAbove50SMA: niftySMA50 && niftyLast ? niftyLast > niftySMA50 : true,
        regime,
      };

      // stocks data up to previous week
      const stocksForScan = [];
      for (const [symbol, candles] of Object.entries(stocksHistory)) {
        const prior = candles.filter(c => c.date < weekStart);
        if (prior.length < this.lookbackWeeks * 5) continue;
        stocksForScan.push({
          symbol, sector: null, isFnO: fnoList.includes(symbol),
          dailyCandles: prior.slice(-this.lookbackWeeks * 6),
        });
      }

      // weekend scan
      const { breakoutWatchlist, breakdownWatchlist } = scanner.weekendScan(stocksForScan, market);
      const watchlist = [...breakoutWatchlist.slice(0, 8), ...breakdownWatchlist.slice(0, 4)];

      // get this week's daily candles
      const nextWeekStart = wi + 1 < allWeekStarts.length ? allWeekStarts[wi + 1] : '9999-12-31';
      const weekCandles = {};
      for (const [symbol, candles] of Object.entries(stocksHistory)) {
        const wc = candles.filter(c => c.date >= weekStart && c.date < nextWeekStart);
        if (wc.length > 0) weekCandles[symbol] = wc;
      }

      // simulate Mon-Tue-Wed for confirmation
      let takenThisWeek = 0;

      for (const candidate of watchlist) {
        if (takenThisWeek >= this.maxPositions) break;
        const sym = candidate.symbol;
        const wDays = weekCandles[sym];
        if (!wDays || wDays.length < 2) continue;

        const isLong = candidate.direction === 'LONG';
        const pw = candidate.pw;
        const avgVol = candidate.indicators.avgDailyVolume;

        let entryDay = null;
        let entryDayIdx = -1;
        let entryType = 'BREAKOUT';

        // check first 3 days for breakout or retest
        for (let d = 0; d < Math.min(3, wDays.length); d++) {
          const day = wDays[d];
          const det = ind.detectBreakout(day, pw, avgVol);
          if (!det) continue;

          if (isLong && det.breakout.occurred) {
            entryDay = day; entryDayIdx = d; entryType = 'BREAKOUT'; break;
          }
          if (!isLong && det.breakdown.occurred) {
            entryDay = day; entryDayIdx = d; entryType = 'BREAKOUT'; break;
          }

          // retest check (day after a breakout)
          if (this.allowRetest && d > 0) {
            const prevDay = wDays[d - 1];
            const prevDet = ind.detectBreakout(prevDay, pw, avgVol);
            if (prevDet) {
              const prevBroke = isLong ? prevDet.breakout.occurred : prevDet.breakdown.occurred;
              if (prevBroke) {
                const retest = ind.detectRetest(prevDay, day, isLong ? pw.pwHigh : pw.pwLow, candidate.direction);
                if (retest?.isValidRetest) {
                  entryDay = day;
                  entryDayIdx = d;
                  entryType = 'RETEST';
                  break;
                }
              }
            }
          }
        }

        if (!entryDay) continue;

        // entry and stop
        const entryPrice = entryType === 'RETEST'
          ? (() => {
              const retest = ind.detectRetest(
                wDays[entryDayIdx - 1], entryDay,
                isLong ? pw.pwHigh : pw.pwLow, candidate.direction
              );
              return retest?.retestEntry || entryDay.c;
            })()
          : entryDay.c;

        let stopLoss;
        if (this.stopLossMode === 'BASE_LOW' && candidate.base?.exists) {
          stopLoss = isLong ? candidate.base.baseLowest : candidate.base.baseHighest;
        } else {
          stopLoss = isLong ? pw.pwLow : pw.pwHigh;
        }

        // cap at 7%
        if (Math.abs((entryPrice - stopLoss) / entryPrice) > 0.07) {
          stopLoss = isLong ? entryPrice * 0.93 : entryPrice * 1.07;
        }

        // simulate hold
        const remaining = wDays.slice(entryDayIdx + 1);
        const exitResult = this._simulate(
          entryPrice, stopLoss, remaining, isLong, pw, signalEngine
        );

        const pnl = isLong
          ? ((exitResult.exitPrice - entryPrice) / entryPrice) * 100
          : ((entryPrice - exitResult.exitPrice) / entryPrice) * 100;

        trades.push({
          weekStart,
          entryDate: entryDay.date,
          exitDate: exitResult.exitDate,
          symbol: sym,
          direction: candidate.direction,
          entryType,
          entryPrice: ind.r(entryPrice),
          stopLoss: ind.r(stopLoss),
          exitPrice: ind.r(exitResult.exitPrice),
          exitReason: exitResult.reason,
          pnlPercent: ind.r(pnl),
          result: pnl > 0.1 ? 'WIN' : pnl < -0.1 ? 'LOSS' : 'FLAT',
          holdDays: exitResult.holdDays,
          peakProfit: ind.r(exitResult.peakProfit),
          isMondayEntry: ind.getDow(entryDay.date) === 1,
          weekendScore: candidate.weekendScore,
          baseClassification: candidate.base?.classification || 'NONE',
          rsRank: candidate.relativeStrength?.rsRank || 'UNKNOWN',
        });

        takenThisWeek++;
      }

      weeklyLog.push({
        weekStart,
        regime: regime?.regime,
        breakoutCandidates: breakoutWatchlist.length,
        breakdownCandidates: breakdownWatchlist.length,
        taken: takenThisWeek,
      });
    }

    return {
      trades,
      weeklyLog,
      summary: this._summary(trades),
      analysis: this._analysis(trades),
    };
  }

  _simulate(entryPrice, stopLoss, days, isLong, pw, engine) {
    let highSince = entryPrice, lowSince = entryPrice;
    let daysInside = 0;

    for (let i = 0; i < days.length; i++) {
      const d = days[i];
      highSince = Math.max(highSince, d.h);
      lowSince = Math.min(lowSince, d.l);

      const profit = isLong
        ? ((d.c - entryPrice) / entryPrice) * 100
        : ((entryPrice - d.c) / entryPrice) * 100;

      const peakProfit = isLong
        ? ((highSince - entryPrice) / entryPrice) * 100
        : ((entryPrice - lowSince) / entryPrice) * 100;

      // stop loss
      if (isLong ? d.l <= stopLoss : d.h >= stopLoss) {
        return { exitPrice: stopLoss, exitDate: d.date, reason: 'STOP_LOSS', holdDays: i + 1, peakProfit };
      }

      // failed breakout
      const inside = isLong
        ? d.c < pw.pwHigh && d.c > pw.pwLow
        : d.c > pw.pwLow && d.c < pw.pwHigh;
      if (inside) daysInside++; else daysInside = 0;
      if (daysInside >= 2) {
        return { exitPrice: d.c, exitDate: d.date, reason: 'FAILED_BREAKOUT', holdDays: i + 1, peakProfit };
      }

      // trailing (after 3%)
      if (profit >= 3) {
        const trail = isLong
          ? entryPrice + (highSince - entryPrice) * 0.6
          : entryPrice - (entryPrice - lowSince) * 0.6;
        if (isLong ? d.c <= trail : d.c >= trail) {
          return { exitPrice: trail, exitDate: d.date, reason: 'TRAILING_STOP', holdDays: i + 1, peakProfit };
        }
      }

      // last day = Friday exit
      if (i === days.length - 1) {
        return { exitPrice: d.c, exitDate: d.date, reason: 'FRIDAY_EXIT', holdDays: i + 1, peakProfit };
      }
    }

    return { exitPrice: entryPrice, exitDate: null, reason: 'NO_DATA', holdDays: 0, peakProfit: 0 };
  }

  _summary(trades) {
    if (!trades.length) return { totalTrades: 0 };

    const w = trades.filter(t => t.result === 'WIN');
    const l = trades.filter(t => t.result === 'LOSS');
    const totalPnl = trades.reduce((s, t) => s + t.pnlPercent, 0);
    const gp = w.reduce((s, t) => s + t.pnlPercent, 0);
    const gl = Math.abs(l.reduce((s, t) => s + t.pnlPercent, 0));

    let maxStreak = 0, streak = 0;
    for (const t of trades) {
      if (t.result === 'LOSS') { streak++; maxStreak = Math.max(maxStreak, streak); }
      else streak = 0;
    }

    return {
      totalTrades: trades.length,
      wins: w.length, losses: l.length,
      winRate: ind.r((w.length / trades.length) * 100),
      totalPnl: ind.r(totalPnl),
      avgPnl: ind.r(totalPnl / trades.length),
      avgWin: w.length ? ind.r(gp / w.length) : 0,
      avgLoss: l.length ? ind.r(-gl / l.length) : 0,
      profitFactor: gl > 0 ? ind.r(gp / gl) : Infinity,
      maxConsecLosses: maxStreak,
      avgHold: ind.r(trades.reduce((s, t) => s + t.holdDays, 0) / trades.length, 1),
      best: ind.r(Math.max(...trades.map(t => t.pnlPercent))),
      worst: ind.r(Math.min(...trades.map(t => t.pnlPercent))),
    };
  }

  _analysis(trades) {
    if (trades.length < 10) return null;

    const avg = arr => arr.length ? ind.r(arr.reduce((s, t) => s + t.pnlPercent, 0) / arr.length) : 0;
    const wr = arr => arr.length ? ind.r((arr.filter(t => t.result === 'WIN').length / arr.length) * 100) : 0;

    const longs = trades.filter(t => t.direction === 'LONG');
    const shorts = trades.filter(t => t.direction === 'SHORT');
    const mondays = trades.filter(t => t.isMondayEntry);
    const nonMondays = trades.filter(t => !t.isMondayEntry);
    const breakouts = trades.filter(t => t.entryType === 'BREAKOUT');
    const retests = trades.filter(t => t.entryType === 'RETEST');
    const withBase = trades.filter(t => t.baseClassification !== 'NONE' && t.baseClassification !== 'NO_BASE');
    const noBase = trades.filter(t => t.baseClassification === 'NONE' || t.baseClassification === 'NO_BASE');
    const strongRS = trades.filter(t => t.rsRank === 'STRONG' || t.rsRank === 'VERY_STRONG');
    const weakRS = trades.filter(t => t.rsRank === 'WEAK' || t.rsRank === 'VERY_WEAK');

    const byReason = {};
    for (const t of trades) {
      if (!byReason[t.exitReason]) byReason[t.exitReason] = { count: 0, pnl: 0 };
      byReason[t.exitReason].count++;
      byReason[t.exitReason].pnl += t.pnlPercent;
    }

    // drawdown
    let cum = 0, peak = 0, maxDD = 0;
    for (const t of trades) { cum += t.pnlPercent; peak = Math.max(peak, cum); maxDD = Math.max(maxDD, peak - cum); }

    return {
      byDirection: {
        long:  { count: longs.length,  avgPnl: avg(longs),  winRate: wr(longs) },
        short: { count: shorts.length, avgPnl: avg(shorts), winRate: wr(shorts) },
      },
      mondayEffect: {
        monday:    { count: mondays.length,    avgPnl: avg(mondays),    winRate: wr(mondays) },
        nonMonday: { count: nonMondays.length, avgPnl: avg(nonMondays), winRate: wr(nonMondays) },
        edge: mondays.length >= 5 ? ind.r(avg(mondays) - avg(nonMondays)) : null,
      },
      entryType: {
        breakout: { count: breakouts.length, avgPnl: avg(breakouts), winRate: wr(breakouts) },
        retest:   { count: retests.length,   avgPnl: avg(retests),   winRate: wr(retests) },
      },
      baseEffect: {
        withBase: { count: withBase.length, avgPnl: avg(withBase), winRate: wr(withBase) },
        noBase:   { count: noBase.length,   avgPnl: avg(noBase),   winRate: wr(noBase) },
        edge: withBase.length >= 5 ? ind.r(avg(withBase) - avg(noBase)) : null,
      },
      relativeStrengthEffect: {
        strong: { count: strongRS.length, avgPnl: avg(strongRS), winRate: wr(strongRS) },
        weak:   { count: weakRS.length,   avgPnl: avg(weakRS),   winRate: wr(weakRS) },
      },
      byExitReason: Object.entries(byReason).map(([r, d]) => ({
        reason: r, count: d.count, avgPnl: ind.r(d.pnl / d.count),
      })),
      maxDrawdown: ind.r(maxDD),
      finalCumPnl: ind.r(cum),
    };
  }
}

module.exports = { WeeklyBreakoutBacktester };
```
