```javascript
// ═══════════════════════════════════════════════
//  CORE RSI (both periods)
// ═══════════════════════════════════════════════

function rsi(closes, period = 14) {
  if (!closes || closes.length < period + 1) return null;
  let ag = 0, al = 0;
  for (let i = 1; i <= period; i++) {
    const d = closes[i] - closes[i - 1];
    if (d >= 0) ag += d; else al -= d;
  }
  ag /= period; al /= period;
  for (let i = period + 1; i < closes.length; i++) {
    const d = closes[i] - closes[i - 1];
    ag = (ag * (period - 1) + Math.max(d, 0)) / period;
    al = (al * (period - 1) + Math.max(-d, 0)) / period;
  }
  return al === 0 ? 100 : r(100 - 100 / (1 + ag / al));
}

function rsiSeries(closes, period = 14) {
  if (!closes || closes.length < period + 1) return [];
  const result = new Array(period).fill(null);
  let ag = 0, al = 0;
  for (let i = 1; i <= period; i++) {
    const d = closes[i] - closes[i - 1];
    if (d >= 0) ag += d; else al -= d;
  }
  ag /= period; al /= period;
  result.push(al === 0 ? 100 : r(100 - 100 / (1 + ag / al)));
  for (let i = period + 1; i < closes.length; i++) {
    const d = closes[i] - closes[i - 1];
    ag = (ag * (period - 1) + Math.max(d, 0)) / period;
    al = (al * (period - 1) + Math.max(-d, 0)) / period;
    result.push(al === 0 ? 100 : r(100 - 100 / (1 + ag / al)));
  }
  return result;
}

function sma(values, period) {
  if (!values || values.length < period) return null;
  return r(values.slice(-period).reduce((a, b) => a + b, 0) / period);
}

function ema(values, period) {
  if (!values || values.length < period) return null;
  const k = 2 / (period + 1);
  let val = values.slice(0, period).reduce((a, b) => a + b, 0) / period;
  for (let i = period; i < values.length; i++) val = values[i] * k + val * (1 - k);
  return r(val);
}

function atr(candles, period = 14) {
  if (!candles || candles.length < period + 1) return null;
  const trs = [];
  for (let i = 1; i < candles.length; i++) {
    trs.push(Math.max(
      candles[i].h - candles[i].l,
      Math.abs(candles[i].h - candles[i - 1].c),
      Math.abs(candles[i].l - candles[i - 1].c)
    ));
  }
  let val = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
  for (let i = period; i < trs.length; i++) val = (val * (period - 1) + trs[i]) / period;
  return r(val);
}

function avgVolume(candles, period = 20) {
  if (!candles || candles.length < period) return null;
  return Math.round(candles.slice(-period).reduce((s, c) => s + c.v, 0) / period);
}


// ═══════════════════════════════════════════════
//  BOLLINGER BANDS
// ═══════════════════════════════════════════════

function bollingerBands(closes, period = 20, mult = 2) {
  if (!closes || closes.length < period) return null;

  const slice = closes.slice(-period);
  const mean = slice.reduce((a, b) => a + b, 0) / period;
  const stdDev = Math.sqrt(slice.reduce((s, v) => s + (v - mean) ** 2, 0) / period);

  const upper = mean + mult * stdDev;
  const lower = mean - mult * stdDev;
  const close = closes[closes.length - 1];

  const percentB = (upper - lower) > 0 ? (close - lower) / (upper - lower) : 0.5;
  const bandwidth = mean > 0 ? ((upper - lower) / mean) * 100 : 0;

  // historical bandwidth for squeeze detection
  const bwHistory = [];
  for (let i = period; i <= closes.length; i++) {
    const s = closes.slice(i - period, i);
    const m = s.reduce((a, b) => a + b, 0) / period;
    const sd = Math.sqrt(s.reduce((sm, v) => sm + (v - m) ** 2, 0) / period);
    bwHistory.push(m > 0 ? ((2 * mult * sd) / m) * 100 : 0);
  }
  const avgBW = bwHistory.length > 0
    ? bwHistory.reduce((a, b) => a + b, 0) / bwHistory.length
    : bandwidth;

  return {
    upper: r(upper), middle: r(mean), lower: r(lower),
    stdDev: r(stdDev), percentB: r(percentB, 3),
    bandwidth: r(bandwidth), avgBandwidth: r(avgBW),
    close: r(close),
    belowLower: close < lower,
    aboveMiddle: close > mean,
    distBelowLower: close < lower ? r(((lower - close) / lower) * 100) : 0,
    bandsWideEnough: bandwidth > 10,
    isSqueeze: bandwidth < avgBW * 0.6,
  };
}

function distanceFromMeans(closes) {
  if (!closes || closes.length < 200) return null;
  const c = closes[closes.length - 1];

  const sma5v = sma(closes, 5);
  const sma10v = sma(closes, 10);
  const sma20v = sma(closes, 20);
  const sma50v = sma(closes, 50);
  const sma100v = sma(closes, 100);
  const sma200v = sma(closes, 200);

  const dist = (avg) => avg ? r(((c - avg) / avg) * 100) : null;

  return {
    sma5: { value: sma5v, distance: dist(sma5v) },
    sma10: { value: sma10v, distance: dist(sma10v) },
    sma20: { value: sma20v, distance: dist(sma20v) },
    sma50: { value: sma50v, distance: dist(sma50v) },
    sma100: { value: sma100v, distance: dist(sma100v) },
    sma200: { value: sma200v, distance: dist(sma200v) },
    above200SMA: sma200v ? c > sma200v : null,
    above100SMA: sma100v ? c > sma100v : null,
    above50SMA: sma50v ? c > sma50v : null,
    below5SMA: sma5v ? c < sma5v : null,
    below5SMAby3pct: dist(sma5v) !== null && dist(sma5v) <= -3,
    close: r(c),
  };
}

/**
 * Classify the TYPE of decline — each has different MR probability
 *
 * Types:
 *   ORDERLY_PULLBACK:    Gradual drift lower, low volume, no gaps → 80% MR success
 *   SECTOR_ROTATION:     Whole sector falling, stock follows → 65% MR (needs sector recovery)
 *   NEWS_DRIVEN_GAP:     Big gap down on news → 45% MR (fundamental change possible)
 *   EARNINGS_MISS:       Post-results dump → 35% MR (don't mean-revert fundamentals)
 *   PANIC_CAPITULATION:  Massive volume spike, deep drop → 70% MR (capitulation = near bottom)
 *   WATERFALL:           Multi-day crash, accelerating → 25% MR (AVOID)
 *   MEAN_DRIFT:          Slow persistent decline, below all MAs → 40% MR (trend change)
 */
function classifyDecline(dailyCandles, sectorReturn5d = null, hasEarnings = false) {
  if (!dailyCandles || dailyCandles.length < 20) return null;

  const recent = dailyCandles.slice(-10);
  const today = recent[recent.length - 1];
  const avgVol = avgVolume(dailyCandles.slice(-30), 20);

  // count consecutive down days
  let downDays = 0;
  let totalDecline = 0;
  let maxSingleDrop = 0;
  let hasGapDown = false;
  let gapDownFilled = false;
  let volumeOnWorstDay = 0;

  for (let i = recent.length - 1; i > 0; i--) {
    const curr = recent[i];
    const prev = recent[i - 1];
    if (curr.c < prev.c) {
      downDays++;
      const drop = ((prev.c - curr.c) / prev.c) * 100;
      totalDecline += drop;
      if (drop > maxSingleDrop) {
        maxSingleDrop = drop;
        volumeOnWorstDay = curr.v;
      }
      if (curr.o < prev.l) {
        hasGapDown = true;
        gapDownFilled = curr.h >= prev.l;
      }
    } else {
      break;
    }
  }

  // volume pattern during decline
  const downCandles = [];
  for (let i = recent.length - downDays; i < recent.length; i++) {
    if (i >= 0) downCandles.push(recent[i]);
  }
  const downVolumes = downCandles.map(c => c.v);
  const todayVolRatio = avgVol > 0 ? today.v / avgVol : 1;
  const maxVolRatio = avgVol > 0 ? Math.max(...downVolumes) / avgVol : 1;

  // is volume accelerating (panic) or declining (orderly)?
  const volSlope = linearSlope(downVolumes);
  const volumeAccelerating = volSlope > 0 && downVolumes.length >= 3;
  const volumeDeclining = volSlope < 0;

  // is today a climax day? (spike volume + reversal candle)
  const todayRange = today.h - today.l;
  const todayLowerWick = Math.min(today.o, today.c) - today.l;
  const isClimaxCandle = todayVolRatio > 2.0
    && todayRange > 0
    && todayLowerWick / todayRange > 0.4
    && downDays >= 3;

  // 20-day drawdown
  const high20 = Math.max(...dailyCandles.slice(-20).map(c => c.h));
  const drawdown20 = ((high20 - today.c) / high20) * 100;

  // acceleration: are drops getting bigger each day?
  let accelerating = false;
  if (downCandles.length >= 3) {
    const drops = [];
    for (let i = 1; i < downCandles.length; i++) {
      drops.push(Math.abs((downCandles[i].c - downCandles[i - 1].c) / downCandles[i - 1].c) * 100);
    }
    accelerating = drops.length >= 2 && drops[drops.length - 1] > drops[0] * 1.3;
  }

  // ── CLASSIFICATION ──
  let type, mrProbability, score;

  if (hasEarnings && maxSingleDrop > 3) {
    type = 'EARNINGS_MISS';
    mrProbability = 0.35;
    score = 20;
  } else if (drawdown20 > 15 || (accelerating && totalDecline > 10)) {
    type = 'WATERFALL';
    mrProbability = 0.25;
    score = 10;
  } else if (hasGapDown && !gapDownFilled && maxSingleDrop > 4) {
    type = 'NEWS_DRIVEN_GAP';
    mrProbability = 0.45;
    score = 35;
  } else if (isClimaxCandle) {
    type = 'PANIC_CAPITULATION';
    mrProbability = 0.70;
    score = 75;
  } else if (sectorReturn5d !== null && sectorReturn5d < -3 && totalDecline < sectorReturn5d * 1.5) {
    type = 'SECTOR_ROTATION';
    mrProbability = 0.65;
    score = 55;
  } else if (downDays >= 6 && !volumeAccelerating && drawdown20 > 8) {
    type = 'MEAN_DRIFT';
    mrProbability = 0.40;
    score = 30;
  } else if (downDays >= 3 && maxSingleDrop < 3 && (volumeDeclining || maxVolRatio < 2.0)) {
    type = 'ORDERLY_PULLBACK';
    mrProbability = 0.80;
    score = 85;
  } else {
    type = 'MIXED';
    mrProbability = 0.55;
    score = 50;
  }

  return {
    type,
    mrProbability: r(mrProbability, 2),
    score: r(score),
    downDays,
    totalDecline: r(totalDecline),
    maxSingleDrop: r(maxSingleDrop),
    drawdown20: r(drawdown20),
    hasGapDown,
    gapDownFilled,
    isClimaxCandle,
    accelerating,
    volumeAccelerating,
    volumeDeclining,
    todayVolRatio: r(todayVolRatio),
    maxVolRatio: r(maxVolRatio),
    // actionable guidance
    safe: type === 'ORDERLY_PULLBACK' || type === 'PANIC_CAPITULATION',
    dangerous: type === 'WATERFALL' || type === 'EARNINGS_MISS',
    caution: type === 'NEWS_DRIVEN_GAP' || type === 'MEAN_DRIFT',
  };
}

/**
 * Build weekly candles and check if weekly RSI is also oversold
 * Weekly + daily both oversold = 2× the edge
 */
function multiTimeframeOversold(dailyCandles) {
  if (!dailyCandles || dailyCandles.length < 100) return null;

  const closes = dailyCandles.map(c => c.c);
  const rsi2Daily = rsi(closes, 2);
  const rsi14Daily = rsi(closes, 14);

  // build weekly closes (5-day buckets from end)
  const weeklyCloses = [];
  for (let i = dailyCandles.length; i >= 5; i -= 5) {
    weeklyCloses.unshift(dailyCandles[i - 1].c);
  }

  const rsi2Weekly = weeklyCloses.length >= 4 ? rsi(weeklyCloses, 2) : null;
  const rsi14Weekly = weeklyCloses.length >= 15 ? rsi(weeklyCloses, 14) : null;

  const dailyOversold = (rsi2Daily !== null && rsi2Daily <= 10)
    || (rsi14Daily !== null && rsi14Daily < 30);
  const weeklyOversold = (rsi2Weekly !== null && rsi2Weekly <= 15)
    || (rsi14Weekly !== null && rsi14Weekly < 35);

  const bothOversold = dailyOversold && weeklyOversold;

  return {
    daily: { rsi2: rsi2Daily, rsi14: rsi14Daily, oversold: dailyOversold },
    weekly: { rsi2: rsi2Weekly, rsi14: rsi14Weekly, oversold: weeklyOversold },
    bothOversold,
    multiTFScore: r(clamp(
      (dailyOversold ? 40 : 0) + (weeklyOversold ? 40 : 0) + (bothOversold ? 20 : 0)
    )),
    edge: bothOversold ? 'STRONG — weekly + daily aligned' : dailyOversold ? 'MODERATE — daily only' : 'WEAK',
  };
}

/**
 * Has the stock bounced from this price level before?
 * Multiple successful bounces = proven support = highest probability MR
 */
function priorBounceAtLevel(dailyCandles, currentPrice, tolerance = 0.03) {
  if (!dailyCandles || dailyCandles.length < 100) return null;

  const bounces = [];

  // find swing lows in last 200 days
  for (let i = 2; i < dailyCandles.length - 5; i++) {
    const c = dailyCandles[i];
    const isSwingLow = c.l <= dailyCandles[i - 1].l
      && c.l <= dailyCandles[i - 2].l
      && c.l <= dailyCandles[i + 1].l
      && c.l <= dailyCandles[i + 2].l;

    if (!isSwingLow) continue;

    // is this swing low near the current price?
    const proximity = Math.abs((c.l - currentPrice) / currentPrice);
    if (proximity > tolerance) continue;

    // how much did it bounce from this level?
    let maxBounce = 0;
    for (let j = i + 1; j < Math.min(i + 15, dailyCandles.length); j++) {
      const bouncePercent = ((dailyCandles[j].h - c.l) / c.l) * 100;
      maxBounce = Math.max(maxBounce, bouncePercent);
    }

    bounces.push({
      date: c.date,
      level: r(c.l),
      proximity: r(proximity * 100),
      maxBounce: r(maxBounce),
      bounced: maxBounce > 2,
    });
  }

  const successfulBounces = bounces.filter(b => b.bounced);

  return {
    totalTests: bounces.length,
    successfulBounces: successfulBounces.length,
    failedBounces: bounces.length - successfulBounces.length,
    avgBounce: successfulBounces.length > 0
      ? r(successfulBounces.reduce((s, b) => s + b.maxBounce, 0) / successfulBounces.length)
      : 0,
    levels: bounces,
    isProvenSupport: successfulBounces.length >= 2,
    supportStrength: bounces.length > 0
      ? r((successfulBounces.length / bounces.length) * 100)
      : 0,
    priorBounceScore: r(clamp(
      successfulBounces.length >= 3 ? 95 :
      successfulBounces.length >= 2 ? 80 :
      successfulBounces.length >= 1 ? 55 :
      bounces.length > 0 ? 20 : 0
    )),
  };
}

/**
 * Check if the reversal candle forms exactly at a key MA
 * Hammer at 200 SMA = one of the highest-probability setups in existence
 */
function patternAtKeyMA(todayCandle, distances) {
  if (!todayCandle || !distances) return null;

  const checks = [];

  // check each major MA
  const mas = [
    { name: 'SMA200', value: distances.sma200?.value, importance: 100 },
    { name: 'SMA100', value: distances.sma100?.value, importance: 85 },
    { name: 'SMA50', value: distances.sma50?.value, importance: 70 },
    { name: 'SMA20', value: distances.sma20?.value, importance: 50 },
  ];

  for (const ma of mas) {
    if (!ma.value) continue;

    // did today's LOW touch/pierce the MA and then close back above it?
    const touchedMA = todayCandle.l <= ma.value * 1.005;  // within 0.5%
    const closedAboveMA = todayCandle.c >= ma.value * 0.995;
    const wickThroughMA = todayCandle.l < ma.value && todayCandle.c > ma.value;

    const distFromMA = Math.abs((todayCandle.c - ma.value) / ma.value) * 100;
    const isNearMA = distFromMA < 1.5;

    if (touchedMA || isNearMA) {
      checks.push({
        ma: ma.name,
        maValue: r(ma.value),
        touchedMA,
        closedAboveMA,
        wickThroughMA,
        distFromMA: r(distFromMA),
        importance: ma.importance,
        // wick through MA + close above = perfect reversal at support
        isPerfect: wickThroughMA && closedAboveMA,
      });
    }
  }

  const best = checks.sort((a, b) => b.importance - a.importance)[0] || null;

  return {
    atKeyMA: checks.length > 0,
    nearestMA: best,
    allMAs: checks,
    maScore: best
      ? r(clamp(best.importance * (best.isPerfect ? 1.0 : best.touchedMA ? 0.8 : 0.5)))
      : 0,
  };
}

//  CANDLE PATTERN DETECTION (enhanced)
// ═══════════════════════════════════════════════

function detectReversalCandle(todayCandle, yesterdayCandle) {
  if (!todayCandle) return null;

  const t = todayCandle;
  const y = yesterdayCandle;
  const range = t.h - t.l;
  if (range === 0) return { type: 'DOJI', score: 0, entryTrigger: null };

  const body = Math.abs(t.c - t.o);
  const isBullish = t.c >= t.o;
  const upperWick = isBullish ? t.h - t.c : t.h - t.o;
  const lowerWick = isBullish ? t.o - t.l : t.c - t.l;
  const bodyRatio = body / range;
  const lowerWickRatio = lowerWick / range;
  const upperWickRatio = upperWick / range;

  const result = {
    type: 'NONE',
    isHammer: false,
    isBullishEngulfing: false,
    isDragonfly: false,
    isPinBar: false,
    isPiercing: false,
    score: 0,
    details: {},
    // CRITICAL: the entry trigger for Strategy C
    // "Buy above the high of the hammer/engulfing candle"
    entryTrigger: null,     // price level that must be breached next day
    stopLevel: null,        // stop below the pattern low
    candleHigh: r(t.h),
    candleLow: r(t.l),
  };

  // ── HAMMER ──
  if (lowerWick >= body * 2 && upperWickRatio < 0.15 && bodyRatio < 0.35) {
    result.isHammer = true;
    result.type = 'HAMMER';
    let s = 50;
    if (lowerWick >= body * 3) s += 15;
    if (isBullish) s += 15;
    if (upperWickRatio < 0.05) s += 10;
    if (bodyRatio < 0.20) s += 10;
    result.score = clamp(s);
    result.details.wickBodyRatio = r(lowerWick / Math.max(body, 0.01));
    result.entryTrigger = r(t.h * 1.001); // just above candle high
    result.stopLevel = r(t.l * 0.998);    // just below candle low
  }

  // ── BULLISH ENGULFING ──
  if (y && y.c < y.o && t.c > t.o) {
    if (t.o <= y.c && t.c >= y.o) {
      result.isBullishEngulfing = true;
      result.type = result.isHammer ? 'HAMMER_ENGULFING_COMBO' : 'BULLISH_ENGULFING';
      let s = 55;
      const todayBody = t.c - t.o;
      const yesterBody = y.o - y.c;
      const engulfRatio = yesterBody > 0 ? todayBody / yesterBody : 1;
      if (engulfRatio > 2) s += 20;
      else if (engulfRatio > 1.5) s += 12;
      if (t.c > y.h) s += 10;
      if (t.l < y.l) s += 5;  // new low then reversal
      result.score = Math.max(result.score, clamp(s));
      result.details.engulfRatio = r(engulfRatio);
      // engulfing entry: above engulfing candle high
      result.entryTrigger = r(t.h * 1.001);
      result.stopLevel = r(Math.min(t.l, y.l) * 0.998);
    }
  }

  // ── DRAGONFLY DOJI ──
  if (bodyRatio < 0.05 && lowerWickRatio > 0.70 && upperWickRatio < 0.10) {
    result.isDragonfly = true;
    if (result.score < 65) {
      result.type = 'DRAGONFLY_DOJI';
      result.score = 65;
      result.entryTrigger = r(t.h * 1.001);
      result.stopLevel = r(t.l * 0.998);
    }
  }

  // ── BULLISH PIN BAR ──
  if (lowerWickRatio > 0.66 && bodyRatio < 0.25 && upperWickRatio < 0.15) {
    result.isPinBar = true;
    if (result.score < 70) {
      result.type = 'BULLISH_PIN_BAR';
      result.score = 70;
      result.entryTrigger = r(t.h * 1.001);
      result.stopLevel = r(t.l * 0.998);
    }
  }

  // ── PIERCING PATTERN ──
  if (y && y.c < y.o) {
    const yMid = (y.o + y.c) / 2;
    if (t.o < y.l && t.c > yMid && t.c < y.o && t.c > t.o) {
      result.isPiercing = true;
      if (result.score < 60) {
        result.type = 'PIERCING';
        result.score = 60;
        result.entryTrigger = r(t.h * 1.001);
        result.stopLevel = r(t.l * 0.998);
      }
    }
  }

  // set fallback triggers if none set
  if (!result.entryTrigger && result.type !== 'NONE') {
    result.entryTrigger = r(t.h * 1.001);
    result.stopLevel = r(t.l * 0.998);
  }

  return result;
}


// ═══════════════════════════════════════════════
//  GAP-AT-ENTRY FILTER
// ═══════════════════════════════════════════════

/**
 * On the ENTRY DAY (next morning), check if the gap is acceptable
 *
 * @param {number} prevClose - signal day's close (yesterday)
 * @param {number} todayOpen - entry day's open (this morning)
 * @param {number} stopLoss - planned stop loss
 * @param {string} strategy
 */
function gapAtEntryFilter(prevClose, todayOpen, stopLoss, strategy = 'RSI2') {
  const gapPercent = ((todayOpen - prevClose) / prevClose) * 100;
  const distToStop = stopLoss > 0 ? ((todayOpen - stopLoss) / todayOpen) * 100 : 999;

  // max acceptable gap DOWN limits per strategy
  const maxGapDown = { RSI2: -3, BOLLINGER: -2.5, PATTERN: -2, CONFLUENCE: -2 };
  const maxGap = maxGapDown[strategy] || -3;

  // max gap UP (if stock already bounced 3% overnight, the edge is gone)
  const maxGapUp = 3;

  const gapTooLarge = gapPercent < maxGap;
  const alreadyBounced = gapPercent > maxGapUp;
  const stopTooClose = distToStop < 1.0; // less than 1% to stop after gap

  let verdict;
  if (gapTooLarge) {
    verdict = 'SKIP_GAP_DOWN_TOO_LARGE';
  } else if (alreadyBounced) {
    verdict = 'SKIP_ALREADY_BOUNCED';
  } else if (stopTooClose) {
    verdict = 'SKIP_STOP_TOO_CLOSE';
  } else {
    verdict = 'OK';
  }

  return {
    gapPercent: r(gapPercent),
    distToStop: r(distToStop),
    verdict,
    canEnter: verdict === 'OK',
    adjustedEntry: todayOpen,
    adjustedStop: gapPercent < -1 ? r(todayOpen * (1 - Math.abs(maxGap) / 100)) : stopLoss,
    notes: gapPercent < -1
      ? 'Gap down reduces effective R:R — consider tighter stop or skip'
      : gapPercent > 1
        ? 'Positive gap — edge partially consumed but still valid'
        : 'Normal open — full edge available',
  };
}

function bounceHistory(dailyCandles, lookback = 252) {
  if (!dailyCandles || dailyCandles.length < lookback) return null;

  const candles = dailyCandles.slice(-lookback);
  const closes = candles.map(c => c.c);
  const rsi2s = rsiSeries(closes, 2);

  const bounces = [];
  for (let i = 10; i < rsi2s.length - 6; i++) {
    if (rsi2s[i] === null || rsi2s[i] > 10) continue;

    const entryOpen = i + 1 < candles.length ? candles[i + 1].o : null;
    if (!entryOpen) continue;

    const returns = [];
    let maxGain = 0, maxLoss = 0;
    for (let d = 1; d <= 5 && i + 1 + d < closes.length; d++) {
      const ret = ((closes[i + 1 + d] - entryOpen) / entryOpen) * 100;
      returns.push(ret);
      maxGain = Math.max(maxGain, ret);
      maxLoss = Math.min(maxLoss, ret);
    }

    const finalRet = returns.length > 0 ? returns[returns.length - 1] : 0;
    bounces.push({
      date: candles[i].date, rsi2: rsi2s[i],
      entryOpen: r(entryOpen), finalReturn: r(finalRet),
      maxGain: r(maxGain), maxLoss: r(maxLoss),
      won: finalRet > 0,
    });
  }

  if (bounces.length === 0) return { hasData: false, bounceScore: 50 };

  const wins = bounces.filter(b => b.won);
  const winRate = (wins.length / bounces.length) * 100;
  const avgRet = bounces.reduce((s, b) => s + b.finalReturn, 0) / bounces.length;
  const avgWin = wins.length ? wins.reduce((s, b) => s + b.finalReturn, 0) / wins.length : 0;
  const losses = bounces.filter(b => !b.won);
  const avgLoss = losses.length ? losses.reduce((s, b) => s + b.finalReturn, 0) / losses.length : 0;

  return {
    hasData: true, instances: bounces.length,
    winRate: r(winRate), avgReturn: r(avgRet),
    avgWin: r(avgWin), avgLoss: r(avgLoss),
    avgMaxGain: r(bounces.reduce((s, b) => s + b.maxGain, 0) / bounces.length),
    avgMaxLoss: r(bounces.reduce((s, b) => s + b.maxLoss, 0) / bounces.length),
    isReliableBouncer: winRate > 65 && bounces.length >= 3,
    isPoorBouncer: winRate < 45 && bounces.length >= 3,
    bounceScore: r(clamp(mapRange(winRate, 40, 85, 15, 100))),
  };
}

function meanReversionRegime(dailyCandles, niftyCandles, vix = null) {
  if (!dailyCandles || dailyCandles.length < 60) return null;

  const closes = dailyCandles.slice(-60).map(c => c.c);
  const sma50v = sma(closes, 50);
  const close = closes[closes.length - 1];

  const above50 = sma50v ? close > sma50v : null;
  const high20 = Math.max(...closes.slice(-20));
  const drawdown = ((high20 - close) / high20) * 100;
  const isWaterfall = drawdown > 15;
  const isSevere = drawdown > 10;

  let niftyOk = true, niftyAbove200 = true;
  if (niftyCandles && niftyCandles.length >= 200) {
    const nc = niftyCandles.map(c => c.c);
    const n200 = sma(nc, 200);
    niftyAbove200 = n200 ? nc[nc.length - 1] > n200 : true;
    const nHigh = Math.max(...nc.slice(-20));
    niftyOk = ((nHigh - nc[nc.length - 1]) / nHigh) * 100 < 8;
  }

  let vixOk = true, vixLevel = 'NORMAL';
  if (vix != null) {
    if (vix > 25) { vixOk = false; vixLevel = 'CRISIS'; }
    else if (vix > 22) { vixOk = false; vixLevel = 'ELEVATED'; }
    else if (vix > 18) { vixLevel = 'CAUTIOUS'; }
  }

  let regime;
  if (isWaterfall || !vixOk) regime = 'AVOID';
  else if (isSevere && !above50) regime = 'DANGEROUS';
  else if (above50 && niftyOk) regime = 'IDEAL';
  else if (above50) regime = 'GOOD';
  else if (niftyAbove200 && !isSevere) regime = 'ACCEPTABLE';
  else regime = 'RISKY';

  const scores = { IDEAL: 95, GOOD: 75, ACCEPTABLE: 55, RISKY: 30, DANGEROUS: 10, AVOID: 0 };

  return {
    regime, regimeScore: scores[regime],
    above50SMA: above50, drawdown20: r(drawdown),
    isWaterfall, isSevere,
    niftyOk, niftyAbove200, vix, vixLevel, vixOk,
    isSafe: regime === 'IDEAL' || regime === 'GOOD',
    warnings: [
      ...(isWaterfall ? ['WATERFALL: >15% decline — DO NOT buy dip'] : []),
      ...(vix > 22 ? [`VIX ${vix} > 22`] : []),
      ...(!niftyAbove200 ? ['Nifty below 200 SMA'] : []),
      ...(!above50 ? ['Stock below 50 SMA'] : []),
    ],
  };
}

function sectorCorrelation(stockRet5d, sectorRet5d, niftyRet5d) {
  if (stockRet5d == null) return null;

  const stockSpecific = stockRet5d - (sectorRet5d || 0);
  const isStockSpecific = stockSpecific < -2 && (sectorRet5d || 0) > -1;
  const isSectorWide = (sectorRet5d || 0) < -3;
  const isMarketWide = (niftyRet5d || 0) < -3;

  return {
    stockReturn5d: r(stockRet5d), sectorReturn5d: r(sectorRet5d), niftyReturn5d: r(niftyRet5d),
    isStockSpecific, isSectorWide, isMarketWide,
    sectorScore: r(clamp(isStockSpecific ? 85 : !isSectorWide ? 65 : isMarketWide ? 40 : 30)),
  };
}

//  SIGNAL STALENESS (re-entry check)
// ═══════════════════════════════════════════════

/**
 * If RSI(2) was 3 yesterday and is 8 today, is the signal still valid?
 */
function signalFreshness(rsi2Today, rsi2Yesterday, downDaysToday, downDaysYesterday) {
  // signal is freshest on the first day RSI(2) goes below threshold
  const justTriggered = rsi2Yesterday !== null && rsi2Yesterday > 10 && rsi2Today <= 5;

  // signal is still valid if:
  // 1. RSI(2) is still below 10 (even if above 5)
  // 2. AND stock is still in its down-streak
  const stillValid = rsi2Today <= 10 && downDaysToday >= 3;

  // signal is stale if:
  // 1. RSI(2) has risen above 15
  // 2. OR down-streak is broken
  const stale = rsi2Today > 15 || downDaysToday < 2;

  return {
    justTriggered,
    stillValid: !stale && (justTriggered || stillValid),
    stale,
    freshness: justTriggered ? 'FRESH' : stillValid ? 'VALID' : 'STALE',
    score: justTriggered ? 100 : stillValid ? 70 : stale ? 10 : 40,
  };
}

function linearSlope(values) {
  const n = values.length;
  if (n < 2) return 0;
  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
  for (let i = 0; i < n; i++) { sx += i; sy += values[i]; sxy += i * values[i]; sx2 += i * i; }
  const d = n * sx2 - sx * sx;
  return d !== 0 ? (n * sxy - sx * sy) / d : 0;
}

function r(val, d = 2) {
  if (val == null || isNaN(val)) return null;
  return Math.round(val * 10 ** d) / 10 ** d;
}

function clamp(val, min = 0, max = 100) {
  return Math.min(max, Math.max(min, val));
}

function mapRange(val, inMin, inMax, outMin, outMax) {
  return ((val - inMin) / (inMax - inMin)) * (outMax - outMin) + outMin;
}
```

```javascript
const ind = require('./mr-indicators');

/**
 * Three sub-strategy scanners + confluence detection
 *
 * KEY CHANGE from v1: Scanner produces SIGNALS (scan-day analysis),
 * but entry happens NEXT DAY with separate validation.
 * The scan day ≠ entry day.
 */
class MeanReversionScanner {
  constructor(config = {}) {
    this.A = {
      rsi2Max: config.rsi2Max ?? 5,
      minDownDays: config.minDownDays ?? 3,
      minDistBelow5SMA: config.minDistBelow5SMA ?? -3,
      maxVix: config.maxVix ?? 22,
    };
    this.B = {
      bbPeriod: config.bbPeriod ?? 20,
      bbStdDev: config.bbStdDev ?? 2,
      minBandwidth: config.minBandwidth ?? 10,
      maxVolumeRatio: config.maxVolumeRatio ?? 2.5,
    };
    this.C = {
      rsi14Max: config.rsi14Max ?? 30,
      minPatternScore: config.minPatternScore ?? 40,
      minCandleVolMultiple: config.minCandleVolMultiple ?? 1.2,
    };
  }

  /**
   * SCAN DAY analysis — run at market close
   * Produces signal candidates for NEXT DAY entry
   */
  scan(stocks, market) {
    const results = {
      strategyA: [], strategyB: [], strategyC: [],
      confluence: [], rejected: [],
    };
    const vix = market.indiaVix;

    for (const stock of stocks) {
      try {
        const eval_ = this._evaluate(stock, market);
        if (eval_.blocked) {
          results.rejected.push({ symbol: stock.symbol, reason: eval_.blockReason });
          continue;
        }

        if (eval_.strategyA.pass) results.strategyA.push(eval_);
        if (eval_.strategyB.pass) results.strategyB.push(eval_);
        if (eval_.strategyC.pass) results.strategyC.push(eval_);

        const passCount = [eval_.strategyA.pass, eval_.strategyB.pass, eval_.strategyC.pass]
          .filter(Boolean).length;
        if (passCount >= 2) {
          results.confluence.push({
            ...eval_,
            confluenceCount: passCount,
            activeStrategies: [
              eval_.strategyA.pass ? 'RSI2' : null,
              eval_.strategyB.pass ? 'BOLLINGER' : null,
              eval_.strategyC.pass ? 'PATTERN' : null,
            ].filter(Boolean),
          });
        }
      } catch (err) {
        results.rejected.push({ symbol: stock.symbol, error: err.message });
      }
    }

    for (const key of ['strategyA', 'strategyB', 'strategyC', 'confluence']) {
      results[key].sort((a, b) => b.compositeScore - a.compositeScore);
    }

    results.meta = {
      scanned: stocks.length, vix,
      a: results.strategyA.length, b: results.strategyB.length,
      c: results.strategyC.length, confluence: results.confluence.length,
    };
    return results;
  }

  _evaluate(stock, market) {
    const daily = stock.dailyCandles;
    if (!daily || daily.length < 210) {
      return { blocked: true, blockReason: 'Insufficient data' };
    }

    const closes = daily.map(c => c.c);
    const today = daily[daily.length - 1];
    const yesterday = daily[daily.length - 2];
    const vix = market.indiaVix;

    // ── compute indicators ──
    const rsi2 = ind.rsi(closes, 2);
    const rsi14 = ind.rsi(closes, 14);
    const bb = ind.bollingerBands(closes, this.B.bbPeriod, this.B.bbStdDev);
    const distances = ind.distanceFromMeans(closes);
    const atrVal = ind.atr(daily, 14);
    const avgVol = ind.avgVolume(daily, 20);
    const todayVolRatio = avgVol > 0 ? today.v / avgVol : 1;

    // ── enhanced analysis ──
    const decline = ind.classifyDecline(
      daily,
      market.sectorReturns5d?.[stock.sector] ?? null,
      stock.hasEarningsRecently ?? false
    );
    const multiTF = ind.multiTimeframeOversold(daily);
    const pattern = ind.detectReversalCandle(today, yesterday);
    const priorBounce = ind.priorBounceAtLevel(daily, today.c, 0.03);
    const maPattern = ind.patternAtKeyMA(today, distances);
    const regime = ind.meanReversionRegime(daily, market.niftyDailyCandles, vix);
    const bounceHist = ind.bounceHistory(daily, 252);
    const sectorCorr = ind.sectorCorrelation(
      closes.length >= 6 ? ((closes[closes.length - 1] - closes[closes.length - 6]) / closes[closes.length - 6]) * 100 : null,
      market.sectorReturns5d?.[stock.sector] ?? null,
      market.niftyReturn5d ?? null
    );

    // freshness: did RSI(2) JUST go below threshold or was it there yesterday too?
    const prevRsi2 = closes.length >= 4 ? ind.rsi(closes.slice(0, -1), 2) : null;
    const prevDown = daily.length >= 4 ? (() => {
      let d = 0;
      for (let i = daily.length - 2; i > 0; i--) {
        if (daily[i].c < daily[i - 1].c) d++; else break;
      }
      return d;
    })() : 0;
    const freshness = ind.signalFreshness(
      rsi2, prevRsi2,
      decline?.downDays || 0, prevDown
    );

    // ── BLOCKS ──
    if (regime?.regime === 'AVOID') {
      return { blocked: true, blockReason: `Regime AVOID: ${regime.warnings.join('; ')}` };
    }
    if (decline?.type === 'WATERFALL') {
      return { blocked: true, blockReason: `Waterfall decline: ${decline.drawdown20}% drawdown` };
    }
    if (decline?.type === 'EARNINGS_MISS') {
      return { blocked: true, blockReason: 'Post-earnings decline — don\'t mean-revert fundamentals' };
    }

    // ── Strategy A: RSI(2) ≤ 5 ──
    const stratA = this._evalA(rsi2, decline, distances, regime, vix);

    // ── Strategy B: Bollinger ──
    const stratB = this._evalB(bb, distances, decline, regime, todayVolRatio);

    // ── Strategy C: RSI(14) + Pattern ──
    const stratC = this._evalC(rsi14, pattern, distances, todayVolRatio, regime);

    // ── composite score with all enhancements ──
    const bestScore = Math.max(
      stratA.pass ? stratA.score : 0,
      stratB.pass ? stratB.score : 0,
      stratC.pass ? stratC.score : 0
    );

    let bonus = 0;
    // decline type
    bonus += (decline?.score || 50) * 0.12;
    // multi-timeframe
    if (multiTF?.bothOversold) bonus += 8;
    // prior bounce at this level
    if (priorBounce?.isProvenSupport) bonus += 10;
    else if (priorBounce?.successfulBounces >= 1) bonus += 5;
    // pattern at key MA
    if (maPattern?.atKeyMA) bonus += (maPattern.maScore / 100) * 8;
    // historical bounce reliability
    if (bounceHist?.isReliableBouncer) bonus += 6;
    // sector isolation
    if (sectorCorr?.isStockSpecific) bonus += 5;
    // freshness
    if (freshness.justTriggered) bonus += 4;
    // climax candle
    if (decline?.isClimaxCandle) bonus += 5;

    // penalties
    if (decline?.dangerous) bonus -= 15;
    if (decline?.caution) bonus -= 5;
    if (bounceHist?.isPoorBouncer) bonus -= 8;
    if (freshness.stale) bonus -= 10;

    const compositeScore = ind.r(clamp(bestScore + bonus));

    // ── build entry plan for NEXT DAY ──
    // (scanner prepares the entry parameters, actual entry happens next morning)
    const entryPlan = this._buildEntryPlan(stratA, stratB, stratC, today, pattern, bb, distances, atrVal);

    return {
      blocked: false,
      symbol: stock.symbol,
      sector: stock.sector,

      strategyA: stratA,
      strategyB: stratB,
      strategyC: stratC,
      compositeScore,

      // next-day entry plan
      entryPlan,

      // all computed data
      todayCandle: today,
      indicators: {
        close: today.c, rsi2, rsi14,
        atr14: atrVal, avgVolume: avgVol,
        todayVolRatio: ind.r(todayVolRatio),
      },
      bb, distances, decline, multiTF,
      pattern, priorBounce, maPattern,
      regime, bounceHistory: bounceHist,
      sectorCorrelation: sectorCorr,
      freshness,
    };
  }

  _evalA(rsi2, decline, dist, regime, vix) {
    const c = {};
    c.above200SMA = { pass: dist?.above200SMA === true };
    c.rsi2 = { value: rsi2, pass: rsi2 !== null && rsi2 <= this.A.rsi2Max };
    c.downDays = { value: decline?.downDays, pass: decline?.downDays >= this.A.minDownDays };
    c.below5SMA = { value: dist?.sma5?.distance, pass: dist?.below5SMAby3pct === true };
    c.vix = { value: vix, pass: vix == null || vix < this.A.maxVix };

    const pass = Object.values(c).every(x => x.pass);
    let score = 0;
    if (pass) {
      score += clamp(mapRange(rsi2, 0, 5, 100, 50)) * 0.30;
      score += clamp(mapRange(decline?.downDays || 3, 3, 7, 50, 100)) * 0.20;
      score += clamp(mapRange(Math.abs(dist?.sma5?.distance || 3), 3, 8, 50, 100)) * 0.20;
      score += (regime?.regimeScore || 50) * 0.15;
      score += (vix != null ? clamp(mapRange(vix, 10, 22, 100, 40)) : 70) * 0.15;
    }
    return { pass, score: ind.r(score), checks: c, strategy: 'RSI2' };
  }

  _evalB(bb, dist, decline, regime, todayVolRatio) {
    const c = {};
    c.belowLowerBB = { pass: bb?.belowLower === true };
    c.above200SMA = { pass: dist?.above200SMA === true };
    c.bandwidth = { pass: bb?.bandsWideEnough === true };
    c.volumeNotExtreme = { pass: todayVolRatio < this.B.maxVolumeRatio };
    c.notSqueeze = { pass: bb?.isSqueeze !== true };

    const pass = Object.values(c).every(x => x.pass);
    let score = 0;
    if (pass) {
      score += clamp(mapRange(bb?.distBelowLower || 0, 0, 4, 40, 100)) * 0.25;
      score += clamp(mapRange(bb?.bandwidth || 10, 10, 25, 40, 90)) * 0.15;
      score += (decline?.score || 50) * 0.20;
      score += (regime?.regimeScore || 50) * 0.20;
      score += clamp(mapRange(bb?.percentB || 0, -0.2, 0, 100, 50)) * 0.20;
    }
    return { pass, score: ind.r(score), checks: c, strategy: 'BOLLINGER' };
  }

  _evalC(rsi14, pattern, dist, todayVolRatio, regime) {
    const hasPattern = pattern && (pattern.isHammer || pattern.isBullishEngulfing
      || pattern.isDragonfly || pattern.isPinBar || pattern.isPiercing);
    const c = {};
    c.rsi14 = { value: rsi14, pass: rsi14 !== null && rsi14 < this.C.rsi14Max };
    c.pattern = { value: pattern?.type, pass: hasPattern && (pattern?.score || 0) >= this.C.minPatternScore };
    c.above100SMA = { pass: dist?.above100SMA === true };
    c.volume = { value: ind.r(todayVolRatio), pass: todayVolRatio >= this.C.minCandleVolMultiple };

    const pass = Object.values(c).every(x => x.pass);
    let score = 0;
    if (pass) {
      score += (pattern?.score || 50) * 0.30;
      score += clamp(mapRange(rsi14, 15, 30, 100, 50)) * 0.25;
      score += clamp(mapRange(todayVolRatio, 1.2, 3.0, 50, 100)) * 0.20;
      score += (regime?.regimeScore || 50) * 0.15;
      score += (dist?.above200SMA ? 80 : dist?.above100SMA ? 60 : 30) * 0.10;
    }
    return { pass, score: ind.r(score), checks: c, strategy: 'PATTERN' };
  }

  /**
   * Build the NEXT-DAY entry plan
   * Separates scan-day analysis from entry-day execution
   */
  _buildEntryPlan(stratA, stratB, stratC, today, pattern, bb, distances, atrVal) {
    const strategies = [];
    if (stratA.pass) strategies.push('RSI2');
    if (stratB.pass) strategies.push('BOLLINGER');
    if (stratC.pass) strategies.push('PATTERN');

    const isConfluence = strategies.length >= 2;
    const primary = isConfluence ? 'CONFLUENCE' : strategies[0];

    // ── ENTRY TYPE ──
    // Strategy A & B: market order at next day open
    // Strategy C: stop-buy above candle high (conditional)
    let entryType, triggerPrice, maxEntryPrice;

    if (strategies.includes('PATTERN') && pattern?.entryTrigger) {
      entryType = 'STOP_BUY_ABOVE_HIGH';
      triggerPrice = pattern.entryTrigger;
      // if it never reaches the trigger, NO entry
    } else {
      entryType = 'MARKET_AT_OPEN';
      triggerPrice = null;
    }

    // max acceptable entry price (prevent buying after 4% gap up)
    maxEntryPrice = ind.r(today.c * 1.03); // max 3% above scan-day close

    // ── STOP LOSS ──
    let stopLoss, stopReason;
    if (primary === 'PATTERN' && pattern?.stopLevel) {
      stopLoss = pattern.stopLevel;
      stopReason = 'Below pattern candle low';
    } else if (primary === 'BOLLINGER') {
      // 3% below scan-day close
      stopLoss = ind.r(today.c * 0.97);
      stopReason = '3% below entry (BB strategy)';
    } else {
      // 4% below scan-day close (RSI2)
      stopLoss = ind.r(today.c * 0.96);
      stopReason = '4% below entry (RSI2 strategy)';
    }

    // ATR-adjusted stop (don't let stop be too tight for volatile stocks)
    if (atrVal) {
      const atrStop = ind.r(today.c - atrVal * 1.5);
      if (atrStop < stopLoss) {
        // use wider ATR stop if stock is volatile
        stopLoss = atrStop;
        stopReason += ' (widened by ATR)';
      }
    }

    // ── TARGETS ──
    // T1: first mean (5 SMA for RSI2, middle BB for Bollinger)
    // T2: full reversion (20 SMA)
    const target1 = primary === 'BOLLINGER' ? bb?.middle : distances?.sma5?.value;
    const target2 = distances?.sma20?.value;

    // ── EXIT RULES ──
    const exitRules = [];
    if (strategies.includes('RSI2')) {
      exitRules.push({ type: 'RSI2_GE_65', description: 'RSI(2) ≥ 65' });
      exitRules.push({ type: 'CLOSE_ABOVE_5SMA', description: 'Close > 5-day SMA' });
    }
    if (strategies.includes('BOLLINGER')) {
      exitRules.push({ type: 'CLOSE_ABOVE_MIDDLE_BB', description: 'Close > middle Bollinger Band' });
    }
    if (strategies.includes('PATTERN')) {
      exitRules.push({ type: 'RSI14_GT_50', description: 'RSI(14) > 50' });
    }
    exitRules.push({ type: 'TIME_5_DAYS', description: '5 trading days max hold' });
    exitRules.push({ type: 'STOP_LOSS', description: `Stop at ₹${stopLoss}`, level: stopLoss });

    // max acceptable gap down at entry
    const maxGapDown = primary === 'RSI2' ? -3 : primary === 'BOLLINGER' ? -2.5 : -2;

    return {
      primaryStrategy: primary,
      activeStrategies: strategies,
      isConfluence,
      entryType,
      triggerPrice,          // null for market orders, price for stop-buy
      maxEntryPrice,
      maxGapDown,            // skip if open gaps down more than this %
      stopLoss,
      stopReason,
      target1: target1 ? ind.r(target1) : null,
      target2: target2 ? ind.r(target2) : null,
      exitRules,
      maxHoldDays: 5,
      scanDayClose: ind.r(today.c),
      scanDayHigh: ind.r(today.h),
      scanDayLow: ind.r(today.l),
      patternType: pattern?.type || null,
    };
  }
}

function clamp(v, min = 0, max = 100) { return Math.min(max, Math.max(min, v)); }
function mapRange(v, iMin, iMax, oMin, oMax) { return ((v - iMin) / (iMax - iMin)) * (oMax - oMin) + oMin; }
```

```javascript
const ind = require('./mr-indicators');

/**
 * Entry / exit signal engine
 *
 * KEY CHANGES:
 * 1. Entry validation on ENTRY DAY (next morning) — gap filter, trigger check
 * 2. Conditional entry for Strategy C (stop-buy above candle high)
 * 3. Scaling out: sell 50% at T1, trail rest to T2
 * 4. Entry-day intraday management
 */
class MeanReversionSignalEngine {
  constructor(config = {}) {
    this.scaleOut = config.scaleOut ?? true;       // sell half at T1
    this.scaleOutRatio = config.scaleOutRatio ?? 0.5;
    this.breakevenAfterT1 = config.breakevenAfterT1 ?? true;

    this.exits = {
      RSI2:       { rsi2Exit: 65, above5SMA: true, maxDays: 5, stopPct: 4 },
      BOLLINGER:  { aboveMiddleBB: true, maxDays: 5, stopPct: 3 },
      PATTERN:    { rsi14Exit: 50, maxDays: 5, usePatternStop: true },
      CONFLUENCE: { maxDays: 5, stopPct: 3 },
    };
  }

  // ═══════════════════════════════════════
  //  ENTRY-DAY VALIDATION
  //  (Called at 9:15 AM on the entry day)
  // ═══════════════════════════════════════

  /**
   * Validate entry on the ACTUAL entry day
   * This is the critical step I missed before —
   * scan day ≠ entry day, and a lot can change overnight
   *
   * @param {Object} signal - from scanner (previous day)
   * @param {Object} entryDayOpen - { open, high, low, close, time }
   * @param {Object} positionSize - from risk manager
   */
  validateEntry(signal, entryDayOpen, positionSize = null) {
    const plan = signal.entryPlan;
    const todayOpen = entryDayOpen.open;

    // ── GAP FILTER ──
    const gapCheck = ind.gapAtEntryFilter(
      plan.scanDayClose, todayOpen, plan.stopLoss, plan.primaryStrategy
    );

    if (!gapCheck.canEnter) {
      return {
        action: 'SKIP_ENTRY',
        symbol: signal.symbol,
        reason: gapCheck.verdict,
        gapPercent: gapCheck.gapPercent,
        notes: gapCheck.notes,
      };
    }

    // ── CONDITIONAL TRIGGER FOR STRATEGY C ──
    if (plan.entryType === 'STOP_BUY_ABOVE_HIGH') {
      return {
        action: 'PLACE_STOP_BUY_ORDER',
        symbol: signal.symbol,
        orderType: 'SL_BUY',
        triggerPrice: plan.triggerPrice,
        limitPrice: ind.r(plan.triggerPrice * 1.003), // 0.3% above trigger
        validUntil: 'TODAY_CLOSE',
        reason: `Strategy C: buy above pattern high ₹${plan.triggerPrice}`,
        ifNotTriggered: 'Cancel order at 3:15 PM — signal expires',
        stopLoss: plan.stopLoss,
        notes: `If stock never reaches ₹${plan.triggerPrice} today, NO entry.`,
        _pendingSignal: signal,
        _positionSize: positionSize,
      };
    }

    // ── MARKET ORDER ENTRY (Strategy A & B) ──
    const entryPrice = todayOpen;

    // adjust stop loss based on actual entry price
    const adjustedStop = plan.primaryStrategy === 'PATTERN' && plan.stopLoss
      ? plan.stopLoss  // pattern stop is absolute (below candle low)
      : ind.r(entryPrice * (1 - (this.exits[plan.primaryStrategy]?.stopPct || 4) / 100));

    return this._buildEntry(signal, entryPrice, adjustedStop, positionSize, gapCheck);
  }

  /**
   * For Strategy C: called when stop-buy order triggers intraday
   */
  onConditionalTrigger(pendingSignal, triggerPrice, positionSize) {
    const plan = pendingSignal.entryPlan;
    const entryPrice = triggerPrice;
    const adjustedStop = plan.stopLoss; // below pattern candle low

    return this._buildEntry(pendingSignal, entryPrice, adjustedStop, positionSize, {
      gapPercent: 0, canEnter: true,
    });
  }

  _buildEntry(signal, entryPrice, stopLoss, positionSize, gapCheck) {
    const plan = signal.entryPlan;
    const stopPercent = ind.r(((entryPrice - stopLoss) / entryPrice) * 100);

    // recalculate targets based on actual entry
    const t1Percent = plan.target1 ? ind.r(((plan.target1 - entryPrice) / entryPrice) * 100) : null;
    const t2Percent = plan.target2 ? ind.r(((plan.target2 - entryPrice) / entryPrice) * 100) : null;
    const rr = stopPercent > 0 && t1Percent ? ind.r(t1Percent / stopPercent) : null;

    // quantity split for scaling
    const totalQty = positionSize?.quantity || null;
    const scaleQty1 = totalQty ? Math.ceil(totalQty * this.scaleOutRatio) : null;
    const scaleQty2 = totalQty ? totalQty - scaleQty1 : null;

    return {
      action: 'ENTER',
      symbol: signal.symbol,
      direction: 'LONG',
      orderType: 'CNC',
      primaryStrategy: plan.primaryStrategy,
      activeStrategies: plan.activeStrategies,
      isConfluence: plan.isConfluence,

      entryPrice: ind.r(entryPrice),
      gapAtEntry: gapCheck.gapPercent,
      stopLoss: ind.r(stopLoss),
      stopLossPercent: stopPercent,

      target1: plan.target1,
      target1Percent: t1Percent,
      target2: plan.target2,
      target2Percent: t2Percent,
      riskRewardRatio: rr,

      // scaling plan
      scaling: this.scaleOut ? {
        enabled: true,
        tranche1: { quantity: scaleQty1, exitAt: 'TARGET_1', target: plan.target1 },
        tranche2: { quantity: scaleQty2, exitAt: 'TARGET_2_OR_TRAIL', target: plan.target2 },
        moveStopToBreakevenAfterT1: this.breakevenAfterT1,
      } : { enabled: false },

      exitRules: plan.exitRules,
      maxHoldDays: plan.maxHoldDays,
      entryDate: new Date().toISOString().split('T')[0],
      sector: signal.sector,

      _context: {
        rsi2AtScan: signal.indicators.rsi2,
        rsi14AtScan: signal.indicators.rsi14,
        bbMiddle: signal.bb?.middle,
        sma5: signal.distances?.sma5?.value,
        sma20: signal.distances?.sma20?.value,
        declineType: signal.decline?.type,
        regime: signal.regime?.regime,
        patternType: signal.pattern?.type,
        patternLow: plan.scanDayLow,
        priorBounce: signal.priorBounce?.isProvenSupport,
        multiTFOversold: signal.multiTF?.bothOversold,
      },

      _state: {
        daysSinceEntry: 0,
        highSinceEntry: entryPrice,
        lowSinceEntry: entryPrice,
        t1Hit: false,
        t1ExitQty: 0,
        remainingQty: totalQty,
        stopMovedToBreakeven: false,
      },

      compositeScore: signal.compositeScore,
      quantity: totalQty,
      investmentAmount: totalQty ? ind.r(totalQty * entryPrice) : null,
      maxLoss: totalQty ? ind.r(totalQty * Math.abs(entryPrice - stopLoss)) : null,

      timestamp: new Date().toISOString(),
    };
  }


  // ═══════════════════════════════════════
  //  DAILY EXIT CHECK (with scaling)
  // ═══════════════════════════════════════

  /**
   * @param {Object} entry - from validateEntry/buildEntry
   * @param {Object} todayCandle - {date, o, h, l, c, v}
   * @param {Object} liveIndicators - computed with today's data appended
   *   { rsi2, rsi14, sma5, sma20, bbMiddle }
   * @param {number} dayNumber - 1-based
   */
  dailyExitCheck(entry, todayCandle, liveIndicators, dayNumber) {
    const state = entry._state;
    const price = todayCandle.c;

    state.daysSinceEntry = dayNumber;
    state.highSinceEntry = Math.max(state.highSinceEntry, todayCandle.h);
    state.lowSinceEntry = Math.min(state.lowSinceEntry, todayCandle.l);

    const profitPercent = ((price - entry.entryPrice) / entry.entryPrice) * 100;

    // ── 1. STOP LOSS ──
    if (todayCandle.l <= entry.stopLoss) {
      return {
        action: 'EXIT_STOP_LOSS',
        exitPrice: entry.stopLoss,
        profitPercent: ind.r(-entry.stopLossPercent),
        quantity: state.remainingQty || entry.quantity,
        reason: `Stop ₹${entry.stopLoss} breached`,
        dayNumber,
      };
    }

    // ── 2. T1 HIT — SCALE OUT (sell half) ──
    if (entry.scaling?.enabled && !state.t1Hit && entry.target1) {
      if (todayCandle.h >= entry.target1) {
        state.t1Hit = true;
        const scaleQty = entry.scaling.tranche1.quantity;
        state.t1ExitQty = scaleQty;
        state.remainingQty = (entry.quantity || 0) - scaleQty;

        // move stop to breakeven for remainder
        if (this.breakevenAfterT1) {
          entry.stopLoss = ind.r(entry.entryPrice * 1.002);
          state.stopMovedToBreakeven = true;
        }

        return {
          action: 'SCALE_OUT_T1',
          exitPrice: entry.target1,
          profitPercent: entry.target1Percent,
          quantity: scaleQty,
          remainingQuantity: state.remainingQty,
          newStopLoss: entry.stopLoss,
          reason: `Target 1 ₹${entry.target1} hit — sold ${scaleQty}, holding ${state.remainingQty}`,
          dayNumber,
          // remaining position continues with exit rules
          holdingRemainder: true,
        };
      }
    }

    // ── 3. STRATEGY-SPECIFIC EXITS ──

    // RSI(2) ≥ 65
    if (entry.activeStrategies.includes('RSI2') && liveIndicators.rsi2 != null) {
      if (liveIndicators.rsi2 >= this.exits.RSI2.rsi2Exit) {
        return {
          action: 'EXIT_RSI2_TARGET',
          exitPrice: price,
          profitPercent: ind.r(profitPercent),
          quantity: state.remainingQty || entry.quantity,
          reason: `RSI(2) = ${liveIndicators.rsi2} ≥ ${this.exits.RSI2.rsi2Exit}`,
          dayNumber,
        };
      }
    }

    // Close > 5 SMA
    if (entry.activeStrategies.includes('RSI2') && liveIndicators.sma5 != null) {
      if (price > liveIndicators.sma5) {
        return {
          action: 'EXIT_ABOVE_5SMA',
          exitPrice: price,
          profitPercent: ind.r(profitPercent),
          quantity: state.remainingQty || entry.quantity,
          reason: `Close ₹${ind.r(price)} > 5SMA ₹${ind.r(liveIndicators.sma5)}`,
          dayNumber,
        };
      }
    }

    // Close > middle BB
    if (entry.activeStrategies.includes('BOLLINGER') && liveIndicators.bbMiddle != null) {
      if (price > liveIndicators.bbMiddle) {
        return {
          action: 'EXIT_ABOVE_MIDDLE_BB',
          exitPrice: price,
          profitPercent: ind.r(profitPercent),
          quantity: state.remainingQty || entry.quantity,
          reason: `Close > middle BB ₹${ind.r(liveIndicators.bbMiddle)}`,
          dayNumber,
        };
      }
    }

    // RSI(14) > 50
    if (entry.activeStrategies.includes('PATTERN') && liveIndicators.rsi14 != null) {
      if (liveIndicators.rsi14 > this.exits.PATTERN.rsi14Exit) {
        return {
          action: 'EXIT_RSI14_RECOVERED',
          exitPrice: price,
          profitPercent: ind.r(profitPercent),
          quantity: state.remainingQty || entry.quantity,
          reason: `RSI(14) = ${liveIndicators.rsi14} > ${this.exits.PATTERN.rsi14Exit}`,
          dayNumber,
        };
      }
    }

    // ── 4. TIME EXIT ──
    if (dayNumber >= entry.maxHoldDays) {
      return {
        action: 'EXIT_TIME_LIMIT',
        exitPrice: price,
        profitPercent: ind.r(profitPercent),
        quantity: state.remainingQty || entry.quantity,
        reason: `${entry.maxHoldDays}-day hold limit`,
        dayNumber,
      };
    }

    // ── 5. PROFIT PROTECTION ──
    if (profitPercent >= 5 && !state.stopMovedToBreakeven) {
      entry.stopLoss = ind.r(entry.entryPrice * 1.01);
      state.stopMovedToBreakeven = true;
    }

    // ── HOLD ──
    return {
      action: 'HOLD',
      currentProfit: ind.r(profitPercent),
      stopLoss: entry.stopLoss,
      daysRemaining: entry.maxHoldDays - dayNumber,
      dayNumber,
      t1Hit: state.t1Hit,
      exitProgress: {
        rsi2: entry.activeStrategies.includes('RSI2')
          ? { current: liveIndicators.rsi2, target: this.exits.RSI2.rsi2Exit, desc: `${liveIndicators.rsi2} → ${this.exits.RSI2.rsi2Exit}` }
          : null,
        sma5: entry.activeStrategies.includes('RSI2') && liveIndicators.sma5
          ? { price, level: liveIndicators.sma5, above: price > liveIndicators.sma5, desc: `₹${ind.r(price)} vs ₹${ind.r(liveIndicators.sma5)}` }
          : null,
        bbMiddle: entry.activeStrategies.includes('BOLLINGER') && liveIndicators.bbMiddle
          ? { price, level: liveIndicators.bbMiddle, above: price > liveIndicators.bbMiddle, desc: `₹${ind.r(price)} vs ₹${ind.r(liveIndicators.bbMiddle)}` }
          : null,
        rsi14: entry.activeStrategies.includes('PATTERN')
          ? { current: liveIndicators.rsi14, target: this.exits.PATTERN.rsi14Exit, desc: `${liveIndicators.rsi14} → ${this.exits.PATTERN.rsi14Exit}` }
          : null,
      },
    };
  }
}
```

## `mr-backtest.js`

```javascript
const ind = require('./mr-indicators');
const { MeanReversionScanner } = require('./mr-scanner');

/**
 * Walk-forward backtester
 *
 * KEY CHANGES:
 * 1. Entry at NEXT DAY OPEN (not scan-day close)
 * 2. Gap filter applied on entry day
 * 3. Strategy C: conditional entry only if high > trigger price
 * 4. Scaling out tracked
 * 5. Per-strategy AND per-decline-type analysis
 */
class MeanReversionBacktester {
  constructor(config = {}) {
    this.maxPositions = config.maxPositions ?? 3;
    this.lookback = config.lookback ?? 210;
    this.scannerConfig = config.scannerConfig ?? {};
    this.scaleOut = config.scaleOut ?? true;
    this.maxGapDown = config.maxGapDown ?? -3;
    this.maxGapUp = config.maxGapUp ?? 3;

    this.exits = {
      RSI2: { rsi2Exit: 65, above5SMA: true, maxDays: 5, stopPct: 4 },
      BOLLINGER: { aboveMiddleBB: true, maxDays: 5, stopPct: 3 },
      PATTERN: { rsi14Exit: 50, maxDays: 5, stopPct: 4 },
      CONFLUENCE: { maxDays: 5, stopPct: 3 },
    };
  }

  run(stocksHistory, niftyDaily, options = {}) {
    const scanner = new MeanReversionScanner(this.scannerConfig);
    const dates = niftyDaily.map(c => c.date);

    const startIdx = options.startDate
      ? Math.max(dates.findIndex(d => d >= options.startDate), this.lookback)
      : this.lookback;
    const endIdx = options.endDate
      ? dates.findIndex(d => d > options.endDate)
      : dates.length - 7;

    const trades = [];

    for (let i = startIdx; i < endIdx; i++) {
      const scanDate = dates[i];
      const entryDate = dates[i + 1];
      if (!entryDate) continue;

      // market context
      const niftyPrior = niftyDaily.slice(0, i + 1);
      const nCloses = niftyPrior.map(c => c.c);
      const niftyRet5d = nCloses.length >= 6
        ? ((nCloses[nCloses.length - 1] - nCloses[nCloses.length - 6]) / nCloses[nCloses.length - 6]) * 100 : 0;

      const market = {
        indiaVix: options.vixSeries?.[scanDate] ?? 14,
        niftyDailyCandles: niftyPrior.slice(-210),
        niftyReturn5d: ind.r(niftyRet5d),
        sectorReturns5d: {},
      };

      // build snapshots
      const snapshots = [];
      for (const [symbol, candles] of Object.entries(stocksHistory)) {
        const idx = candles.findIndex(c => c.date === scanDate);
        if (idx < this.lookback) continue;
        snapshots.push({
          symbol, sector: null,
          dailyCandles: candles.slice(idx - this.lookback, idx + 1),
        });
      }

      // scan
      const results = scanner.scan(snapshots, market);

      // collect unique candidates
      const seen = new Set();
      const candidates = [];
      for (const list of [results.confluence, results.strategyA, results.strategyB, results.strategyC]) {
        for (const c of list) {
          if (!seen.has(c.symbol)) { seen.add(c.symbol); candidates.push(c); }
        }
      }
      candidates.sort((a, b) => b.compositeScore - a.compositeScore);

      // simulate entries
      let taken = 0;
      for (const cand of candidates) {
        if (taken >= this.maxPositions) break;

        const sym = cand.symbol;
        const candles = stocksHistory[sym];
        if (!candles) continue;

        const scanIdx = candles.findIndex(c => c.date === scanDate);
        const entryIdx = candles.findIndex(c => c.date === entryDate);
        if (scanIdx < 0 || entryIdx < 0 || entryIdx + 6 >= candles.length) continue;

        const plan = cand.entryPlan;
        const entryDayCandle = candles[entryIdx];

        // ── GAP FILTER ──
        const gapPct = ((entryDayCandle.o - plan.scanDayClose) / plan.scanDayClose) * 100;
        if (gapPct < this.maxGapDown || gapPct > this.maxGapUp) continue;

        // ── ENTRY PRICE ──
        let entryPrice;
        if (plan.entryType === 'STOP_BUY_ABOVE_HIGH') {
          // Strategy C: only enter if day's high reaches trigger
          if (entryDayCandle.h < plan.triggerPrice) continue; // trigger not reached
          entryPrice = plan.triggerPrice;
        } else {
          entryPrice = entryDayCandle.o; // NEXT DAY OPEN (the correct behavior)
        }

        // ── STOP LOSS ──
        const stopLoss = plan.entryType === 'STOP_BUY_ABOVE_HIGH'
          ? plan.stopLoss  // pattern low
          : entryPrice * (1 - (this.exits[plan.primaryStrategy]?.stopPct || 4) / 100);

        // ── SIMULATE HOLD ──
        const exitResult = this._simulateHold(
          candles, entryIdx, entryPrice, stopLoss, plan
        );

        const pnl = ((exitResult.exitPrice - entryPrice) / entryPrice) * 100;

        trades.push({
          scanDate,
          entryDate: candles[entryIdx]?.date,
          exitDate: exitResult.exitDate,
          symbol: sym,
          strategy: plan.primaryStrategy,
          activeStrategies: plan.activeStrategies,
          isConfluence: plan.isConfluence,
          entryType: plan.entryType,
          entryPrice: ind.r(entryPrice),
          gapAtEntry: ind.r(gapPct),
          stopLoss: ind.r(stopLoss),
          exitPrice: ind.r(exitResult.exitPrice),
          exitReason: exitResult.reason,
          pnlPercent: ind.r(pnl),
          result: pnl > 0.1 ? 'WIN' : pnl < -0.1 ? 'LOSS' : 'FLAT',
          holdDays: exitResult.holdDays,
          t1Hit: exitResult.t1Hit,
          compositeScore: cand.compositeScore,
          declineType: cand.decline?.type,
          regime: cand.regime?.regime,
          rsi2AtScan: cand.indicators.rsi2,
          multiTFOversold: cand.multiTF?.bothOversold,
          priorBounceSupport: cand.priorBounce?.isProvenSupport,
          patternAtMA: cand.maPattern?.atKeyMA,
        });

        taken++;
      }
    }

    return {
      trades,
      summary: this._summary(trades),
      strategyComparison: this._compareStrategies(trades),
      analysis: this._deepAnalysis(trades),
    };
  }

  _simulateHold(candles, entryIdx, entryPrice, stopLoss, plan) {
    const maxDays = plan.maxHoldDays || 5;
    let t1Hit = false;
    let currentStop = stopLoss;

    for (let d = 1; d <= maxDays && entryIdx + d < candles.length; d++) {
      const day = candles[entryIdx + d];
      const dayIdx = entryIdx + d;

      // stop loss
      if (day.l <= currentStop) {
        return { exitPrice: currentStop, exitDate: day.date, reason: 'STOP_LOSS', holdDays: d, t1Hit };
      }

      // compute indicators with data up to today
      const allCloses = candles.slice(0, dayIdx + 1).map(c => c.c);

      // strategy exits
      const primary = plan.primaryStrategy;

      if (primary === 'RSI2' || primary === 'CONFLUENCE') {
        const r2 = ind.rsi(allCloses, 2);
        if (r2 !== null && r2 >= (this.exits.RSI2.rsi2Exit || 65)) {
          return { exitPrice: day.c, exitDate: day.date, reason: 'RSI2_EXIT', holdDays: d, t1Hit };
        }
        const s5 = ind.sma(allCloses, 5);
        if (s5 && day.c > s5) {
          return { exitPrice: day.c, exitDate: day.date, reason: 'ABOVE_5SMA', holdDays: d, t1Hit };
        }
      }

      if (primary === 'BOLLINGER' || primary === 'CONFLUENCE') {
        const bb = ind.bollingerBands(allCloses, 20, 2);
        if (bb && day.c > bb.middle) {
          return { exitPrice: day.c, exitDate: day.date, reason: 'ABOVE_MIDDLE_BB', holdDays: d, t1Hit };
        }
      }

      if (primary === 'PATTERN' || primary === 'CONFLUENCE') {
        const r14 = ind.rsi(allCloses, 14);
        if (r14 !== null && r14 > (this.exits.PATTERN.rsi14Exit || 50)) {
          return { exitPrice: day.c, exitDate: day.date, reason: 'RSI14_EXIT', holdDays: d, t1Hit };
        }
      }

      // T1 hit (for scaling tracking)
      if (!t1Hit && plan.target1 && day.h >= plan.target1) {
        t1Hit = true;
        if (this.scaleOut) {
          // move stop to breakeven for remainder
          currentStop = Math.max(currentStop, entryPrice * 1.002);
        }
      }

      // time exit
      if (d === maxDays) {
        return { exitPrice: day.c, exitDate: day.date, reason: 'TIME_EXIT', holdDays: d, t1Hit };
      }
    }

    const lastDay = candles[Math.min(entryIdx + maxDays, candles.length - 1)];
    return { exitPrice: lastDay.c, exitDate: lastDay.date, reason: 'TIME_EXIT', holdDays: maxDays, t1Hit };
  }

  _summary(trades) {
    if (!trades.length) return { totalTrades: 0 };
    const w = trades.filter(t => t.result === 'WIN');
    const l = trades.filter(t => t.result === 'LOSS');
    const tot = trades.reduce((s, t) => s + t.pnlPercent, 0);
    const gp = w.reduce((s, t) => s + t.pnlPercent, 0);
    const gl = Math.abs(l.reduce((s, t) => s + t.pnlPercent, 0));

    return {
      totalTrades: trades.length,
      wins: w.length, losses: l.length,
      winRate: ind.r((w.length / trades.length) * 100),
      totalPnl: ind.r(tot),
      avgPnl: ind.r(tot / trades.length),
      avgWin: w.length ? ind.r(gp / w.length) : 0,
      avgLoss: l.length ? ind.r(-gl / l.length) : 0,
      profitFactor: gl > 0 ? ind.r(gp / gl) : Infinity,
      avgHold: ind.r(trades.reduce((s, t) => s + t.holdDays, 0) / trades.length, 1),
      best: ind.r(Math.max(...trades.map(t => t.pnlPercent))),
      worst: ind.r(Math.min(...trades.map(t => t.pnlPercent))),
    };
  }

  _compareStrategies(trades) {
    const groups = {};
    for (const t of trades) {
      const k = t.strategy;
      if (!groups[k]) groups[k] = [];
      groups[k].push(t);
    }
    const result = {};
    for (const [k, v] of Object.entries(groups)) result[k] = this._summary(v);

    const conf = trades.filter(t => t.isConfluence);
    const nonConf = trades.filter(t => !t.isConfluence);
    result.CONFLUENCE_EDGE = {
      confluence: this._summary(conf),
      individual: this._summary(nonConf),
      edge: conf.length >= 5 && nonConf.length >= 5
        ? ind.r((conf.reduce((s, t) => s + t.pnlPercent, 0) / conf.length)
          - (nonConf.reduce((s, t) => s + t.pnlPercent, 0) / nonConf.length))
        : null,
    };
    return result;
  }

  _deepAnalysis(trades) {
    if (trades.length < 10) return null;

    const avg = arr => arr.length ? ind.r(arr.reduce((s, t) => s + t.pnlPercent, 0) / arr.length) : 0;
    const wr = arr => arr.length ? ind.r((arr.filter(t => t.result === 'WIN').length / arr.length) * 100) : 0;

    // by decline type (THE key insight)
    const byDecline = {};
    for (const t of trades) {
      const d = t.declineType || 'UNKNOWN';
      if (!byDecline[d]) byDecline[d] = [];
      byDecline[d].push(t);
    }

    // by entry type
    const marketOrders = trades.filter(t => t.entryType === 'MARKET_AT_OPEN');
    const stopBuys = trades.filter(t => t.entryType === 'STOP_BUY_ABOVE_HIGH');

    // by gap at entry
    const gapDown = trades.filter(t => t.gapAtEntry < -0.5);
    const gapUp = trades.filter(t => t.gapAtEntry > 0.5);
    const flatOpen = trades.filter(t => Math.abs(t.gapAtEntry) <= 0.5);

    // multi-TF oversold
    const multiTF = trades.filter(t => t.multiTFOversold);
    const singleTF = trades.filter(t => !t.multiTFOversold);

    // prior bounce support
    const withSupport = trades.filter(t => t.priorBounceSupport);
    const withoutSupport = trades.filter(t => !t.priorBounceSupport);

    // pattern at MA
    const atMA = trades.filter(t => t.patternAtMA);
    const notAtMA = trades.filter(t => !t.patternAtMA);

    // by exit reason
    const byExit = {};
    for (const t of trades) {
      if (!byExit[t.exitReason]) byExit[t.exitReason] = [];
      byExit[t.exitReason].push(t);
    }

    // by regime
    const byRegime = {};
    for (const t of trades) {
      const r = t.regime || 'UNKNOWN';
      if (!byRegime[r]) byRegime[r] = [];
      byRegime[r].push(t);
    }

    // drawdown
    let cum = 0, peak = 0, maxDD = 0;
    for (const t of trades) { cum += t.pnlPercent; peak = Math.max(peak, cum); maxDD = Math.max(maxDD, peak - cum); }

    return {
      byDeclineType: Object.fromEntries(
        Object.entries(byDecline).map(([k, v]) => [k, { count: v.length, avgPnl: avg(v), winRate: wr(v) }])
      ),
      byEntryType: {
        marketOrder: { count: marketOrders.length, avgPnl: avg(marketOrders), winRate: wr(marketOrders) },
        stopBuy: { count: stopBuys.length, avgPnl: avg(stopBuys), winRate: wr(stopBuys) },
      },
      byGapAtEntry: {
        gapDown: { count: gapDown.length, avgPnl: avg(gapDown), winRate: wr(gapDown) },
        flatOpen: { count: flatOpen.length, avgPnl: avg(flatOpen), winRate: wr(flatOpen) },
        gapUp: { count: gapUp.length, avgPnl: avg(gapUp), winRate: wr(gapUp) },
      },
      multiTimeframe: {
        bothOversold: { count: multiTF.length, avgPnl: avg(multiTF), winRate: wr(multiTF) },
        dailyOnly: { count: singleTF.length, avgPnl: avg(singleTF), winRate: wr(singleTF) },
        edge: multiTF.length >= 3 ? ind.r(avg(multiTF) - avg(singleTF)) : null,
      },
      priorBounceEffect: {
        withSupport: { count: withSupport.length, avgPnl: avg(withSupport), winRate: wr(withSupport) },
        without: { count: withoutSupport.length, avgPnl: avg(withoutSupport), winRate: wr(withoutSupport) },
        edge: withSupport.length >= 3 ? ind.r(avg(withSupport) - avg(withoutSupport)) : null,
      },
      patternAtMA: {
        atMA: { count: atMA.length, avgPnl: avg(atMA), winRate: wr(atMA) },
        notAtMA: { count: notAtMA.length, avgPnl: avg(notAtMA), winRate: wr(notAtMA) },
        edge: atMA.length >= 3 ? ind.r(avg(atMA) - avg(notAtMA)) : null,
      },
      byRegime: Object.fromEntries(
        Object.entries(byRegime).map(([k, v]) => [k, { count: v.length, avgPnl: avg(v), winRate: wr(v) }])
      ),
      byExitReason: Object.fromEntries(
        Object.entries(byExit).map(([k, v]) => [k, { count: v.length, avgPnl: avg(v) }])
      ),
      maxDrawdown: ind.r(maxDD),
      finalCumPnl: ind.r(cum),
    };
  }
}

module.exports = { MeanReversionBacktester };
```
