```javascript
/**
 * Pure indicator functions for BTST/STBT strategy
 * Zero dependencies — all math, no state
 */

function ema(closes, period) {
  if (!closes || closes.length < period) return null;
  const k = 2 / (period + 1);
  let val = 0;
  for (let i = 0; i < period; i++) val += closes[i];
  val /= period;
  for (let i = period; i < closes.length; i++) {
    val = closes[i] * k + val * (1 - k);
  }
  return r(val);
}

function emaSeries(closes, period) {
  if (!closes || closes.length < period) return [];
  const k = 2 / (period + 1);
  const result = new Array(period - 1).fill(null);
  let val = 0;
  for (let i = 0; i < period; i++) val += closes[i];
  val /= period;
  result.push(r(val));
  for (let i = period; i < closes.length; i++) {
    val = closes[i] * k + val * (1 - k);
    result.push(r(val));
  }
  return result;
}

function sma(values, period) {
  if (!values || values.length < period) return null;
  let sum = 0;
  for (let i = values.length - period; i < values.length; i++) sum += values[i];
  return r(sum / period);
}

/**
 * RSI using Wilder's smoothing (standard)
 */
function rsi(closes, period = 14) {
  if (!closes || closes.length < period + 1) return null;

  let avgGain = 0, avgLoss = 0;
  for (let i = 1; i <= period; i++) {
    const diff = closes[i] - closes[i - 1];
    if (diff >= 0) avgGain += diff;
    else avgLoss -= diff;
  }
  avgGain /= period;
  avgLoss /= period;

  for (let i = period + 1; i < closes.length; i++) {
    const diff = closes[i] - closes[i - 1];
    avgGain = (avgGain * (period - 1) + Math.max(diff, 0)) / period;
    avgLoss = (avgLoss * (period - 1) + Math.max(-diff, 0)) / period;
  }

  if (avgLoss === 0) return 100;
  return r(100 - 100 / (1 + avgGain / avgLoss));
}

/**
 * VWAP from intraday candles
 * @param {Array} candles - [{time, o, h, l, c, v}, ...]
 */
function vwap(candles) {
  if (!candles || candles.length === 0) return null;
  let cumTPV = 0, cumVol = 0;
  for (const c of candles) {
    const tp = (c.h + c.l + c.c) / 3;
    cumTPV += tp * c.v;
    cumVol += c.v;
  }
  return cumVol > 0 ? r(cumTPV / cumVol) : null;
}

/**
 * VWAP with bands (±1σ, ±2σ)
 */
function vwapWithBands(candles) {
  if (!candles || candles.length === 0) return null;
  let cumTPV = 0, cumVol = 0, cumTPV2 = 0;

  const vwapValue = vwap(candles);
  if (!vwapValue) return null;

  for (const c of candles) {
    const tp = (c.h + c.l + c.c) / 3;
    cumTPV += tp * c.v;
    cumTPV2 += tp * tp * c.v;
    cumVol += c.v;
  }

  const variance = cumTPV2 / cumVol - vwapValue * vwapValue;
  const stdDev = Math.sqrt(Math.max(variance, 0));

  return {
    vwap: vwapValue,
    upperBand1: r(vwapValue + stdDev),
    lowerBand1: r(vwapValue - stdDev),
    upperBand2: r(vwapValue + 2 * stdDev),
    lowerBand2: r(vwapValue - 2 * stdDev),
    stdDev: r(stdDev),
  };
}

/**
 * ATR — Average True Range (Wilder's smoothing)
 * @param {Array} candles - [{o, h, l, c}, ...] daily candles
 */
function atr(candles, period = 14) {
  if (!candles || candles.length < period + 1) return null;

  const trs = [];
  for (let i = 1; i < candles.length; i++) {
    trs.push(Math.max(
      candles[i].h - candles[i].l,
      Math.abs(candles[i].h - candles[i - 1].c),
      Math.abs(candles[i].l - candles[i - 1].c)
    ));
  }

  let val = 0;
  for (let i = 0; i < period; i++) val += trs[i];
  val /= period;

  for (let i = period; i < trs.length; i++) {
    val = (val * (period - 1) + trs[i]) / period;
  }
  return r(val);
}

/**
 * Volume analysis suite
 */
function avgVolume(volumes, period = 20) {
  if (!volumes || volumes.length < period) return null;
  return r(volumes.slice(-period).reduce((a, b) => a + b, 0) / period);
}

/**
 * Volume acceleration — compares last-hour volume rate to rest-of-day rate
 * @param {Array} intradayCandles - 5-min candles with time field
 * @param {string} cutoffTime - 'HH:MM' format, default '14:30'
 */
function volumeAcceleration(intradayCandles, cutoffTime = '14:30') {
  if (!intradayCandles || intradayCandles.length < 4) return null;

  const before = [], after = [];
  for (const c of intradayCandles) {
    const t = extractTime(c.time);
    if (t < cutoffTime) before.push(c);
    else after.push(c);
  }

  if (before.length === 0 || after.length === 0) return null;

  const beforeRate = before.reduce((s, c) => s + c.v, 0) / before.length;
  const afterRate = after.reduce((s, c) => s + c.v, 0) / after.length;

  return {
    ratio: beforeRate > 0 ? r(afterRate / beforeRate) : null,
    lastPeriodVolume: after.reduce((s, c) => s + c.v, 0),
    restOfDayRate: r(beforeRate),
    lastHourRate: r(afterRate),
    isAccelerating: afterRate > beforeRate * 1.3,
  };
}

/**
 * Last-hour momentum — price/volume behavior after cutoff
 */
function lastHourMomentum(intradayCandles, cutoffTime = '14:30') {
  if (!intradayCandles || intradayCandles.length < 4) return null;

  const lastHour = intradayCandles.filter(c => extractTime(c.time) >= cutoffTime);
  if (lastHour.length < 2) return null;

  const first = lastHour[0];
  const last = lastHour[lastHour.length - 1];
  const priceChange = ((last.c - first.o) / first.o) * 100;

  // higher highs count
  let hh = 0;
  for (let i = 1; i < lastHour.length; i++) {
    if (lastHour[i].h > lastHour[i - 1].h) hh++;
  }

  // higher lows count (support building)
  let hl = 0;
  for (let i = 1; i < lastHour.length; i++) {
    if (lastHour[i].l > lastHour[i - 1].l) hl++;
  }

  const total = lastHour.length - 1;
  const higherHighRatio = total > 0 ? hh / total : 0;
  const higherLowRatio = total > 0 ? hl / total : 0;

  // volume trend within last hour (split into halves)
  const mid = Math.floor(lastHour.length / 2);
  const firstHalfVol = lastHour.slice(0, mid).reduce((s, c) => s + c.v, 0);
  const secondHalfVol = lastHour.slice(mid).reduce((s, c) => s + c.v, 0);
  const volumeTrend = firstHalfVol > 0 ? secondHalfVol / firstHalfVol : 1;

  // did it make a new day high in last hour?
  const allCandles = intradayCandles;
  const dayHighBeforeCutoff = Math.max(
    ...allCandles.filter(c => extractTime(c.time) < cutoffTime).map(c => c.h)
  );
  const lastHourHigh = Math.max(...lastHour.map(c => c.h));
  const newDayHigh = lastHourHigh > dayHighBeforeCutoff;

  return {
    priceChange: r(priceChange),
    higherHighRatio: r(higherHighRatio),
    higherLowRatio: r(higherLowRatio),
    volumeTrend: r(volumeTrend),
    newDayHigh,
    candleCount: lastHour.length,
    isStrongClose: priceChange > 0.3 && higherHighRatio > 0.5 && volumeTrend > 1,
    isBuyingClimax: priceChange > 0.5 && volumeTrend > 1.5 && newDayHigh,
  };
}

/**
 * Candle body analysis
 */
function candleBody(candle) {
  const range = candle.h - candle.l;
  if (range === 0) {
    return { bodyPercent: 0, upperWick: 0, lowerWick: 0, isBullish: true, isMarubozu: false, isDoji: true };
  }

  const body = Math.abs(candle.c - candle.o);
  const isBullish = candle.c >= candle.o;
  const upper = isBullish ? candle.h - candle.c : candle.h - candle.o;
  const lower = isBullish ? candle.o - candle.l : candle.c - candle.l;

  return {
    bodyPercent: r((body / range) * 100),
    upperWickPercent: r((upper / range) * 100),
    lowerWickPercent: r((lower / range) * 100),
    isBullish,
    isMarubozu: body / range > 0.80,
    isBullishMarubozu: isBullish && body / range > 0.80,
    isBearishMarubozu: !isBullish && body / range > 0.80,
    isDoji: body / range < 0.10,
    isHammer: isBullish && lower / range > 0.60 && upper / range < 0.10,
    isShootingStar: !isBullish && upper / range > 0.60 && lower / range < 0.10,
  };
}

/**
 * Closing strength: where close sits in day's range (0 = at low, 1 = at high)
 */
function closingStrength(candle) {
  const range = candle.h - candle.l;
  if (range === 0) return 0.5;
  return r((candle.c - candle.l) / range);
}

/**
 * Delivery percentage estimation heuristic
 * Real delivery data comes post-market; this estimates during market hours
 */
function estimateDelivery(rangePosition, volumeRatio, bodyAnalysis) {
  let score = 0;

  // Close near high → buyers not selling, holding for delivery
  score += Math.min(rangePosition, 1) * 30;

  // High volume with bullish bias → institutional delivery buying
  score += Math.min(volumeRatio / 3, 1) * 25;

  // Full-body bullish candle → decisive, less speculative
  if (bodyAnalysis.isBullishMarubozu) score += 25;
  else if (bodyAnalysis.isBullish && bodyAnalysis.bodyPercent > 60) score += 18;
  else if (bodyAnalysis.isBullish) score += 10;

  // No upper wick → no selling pressure at highs
  if (bodyAnalysis.upperWickPercent < 5) score += 12;
  else if (bodyAnalysis.upperWickPercent < 15) score += 6;

  // Volume ratio extreme → block deal / institutional
  if (volumeRatio > 3) score += 8;

  return r(Math.min(score, 100));
}

/**
 * Beta — stock sensitivity to index
 * Uses daily returns over given period
 */
function beta(stockCloses, indexCloses, period = 60) {
  const len = Math.min(stockCloses.length, indexCloses.length, period + 1);
  if (len < 21) return null; // need at least 20 data points

  const stockReturns = [], indexReturns = [];
  const start = Math.max(stockCloses.length - len, 0);
  const idxStart = Math.max(indexCloses.length - len, 0);

  for (let i = 1; i < len; i++) {
    stockReturns.push((stockCloses[start + i] - stockCloses[start + i - 1]) / stockCloses[start + i - 1]);
    indexReturns.push((indexCloses[idxStart + i] - indexCloses[idxStart + i - 1]) / indexCloses[idxStart + i - 1]);
  }

  const n = stockReturns.length;
  const mS = stockReturns.reduce((a, b) => a + b, 0) / n;
  const mI = indexReturns.reduce((a, b) => a + b, 0) / n;

  let cov = 0, varI = 0;
  for (let i = 0; i < n; i++) {
    cov += (stockReturns[i] - mS) * (indexReturns[i] - mI);
    varI += (indexReturns[i] - mI) ** 2;
  }

  return varI > 0 ? r(cov / varI) : null;
}

/**
 * Detect upper/lower circuit — untradeable stocks
 */
function detectCircuit(candle, prevClose) {
  const dayChange = Math.abs((candle.c - prevClose) / prevClose) * 100;
  const range = candle.h - candle.l;
  const rangePercent = (range / candle.c) * 100;

  // Circuit = significant move with almost zero range (locked)
  const isCircuit = dayChange > 4.5 && rangePercent < 0.3;
  const isUpperCircuit = isCircuit && candle.c > prevClose;
  const isLowerCircuit = isCircuit && candle.c < prevClose;

  return { isCircuit, isUpperCircuit, isLowerCircuit, dayChange: r(dayChange), rangePercent: r(rangePercent) };
}

/**
 * Analyze historical gaps for a stock
 * Returns gap tendencies after strong bullish/bearish days
 */
function gapHistory(dailyCandles, lookback = 120) {
  const candles = dailyCandles.slice(-Math.min(lookback + 1, dailyCandles.length));
  if (candles.length < 30) return null;

  const allGaps = [];
  const gapsAfterBullish = [];
  const gapsAfterBearish = [];

  for (let i = 1; i < candles.length; i++) {
    const prev = candles[i - 1];
    const curr = candles[i];
    const gap = ((curr.o - prev.c) / prev.c) * 100;

    const prevDayChange = ((prev.c - prev.o) / prev.o) * 100;
    const prevRange = prev.h - prev.l;
    const prevCloseStrength = prevRange > 0 ? (prev.c - prev.l) / prevRange : 0.5;

    allGaps.push(gap);

    // BTST-like condition: prev day up >2%, close near high
    if (prevDayChange > 2 && prevCloseStrength > 0.85) {
      gapsAfterBullish.push(gap);
    }

    // STBT-like condition: prev day down >2%, close near low
    if (prevDayChange < -2 && prevCloseStrength < 0.15) {
      gapsAfterBearish.push(gap);
    }
  }

  const analyze = (arr) => {
    if (arr.length === 0) return { avg: 0, median: 0, upProb: 0.5, count: 0, stdDev: 0 };
    const sorted = [...arr].sort((a, b) => a - b);
    const avg = arr.reduce((a, b) => a + b, 0) / arr.length;
    const median = sorted[Math.floor(sorted.length / 2)];
    const upCount = arr.filter(g => g > 0.1).length;
    const variance = arr.reduce((s, g) => s + (g - avg) ** 2, 0) / arr.length;
    return {
      avg: r(avg),
      median: r(median),
      upProb: r(upCount / arr.length),
      downProb: r(1 - upCount / arr.length),
      count: arr.length,
      stdDev: r(Math.sqrt(variance)),
      max: r(Math.max(...arr)),
      min: r(Math.min(...arr)),
    };
  };

  return {
    overall: analyze(allGaps),
    afterBullishDay: analyze(gapsAfterBullish),
    afterBearishDay: analyze(gapsAfterBearish),
    avgAbsGap: r(allGaps.reduce((s, g) => s + Math.abs(g), 0) / allGaps.length),
    isGapProne: Math.abs(analyze(allGaps).avg) > 0.3 || analyze(allGaps).stdDev > 1.0,
  };
}

// ── Utilities ──

function extractTime(timeStr) {
  if (!timeStr) return '00:00';
  if (timeStr.includes('T')) {
    // ISO format: 2024-01-15T09:15:00
    return timeStr.split('T')[1].substring(0, 5);
  }
  // already HH:MM
  return timeStr.substring(0, 5);
}

function r(val, d = 2) {
  if (val === null || val === undefined || isNaN(val)) return null;
  const m = 10 ** d;
  return Math.round(val * m) / m;
}

  beta, detectCircuit, gapHistory,
```

## `btst-scanner.js`

```javascript
const ind = require('./btst-indicators');

/**
 * BTST Scanner — evaluates stocks at 2:30 PM and 3:00 PM
 *
 * Two-phase scanning:
 *   Phase 1 (2:30 PM): Preliminary scan → builds watchlist
 *   Phase 2 (3:00 PM): Confirms with closing behavior → final candidates
 */
class BTSTScanner {
  constructor(config = {}) {
    this.c = {
      minDayChange: config.minDayChange ?? 2.0,
      minClosingStrength: config.minClosingStrength ?? 0.85,
      minVolumeMultiple: config.minVolumeMultiple ?? 1.5,
      emaPeriod: config.emaPeriod ?? 20,
      rsiMin: config.rsiMin ?? 55,
      rsiMax: config.rsiMax ?? 75,
      requireFnO: config.requireFnO ?? true,
      requireNiftyPositive: config.requireNiftyPositive ?? true,

      // intraday enhancements
      minVolumeAcceleration: config.minVolumeAcceleration ?? 1.2,
      requireAboveVwap: config.requireAboveVwap ?? true,
      minDeliveryEstimate: config.minDeliveryEstimate ?? 40,
      lastHourCutoff: config.lastHourCutoff ?? '14:30',
    };
  }

  /**
   * Full scan — takes pre-built stock data and market context
   *
   * @param {Array<StockData>} stocks
   *   Each stock: {
   *     symbol, sector, isFnO,
   *     dailyCandles: [{date, o, h, l, c, v}, ...],  // last 30+ days
   *     today: { open, high, low, close, volume, prevClose },
   *     intradayCandles: [{time, o, h, l, c, v}, ...] | null  // 5-min candles today
   *   }
   *
   * @param {MarketContext} market
   *   {
   *     niftyChangePercent, niftyPositive,
   *     indiaVix, usFuturesChangePercent,
   *     niftyDailyCandles: [{date, o, h, l, c, v}, ...]  // for beta calc
   *   }
   */
  scan(stocks, market) {
    // market-level gate
    if (this.c.requireNiftyPositive && !market.niftyPositive) {
      return {
        candidates: [],
        rejected: [],
        meta: { skippedReason: `Nifty negative (${market.niftyChangePercent}%)`, scanned: 0 },
      };
    }

    const candidates = [];
    const rejected = [];

    for (const stock of stocks) {
      try {
        const result = this.evaluate(stock, market);
        if (result.passed) {
          candidates.push(result);
        } else {
          rejected.push({ symbol: stock.symbol, failedAt: result.firstFailure, passCount: result.passCount });
        }
      } catch (err) {
        rejected.push({ symbol: stock.symbol, error: err.message });
      }
    }

    candidates.sort((a, b) => b.compositeScore - a.compositeScore);

    return {
      candidates,
      rejected,
      meta: {
        scanned: stocks.length,
        passed: candidates.length,
        niftyChange: market.niftyChangePercent,
        timestamp: new Date().toISOString(),
      },
    };
  }

  /**
   * Evaluate a single stock — all criteria + intraday analysis
   */
  evaluate(stock, market) {
    const t = stock.today;
    const dailyCloses = stock.dailyCandles.map(c => c.c);
    const dailyVolumes = stock.dailyCandles.map(c => c.v);

    // append today for indicator computation
    const allCloses = [...dailyCloses, t.close];
    const allVolumes = [...dailyVolumes, t.volume];

    // ── compute indicators ──
    const ema20 = ind.ema(allCloses, this.c.emaPeriod);
    const rsi14 = ind.rsi(allCloses, 14);
    const atr14 = ind.atr([...stock.dailyCandles, { o: t.open, h: t.high, l: t.low, c: t.close }], 14);
    const avgVol20 = ind.avgVolume(dailyVolumes, 20);
    const cs = ind.closingStrength(t);
    const body = ind.candleBody(t);
    const circuit = ind.detectCircuit(t, t.prevClose);

    // ── intraday analysis (if available) ──
    let intradayData = null;
    if (stock.intradayCandles && stock.intradayCandles.length > 0) {
      const vwapVal = ind.vwap(stock.intradayCandles);
      const vwapBands = ind.vwapWithBands(stock.intradayCandles);
      const volAccel = ind.volumeAcceleration(stock.intradayCandles, this.c.lastHourCutoff);
      const lastHour = ind.lastHourMomentum(stock.intradayCandles, this.c.lastHourCutoff);

      intradayData = { vwap: vwapVal, vwapBands, volAccel, lastHour };
    }

    // ── delivery estimate ──
    const volRatio = avgVol20 > 0 ? t.volume / avgVol20 : 0;
    const deliveryEst = ind.estimateDelivery(cs, volRatio, body);

    // ── beta ──
    let betaVal = null;
    if (market.niftyDailyCandles) {
      const niftyCloses = market.niftyDailyCandles.map(c => c.c);
      betaVal = ind.beta(dailyCloses, niftyCloses, 60);
    }

    // ── gap history ──
    const gaps = ind.gapHistory(stock.dailyCandles, 120);

    // ── run the 7 core criteria + enhanced checks ──
    const checks = {};
    let passCount = 0;
    let firstFailure = null;

    const dayChange = ((t.close - t.prevClose) / t.prevClose) * 100;

    // 1. Day change > 2%
    checks.dayChange = {
      value: ind.r(dayChange), threshold: this.c.minDayChange,
      pass: dayChange >= this.c.minDayChange,
    };

    // 2. Closing near day high
    checks.closingStrength = {
      value: ind.r(cs, 3), threshold: this.c.minClosingStrength,
      pass: cs >= this.c.minClosingStrength,
    };

    // 3. Volume > 1.5× average
    checks.volume = {
      value: ind.r(volRatio), threshold: this.c.minVolumeMultiple,
      pass: volRatio >= this.c.minVolumeMultiple,
    };

    // 4. Close above 20 EMA
    checks.aboveEma = {
      value: t.close, ema: ema20,
      pass: ema20 !== null && t.close > ema20,
    };

    // 5. Nifty positive
    checks.niftyPositive = {
      value: market.niftyChangePercent,
      pass: market.niftyPositive,
    };

    // 6. RSI 55-75
    checks.rsi = {
      value: rsi14, range: [this.c.rsiMin, this.c.rsiMax],
      pass: rsi14 !== null && rsi14 >= this.c.rsiMin && rsi14 <= this.c.rsiMax,
    };

    // 7. F&O segment
    checks.fno = {
      value: stock.isFnO,
      pass: this.c.requireFnO ? stock.isFnO === true : true,
    };

    // ── Enhanced checks (don't block, but affect score) ──

    // 8. NOT at upper circuit (can't buy)
    checks.notCircuit = {
      value: circuit,
      pass: !circuit.isUpperCircuit,
      critical: true, // blocks even if core criteria pass
    };

    // 9. Close above VWAP (if intraday available)
    checks.aboveVwap = {
      value: intradayData?.vwap,
      pass: intradayData ? t.close > intradayData.vwap : true, // pass if no intraday data
      hasData: !!intradayData,
    };

    // 10. Volume acceleration in last hour
    checks.volumeAcceleration = {
      value: intradayData?.volAccel?.ratio,
      pass: intradayData?.volAccel
        ? intradayData.volAccel.ratio >= this.c.minVolumeAcceleration
        : true,
      hasData: !!intradayData?.volAccel,
    };

    // 11. Bullish candle body
    checks.candleBody = {
      value: body,
      pass: body.isBullish && body.bodyPercent > 40,
    };

    // count core criteria (1-7)
    const coreChecks = ['dayChange', 'closingStrength', 'volume', 'aboveEma', 'niftyPositive', 'rsi', 'fno'];
    for (const key of coreChecks) {
      if (checks[key].pass) passCount++;
      else if (!firstFailure) firstFailure = key;
    }

    // circuit is a hard block
    if (checks.notCircuit.pass === false) {
      firstFailure = 'upperCircuit';
    }

    const corePass = passCount === coreChecks.length && checks.notCircuit.pass;

    // ── composite score for ranking ──
    let compositeScore = 0;
    if (corePass) {
      compositeScore = this._score(checks, intradayData, deliveryEst, gaps, body);
    }

    return {
      symbol: stock.symbol,
      sector: stock.sector,
      passed: corePass,
      passCount,
      firstFailure,
      checks,
      compositeScore: ind.r(compositeScore),
      indicators: {
        close: t.close,
        dayChange: ind.r(dayChange),
        closingStrength: ind.r(cs, 3),
        volumeRatio: ind.r(volRatio),
        ema20,
        rsi14,
        atr14,
        beta: betaVal,
        deliveryEstimate: deliveryEst,
      },
      intraday: intradayData,
      gapHistory: gaps,
      candle: body,
    };
  }

  _score(checks, intraday, delivery, gaps, body) {
    let s = 0;

    // momentum strength (0-20)
    s += Math.min(checks.dayChange.value / 6, 1) * 20;

    // closing strength (0-20)
    s += checks.closingStrength.value * 20;

    // volume conviction (0-15)
    s += Math.min(checks.volume.value / 4, 1) * 15;

    // RSI sweet spot — 65 ideal (0-10)
    const rsiDev = Math.abs(checks.rsi.value - 65);
    s += Math.max(0, 1 - rsiDev / 15) * 10;

    // EMA distance — moderate is best (0-10)
    if (checks.aboveEma.ema) {
      const emaDist = ((checks.aboveEma.value - checks.aboveEma.ema) / checks.aboveEma.ema) * 100;
      if (emaDist > 0 && emaDist <= 4) s += 10;
      else if (emaDist <= 8) s += 6;
      else s += 3;
    }

    // delivery estimate (0-10)
    s += (delivery / 100) * 10;

    // intraday bonuses (0-15)
    if (intraday) {
      if (intraday.volAccel?.isAccelerating) s += 5;
      if (intraday.lastHour?.isBuyingClimax) s += 5;
      else if (intraday.lastHour?.isStrongClose) s += 3;
      if (checks.aboveVwap.pass && checks.aboveVwap.hasData) s += 5;
    }

    // gap history bonus — stock tends to gap up after strong days
    if (gaps?.afterBullishDay?.upProb > 0.6) s += 5;

    // candle quality bonus
    if (body.isBullishMarubozu) s += 5;

    return s;
  }
}

/**
 * STBT Scanner — mirror of BTST for short setups
 */
class STBTScanner {
  constructor(config = {}) {
    this.c = {
      maxDayChange: config.maxDayChange ?? -2.0,
      maxClosingStrength: config.maxClosingStrength ?? 0.15,
      minVolumeMultiple: config.minVolumeMultiple ?? 1.5,
      emaPeriod: config.emaPeriod ?? 20,
      rsiMin: config.rsiMin ?? 25,
      rsiMax: config.rsiMax ?? 45,
      requireFnO: true,
      requireNiftyNegative: config.requireNiftyNegative ?? true,
      lastHourCutoff: config.lastHourCutoff ?? '14:30',
    };
  }

  scan(stocks, market) {
    if (this.c.requireNiftyNegative && market.niftyChangePercent > 0) {
      return { candidates: [], meta: { skippedReason: 'Nifty positive — STBT needs bearish market' } };
    }

    const candidates = [];

    for (const stock of stocks) {
      const result = this.evaluateShort(stock, market);
      if (result.passed) candidates.push(result);
    }

    candidates.sort((a, b) => b.compositeScore - a.compositeScore);
    return { candidates, meta: { scanned: stocks.length, passed: candidates.length } };
  }

  evaluateShort(stock, market) {
    const t = stock.today;
    const dailyCloses = [...stock.dailyCandles.map(c => c.c), t.close];
    const dailyVolumes = stock.dailyCandles.map(c => c.v);

    const ema20 = ind.ema(dailyCloses, this.c.emaPeriod);
    const rsi14 = ind.rsi(dailyCloses, 14);
    const avgVol20 = ind.avgVolume(dailyVolumes, 20);
    const cs = ind.closingStrength(t);
    const body = ind.candleBody(t);
    const circuit = ind.detectCircuit(t, t.prevClose);
    const dayChange = ((t.close - t.prevClose) / t.prevClose) * 100;
    const volRatio = avgVol20 > 0 ? t.volume / avgVol20 : 0;

    let intradayData = null;
    if (stock.intradayCandles && stock.intradayCandles.length > 0) {
      intradayData = {
        vwap: ind.vwap(stock.intradayCandles),
        volAccel: ind.volumeAcceleration(stock.intradayCandles, this.c.lastHourCutoff),
        lastHour: ind.lastHourMomentum(stock.intradayCandles, this.c.lastHourCutoff),
      };
    }

    // STBT criteria (MIRROR of BTST)
    const checks = {};

    // 1. Stock DOWN > 2%
    checks.dayChange = { value: ind.r(dayChange), pass: dayChange <= this.c.maxDayChange };

    // 2. Close near day LOW
    checks.closingWeakness = { value: ind.r(cs, 3), pass: cs <= this.c.maxClosingStrength };

    // 3. High volume
    checks.volume = { value: ind.r(volRatio), pass: volRatio >= this.c.minVolumeMultiple };

    // 4. Below 20 EMA
    checks.belowEma = { value: t.close, ema: ema20, pass: ema20 !== null && t.close < ema20 };

    // 5. Nifty negative
    checks.niftyNegative = { value: market.niftyChangePercent, pass: market.niftyChangePercent < 0 };

    // 6. RSI 25-45
    checks.rsi = { value: rsi14, pass: rsi14 !== null && rsi14 >= this.c.rsiMin && rsi14 <= this.c.rsiMax };

    // 7. F&O (mandatory for shorting)
    checks.fno = { value: stock.isFnO, pass: stock.isFnO === true };

    // 8. Not at lower circuit
    checks.notCircuit = { value: circuit, pass: !circuit.isLowerCircuit };

    // 9. Below VWAP
    checks.belowVwap = {
      value: intradayData?.vwap,
      pass: intradayData ? t.close < intradayData.vwap : true,
    };

    const coreKeys = ['dayChange', 'closingWeakness', 'volume', 'belowEma', 'niftyNegative', 'rsi', 'fno'];
    let passCount = 0;
    for (const k of coreKeys) { if (checks[k].pass) passCount++; }

    const passed = passCount === coreKeys.length && checks.notCircuit.pass;

    let compositeScore = 0;
    if (passed) {
      compositeScore += Math.min(Math.abs(dayChange) / 6, 1) * 25;
      compositeScore += (1 - cs) * 25;
      compositeScore += Math.min(volRatio / 4, 1) * 20;
      compositeScore += (rsi14 < 35 ? 15 : 10);
      if (body.isBearishMarubozu) compositeScore += 10;
      if (intradayData?.lastHour?.priceChange < -0.3) compositeScore += 5;
    }

    return {
      symbol: stock.symbol,
      sector: stock.sector,
      passed,
      passCount,
      checks,
      compositeScore: ind.r(compositeScore),
      direction: 'SHORT',
      indicators: {
        close: t.close, dayChange: ind.r(dayChange), closingStrength: ind.r(cs, 3),
        volumeRatio: ind.r(volRatio), ema20, rsi14,
      },
      intraday: intradayData,
      candle: body,
    };
  }
}

module.exports = { BTSTScanner, STBTScanner };
```

```javascript
const ind = require('./btst-indicators');

/**
 * Entry/Exit signal engine for BTST and STBT
 *
 * Entry: 3:00-3:15 PM CNC order
 * Exit: Next morning based on opening scenario
 */
class BTSTSignalEngine {
  constructor(config = {}) {
    this.stopLossATRMultiple = config.stopLossATRMultiple ?? 1.5;
    this.fallbackStopPercent = config.fallbackStopPercent ?? 1.5;
    this.targetATRMultiple = config.targetATRMultiple ?? 2.5;
    this.gapUpThreshold = config.gapUpThreshold ?? 0.5;      // % for gap-up classification
    this.gapDownThreshold = config.gapDownThreshold ?? -0.3;  // % for gap-down classification
    this.maxHoldMinutes = config.maxHoldMinutes ?? 90;        // sell within 90 min of open
  }

  // ═══════════════════════════════════════
  //  ENTRY SIGNAL
  // ═══════════════════════════════════════

  /**
   * Generate a complete entry signal with ATR-based levels
   */
  generateEntry(candidate, positionSize) {
    const price = candidate.indicators.close;
    const atrVal = candidate.indicators.atr14;
    const direction = candidate.direction || 'LONG';

    // ATR-based stop loss (adapts to stock volatility)
    let stopLoss, target;
    if (atrVal && atrVal > 0) {
      if (direction === 'LONG') {
        stopLoss = ind.r(price - atrVal * this.stopLossATRMultiple);
        target = ind.r(price + atrVal * this.targetATRMultiple);
      } else {
        stopLoss = ind.r(price + atrVal * this.stopLossATRMultiple);
        target = ind.r(price - atrVal * this.targetATRMultiple);
      }
    } else {
      // fallback to percentage-based
      if (direction === 'LONG') {
        stopLoss = ind.r(price * (1 - this.fallbackStopPercent / 100));
        target = ind.r(price * (1 + this.fallbackStopPercent * 1.5 / 100));
      } else {
        stopLoss = ind.r(price * (1 + this.fallbackStopPercent / 100));
        target = ind.r(price * (1 - this.fallbackStopPercent * 1.5 / 100));
      }
    }

    const riskPerShare = Math.abs(price - stopLoss);
    const rewardPerShare = Math.abs(target - price);

    return {
      symbol: candidate.symbol,
      direction,
      action: direction === 'LONG' ? 'BUY' : 'SELL',
      orderType: direction === 'LONG' ? 'CNC' : 'MIS/NRML', // CNC for delivery, NRML for futures
      entryPrice: price,
      stopLoss,
      target,
      riskPerShare: ind.r(riskPerShare),
      rewardPerShare: ind.r(rewardPerShare),
      riskRewardRatio: ind.r(rewardPerShare / riskPerShare),
      stopLossPercent: ind.r((riskPerShare / price) * 100),
      targetPercent: ind.r((rewardPerShare / price) * 100),
      atrUsed: atrVal,
      quantity: positionSize?.quantity ?? null,
      investmentAmount: positionSize ? ind.r(positionSize.quantity * price) : null,
      maxLoss: positionSize ? ind.r(positionSize.quantity * riskPerShare) : null,
      entryWindow: { start: '15:00', end: '15:15' },
      score: candidate.compositeScore,
      sector: candidate.sector,

      // data for exit engine
      _meta: {
        closingStrength: candidate.indicators.closingStrength,
        volumeRatio: candidate.indicators.volumeRatio,
        rsi: candidate.indicators.rsi14,
        beta: candidate.indicators.beta,
        gapHistory: candidate.gapHistory,
        lastHourMomentum: candidate.intraday?.lastHour,
      },

      timestamp: new Date().toISOString(),
    };
  }

  // ═══════════════════════════════════════
  //  PRE-MARKET ASSESSMENT
  // ═══════════════════════════════════════

  /**
   * Before market opens — assess GIFT Nifty and globals to prepare exit plan
   * Call this at 8:30-9:00 AM with pre-market data
   */
  preMarketAssessment(entry, preMarketData) {
    const {
      giftNiftyChangePercent = 0,
      usFuturesChangePercent = 0,
      sgxNiftyChangePercent = 0,
      asianMarketsChangePercent = 0,
    } = preMarketData;

    // weighted global sentiment
    const globalSentiment =
      giftNiftyChangePercent * 0.50 +
      usFuturesChangePercent * 0.25 +
      sgxNiftyChangePercent * 0.15 +
      asianMarketsChangePercent * 0.10;

    const isLong = entry.direction === 'LONG';

    let outlook, plan;

    if ((isLong && globalSentiment > 0.5) || (!isLong && globalSentiment < -0.5)) {
      outlook = 'FAVORABLE';
      plan = 'Expect profitable open. Prepare to book profits in first 15 min or trail stop.';
    } else if ((isLong && globalSentiment < -0.8) || (!isLong && globalSentiment > 0.8)) {
      outlook = 'ADVERSE';
      plan = 'Expect adverse gap. Place market order at 9:15 AM to exit immediately.';
    } else {
      outlook = 'NEUTRAL';
      plan = 'Flat open expected. Wait 15-30 min for direction, then decide.';
    }

    // beta-adjusted expected gap
    const betaVal = entry._meta?.beta ?? 1;
    const expectedStockGap = ind.r(globalSentiment * betaVal);

    return {
      symbol: entry.symbol,
      direction: entry.direction,
      globalSentiment: ind.r(globalSentiment),
      outlook,
      plan,
      expectedGap: expectedStockGap,
      inputs: {
        giftNifty: giftNiftyChangePercent,
        usFutures: usFuturesChangePercent,
        sgxNifty: sgxNiftyChangePercent,
        asianMarkets: asianMarketsChangePercent,
      },
      beta: betaVal,
    };
  }

  // ═══════════════════════════════════════
  //  EXIT DECISION (9:15 AM - 10:45 AM)
  // ═══════════════════════════════════════

  /**
   * Primary exit decision at market open
   *
   * @param {Object} entry - from generateEntry()
   * @param {Object} openData - { open, high, low, close, volume }
   *   first candle or snapshot at 9:15-9:20 AM
   * @param {Object} preMarket - from preMarketAssessment() (optional)
   */
  exitDecision(entry, openData, preMarket = null) {
    const openPrice = openData.open;
    const isLong = entry.direction === 'LONG';

    const gapPercent = isLong
      ? ((openPrice - entry.entryPrice) / entry.entryPrice) * 100
      : ((entry.entryPrice - openPrice) / entry.entryPrice) * 100;
    // gapPercent > 0 = favorable, < 0 = adverse (for both long/short)

    // ── SCENARIO A: Favorable gap ──
    if (gapPercent >= this.gapUpThreshold) {
      return {
        symbol: entry.symbol,
        scenario: 'FAVORABLE_GAP',
        gapPercent: ind.r(gapPercent),
        action: 'BOOK_PARTIAL_OR_TRAIL',
        urgency: 'FIRST_15_MIN',
        plan: {
          option1: {
            label: 'Book full profit at open',
            exitPrice: openPrice,
            profitPercent: ind.r(gapPercent),
            when: 'Immediately at 9:15-9:16',
          },
          option2: {
            label: 'Trail with tight stop',
            trailingStop: isLong
              ? ind.r(openPrice * 0.995)  // 0.5% below open
              : ind.r(openPrice * 1.005),
            holdUntil: 'Stop hit or target reached',
            target: entry.target,
          },
          option3: {
            label: 'Book 50%, trail rest',
            sellQty: entry.quantity ? Math.ceil(entry.quantity / 2) : 'half',
            trailingStop: isLong
              ? ind.r(openPrice * 0.997)
              : ind.r(openPrice * 1.003),
          },
        },
        recommended: gapPercent > 2 ? 'option1' : 'option3',
      };
    }

    // ── SCENARIO D: Hard adverse gap (beyond stop loss) ──
    if (gapPercent <= -(entry.stopLossPercent || this.fallbackStopPercent)) {
      return {
        symbol: entry.symbol,
        scenario: 'HARD_ADVERSE_GAP',
        gapPercent: ind.r(gapPercent),
        action: 'EXIT_IMMEDIATELY',
        urgency: 'MARKET_ORDER_AT_OPEN',
        plan: {
          exitPrice: openPrice,
          lossPercent: ind.r(Math.abs(gapPercent)),
          orderType: 'MARKET',
          notes: 'Do NOT wait. Do NOT average. Cut loss NOW.',
        },
        recommended: 'market_sell_9:15',
      };
    }

    // ── SCENARIO C: Mild adverse gap (within stop) ──
    if (gapPercent < this.gapDownThreshold) {
      return {
        symbol: entry.symbol,
        scenario: 'MILD_ADVERSE_GAP',
        gapPercent: ind.r(gapPercent),
        action: 'HOLD_WITH_TIGHT_STOP',
        urgency: 'MONITOR_CLOSELY',
        plan: {
          stopLoss: entry.stopLoss,
          waitMinutes: 15,
          rules: [
            `If price recovers above ${isLong ? entry.entryPrice : entry.entryPrice} within 15 min → hold for target`,
            `If price breaks ${entry.stopLoss} → exit immediately`,
            `If sideways after 15 min → exit at market (don't hold a losing BTST)`,
          ],
          maxHold: `${this.maxHoldMinutes} minutes from open`,
        },
        recommended: 'place_stop_loss_order',
      };
    }

    // ── SCENARIO B: Flat open (within ±0.5%) ──
    return {
      symbol: entry.symbol,
      scenario: 'FLAT_OPEN',
      gapPercent: ind.r(gapPercent),
      action: 'WAIT_FOR_DIRECTION',
      urgency: 'WAIT_15_TO_30_MIN',
      plan: {
        stopLoss: entry.stopLoss,
        target: entry.target,
        waitMinutes: 30,
        breakoutLevel: isLong
          ? ind.r(entry.entryPrice * 1.005)
          : ind.r(entry.entryPrice * 0.995),
        rules: [
          `If price moves favorably > 0.5% in first 30 min → trail stop`,
          `If first 15-min candle is strong → hold for target`,
          `If sideways → exit at 30 min mark`,
          `Hard stop at ${entry.stopLoss} regardless`,
        ],
        deadlineExit: `Must exit by ${this.maxHoldMinutes} min after open`,
      },
      recommended: 'wait_with_stop_loss',
    };
  }

  // ═══════════════════════════════════════
  //  TRAILING EXIT (live during exit morning)
  // ═══════════════════════════════════════

  /**
   * Call with each new candle/tick on exit morning
   */
  trailingExit(entry, currentPrice, highSinceOpen, lowSinceOpen, minutesSinceOpen) {
    const isLong = entry.direction === 'LONG';

    const profitPercent = isLong
      ? ((currentPrice - entry.entryPrice) / entry.entryPrice) * 100
      : ((entry.entryPrice - currentPrice) / entry.entryPrice) * 100;

    const peakProfit = isLong
      ? ((highSinceOpen - entry.entryPrice) / entry.entryPrice) * 100
      : ((entry.entryPrice - lowSinceOpen) / entry.entryPrice) * 100;

    const drawdownFromPeak = peakProfit - profitPercent;

    // ── TIME-BASED EXIT ──
    if (minutesSinceOpen >= this.maxHoldMinutes) {
      return {
        action: 'EXIT_TIME_LIMIT',
        exitPrice: currentPrice,
        profitPercent: ind.r(profitPercent),
        reason: `Held for ${minutesSinceOpen} min — BTST should not become a swing trade`,
      };
    }

    // ── STOP LOSS HIT ──
    const stopHit = isLong
      ? currentPrice <= entry.stopLoss
      : currentPrice >= entry.stopLoss;
    if (stopHit) {
      return {
        action: 'EXIT_STOP_LOSS',
        exitPrice: currentPrice,
        lossPercent: ind.r(Math.abs(profitPercent)),
        reason: 'Stop loss breached',
      };
    }

    // ── TARGET HIT ──
    const targetHit = isLong
      ? currentPrice >= entry.target
      : currentPrice <= entry.target;
    if (targetHit) {
      return {
        action: 'EXIT_TARGET',
        exitPrice: currentPrice,
        profitPercent: ind.r(profitPercent),
        reason: 'Target reached',
      };
    }

    // ── TRAILING STOP LOGIC ──
    // Tiered trailing based on profit level

    let trailingStop = entry.stopLoss;

    if (profitPercent >= 3.0) {
      // lock in at least 2%
      trailingStop = isLong
        ? ind.r(entry.entryPrice * 1.02)
        : ind.r(entry.entryPrice * 0.98);
      // also trail from peak
      const peakTrail = isLong
        ? ind.r(highSinceOpen * 0.993)   // 0.7% from high
        : ind.r(lowSinceOpen * 1.007);
      trailingStop = isLong
        ? Math.max(trailingStop, peakTrail)
        : Math.min(trailingStop, peakTrail);
    } else if (profitPercent >= 2.0) {
      // lock in at least 1%
      trailingStop = isLong
        ? ind.r(entry.entryPrice * 1.01)
        : ind.r(entry.entryPrice * 0.99);
      const peakTrail = isLong
        ? ind.r(highSinceOpen * 0.995)
        : ind.r(lowSinceOpen * 1.005);
      trailingStop = isLong
        ? Math.max(trailingStop, peakTrail)
        : Math.min(trailingStop, peakTrail);
    } else if (profitPercent >= 1.0) {
      // move stop to breakeven
      trailingStop = isLong
        ? ind.r(entry.entryPrice * 1.001)
        : ind.r(entry.entryPrice * 0.999);
    }

    // check if trailing stop hit
    const trailHit = isLong
      ? currentPrice <= trailingStop
      : currentPrice >= trailingStop;
    if (trailHit && profitPercent > 0) {
      return {
        action: 'EXIT_TRAILING_STOP',
        exitPrice: currentPrice,
        profitPercent: ind.r(profitPercent),
        peakProfit: ind.r(peakProfit),
        reason: `Trailing stop at ${trailingStop} hit after ${ind.r(peakProfit)}% peak profit`,
      };
    }

    return {
      action: 'HOLD',
      currentProfit: ind.r(profitPercent),
      peakProfit: ind.r(peakProfit),
      trailingStop,
      hardStop: entry.stopLoss,
      target: entry.target,
      minutesRemaining: this.maxHoldMinutes - minutesSinceOpen,
    };
  }
}
```

```javascript
const ind = require('./btst-indicators');

/**
 * Risk management: pre-trade gates, ATR-based sizing,
 * beta-adjusted allocation, portfolio-level risk
 */
class BTSTRiskManager {
  constructor(config = {}) {
    this.totalCapital = config.totalCapital ?? 500000;
    this.maxBTSTAllocation = config.maxBTSTAllocation ?? 0.30; // 30% max in BTST
    this.maxPositions = config.maxPositions ?? 3;
    this.maxPerStock = config.maxPerStock ?? 0.12;             // 12% per stock
    this.maxRiskPerTrade = config.maxRiskPerTrade ?? 0.01;     // 1% of capital
    this.maxTotalRisk = config.maxTotalRisk ?? 0.03;           // 3% total overnight risk
    this.blockedDays = new Set(config.blockedDays ?? [5]);     // Friday
  }

  // ═══════════════════════════════════════
  //  PRE-TRADE GATES
  // ═══════════════════════════════════════

  /**
   * Full pre-trade risk check — must pass before ANY BTST trade
   */
  preTradeGate(market, date = new Date()) {
    const blocks = [];
    const warnings = [];

    // 1. Day of week
    const dow = date.getDay();
    if (this.blockedDays.has(dow)) {
      blocks.push({
        rule: 'WEEKEND_RISK',
        severity: 'BLOCK',
        msg: 'BTST blocked on Fridays — 2 nights of gap risk unacceptable',
      });
    }

    // 2. India VIX
    if (market.indiaVix != null) {
      if (market.indiaVix > 22) {
        blocks.push({
          rule: 'VIX_EXTREME',
          severity: 'BLOCK',
          msg: `India VIX ${market.indiaVix} > 22 — overnight risk too high`,
        });
      } else if (market.indiaVix > 18) {
        warnings.push({
          rule: 'VIX_ELEVATED',
          severity: 'WARN',
          msg: `India VIX ${market.indiaVix} — reduce position sizes by 50%`,
          sizeMultiplier: 0.5,
        });
      }
    }

    // 3. US futures
    if (market.usFuturesChangePercent != null) {
      if (market.usFuturesChangePercent < -1.5) {
        blocks.push({
          rule: 'US_CRASHING',
          severity: 'BLOCK',
          msg: `US futures down ${market.usFuturesChangePercent}% — global selloff risk`,
        });
      } else if (market.usFuturesChangePercent < -0.8) {
        warnings.push({
          rule: 'US_WEAK',
          severity: 'WARN',
          msg: `US futures down ${market.usFuturesChangePercent}% — be cautious`,
          sizeMultiplier: 0.7,
        });
      }
    }

    // 4. Major events
    if (market.majorEventTomorrow) {
      blocks.push({
        rule: 'EVENT_RISK',
        severity: 'BLOCK',
        msg: `Event tomorrow: ${market.majorEventTomorrow}`,
      });
    }

    // 5. Market breadth (if Nifty down > 1%, panic day)
    if (market.niftyChangePercent < -1.0) {
      blocks.push({
        rule: 'MARKET_PANIC',
        severity: 'BLOCK',
        msg: `Nifty down ${market.niftyChangePercent}% — not a BTST day`,
      });
    }

    // 6. RBI/Fed decision days
    if (market.isCentralBankDay) {
      blocks.push({
        rule: 'CENTRAL_BANK',
        severity: 'BLOCK',
        msg: 'Central bank decision day — skip BTST',
      });
    }

    // 7. Results season for specific stock (handled per-stock, not here)

    // compute size multiplier from warnings
    let sizeMultiplier = 1.0;
    for (const w of warnings) {
      if (w.sizeMultiplier) sizeMultiplier = Math.min(sizeMultiplier, w.sizeMultiplier);
    }

    return {
      allowed: blocks.length === 0,
      blocks,
      warnings,
      sizeMultiplier,
      timestamp: date.toISOString(),
    };
  }

  /**
   * Per-stock risk check (results announcement, ex-date, etc.)
   */
  stockSpecificCheck(symbol, events = []) {
    const blocks = [];

    for (const event of events) {
      if (event.symbol === symbol) {
        if (event.type === 'RESULTS' && event.isWithin2Days) {
          blocks.push({
            rule: 'RESULTS_RISK',
            msg: `${symbol} results on ${event.date} — skip BTST (gap risk ±10%+)`,
          });
        }
        if (event.type === 'EX_DIVIDEND' || event.type === 'EX_BONUS') {
          blocks.push({
            rule: 'CORPORATE_ACTION',
            msg: `${symbol} ${event.type} on ${event.date} — price adjustment will occur`,
          });
        }
      }
    }

    return { allowed: blocks.length === 0, blocks };
  }

  // ═══════════════════════════════════════
  //  POSITION SIZING
  // ═══════════════════════════════════════

  /**
   * ATR-based position sizing with multiple constraints
   *
   * @param {number} entryPrice
   * @param {number} atrValue - stock's ATR(14)
   * @param {number} atrStopMultiple - how many ATRs for stop loss
   * @param {number} existingPositions - how many BTST positions already open
   * @param {number} existingExposure - ₹ already allocated to BTST tonight
   * @param {number} sizeMultiplier - from pre-trade gate (1.0 = normal, 0.5 = half)
   * @param {number|null} betaValue - stock beta for beta-adjusted sizing
   */
  calculatePosition(
    entryPrice, atrValue, atrStopMultiple = 1.5,
    existingPositions = 0, existingExposure = 0,
    sizeMultiplier = 1.0, betaValue = null
  ) {
    if (existingPositions >= this.maxPositions) {
      return { allowed: false, reason: `Max ${this.maxPositions} BTST positions reached` };
    }

    const btstPool = this.totalCapital * this.maxBTSTAllocation * sizeMultiplier;
    const remainingPool = btstPool - existingExposure;
    const remainingSlots = this.maxPositions - existingPositions;

    if (remainingPool <= 0) {
      return { allowed: false, reason: 'BTST capital pool exhausted' };
    }

    // ── Method 1: Risk-based (max 1% of capital at risk per trade) ──
    const riskBudget = this.totalCapital * this.maxRiskPerTrade * sizeMultiplier;
    const riskPerShare = atrValue && atrValue > 0
      ? atrValue * atrStopMultiple
      : entryPrice * (this.maxRiskPerTrade * 100 / 100); // fallback
    const sharesByRisk = Math.floor(riskBudget / riskPerShare);

    // ── Method 2: Capital-based (max 12% per stock) ──
    const maxAllocation = this.totalCapital * this.maxPerStock * sizeMultiplier;
    const sharesByCapital = Math.floor(maxAllocation / entryPrice);

    // ── Method 3: Pool-based (equal distribution of remaining pool) ──
    const perSlotAllocation = remainingPool / remainingSlots;
    const sharesByPool = Math.floor(perSlotAllocation / entryPrice);

    // ── Method 4: Beta-adjusted (high beta → smaller position) ──
    let sharesByBeta = Infinity;
    if (betaValue && betaValue > 0) {
      const betaAdjustedAllocation = maxAllocation / Math.max(betaValue, 0.5);
      sharesByBeta = Math.floor(betaAdjustedAllocation / entryPrice);
    }

    // Take the MINIMUM of all methods
    const quantity = Math.max(1, Math.min(sharesByRisk, sharesByCapital, sharesByPool, sharesByBeta));
    const investment = ind.r(quantity * entryPrice);
    const maxLoss = ind.r(quantity * riskPerShare);
    const stopLoss = ind.r(entryPrice - riskPerShare);

    // determine limiting factor
    const limits = { risk: sharesByRisk, capital: sharesByCapital, pool: sharesByPool, beta: sharesByBeta };
    const limitingFactor = Object.entries(limits)
      .sort((a, b) => a[1] - b[1])[0][0]
      .toUpperCase();

    return {
      allowed: true,
      quantity,
      entryPrice: ind.r(entryPrice),
      stopLoss,
      investmentAmount: investment,
      maxLoss,
      maxLossPercent: ind.r((maxLoss / this.totalCapital) * 100),
      capitalUtilization: ind.r((investment / this.totalCapital) * 100),
      remainingSlots: remainingSlots - 1,
      remainingPool: ind.r(remainingPool - investment),
      limitingFactor,
      sizeMultiplier,
      breakdown: {
        byRisk: sharesByRisk,
        byCapital: sharesByCapital,
        byPool: sharesByPool,
        byBeta: sharesByBeta === Infinity ? 'N/A' : sharesByBeta,
      },
    };
  }

  // ═══════════════════════════════════════
  //  PORTFOLIO-LEVEL RISK
  // ═══════════════════════════════════════

  /**
   * Overnight risk summary for all active BTST positions
   */
  portfolioRisk(activePositions) {
    if (activePositions.length === 0) {
      return { positionCount: 0, riskRating: 'NONE', totalExposure: 0 };
    }

    const totalExposure = activePositions.reduce(
      (sum, p) => sum + (p.quantity * p.entryPrice), 0
    );

    const totalMaxLoss = activePositions.reduce(
      (sum, p) => sum + (p.maxLoss || p.quantity * p.entryPrice * 0.015), 0
    );

    const worstCasePercent = ind.r((totalMaxLoss / this.totalCapital) * 100);

    // sector concentration
    const sectorCounts = {};
    for (const p of activePositions) {
      sectorCounts[p.sector || 'UNKNOWN'] = (sectorCounts[p.sector || 'UNKNOWN'] || 0) + 1;
    }
    const maxSectorConcentration = Math.max(...Object.values(sectorCounts));
    const sectorWarning = maxSectorConcentration >= 2 && activePositions.length > 1;

    // correlation risk (simplified: same sector = correlated)
    const uniqueSectors = Object.keys(sectorCounts).length;
    const diversificationScore = uniqueSectors / activePositions.length;

    // beta-weighted exposure
    const betaWeightedExposure = activePositions.reduce((sum, p) => {
      const b = p.beta ?? 1;
      return sum + (p.quantity * p.entryPrice * b);
    }, 0);

    let riskRating;
    if (worstCasePercent > this.maxTotalRisk * 100) riskRating = 'EXCESSIVE';
    else if (worstCasePercent > 2) riskRating = 'HIGH';
    else if (worstCasePercent > 1) riskRating = 'MODERATE';
    else riskRating = 'LOW';

    return {
      positionCount: activePositions.length,
      symbols: activePositions.map(p => p.symbol),
      totalExposure: ind.r(totalExposure),
      exposurePercent: ind.r((totalExposure / this.totalCapital) * 100),
      totalMaxLoss: ind.r(totalMaxLoss),
      worstCaseDrawdown: worstCasePercent,
      betaWeightedExposure: ind.r(betaWeightedExposure),
      sectorBreakdown: sectorCounts,
      sectorConcentrationWarning: sectorWarning,
      diversificationScore: ind.r(diversificationScore),
      riskRating,
      withinLimits: worstCasePercent <= this.maxTotalRisk * 100,
    };
  }
}

module.exports = { BTSTRiskManager };
```

```javascript
const ind = require('./btst-indicators');

/**
 * Multi-factor scoring and ranking for BTST/STBT candidates
 * Integrates gap history, sector strength, intraday quality
 */
class BTSTScorer {
  constructor(weights = {}) {
    // weights sum to 100
    this.w = {
      momentum:        weights.momentum        ?? 15,
      closingStrength: weights.closingStrength  ?? 15,
      volumeConviction: weights.volumeConviction ?? 12,
      lastHourQuality: weights.lastHourQuality  ?? 15,
      vwapRelation:    weights.vwapRelation     ?? 8,
      deliveryEstimate: weights.deliveryEstimate ?? 10,
      trendAlignment:  weights.trendAlignment   ?? 8,
      rsiOptimality:   weights.rsiOptimality    ?? 5,
      gapTendency:     weights.gapTendency      ?? 7,
      sectorStrength:  weights.sectorStrength   ?? 5,
    };
  }

  /**
   * Score and rank a list of scanner candidates
   *
   * @param {Array} candidates - from BTSTScanner.scan().candidates
   * @param {Object} sectorData - { 'AUTO': 2.1, 'IT': -0.5, ... }
   */
  rank(candidates, sectorData = {}) {
    const scored = candidates.map(c => {
      const factors = this._factors(c, sectorData);
      const totalScore = this._weightedTotal(factors);
      const grade = this._grade(totalScore);

      return {
        ...c,
        scoring: {
          factors,
          totalScore: ind.r(totalScore),
          grade,
          recommendation: this._recommend(totalScore, grade, factors),
        },
      };
    });

    scored.sort((a, b) => b.scoring.totalScore - a.scoring.totalScore);
    scored.forEach((s, i) => { s.rank = i + 1; });

    return scored;
  }

  _factors(c, sectorData) {
    const i = c.indicators;
    const id = c.intraday;
    const g = c.gapHistory;
    const body = c.candle;

    // 1. Momentum — day change magnitude
    const momentum = clamp(mapRange(i.dayChange, 2, 7, 30, 100));

    // 2. Closing strength — how close to day high
    const closing = clamp(mapRange(i.closingStrength, 0.85, 1.0, 40, 100));

    // 3. Volume conviction
    const volume = clamp(mapRange(i.volumeRatio, 1.5, 5.0, 30, 100));

    // 4. Last hour quality (KEY differentiator)
    let lastHour = 50; // neutral if no intraday data
    if (id?.lastHour) {
      const lh = id.lastHour;
      lastHour = 0;
      if (lh.isBuyingClimax) lastHour = 100;
      else if (lh.isStrongClose) lastHour = 75;
      else if (lh.priceChange > 0) lastHour = 55;
      else lastHour = 25;

      // bonus for new day high in last hour
      if (lh.newDayHigh) lastHour = Math.min(lastHour + 15, 100);

      // bonus for higher-high pattern
      if (lh.higherHighRatio > 0.6) lastHour = Math.min(lastHour + 10, 100);
    }

    // 5. VWAP relation
    let vwapScore = 50;
    if (id?.vwap) {
      const vwapDist = ((i.close - id.vwap) / id.vwap) * 100;
      if (vwapDist > 1.5) vwapScore = 90;       // well above VWAP
      else if (vwapDist > 0.5) vwapScore = 75;
      else if (vwapDist > 0) vwapScore = 60;
      else vwapScore = 20;                        // below VWAP = bad
    }

    // 6. Delivery estimate
    const delivery = clamp(i.deliveryEstimate ?? 50);

    // 7. Trend alignment (EMA distance)
    let trend = 50;
    if (i.ema20) {
      const emaDist = ((i.close - i.ema20) / i.ema20) * 100;
      if (emaDist > 0 && emaDist <= 3) trend = 100;       // tight above EMA — ideal
      else if (emaDist > 3 && emaDist <= 6) trend = 75;
      else if (emaDist > 6 && emaDist <= 10) trend = 50;
      else if (emaDist > 10) trend = 30;                    // overextended
      else trend = 10;                                       // below EMA shouldn't be here
    }

    // 8. RSI sweet spot (65 ideal)
    let rsiScore = 50;
    if (i.rsi14) {
      const dev = Math.abs(i.rsi14 - 65);
      rsiScore = clamp(mapRange(dev, 0, 15, 100, 30));
    }

    // 9. Gap tendency from history
    let gapScore = 50;
    if (g?.afterBullishDay) {
      const gb = g.afterBullishDay;
      if (gb.count >= 5) {
        // enough data to be meaningful
        gapScore = clamp(mapRange(gb.upProb, 0.4, 0.8, 20, 100));
        // bonus for high average gap
        if (gb.avg > 0.5) gapScore = Math.min(gapScore + 15, 100);
      }
    }

    // 10. Sector strength
    const sectorChange = sectorData[c.sector] ?? 0;
    const sector = clamp(mapRange(sectorChange, -1, 3, 20, 100));

    return {
      momentum:         { score: ind.r(momentum),  raw: i.dayChange },
      closingStrength:  { score: ind.r(closing),    raw: i.closingStrength },
      volumeConviction: { score: ind.r(volume),     raw: i.volumeRatio },
      lastHourQuality:  { score: ind.r(lastHour),   raw: id?.lastHour ?? null },
      vwapRelation:     { score: ind.r(vwapScore),  raw: id?.vwap ?? null },
      deliveryEstimate: { score: ind.r(delivery),   raw: i.deliveryEstimate },
      trendAlignment:   { score: ind.r(trend),      raw: i.ema20 },
      rsiOptimality:    { score: ind.r(rsiScore),   raw: i.rsi14 },
      gapTendency:      { score: ind.r(gapScore),   raw: g?.afterBullishDay ?? null },
      sectorStrength:   { score: ind.r(sector),     raw: sectorChange },
    };
  }

  _weightedTotal(factors) {
    let total = 0;
    for (const [key, factor] of Object.entries(factors)) {
      const weight = this.w[key] ?? 0;
      total += factor.score * (weight / 100);
    }
    return total;
  }

  _grade(score) {
    if (score >= 78) return 'A+';
    if (score >= 70) return 'A';
    if (score >= 62) return 'B+';
    if (score >= 55) return 'B';
    if (score >= 45) return 'C';
    return 'D';
  }

  _recommend(score, grade, factors) {
    const weakFactors = Object.entries(factors)
      .filter(([_, f]) => f.score < 40)
      .map(([k]) => k);

    const strongFactors = Object.entries(factors)
      .filter(([_, f]) => f.score >= 80)
      .map(([k]) => k);

    if (grade === 'A+' || grade === 'A') {
      return {
        action: 'STRONG_BUY',
        confidence: grade === 'A+' ? 'VERY_HIGH' : 'HIGH',
        message: `All systems go — ${strongFactors.length} factors above 80`,
        strengths: strongFactors,
        weaknesses: weakFactors,
        suggestedSizeMultiplier: grade === 'A+' ? 1.0 : 0.9,
      };
    }

    if (grade === 'B+' || grade === 'B') {
      return {
        action: 'BUY',
        confidence: 'MODERATE',
        message: `Decent setup — watch ${weakFactors.join(', ') || 'nothing specific'}`,
        strengths: strongFactors,
        weaknesses: weakFactors,
        suggestedSizeMultiplier: grade === 'B+' ? 0.8 : 0.6,
      };
    }

    if (grade === 'C') {
      return {
        action: 'MARGINAL',
        confidence: 'LOW',
        message: 'Only take if nothing better available tonight',
        strengths: strongFactors,
        weaknesses: weakFactors,
        suggestedSizeMultiplier: 0.4,
      };
    }

    return {
      action: 'SKIP',
      confidence: 'NONE',
      message: 'Does not meet quality bar',
      weaknesses: weakFactors,
      suggestedSizeMultiplier: 0,
    };
  }
}

function mapRange(val, inMin, inMax, outMin, outMax) {
  return ((val - inMin) / (inMax - inMin)) * (outMax - outMin) + outMin;
}

function clamp(val, min = 0, max = 100) {
  return Math.min(max, Math.max(min, val));
}

module.exports = { BTSTScorer };
```

```javascript
const ind = require('./btst-indicators');
const { BTSTScanner } = require('./btst-scanner');

/**
 * Walk-forward backtester for BTST strategy
 * Simulates daily scan → entry → overnight → exit
 */
class BTSTBacktester {
  constructor(config = {}) {
    this.maxPositions = config.maxPositions ?? 3;
    this.stopLossATRMultiple = config.stopLossATRMultiple ?? 1.5;
    this.fallbackStopPercent = config.fallbackStopPercent ?? 1.5;
    this.exitMode = config.exitMode ?? 'OPEN';  // OPEN | VWAP | BEST
    this.skipFridays = config.skipFridays ?? true;
    this.scannerConfig = config.scannerConfig ?? {};
    this.lookback = config.lookback ?? 30;
    this.capitalPerTrade = config.capitalPerTrade ?? 100000;
  }

  /**
   * @param {Object} stocksHistory - { 'SYMBOL': [{date,o,h,l,c,v}, ...], ... }
   * @param {Array} niftyHistory - [{date,o,h,l,c,v}, ...]
   * @param {Array} fnoList - ['TATAMOTORS', 'INFY', ...]
   * @param {Object} options - { startDate, endDate }
   */
  run(stocksHistory, niftyHistory, fnoList, options = {}) {
    const scanner = new BTSTScanner(this.scannerConfig);

    // index nifty by date
    const niftyMap = {};
    for (const c of niftyHistory) niftyMap[c.date] = c;
    const dates = niftyHistory.map(c => c.date);

    const startIdx = options.startDate
      ? Math.max(dates.findIndex(d => d >= options.startDate), this.lookback + 1)
      : this.lookback + 1;
    const endIdx = options.endDate
      ? dates.findIndex(d => d > options.endDate)
      : dates.length - 1;

    const trades = [];
    const dailyLog = [];

    for (let i = startIdx; i < endIdx; i++) {
      const scanDate = dates[i];
      const exitDate = dates[i + 1];
      if (!exitDate) break;

      // skip Fridays
      if (this.skipFridays && new Date(scanDate).getDay() === 5) continue;

      const niftyToday = niftyMap[scanDate];
      const niftyChange = ((niftyToday.c - niftyToday.o) / niftyToday.o) * 100;

      // build stock snapshots for this date
      const stockSnapshots = [];
      for (const [symbol, candles] of Object.entries(stocksHistory)) {
        const dateIdx = candles.findIndex(c => c.date === scanDate);
        if (dateIdx < this.lookback) continue;

        const history = candles.slice(dateIdx - this.lookback, dateIdx);
        const today = candles[dateIdx];

        stockSnapshots.push({
          symbol,
          sector: null,
          isFnO: fnoList.includes(symbol),
          dailyCandles: history,
          today: {
            open: today.o, high: today.h, low: today.l,
            close: today.c, volume: today.v,
            prevClose: history[history.length - 1].c,
          },
          intradayCandles: null, // not available in backtest typically
        });
      }

      const market = {
        niftyPositive: niftyChange > 0,
        niftyChangePercent: ind.r(niftyChange),
        niftyDailyCandles: niftyHistory.slice(
          Math.max(0, i - 60), i + 1
        ),
      };

      const { candidates, meta } = scanner.scan(stockSnapshots, market);
      const selected = candidates.slice(0, this.maxPositions);

      for (const candidate of selected) {
        const sym = candidate.symbol;
        const candles = stocksHistory[sym];
        const exitIdx = candles.findIndex(c => c.date === exitDate);
        if (exitIdx < 0) continue;

        const entryPrice = candidate.indicators.close;
        const exitCandle = candles[exitIdx];
        const atrVal = candidate.indicators.atr14;

        // calculate stop loss
        const stopLoss = atrVal
          ? entryPrice - atrVal * this.stopLossATRMultiple
          : entryPrice * (1 - this.fallbackStopPercent / 100);

        // simulate exit
        const exitResult = this._simulateExit(entryPrice, stopLoss, exitCandle);

        const pnl = exitResult.exitPrice - entryPrice;
        const pnlPercent = (pnl / entryPrice) * 100;
        const gapPercent = ((exitCandle.o - entryPrice) / entryPrice) * 100;

        trades.push({
          entryDate: scanDate,
          exitDate,
          symbol: sym,
          entryPrice: ind.r(entryPrice),
          exitPrice: ind.r(exitResult.exitPrice),
          exitReason: exitResult.reason,
          pnl: ind.r(pnl),
          pnlPercent: ind.r(pnlPercent),
          gapPercent: ind.r(gapPercent),
          result: pnl > 0 ? 'WIN' : pnl < 0 ? 'LOSS' : 'FLAT',
          score: candidate.compositeScore,
          stopLoss: ind.r(stopLoss),
          atr: atrVal,
        });
      }

      dailyLog.push({
        date: scanDate,
        niftyChange: market.niftyChangePercent,
        scanned: stockSnapshots.length,
        passed: candidates.length,
        taken: selected.length,
      });
    }

    return {
      trades,
      dailyLog,
      summary: this._summary(trades),
      drawdown: this._drawdownAnalysis(trades),
    };
  }

  _simulateExit(entryPrice, stopLoss, exitCandle) {
    const openPrice = exitCandle.o;

    // Gap below stop → exit at open (slippage)
    if (openPrice <= stopLoss) {
      return { exitPrice: openPrice, reason: 'GAP_BELOW_STOP' };
    }

    // Intraday stop hit
    if (exitCandle.l <= stopLoss) {
      return { exitPrice: stopLoss, reason: 'INTRADAY_STOP' };
    }

    // Normal exit
    if (this.exitMode === 'OPEN') {
      return { exitPrice: openPrice, reason: 'EXIT_AT_OPEN' };
    }

    if (this.exitMode === 'VWAP') {
      // approximate VWAP as (O+H+L+C)/4
      const approxVwap = (exitCandle.o + exitCandle.h + exitCandle.l + exitCandle.c) / 4;
      return { exitPrice: Math.min(approxVwap, exitCandle.h), reason: 'EXIT_AT_VWAP_APPROX' };
    }

    if (this.exitMode === 'BEST') {
      // optimistic: exit at day high (upper bound)
      return { exitPrice: exitCandle.h, reason: 'EXIT_AT_HIGH_OPTIMISTIC' };
    }

    return { exitPrice: openPrice, reason: 'EXIT_AT_OPEN' };
  }

  _summary(trades) {
    if (trades.length === 0) return { totalTrades: 0 };

    const wins = trades.filter(t => t.result === 'WIN');
    const losses = trades.filter(t => t.result === 'LOSS');

    const totalPnl = trades.reduce((s, t) => s + t.pnlPercent, 0);
    const avgPnl = totalPnl / trades.length;
    const avgWin = wins.length ? wins.reduce((s, t) => s + t.pnlPercent, 0) / wins.length : 0;
    const avgLoss = losses.length ? losses.reduce((s, t) => s + t.pnlPercent, 0) / losses.length : 0;
    const winRate = (wins.length / trades.length) * 100;

    // consecutive losses
    let maxStreak = 0, streak = 0;
    for (const t of trades) {
      if (t.result === 'LOSS') { streak++; maxStreak = Math.max(maxStreak, streak); }
      else streak = 0;
    }

    // profit factor
    const grossProfit = wins.reduce((s, t) => s + t.pnlPercent, 0);
    const grossLoss = Math.abs(losses.reduce((s, t) => s + t.pnlPercent, 0));
    const profitFactor = grossLoss > 0 ? ind.r(grossProfit / grossLoss) : Infinity;

    // gap analysis
    const avgGap = trades.reduce((s, t) => s + t.gapPercent, 0) / trades.length;
    const posGaps = trades.filter(t => t.gapPercent > 0.1).length;

    // expectancy in ₹ (per trade per 1L invested)
    const expectancyPerLakh = ind.r(avgPnl * 1000); // 1% of 1L = ₹1000

    // monthly breakdown
    const monthly = {};
    for (const t of trades) {
      const m = t.entryDate.substring(0, 7);
      if (!monthly[m]) monthly[m] = { trades: 0, pnl: 0, wins: 0 };
      monthly[m].trades++;
      monthly[m].pnl += t.pnlPercent;
      if (t.result === 'WIN') monthly[m].wins++;
    }

    // by score tier
    const byGrade = { high: { trades: 0, pnl: 0 }, low: { trades: 0, pnl: 0 } };
    for (const t of trades) {
      const tier = t.score >= 60 ? 'high' : 'low';
      byGrade[tier].trades++;
      byGrade[tier].pnl += t.pnlPercent;
    }

    return {
      totalTrades: trades.length,
      wins: wins.length,
      losses: losses.length,
      winRate: ind.r(winRate),
      totalPnlPercent: ind.r(totalPnl),
      avgPnlPercent: ind.r(avgPnl),
      avgWinPercent: ind.r(avgWin),
      avgLossPercent: ind.r(avgLoss),
      profitFactor,
      expectancyPerLakh,
      maxConsecutiveLosses: maxStreak,
      bestTrade: ind.r(Math.max(...trades.map(t => t.pnlPercent))),
      worstTrade: ind.r(Math.min(...trades.map(t => t.pnlPercent))),
      avgGapPercent: ind.r(avgGap),
      positiveGapRate: ind.r((posGaps / trades.length) * 100),
      monthlyBreakdown: Object.entries(monthly)
        .map(([month, d]) => ({
          month,
          trades: d.trades,
          pnl: ind.r(d.pnl),
          winRate: ind.r((d.wins / d.trades) * 100),
        }))
        .sort((a, b) => a.month.localeCompare(b.month)),
      byScoreTier: {
        highScore: {
          trades: byGrade.high.trades,
          avgPnl: byGrade.high.trades ? ind.r(byGrade.high.pnl / byGrade.high.trades) : 0,
        },
        lowScore: {
          trades: byGrade.low.trades,
          avgPnl: byGrade.low.trades ? ind.r(byGrade.low.pnl / byGrade.low.trades) : 0,
        },
      },
    };
  }

  _drawdownAnalysis(trades) {
    if (trades.length === 0) return null;

    let cumulative = 0;
    let peak = 0;
    let maxDrawdown = 0;
    let currentDrawdown = 0;
    const equityCurve = [];

    for (const t of trades) {
      cumulative += t.pnlPercent;
      peak = Math.max(peak, cumulative);
      currentDrawdown = peak - cumulative;
      maxDrawdown = Math.max(maxDrawdown, currentDrawdown);

      equityCurve.push({
        date: t.entryDate,
        cumPnl: ind.r(cumulative),
        peak: ind.r(peak),
        drawdown: ind.r(currentDrawdown),
      });
    }

    // recovery analysis
    let longestRecovery = 0, currentRecoveryLen = 0;
    for (const point of equityCurve) {
      if (point.drawdown > 0) { currentRecoveryLen++; }
      else { longestRecovery = Math.max(longestRecovery, currentRecoveryLen); currentRecoveryLen = 0; }
    }
    longestRecovery = Math.max(longestRecovery, currentRecoveryLen);

    return {
      maxDrawdownPercent: ind.r(maxDrawdown),
      longestRecoveryTrades: longestRecovery,
      finalCumPnl: ind.r(cumulative),
      equityCurve,
    };
  }
}

module.exports = { BTSTBacktester };
```

```javascript
const { BTSTScanner, STBTScanner } = require('./btst-scanner');
const { BTSTSignalEngine } = require('./btst-signals');
const { BTSTRiskManager } = require('./btst-risk');
const { BTSTScorer } = require('./btst-scoring');
const ind = require('./btst-indicators');

// ═══════════════════════════════════════════════
//  BTST DAILY WORKFLOW (run at 2:30 PM & 3:00 PM)
// ═══════════════════════════════════════════════

function runBTSTWorkflow(stocksData, marketContext, events = []) {

  // ── STEP 1: Initialize ──
  const scanner   = new BTSTScanner();
  const stbtScan  = new STBTScanner();
  const signals   = new BTSTSignalEngine();
  const risk      = new BTSTRiskManager({ totalCapital: 500000 });
  const scorer    = new BTSTScorer();

  // ── STEP 2: Pre-trade risk gate ──
  const gate = risk.preTradeGate(marketContext);

  console.log(`\n${'═'.repeat(60)}`);
  console.log(`  BTST WORKFLOW — ${new Date().toLocaleDateString('en-IN')}`);
  console.log(`${'═'.repeat(60)}`);
  console.log(`  Risk Gate: ${gate.allowed ? '✅ CLEAR' : '🚫 BLOCKED'}`);

  if (!gate.allowed) {
    gate.blocks.forEach(b => console.log(`  ❌ ${b.rule}: ${b.msg}`));
    return { status: 'BLOCKED', blocks: gate.blocks };
  }

  gate.warnings.forEach(w => console.log(`  ⚠️  ${w.rule}: ${w.msg}`));
  console.log(`  Size multiplier: ${gate.sizeMultiplier}x`);

  // ── STEP 3: Run BTST scan ──
  const { candidates, meta } = scanner.scan(stocksData, marketContext);
  console.log(`\n  📊 Scanned ${meta.scanned} stocks → ${meta.passed} passed all criteria`);

  if (candidates.length === 0) {
    // try STBT if market is bearish
    if (marketContext.niftyChangePercent < 0) {
      console.log(`\n  Nifty negative — trying STBT scan...`);
      const stbt = stbtScan.scan(stocksData, marketContext);
      if (stbt.candidates.length > 0) {
        console.log(`  Found ${stbt.candidates.length} STBT candidates`);
        // (process STBT candidates similarly)
      }
    }
    return { status: 'NO_CANDIDATES', scanned: meta.scanned };
  }

  // ── STEP 4: Score and rank ──
  const sectorData = marketContext.sectorPerformance || {};
  const ranked = scorer.rank(candidates, sectorData);

  console.log(`\n  🏆 RANKED CANDIDATES:`);
  console.log(`  ${'─'.repeat(56)}`);

  ranked.slice(0, 8).forEach(r => {
    const rec = r.scoring.recommendation;
    const icon = rec.action === 'STRONG_BUY' ? '🟢' :
                 rec.action === 'BUY' ? '🔵' :
                 rec.action === 'MARGINAL' ? '🟡' : '⚪';
    console.log(`  ${icon} #${r.rank} ${r.symbol.padEnd(15)} `
      + `Score: ${String(r.scoring.totalScore).padStart(5)} `
      + `Grade: ${r.scoring.grade.padEnd(3)} `
      + `| Day: +${r.indicators.dayChange}% `
      + `| Vol: ${r.indicators.volumeRatio}x `
      + `| RSI: ${r.indicators.rsi14}`);
  });

  // ── STEP 5: Generate entries for top candidates ──
  const qualifiedGrades = new Set(['A+', 'A', 'B+', 'B']);
  const actionable = ranked.filter(r => qualifiedGrades.has(r.scoring.grade));
  const toTrade = actionable.slice(0, risk.maxPositions);

  console.log(`\n  📋 ENTRIES (${toTrade.length}):`);
  console.log(`  ${'─'.repeat(56)}`);

  const portfolio = [];
  let totalExposure = 0;

  for (const candidate of toTrade) {
    // per-stock event check
    const stockCheck = risk.stockSpecificCheck(candidate.symbol, events);
    if (!stockCheck.allowed) {
      console.log(`  ⏭️  ${candidate.symbol} skipped: ${stockCheck.blocks[0].msg}`);
      continue;
    }

    // position sizing
    const sizeMulti = gate.sizeMultiplier * (candidate.scoring.recommendation.suggestedSizeMultiplier ?? 1);
    const position = risk.calculatePosition(
      candidate.indicators.close,
      candidate.indicators.atr14,
      1.5,                       // ATR stop multiple
      portfolio.length,          // existing positions
      totalExposure,
      sizeMulti,
      candidate.indicators.beta
    );

    if (!position.allowed) {
      console.log(`  ⏭️  ${candidate.symbol} skipped: ${position.reason}`);
      continue;
    }

    // generate entry signal
    const entry = signals.generateEntry(candidate, position);

    portfolio.push(entry);
    totalExposure += position.investmentAmount;

    console.log(`\n  ✅ ${entry.action} ${entry.symbol}`);
    console.log(`     Price: ₹${entry.entryPrice} × ${position.quantity} = ₹${position.investmentAmount}`);
    console.log(`     SL: ₹${entry.stopLoss} (${entry.stopLossPercent}%) | Target: ₹${entry.target} (${entry.targetPercent}%)`);
    console.log(`     R:R = 1:${entry.riskRewardRatio} | Max Loss: ₹${position.maxLoss}`);
    console.log(`     ATR: ${entry.atrUsed} | Sized by: ${position.limitingFactor}`);
  }

  // ── STEP 6: Portfolio risk summary ──
  const portRisk = risk.portfolioRisk(portfolio.map(e => ({
    symbol: e.symbol, sector: e.sector,
    quantity: e.quantity, entryPrice: e.entryPrice,
    maxLoss: e.maxLoss, beta: e._meta?.beta,
  })));

  console.log(`\n  📊 PORTFOLIO RISK:`);
  console.log(`     Positions: ${portRisk.positionCount}`);
  console.log(`     Exposure: ₹${portRisk.totalExposure} (${portRisk.exposurePercent}% of capital)`);
  console.log(`     Worst-case loss: ₹${portRisk.totalMaxLoss} (${portRisk.worstCaseDrawdown}%)`);
  console.log(`     Risk rating: ${portRisk.riskRating}`);
  if (portRisk.sectorConcentrationWarning) {
    console.log(`     ⚠️  Sector concentration detected!`);
  }

  return {
    status: 'ENTRIES_GENERATED',
    portfolio,
    riskSummary: portRisk,
    allRanked: ranked,
  };
}

// ═══════════════════════════════════════════════
//  NEXT MORNING WORKFLOW (run at 8:45 AM & 9:15 AM)
// ═══════════════════════════════════════════════

function runMorningExitWorkflow(portfolio, preMarketData, openingData) {
  const signals = new BTSTSignalEngine();

  console.log(`\n${'═'.repeat(60)}`);
  console.log(`  BTST EXIT MORNING — ${new Date().toLocaleDateString('en-IN')}`);
  console.log(`${'═'.repeat(60)}`);

  // ── PRE-MARKET (8:45 AM) ──
  console.log(`\n  🌍 PRE-MARKET ASSESSMENT:`);

  const assessments = [];
  for (const entry of portfolio) {
    const assessment = signals.preMarketAssessment(entry, preMarketData);
    assessments.push(assessment);

    const icon = assessment.outlook === 'FAVORABLE' ? '🟢' :
                 assessment.outlook === 'ADVERSE' ? '🔴' : '🟡';
    console.log(`  ${icon} ${entry.symbol}: ${assessment.outlook} `
      + `(expected gap: ${assessment.expectedGap}%)`);
    console.log(`     ${assessment.plan}`);
  }

  // ── AT OPEN (9:15 AM) ──
  if (openingData) {
    console.log(`\n  📈 EXIT DECISIONS (9:15 AM):`);
    console.log(`  ${'─'.repeat(56)}`);

    for (const entry of portfolio) {
      const openData = openingData[entry.symbol];
      if (!openData) continue;

      const exit = signals.exitDecision(entry, openData);

      const icon = exit.scenario === 'FAVORABLE_GAP' ? '💰' :
                   exit.scenario === 'FLAT_OPEN' ? '⏳' :
                   exit.scenario.includes('ADVERSE') ? '🛑' : '❓';

      console.log(`\n  ${icon} ${entry.symbol}: ${exit.scenario} (gap: ${exit.gapPercent}%)`);
      console.log(`     Action: ${exit.action}`);

      if (exit.plan.exitPrice) {
        console.log(`     Exit at: ₹${exit.plan.exitPrice}`);
      }
      if (exit.plan.option1) {
        console.log(`     Option 1: ${exit.plan.option1.label} → ₹${exit.plan.option1.exitPrice}`);
        console.log(`     Option 2: ${exit.plan.option2.label} → trail at ₹${exit.plan.option2.trailingStop}`);
        console.log(`     ✓ Recommended: ${exit.recommended}`);
      }
      if (exit.plan.rules) {
        exit.plan.rules.forEach(r => console.log(`     • ${r}`));
      }
    }
  }

  return { assessments };
}
```
