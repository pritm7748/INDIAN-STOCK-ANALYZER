## `vwap-filters.js`

```javascript
/**
 * VWAP Pre-Trade Filters
 * 
 * Handles:
 *  - Time window validation
 *  - VWAP zone proximity detection
 *  - Candle pattern recognition (rejection, hammer, doji)
 *  - Volume analysis (decay on pullback, surge on breakout)
 *  - EMA trend confirmation
 *  - PDC confluence bias
 */

import { CONFIG, DIRECTION, BIAS, VWAP_SETUP } from './vwap-config.js';

// ════════════════════════════════════════════════
// TIME HELPERS
// ════════════════════════════════════════════════

export function timeOfDay(date) {
  const d = date instanceof Date ? date : new Date(date);
  return { h: d.getHours(), m: d.getMinutes() };
}

export function toMinutes(h, m) {
  return h * 60 + m;
}

export function candleTime(candle) {
  const d = candle.time instanceof Date ? candle.time : new Date(candle.time);
  return toMinutes(d.getHours(), d.getMinutes());
}

/**
 * Is this candle within the tradeable window?
 * - Not before 9:30 (first 15 min excluded)
 * - Not after 14:30 (last entry cutoff)
 */
export function isInTradeWindow(candle) {
  const t = candleTime(candle);
  const earliest = toMinutes(CONFIG.NO_TRADE_BEFORE_H, CONFIG.NO_TRADE_BEFORE_M);
  const latest = toMinutes(CONFIG.LAST_ENTRY_H, CONFIG.LAST_ENTRY_M);
  return t >= earliest && t <= latest;
}

/**
 * Is VWAP reliable yet? (after 10:00 AM)
 */
export function isVWAPReliable(candle) {
  const t = candleTime(candle);
  return t >= toMinutes(CONFIG.VWAP_RELIABLE_H, CONFIG.VWAP_RELIABLE_M);
}

/**
 * Is candle within pullback sweet-spot window? (10:00 - 12:30)
 */
export function isInPullbackWindow(candle) {
  const t = candleTime(candle);
  const start = toMinutes(CONFIG.PULLBACK_WINDOW_START_H, CONFIG.PULLBACK_WINDOW_START_M);
  const end = toMinutes(CONFIG.PULLBACK_WINDOW_END_H, CONFIG.PULLBACK_WINDOW_END_M);
  return t >= start && t <= end;
}

/**
 * Is candle past afternoon cutoff? (2:30 PM)
 */
export function isPastAfternoonCutoff(candle) {
  const t = candleTime(candle);
  return t >= toMinutes(CONFIG.LAST_ENTRY_H, CONFIG.LAST_ENTRY_M);
}

/**
 * Is candle at or past MIS exit time? (3:15 PM)
 */
export function isPastMISExit(candle) {
  const t = candleTime(candle);
  return t >= toMinutes(CONFIG.MIS_EXIT_H, CONFIG.MIS_EXIT_M);
}


// ════════════════════════════════════════════════
// VWAP PROXIMITY & ZONE
// ════════════════════════════════════════════════

/**
 * Is price within the VWAP "zone"?
 * VWAP is not a precise line — treat it as ±tolerance band.
 *
 * @param {number} price
 * @param {number} vwap
 * @returns {{ inZone: boolean, distance: number, distancePct: number, side: string }}
 */
export function vwapProximity(price, vwap) {
  const distance = price - vwap;
  const distancePct = Math.abs(distance) / vwap;
  const inZone = distancePct <= CONFIG.VWAP_TOUCH_TOLERANCE;
  const touching = distancePct <= CONFIG.VWAP_PROXIMITY_ZONE;
  const side = distance > 0 ? 'ABOVE' : distance < 0 ? 'BELOW' : 'AT';

  return { inZone, touching, distance, distancePct, side };
}

/**
 * Determine which VWAP band zone the price is in.
 * Zones: below -2SD, -2SD to -1SD, -1SD to VWAP, VWAP to +1SD, +1SD to +2SD, above +2SD
 */
export function vwapBandZone(price, vwap, sd1Upper, sd1Lower, sd2Upper, sd2Lower) {
  if (price >= sd2Upper) return 'ABOVE_2SD';
  if (price >= sd1Upper) return 'BETWEEN_1SD_2SD_UPPER';
  if (price >= vwap) return 'BETWEEN_VWAP_1SD_UPPER';
  if (price >= sd1Lower) return 'BETWEEN_VWAP_1SD_LOWER';
  if (price >= sd2Lower) return 'BETWEEN_1SD_2SD_LOWER';
  return 'BELOW_2SD';
}


// ════════════════════════════════════════════════
// CANDLE PATTERN DETECTION
// ════════════════════════════════════════════════

/**
 * Detect rejection / bounce candle patterns at VWAP.
 *
 * For LONG pullback:
 *   - Lower wick touches/pierces VWAP
 *   - Closes above VWAP
 *   - Lower wick is large relative to body (rejection)
 *
 * For SHORT pullback:
 *   - Upper wick touches/pierces VWAP
 *   - Closes below VWAP
 *   - Upper wick is large relative to body
 *
 * @param {object} candle  – { open, high, low, close }
 * @param {number} vwap
 * @param {string} direction – DIRECTION.LONG | SHORT
 * @returns {{ isRejection: boolean, pattern: string, details: object }}
 */
export function detectRejectionAtVWAP(candle, vwap, direction) {
  const { open, high, low, close } = candle;
  const body = Math.abs(close - open);
  const range = high - low;
  const isBullish = close > open;
  const isBearish = close < open;

  if (range === 0) {
    return { isRejection: false, pattern: 'DOJI_FLAT', details: {} };
  }

  const upperWick = high - Math.max(open, close);
  const lowerWick = Math.min(open, close) - low;
  const bodyPct = body / range;
  const lowerWickPct = lowerWick / range;
  const upperWickPct = upperWick / range;

  const details = { body, range, upperWick, lowerWick, bodyPct, lowerWickPct, upperWickPct };

  if (direction === DIRECTION.LONG) {
    // Need: wick dips to/below VWAP, candle closes above, long lower wick
    const wickTouchesVWAP = low <= vwap * (1 + CONFIG.VWAP_TOUCH_TOLERANCE);
    const closesAbove = close > vwap;
    const strongLowerWick = lowerWickPct >= CONFIG.PULLBACK_REJECTION_WICK_RATIO;

    // Specific patterns
    if (wickTouchesVWAP && closesAbove && strongLowerWick && isBullish) {
      return { isRejection: true, pattern: 'BULLISH_HAMMER', details };
    }
    if (wickTouchesVWAP && closesAbove && bodyPct < 0.1) {
      return { isRejection: true, pattern: 'DOJI_AT_VWAP', details };
    }
    if (wickTouchesVWAP && closesAbove && isBullish) {
      return { isRejection: true, pattern: 'BULLISH_REJECTION', details };
    }

    return { isRejection: false, pattern: 'NONE', details };
  }

  if (direction === DIRECTION.SHORT) {
    // Mirror: wick pokes above VWAP, closes below, strong upper wick
    const wickTouchesVWAP = high >= vwap * (1 - CONFIG.VWAP_TOUCH_TOLERANCE);
    const closesBelow = close < vwap;
    const strongUpperWick = upperWickPct >= CONFIG.PULLBACK_REJECTION_WICK_RATIO;

    if (wickTouchesVWAP && closesBelow && strongUpperWick && isBearish) {
      return { isRejection: true, pattern: 'BEARISH_SHOOTING_STAR', details };
    }
    if (wickTouchesVWAP && closesBelow && bodyPct < 0.1) {
      return { isRejection: true, pattern: 'DOJI_AT_VWAP', details };
    }
    if (wickTouchesVWAP && closesBelow && isBearish) {
      return { isRejection: true, pattern: 'BEARISH_REJECTION', details };
    }

    return { isRejection: false, pattern: 'NONE', details };
  }

  return { isRejection: false, pattern: 'NONE', details };
}


// ════════════════════════════════════════════════
// VOLUME FILTERS
// ════════════════════════════════════════════════

/**
 * Pullback volume decay filter.
 * During pullback toward VWAP, volume should DECREASE
 * compared to the prior rally/trend move.
 *
 * @param {number[]} rallyVolumes    – volumes during the trending leg
 * @param {number[]} pullbackVolumes – volumes during the pullback leg
 * @returns {{ pass: boolean, rallyAvg: number, pullbackAvg: number, ratio: number }}
 */
export function filterPullbackVolumeDecay(rallyVolumes, pullbackVolumes) {
  if (rallyVolumes.length === 0 || pullbackVolumes.length === 0) {
    return { pass: false, reason: 'Insufficient volume data' };
  }

  const rallyAvg = rallyVolumes.reduce((a, b) => a + b, 0) / rallyVolumes.length;
  const pullbackAvg = pullbackVolumes.reduce((a, b) => a + b, 0) / pullbackVolumes.length;

  const ratio = pullbackAvg / rallyAvg;
  const pass = ratio < CONFIG.PULLBACK_VOLUME_DECAY;

  return {
    pass,
    rallyAvg: Math.round(rallyAvg),
    pullbackAvg: Math.round(pullbackAvg),
    ratio: Math.round(ratio * 100) / 100,
    reason: pass ? null : `Pullback vol ratio ${ratio.toFixed(2)} >= threshold ${CONFIG.PULLBACK_VOLUME_DECAY}`,
  };
}

/**
 * Breakout volume surge filter.
 * The VWAP crossover candle must have volume > 1.5× recent average.
 *
 * @param {number} crossoverVolume
 * @param {number[]} recentVolumes – last N candles
 */
export function filterBreakoutVolumeSurge(crossoverVolume, recentVolumes) {
  const lookback = Math.min(recentVolumes.length, CONFIG.BREAKOUT_AVG_VOLUME_LOOKBACK);
  if (lookback === 0) return { pass: false, reason: 'No volume history' };

  const avgVol = recentVolumes.slice(-lookback).reduce((a, b) => a + b, 0) / lookback;
  const required = avgVol * CONFIG.BREAKOUT_VOLUME_MULTIPLE;
  const pass = crossoverVolume >= required;

  return {
    pass,
    crossoverVolume,
    avgVolume: Math.round(avgVol),
    multiple: Math.round((crossoverVolume / avgVol) * 100) / 100,
    reason: pass ? null : `Vol ${crossoverVolume} < required ${Math.round(required)} (${CONFIG.BREAKOUT_VOLUME_MULTIPLE}× avg)`,
  };
}


// ════════════════════════════════════════════════
// EMA TREND CONFIRMATION
// ════════════════════════════════════════════════

/**
 * For pullback LONG: EMA(9) must be ABOVE VWAP (trend intact).
 * For pullback SHORT: EMA(9) must be BELOW VWAP.
 *
 * @param {number} ema9
 * @param {number} vwap
 * @param {string} direction
 */
export function filterEMATrend(ema9, vwap, direction) {
  if (direction === DIRECTION.LONG && ema9 > vwap) {
    return { pass: true };
  }
  if (direction === DIRECTION.SHORT && ema9 < vwap) {
    return { pass: true };
  }
  return {
    pass: false,
    reason: `EMA9 (${ema9.toFixed(2)}) not ${direction === DIRECTION.LONG ? 'above' : 'below'} VWAP (${vwap.toFixed(2)})`,
  };
}


// ════════════════════════════════════════════════
// PREVIOUS DAY CLOSE (PDC) CONFLUENCE
// ════════════════════════════════════════════════

/**
 * Determine directional bias based on where price sits relative to
 * both VWAP and previous day's close.
 *
 * @param {number} price
 * @param {number} vwap
 * @param {number} prevDayClose
 * @returns {{ bias: string, reason: string }}
 */
export function determinePDCBias(price, vwap, prevDayClose) {
  const aboveVWAP = price > vwap;
  const abovePDC = price > prevDayClose;

  if (aboveVWAP && abovePDC) {
    return { bias: BIAS.BULLISH, reason: 'Price above both VWAP and PDC → LONG only' };
  }
  if (!aboveVWAP && !abovePDC) {
    return { bias: BIAS.BEARISH, reason: 'Price below both VWAP and PDC → SHORT only' };
  }
  return { bias: BIAS.NEUTRAL, reason: 'Price between VWAP and PDC → avoid or wait' };
}

/**
 * Check if a proposed trade direction aligns with PDC bias.
 */
export function filterPDCAlignment(direction, price, vwap, prevDayClose) {
  const { bias, reason } = determinePDCBias(price, vwap, prevDayClose);

  if (bias === BIAS.BULLISH && direction === DIRECTION.LONG) {
    return { pass: true, bias, reason };
  }
  if (bias === BIAS.BEARISH && direction === DIRECTION.SHORT) {
    return { pass: true, bias, reason };
  }
  if (bias === BIAS.NEUTRAL) {
    return { pass: false, bias, reason: 'PDC confluence neutral — skip' };
  }
  return { pass: false, bias, reason: `Direction ${direction} conflicts with bias ${bias}` };
}


// ════════════════════════════════════════════════
// TREND PERSISTENCE (how long above/below VWAP)
// ════════════════════════════════════════════════

/**
 * Count consecutive candles where price has been on one side of VWAP.
 * Used for:
 *   - Pullback: need 6+ candles above VWAP before a pullback is valid
 *   - Breakout: need 12+ candles below VWAP before a crossover counts
 *
 * @param {object[]} candles – recent candles leading up to current
 * @param {string}   side    – 'ABOVE' | 'BELOW'
 * @returns {number} consecutive candle count on that side
 */
export function countConsecutiveSide(candles, side) {
  let count = 0;

  // Walk backwards from most recent
  for (let i = candles.length - 1; i >= 0; i--) {
    const c = candles[i];
    const vwap = c.vwap;
    if (vwap == null) break;

    if (side === 'ABOVE' && c.close > vwap) {
      count++;
    } else if (side === 'BELOW' && c.close < vwap) {
      count++;
    } else {
      break; // streak broken
    }
  }

  return count;
}

/**
 * For breakout: count how long price was on the OPPOSITE side before the cross.
 * E.g., for a bullish breakout, how many candles was price below VWAP?
 *
 * @param {object[]} candles         – all candles up to (but not including) crossover
 * @param {string}   priorSide      – 'BELOW' for bullish breakout, 'ABOVE' for bearish
 * @returns {number}
 */
export function countPriorSideDuration(candles, priorSide) {
  let count = 0;

  for (let i = candles.length - 1; i >= 0; i--) {
    const c = candles[i];
    if (c.vwap == null) break;

    if (priorSide === 'BELOW' && c.close < c.vwap) {
      count++;
    } else if (priorSide === 'ABOVE' && c.close > c.vwap) {
      count++;
    } else {
      break;
    }
  }

  return count;
}


// ════════════════════════════════════════════════
// COMPOSITE FILTER: PULLBACK ENTRY
// ════════════════════════════════════════════════

/**
 * @param {object} ctx
 *   { candle, vwap, ema9, direction, precedingCandles,
 *     rallyVolumes, pullbackVolumes, prevDayClose?, usePDCFilter }
 */
export function filterPullbackEntry(ctx) {
  const results = [];

  // 1. Time window
  results.push({
    name: 'TimeWindow',
    pass: isInPullbackWindow(ctx.candle),
    reason: isInPullbackWindow(ctx.candle) ? null : 'Outside pullback window (10:00-12:30)',
  });

  // 2. VWAP reliable
  results.push({
    name: 'VWAPReliable',
    pass: isVWAPReliable(ctx.candle),
    reason: isVWAPReliable(ctx.candle) ? null : 'VWAP not yet reliable (before 10:00)',
  });

  // 3. Rejection candle at VWAP
  const rejection = detectRejectionAtVWAP(ctx.candle, ctx.vwap, ctx.direction);
  results.push({
    name: 'Rejection',
    pass: rejection.isRejection,
    pattern: rejection.pattern,
    reason: rejection.isRejection ? null : 'No rejection pattern at VWAP',
  });

  // 4. Volume decay on pullback
  const volDecay = filterPullbackVolumeDecay(ctx.rallyVolumes, ctx.pullbackVolumes);
  results.push({ name: 'VolumeDecay', ...volDecay });

  // 5. EMA(9) trend confirmation
  const emaTrend = filterEMATrend(ctx.ema9, ctx.vwap, ctx.direction);
  results.push({ name: 'EMATrend', ...emaTrend });

  // 6. Trend persistence (at least 6 candles on trend side before pullback)
  const trendSide = ctx.direction === DIRECTION.LONG ? 'ABOVE' : 'BELOW';
  // Check preceding candles BEFORE the pullback started
  const trendDuration = ctx.trendDurationCandles || 0;
  results.push({
    name: 'TrendPersistence',
    pass: trendDuration >= CONFIG.PULLBACK_MIN_TREND_CANDLES,
    duration: trendDuration,
    reason: trendDuration >= CONFIG.PULLBACK_MIN_TREND_CANDLES
      ? null
      : `Only ${trendDuration} candles on ${trendSide} side (need ${CONFIG.PULLBACK_MIN_TREND_CANDLES})`,
  });

  // 7. PDC confluence (optional)
  if (ctx.usePDCFilter && ctx.prevDayClose != null) {
    const pdc = filterPDCAlignment(ctx.direction, ctx.candle.close, ctx.vwap, ctx.prevDayClose);
    results.push({ name: 'PDCConfluence', ...pdc });
  }

  const allPassed = results.every(r => r.pass);
  return { pass: allPassed, filters: results };
}


// ════════════════════════════════════════════════
// COMPOSITE FILTER: BREAKOUT ENTRY
// ════════════════════════════════════════════════

/**
 * @param {object} ctx
 *   { candle, vwap, direction, precedingCandles, recentVolumes,
 *     crossoverVolume, prevDayClose?, usePDCFilter }
 */
export function filterBreakoutEntry(ctx) {
  const results = [];

  // 1. Time window
  results.push({
    name: 'TimeWindow',
    pass: isInTradeWindow(ctx.candle),
    reason: isInTradeWindow(ctx.candle) ? null : 'Outside trade window',
  });

  // 2. VWAP reliable
  results.push({
    name: 'VWAPReliable',
    pass: isVWAPReliable(ctx.candle),
    reason: isVWAPReliable(ctx.candle) ? null : 'VWAP not yet reliable',
  });

  // 3. Prior duration on opposite side
  const priorSide = ctx.direction === DIRECTION.LONG ? 'BELOW' : 'ABOVE';
  const priorDuration = ctx.priorSideDuration || 0;
  results.push({
    name: 'PriorDuration',
    pass: priorDuration >= CONFIG.BREAKOUT_MIN_TIME_BELOW_CANDLES,
    duration: priorDuration,
    reason: priorDuration >= CONFIG.BREAKOUT_MIN_TIME_BELOW_CANDLES
      ? null
      : `Only ${priorDuration} candles on ${priorSide} side (need ${CONFIG.BREAKOUT_MIN_TIME_BELOW_CANDLES})`,
  });

  // 4. Volume surge on crossover
  const volSurge = filterBreakoutVolumeSurge(ctx.crossoverVolume, ctx.recentVolumes);
  results.push({ name: 'VolumeSurge', ...volSurge });

  // 5. PDC (optional)
  if (ctx.usePDCFilter && ctx.prevDayClose != null) {
    const pdc = filterPDCAlignment(ctx.direction, ctx.candle.close, ctx.vwap, ctx.prevDayClose);
    results.push({ name: 'PDCConfluence', ...pdc });
  }

  const allPassed = results.every(r => r.pass);
  return { pass: allPassed, filters: results };
}
```

## `vwap-core.js`

```javascript
/**
 * VWAP Core Engine
 *
 * 1. Compute VWAP + standard deviation bands
 * 2. Compute EMA(9) on 5-min candles
 * 3. Identify pullback & breakout setups
 * 4. Manage trades: SL, targets, trail, time stops
 *
 * Candle contract:
 * {
 *   time: Date | string,
 *   open: number,
 *   high: number,
 *   low: number,
 *   close: number,
 *   volume: number,
 *   // Computed by this engine:
 *   vwap?: number,
 *   vwapSD?: number,        // standard deviation at this point
 *   vwapUpper1?: number,    // VWAP + 1SD
 *   vwapLower1?: number,    // VWAP - 1SD
 *   vwapUpper2?: number,    // VWAP + 2SD
 *   vwapLower2?: number,    // VWAP - 2SD
 *   ema9?: number,
 * }
 */

import { CONFIG, DIRECTION, VWAP_SETUP, EXIT_REASON } from './vwap-config.js';
import {
  isInTradeWindow, isVWAPReliable, isPastMISExit, isPastAfternoonCutoff,
  candleTime, toMinutes, vwapProximity,
  countConsecutiveSide, countPriorSideDuration,
  filterPullbackEntry, filterBreakoutEntry,
  detectRejectionAtVWAP,
} from './vwap-filters.js';


// ═══════════════════════════════════════════════
// SECTION 1: VWAP + BANDS COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Compute VWAP with ±1 and ±2 standard deviation bands.
 * Mutates candles in-place (sets vwap, vwapSD, vwapUpper1, etc.)
 *
 * SD computation:
 *   variance = Σ(volume_i × (tp_i - VWAP)²) / Σ(volume_i)
 *   SD = sqrt(variance)
 *   Bands = VWAP ± (multiplier × SD)
 *
 * @param {object[]} candles – day's candles sorted by time
 */
export function computeVWAPWithBands(candles) {
  let cumTPV = 0;      // Σ (typical_price × volume)
  let cumVol = 0;      // Σ volume
  let cumTP2V = 0;     // Σ (typical_price² × volume) — for variance

  for (const c of candles) {
    const tp = (c.high + c.low + c.close) / 3;

    cumTPV += tp * c.volume;
    cumVol += c.volume;
    cumTP2V += tp * tp * c.volume;

    if (cumVol === 0) {
      c.vwap = c.close;
      c.vwapSD = 0;
      c.vwapUpper1 = c.close;
      c.vwapLower1 = c.close;
      c.vwapUpper2 = c.close;
      c.vwapLower2 = c.close;
      continue;
    }

    const vwap = cumTPV / cumVol;

    // Variance = E[X²] - (E[X])²  (volume-weighted)
    const variance = (cumTP2V / cumVol) - (vwap * vwap);
    const sd = Math.sqrt(Math.max(0, variance));

    c.vwap = vwap;
    c.vwapSD = sd;
    c.vwapUpper1 = vwap + CONFIG.BAND_SD_1 * sd;
    c.vwapLower1 = vwap - CONFIG.BAND_SD_1 * sd;
    c.vwapUpper2 = vwap + CONFIG.BAND_SD_2 * sd;
    c.vwapLower2 = vwap - CONFIG.BAND_SD_2 * sd;
  }
}


// ═══════════════════════════════════════════════
// SECTION 2: EMA COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Compute EMA on close prices. Mutates candles (sets candle.ema9).
 *
 * @param {object[]} candles
 * @param {number}   period  – default CONFIG.EMA_PERIOD (9)
 */
export function computeEMA(candles, period = CONFIG.EMA_PERIOD) {
  if (candles.length === 0) return;

  const k = 2 / (period + 1);
  candles[0].ema9 = candles[0].close;

  for (let i = 1; i < candles.length; i++) {
    const prev = candles[i - 1].ema9 ?? candles[i - 1].close;
    candles[i].ema9 = candles[i].close * k + prev * (1 - k);
  }
}


// ═══════════════════════════════════════════════
// SECTION 3: DAY HIGH/LOW TRACKER
// ═══════════════════════════════════════════════

/**
 * Compute running day high and low up to each candle.
 * Mutates candles (sets candle.dayHigh, candle.dayLow).
 */
export function computeDayHighLow(candles) {
  let dayHigh = -Infinity;
  let dayLow = Infinity;

  for (const c of candles) {
    if (c.high > dayHigh) dayHigh = c.high;
    if (c.low < dayLow) dayLow = c.low;
    c.dayHigh = dayHigh;
    c.dayLow = dayLow;
  }
}


// ═══════════════════════════════════════════════
// SECTION 4: PULLBACK DETECTION STATE MACHINE
// ═══════════════════════════════════════════════

/**
 * Walk through candles and identify pullback-to-VWAP opportunities.
 *
 * State machine:
 *   IDLE → TRENDING → PULLING_BACK → SIGNAL (or back to TRENDING)
 *
 * @param {object[]} candles    – full day, pre-computed (VWAP, EMA, dayHigh/Low)
 * @param {object}   opts
 *   { prevDayClose, usePDCFilter }
 * @returns {object[]} – array of pullback signals found
 */
export function detectPullbackSignals(candles, opts = {}) {
  const { prevDayClose = null, usePDCFilter = false } = opts;
  const signals = [];

  // We scan for both LONG and SHORT pullbacks
  for (const direction of [DIRECTION.LONG, DIRECTION.SHORT]) {
    const trendSide = direction === DIRECTION.LONG ? 'ABOVE' : 'BELOW';
    
    let state = 'IDLE';
    let trendStartIdx = -1;
    let pullbackStartIdx = -1;
    let rallyVolumes = [];

    for (let i = 1; i < candles.length; i++) {
      const c = candles[i];
      const prev = candles[i - 1];

      if (!isInTradeWindow(c) || !isVWAPReliable(c)) continue;
      if (c.vwap == null) continue;

      const isOnTrendSide = direction === DIRECTION.LONG
        ? c.close > c.vwap
        : c.close < c.vwap;

      const proximity = vwapProximity(c.close, c.vwap);

      switch (state) {
        case 'IDLE':
          if (isOnTrendSide) {
            state = 'TRENDING';
            trendStartIdx = i;
            rallyVolumes = [c.volume];
          }
          break;

        case 'TRENDING':
          if (isOnTrendSide) {
            rallyVolumes.push(c.volume);

            // Check if we have enough trend duration
            const trendLen = i - trendStartIdx + 1;
            if (trendLen < CONFIG.PULLBACK_MIN_TREND_CANDLES) continue;
          }

          // Detect start of pullback: price moves toward VWAP
          if (proximity.inZone || !isOnTrendSide) {
            state = 'PULLING_BACK';
            pullbackStartIdx = i;
          }
          break;

        case 'PULLING_BACK': {
          // Collect pullback volumes
          const pullbackVolumes = [];
          for (let j = pullbackStartIdx; j <= i; j++) {
            pullbackVolumes.push(candles[j].volume);
          }

          // Check for rejection candle at VWAP
          const rejection = detectRejectionAtVWAP(c, c.vwap, direction);

          if (rejection.isRejection && proximity.inZone) {
            // Build filter context
            const trendDurationCandles = pullbackStartIdx - trendStartIdx;

            const filterCtx = {
              candle: c,
              vwap: c.vwap,
              ema9: c.ema9 ?? c.close,
              direction,
              rallyVolumes,
              pullbackVolumes,
              trendDurationCandles,
              prevDayClose,
              usePDCFilter,
            };

            const filterResult = filterPullbackEntry(filterCtx);

            if (filterResult.pass) {
              // Compute entry, SL, targets
              const entry = computePullbackEntry(c, direction, candles.slice(0, i + 1));

              signals.push({
                setup: VWAP_SETUP.PULLBACK,
                direction,
                entryPrice: entry.entryPrice,
                entryTime: c.time,
                entryIndex: i,
                sl: entry.sl,
                targets: entry.targets,
                rejectionPattern: rejection.pattern,
                filterResults: filterResult.filters,
                trendDuration: trendDurationCandles,
                pullbackCandles: i - pullbackStartIdx + 1,
              });

              // Reset — don't look for another signal in same direction immediately
              state = 'IDLE';
              trendStartIdx = -1;
              rallyVolumes = [];
              break;
            }
          }

          // If price moves back to trend side strongly, reset to TRENDING
          if (isOnTrendSide && !proximity.inZone) {
            state = 'TRENDING';
            pullbackStartIdx = -1;
          }

          // If price crosses through VWAP decisively, pullback failed
          const crossedThrough = direction === DIRECTION.LONG
            ? c.close < c.vwap * (1 - CONFIG.VWAP_TOUCH_TOLERANCE * 2)
            : c.close > c.vwap * (1 + CONFIG.VWAP_TOUCH_TOLERANCE * 2);

          if (crossedThrough) {
            state = 'IDLE';
            trendStartIdx = -1;
            rallyVolumes = [];
          }
          break;
        }
      }
    }
  }

  return signals;
}


// ═══════════════════════════════════════════════
// SECTION 5: BREAKOUT DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect VWAP crossover / breakdown signals.
 *
 * Logic:
 *   1. Price has been below (above) VWAP for 1+ hours
 *   2. A candle crosses above (below) VWAP with strong volume
 *   3. Next 2 candles confirm by closing on the new side
 *   4. Entry after confirmation candle #2
 *
 * @param {object[]} candles
 * @param {object}   opts  – { prevDayClose, usePDCFilter }
 * @returns {object[]} – breakout signals
 */
export function detectBreakoutSignals(candles, opts = {}) {
  const { prevDayClose = null, usePDCFilter = false } = opts;
  const signals = [];

  for (let i = 2; i < candles.length; i++) {
    const c = candles[i];
    if (!isInTradeWindow(c) || !isVWAPReliable(c)) continue;
    if (c.vwap == null) continue;

    // Check for bullish crossover
    if (c.close > c.vwap) {
      // Was previous candle below VWAP?
      const prev = candles[i - 1];
      if (prev.close < prev.vwap) {
        // This is a potential crossover candle
        const crossoverCandle = candles[i];

        // Check prior duration below VWAP
        const priorDuration = countPriorSideDuration(candles.slice(0, i), 'BELOW');

        if (priorDuration >= CONFIG.BREAKOUT_MIN_TIME_BELOW_CANDLES) {
          // Wait for confirmation: next 2 candles must also close above VWAP
          if (i + CONFIG.BREAKOUT_CONFIRMATION_CANDLES < candles.length) {
            let confirmed = true;
            for (let j = 1; j <= CONFIG.BREAKOUT_CONFIRMATION_CANDLES; j++) {
              if (candles[i + j].close <= candles[i + j].vwap) {
                confirmed = false;
                break;
              }
            }

            if (confirmed) {
              const confirmIdx = i + CONFIG.BREAKOUT_CONFIRMATION_CANDLES;
              const confirmCandle = candles[confirmIdx];
              const recentVolumes = candles.slice(Math.max(0, i - CONFIG.BREAKOUT_AVG_VOLUME_LOOKBACK), i).map(cc => cc.volume);

              const filterCtx = {
                candle: confirmCandle,
                vwap: confirmCandle.vwap,
                direction: DIRECTION.LONG,
                crossoverVolume: crossoverCandle.volume,
                recentVolumes,
                priorSideDuration: priorDuration,
                prevDayClose,
                usePDCFilter,
              };

              const filterResult = filterBreakoutEntry(filterCtx);

              if (filterResult.pass) {
                const entry = computeBreakoutEntry(confirmCandle, DIRECTION.LONG);
                signals.push({
                  setup: VWAP_SETUP.BREAKOUT,
                  direction: DIRECTION.LONG,
                  entryPrice: entry.entryPrice,
                  entryTime: confirmCandle.time,
                  entryIndex: confirmIdx,
                  sl: entry.sl,
                  targets: entry.targets,
                  crossoverTime: crossoverCandle.time,
                  priorDurationBelow: priorDuration,
                  volumeMultiple: filterResult.filters.find(f => f.name === 'VolumeSurge')?.multiple,
                  filterResults: filterResult.filters,
                });
              }
            }
          }
        }
      }
    }

    // Check for bearish breakdown (mirror)
    if (c.close < c.vwap) {
      const prev = candles[i - 1];
      if (prev.close > prev.vwap) {
        const crossoverCandle = candles[i];
        const priorDuration = countPriorSideDuration(candles.slice(0, i), 'ABOVE');

        if (priorDuration >= CONFIG.BREAKOUT_MIN_TIME_BELOW_CANDLES) {
          if (i + CONFIG.BREAKOUT_CONFIRMATION_CANDLES < candles.length) {
            let confirmed = true;
            for (let j = 1; j <= CONFIG.BREAKOUT_CONFIRMATION_CANDLES; j++) {
              if (candles[i + j].close >= candles[i + j].vwap) {
                confirmed = false;
                break;
              }
            }

            if (confirmed) {
              const confirmIdx = i + CONFIG.BREAKOUT_CONFIRMATION_CANDLES;
              const confirmCandle = candles[confirmIdx];
              const recentVolumes = candles.slice(Math.max(0, i - CONFIG.BREAKOUT_AVG_VOLUME_LOOKBACK), i).map(cc => cc.volume);

              const filterCtx = {
                candle: confirmCandle,
                vwap: confirmCandle.vwap,
                direction: DIRECTION.SHORT,
                crossoverVolume: crossoverCandle.volume,
                recentVolumes,
                priorSideDuration: priorDuration,
                prevDayClose,
                usePDCFilter,
              };

              const filterResult = filterBreakoutEntry(filterCtx);

              if (filterResult.pass) {
                const entry = computeBreakoutEntry(confirmCandle, DIRECTION.SHORT);
                signals.push({
                  setup: VWAP_SETUP.BREAKOUT,
                  direction: DIRECTION.SHORT,
                  entryPrice: entry.entryPrice,
                  entryTime: confirmCandle.time,
                  entryIndex: confirmIdx,
                  sl: entry.sl,
                  targets: entry.targets,
                  crossoverTime: crossoverCandle.time,
                  priorDurationAbove: priorDuration,
                  volumeMultiple: filterResult.filters.find(f => f.name === 'VolumeSurge')?.multiple,
                  filterResults: filterResult.filters,
                });
              }
            }
          }
        }
      }
    }
  }

  return signals;
}


// ═══════════════════════════════════════════════
// SECTION 6: ENTRY / SL / TARGET COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Compute entry, stop-loss, and targets for a PULLBACK setup.
 */
function computePullbackEntry(candle, direction, candlesUpTo) {
  const entryPrice = candle.close;
  const vwap = candle.vwap;
  const isLong = direction === DIRECTION.LONG;
  let sl;
  const targets = [];

  if (isLong) {
    // SL: below VWAP by 0.3% OR below pullback low (whichever tighter)
    const slBelowVWAP = vwap * (1 - CONFIG.PULLBACK_SL_BELOW_VWAP_PCT);
    const pullbackLow = candle.low;
    sl = Math.max(slBelowVWAP, pullbackLow - 1);

    // Enforce max SL
    const slPct = (entryPrice - sl) / entryPrice;
    if (slPct > CONFIG.MAX_SL_PCT) {
      sl = entryPrice * (1 - CONFIG.MAX_SL_PCT);
    }

    // Target 1: previous day high
    const dayHigh = candle.dayHigh || entryPrice * 1.01;
    targets.push({
      price: dayHigh,
      qtyPct: CONFIG.TARGET_1_QTY_PCT,
      label: 'T1_DAY_HIGH',
    });

    // Target 2: VWAP + 1SD band
    const sd1Target = candle.vwapUpper1 || entryPrice * 1.015;
    targets.push({
      price: Math.max(sd1Target, dayHigh),  // take the further one
      qtyPct: CONFIG.TRAIL_QTY_PCT,
      label: 'T2_VWAP_1SD',
    });

  } else {
    // SHORT
    const slAboveVWAP = vwap * (1 + CONFIG.PULLBACK_SL_BELOW_VWAP_PCT);
    const pullbackHigh = candle.high;
    sl = Math.min(slAboveVWAP, pullbackHigh + 1);

    const slPct = (sl - entryPrice) / entryPrice;
    if (slPct > CONFIG.MAX_SL_PCT) {
      sl = entryPrice * (1 + CONFIG.MAX_SL_PCT);
    }

    const dayLow = candle.dayLow || entryPrice * 0.99;
    targets.push({
      price: dayLow,
      qtyPct: CONFIG.TARGET_1_QTY_PCT,
      label: 'T1_DAY_LOW',
    });

    const sd1Target = candle.vwapLower1 || entryPrice * 0.985;
    targets.push({
      price: Math.min(sd1Target, dayLow),
      qtyPct: CONFIG.TRAIL_QTY_PCT,
      label: 'T2_VWAP_1SD',
    });
  }

  // Validate minimum R:R
  const risk = Math.abs(entryPrice - sl);
  const reward = Math.abs(targets[0].price - entryPrice);
  const rr = risk > 0 ? reward / risk : 0;

  return { entryPrice, sl, targets, riskReward: rr, meetsMinRR: rr >= CONFIG.PULLBACK_MIN_RR };
}

/**
 * Compute entry, stop-loss, and targets for a BREAKOUT setup.
 */
function computeBreakoutEntry(candle, direction) {
  const entryPrice = candle.close;
  const vwap = candle.vwap;
  const isLong = direction === DIRECTION.LONG;
  let sl;
  const targets = [];

  if (isLong) {
    sl = vwap * (1 - CONFIG.BREAKOUT_SL_BELOW_VWAP_PCT);

    const slPct = (entryPrice - sl) / entryPrice;
    if (slPct > CONFIG.MAX_SL_PCT) {
      sl = entryPrice * (1 - CONFIG.MAX_SL_PCT);
    }

    // Target: VWAP + 1SD, then trail
    targets.push({
      price: candle.vwapUpper1 || entryPrice * 1.01,
      qtyPct: CONFIG.TARGET_1_QTY_PCT,
      label: 'T1_VWAP_1SD',
    });
    targets.push({
      price: null, // trail
      qtyPct: CONFIG.TRAIL_QTY_PCT,
      label: 'TRAIL',
    });

  } else {
    sl = vwap * (1 + CONFIG.BREAKOUT_SL_BELOW_VWAP_PCT);

    const slPct = (sl - entryPrice) / entryPrice;
    if (slPct > CONFIG.MAX_SL_PCT) {
      sl = entryPrice * (1 + CONFIG.MAX_SL_PCT);
    }

    targets.push({
      price: candle.vwapLower1 || entryPrice * 0.99,
      qtyPct: CONFIG.TARGET_1_QTY_PCT,
      label: 'T1_VWAP_1SD',
    });
    targets.push({
      price: null,
      qtyPct: CONFIG.TRAIL_QTY_PCT,
      label: 'TRAIL',
    });
  }

  return { entryPrice, sl, targets };
}


// ═══════════════════════════════════════════════
// SECTION 7: TRADE SIMULATION
// ═══════════════════════════════════════════════

/**
 * Given a signal, walk forward through candles managing the trade.
 *
 * Key exits:
 *  - Stop-loss (fixed or trailed)
 *  - T1 hit → partial exit, SL moves to breakeven
 *  - Trail active → use EMA(9) as trailing stop
 *  - VWAP cross against → exit immediately
 *  - Time stop: 2:30 PM last new entry, 3:15 PM forced exit
 */
export function simulateTrade(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets } = signal;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  let currentSL = sl;
  const partialExits = [];
  let t1Hit = false;
  let trailActive = false;

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // ── EOD forced exit ──
    if (isPastMISExit(c)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_EOD,
      });
      remainingQty = 0;
      break;
    }

    // ── Afternoon cutoff (exit remaining if T1 not hit) ──
    if (!t1Hit && isPastAfternoonCutoff(c)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_AFTERNOON,
      });
      remainingQty = 0;
      break;
    }

    // ── Stop-loss ──
    const hitSL = isLong ? c.low <= currentSL : c.high >= currentSL;
    if (hitSL) {
      partialExits.push({
        time: c.time, price: currentSL,
        qtyPct: remainingQty,
        reason: trailActive ? EXIT_REASON.TRAIL_STOP : EXIT_REASON.STOP_LOSS,
      });
      remainingQty = 0;
      break;
    }

    // ── VWAP cross against: if price closes on wrong side of VWAP ──
    // (only if trailing, to avoid premature exits on pullback setups)
    if (trailActive && c.vwap != null) {
      const crossedAgainst = isLong
        ? c.close < c.vwap * (1 - CONFIG.VWAP_TOUCH_TOLERANCE)
        : c.close > c.vwap * (1 + CONFIG.VWAP_TOUCH_TOLERANCE);

      if (crossedAgainst) {
        partialExits.push({
          time: c.time, price: c.close,
          qtyPct: remainingQty, reason: EXIT_REASON.VWAP_CROSS_AGAINST,
        });
        remainingQty = 0;
        break;
      }
    }

    // ── Target 1 ──
    if (!t1Hit && targets[0] && targets[0].price != null) {
      const hitT1 = isLong
        ? c.high >= targets[0].price
        : c.low <= targets[0].price;

      if (hitT1) {
        t1Hit = true;
        const exitQty = Math.min(targets[0].qtyPct, remainingQty);
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: exitQty, reason: EXIT_REASON.TARGET_1,
        });
        remainingQty -= exitQty;

        // Move SL to breakeven
        currentSL = entryPrice;
        trailActive = true;
      }
    }

    // ── Trail using EMA(9) ──
    if (trailActive && remainingQty > 0 && c.ema9 != null) {
      if (isLong) {
        // Trail SL up to EMA(9) of previous candle
        const emaTrail = candles[i - 1]?.ema9 ?? currentSL;
        if (emaTrail > currentSL) currentSL = emaTrail;
      } else {
        const emaTrail = candles[i - 1]?.ema9 ?? currentSL;
        if (emaTrail < currentSL) currentSL = emaTrail;
      }

      // Check if T2 (VWAP band target) hit
      if (targets[1] && targets[1].price != null) {
        const hitT2 = isLong
          ? c.high >= targets[1].price
          : c.low <= targets[1].price;

        if (hitT2) {
          partialExits.push({
            time: c.time, price: targets[1].price,
            qtyPct: remainingQty, reason: EXIT_REASON.TARGET_2_TRAIL,
          });
          remainingQty = 0;
          break;
        }
      }
    }
  }

  return buildTradeRecord(signal, partialExits, entryPrice, direction);
}


// ═══════════════════════════════════════════════
// SECTION 8: TRADE RECORD
// ═══════════════════════════════════════════════

function buildTradeRecord(signal, partialExits, entryPrice, direction) {
  const isLong = direction === DIRECTION.LONG;

  let weightedExit = 0;
  let totalQty = 0;

  for (const e of partialExits) {
    weightedExit += e.price * e.qtyPct;
    totalQty += e.qtyPct;
  }

  const avgExitPrice = totalQty > 0 ? weightedExit / totalQty : entryPrice;
  const grossPnlPct = isLong
    ? (avgExitPrice - entryPrice) / entryPrice
    : (entryPrice - avgExitPrice) / entryPrice;
  const netPnlPct = grossPnlPct - (CONFIG.SLIPPAGE_PER_SIDE * 2);

  const lastExit = partialExits[partialExits.length - 1];

  // Determine primary exit reason (biggest qty chunk)
  const reasonMap = {};
  for (const e of partialExits) {
    reasonMap[e.reason] = (reasonMap[e.reason] || 0) + e.qtyPct;
  }
  let primaryReason = 'NONE';
  let maxQty = 0;
  for (const [r, q] of Object.entries(reasonMap)) {
    if (q > maxQty) { maxQty = q; primaryReason = r; }
  }

  return {
    setup: signal.setup,
    direction,
    entryPrice,
    avgExitPrice,
    entryTime: signal.entryTime,
    exitTime: lastExit?.time ?? null,
    grossPnlPct,
    netPnlPct,
    isWinner: netPnlPct > 0,
    partialExits,
    primaryExitReason: primaryReason,
    rejectionPattern: signal.rejectionPattern || null,
    trendDuration: signal.trendDuration || null,
    volumeMultiple: signal.volumeMultiple || null,
    sl: signal.sl,
  };
}


// ═══════════════════════════════════════════════
// SECTION 9: SINGLE-DAY ORCHESTRATOR
// ═══════════════════════════════════════════════

/**
 * Run all VWAP sub-strategies on one day.
 *
 * @param {object[]} dayCandlesRaw – one day of 5-min candles
 * @param {object}   opts
 *   { prevDayClose, usePDCFilter, enabledSetups: ['PULLBACK','BREAKOUT'] }
 * @returns {object[]} – array of trade records for the day
 */
export function runVWAPForDay(dayCandlesRaw, opts = {}) {
  const {
    prevDayClose = null,
    usePDCFilter = false,
    enabledSetups = [VWAP_SETUP.PULLBACK, VWAP_SETUP.BREAKOUT],
    maxTradesPerDay = 2,
  } = opts;

  // Deep copy & parse dates
  const candles = dayCandlesRaw.map(c => ({
    ...c,
    time: c.time instanceof Date ? c.time : new Date(c.time),
  }));

  // Compute indicators
  computeVWAPWithBands(candles);
  computeEMA(candles);
  computeDayHighLow(candles);

  const allSignals = [];

  // Detect pullback signals
  if (enabledSetups.includes(VWAP_SETUP.PULLBACK)) {
    const pullbacks = detectPullbackSignals(candles, { prevDayClose, usePDCFilter });
    allSignals.push(...pullbacks);
  }

  // Detect breakout signals
  if (enabledSetups.includes(VWAP_SETUP.BREAKOUT)) {
    const breakouts = detectBreakoutSignals(candles, { prevDayClose, usePDCFilter });
    allSignals.push(...breakouts);
  }

  // Sort by entry time
  allSignals.sort((a, b) => {
    const ta = a.entryTime instanceof Date ? a.entryTime : new Date(a.entryTime);
    const tb = b.entryTime instanceof Date ? b.entryTime : new Date(b.entryTime);
    return ta - tb;
  });

  // Simulate trades (limit to maxTradesPerDay, no overlapping)
  const trades = [];
  let lastExitIndex = -1;

  for (const signal of allSignals) {
    if (trades.length >= maxTradesPerDay) break;
    if (signal.entryIndex <= lastExitIndex) continue; // skip overlapping

    // Validate R:R for pullback
    if (signal.setup === VWAP_SETUP.PULLBACK) {
      const risk = Math.abs(signal.entryPrice - signal.sl);
      const reward = signal.targets[0]
        ? Math.abs(signal.targets[0].price - signal.entryPrice)
        : 0;
      const rr = risk > 0 ? reward / risk : 0;
      if (rr < CONFIG.PULLBACK_MIN_RR) continue;
    }

    const trade = simulateTrade(candles, signal);
    trades.push(trade);

    // Find exit index to prevent overlap
    const exitCandle = trade.partialExits[trade.partialExits.length - 1];
    if (exitCandle) {
      const exitTime = exitCandle.time instanceof Date ? exitCandle.time : new Date(exitCandle.time);
      for (let j = signal.entryIndex; j < candles.length; j++) {
        const ct = candles[j].time instanceof Date ? candles[j].time : new Date(candles[j].time);
        if (ct >= exitTime) { lastExitIndex = j; break; }
      }
    }
  }

  return trades;
}
```

```javascript
/**
 * VWAP Strategy Analyzer
 *
 * Builds on generic trade metrics with VWAP-specific breakdowns:
 *  - Pullback vs Breakout performance comparison
 *  - Rejection pattern win rates

// CORE PERFORMANCE
// ═══════════════════════════════════════════════

export function analyzePerformance(trades, capital = CONFIG.DEFAULT_CAPITAL) {
  if (trades.length === 0) return { error: 'No trades' };

  const winners = trades.filter(t => t.isWinner);
  const losers = trades.filter(t => !t.isWinner);
  const pnls = trades.map(t => t.netPnlPct);

  const totalTrades = trades.length;
  const winRate = winners.length / totalTrades;
  const avgWin = safeMean(winners.map(t => t.netPnlPct));
  const avgLoss = safeMean(losers.map(t => t.netPnlPct));
  const avgWinLossRatio = Math.abs(avgLoss) > 0 ? avgWin / Math.abs(avgLoss) : Infinity;
  const expectancy = (winRate * avgWin) + ((1 - winRate) * avgLoss);

  const equity = buildEquityCurve(trades, capital);
  const totalReturn = ((equity[equity.length - 1].equity - capital) / capital) * 100;
  const maxDD = computeMaxDrawdown(equity);
  const { maxWinStreak, maxLossStreak } = computeStreaks(trades);

  const sharpe = computeSharpe(pnls);
  const profitFactor = computeProfitFactor(trades);

  return {
    totalTrades,
    winners: winners.length,
    losers: losers.length,
    winRatePct: r2(winRate * 100),
    avgWinPct: r4(avgWin * 100),
    avgLossPct: r4(avgLoss * 100),
    avgWinLossRatio: r2(avgWinLossRatio),
    expectancyPct: r4(expectancy * 100),
    totalReturnPct: r2(totalReturn),
    totalReturnAbs: r2(equity[equity.length - 1].equity - capital),
    maxDrawdownPct: r2(maxDD.pct),
    maxDrawdownDuration: maxDD.duration,
    sharpeRatio: r2(sharpe),
    profitFactor: r2(profitFactor),
    maxWinStreak,
    maxLossStreak,
    equityCurve: equity,
    pnlArray: pnls.map(p => r4(p * 100)),
  };
}

  }

  return reasons;

// INTERNALS
// ═══════════════════════════════════════════════

function buildEquityCurve(trades, startCapital) {
  const curve = [{ tradeNum: 0, equity: startCapital }];
  let equity = startCapital;

  for (let i = 0; i < trades.length; i++) {
    const t = trades[i];
    const riskAmt = equity * CONFIG.DEFAULT_RISK_PER_TRADE;
    const slDist = Math.abs(t.entryPrice - t.sl) || t.entryPrice * 0.003;
    const qty = Math.floor(riskAmt / slDist);
    const posVal = qty * t.entryPrice;
    const pnl = posVal * t.netPnlPct;
    equity += pnl - CONFIG.COMMISSION_PER_SIDE * 2;

    curve.push({
      tradeNum: i + 1,
      equity: r2(equity),
      date: t.entryTime,
      setup: t.setup,
    });
  }

  return curve;
}

function computeMaxDrawdown(curve) {
  let peak = curve[0].equity;
  let maxDD = 0;
  let ddStart = 0;
  let maxDur = 0;

  for (let i = 1; i < curve.length; i++) {
    if (curve[i].equity > peak) { peak = curve[i].equity; ddStart = i; }
    const dd = (peak - curve[i].equity) / peak;
    if (dd > maxDD) { maxDD = dd; maxDur = i - ddStart; }
  }

  return { pct: maxDD * 100, duration: maxDur };
}

function computeStreaks(trades) {
  let mw = 0, ml = 0, cw = 0, cl = 0;
  for (const t of trades) {
    if (t.isWinner) { cw++; cl = 0; if (cw > mw) mw = cw; }
    else { cl++; cw = 0; if (cl > ml) ml = cl; }
  }
  return { maxWinStreak: mw, maxLossStreak: ml };
}

function computeSharpe(pnls) {
  if (pnls.length < 2) return 0;
  const avg = safeMean(pnls);
  const std = Math.sqrt(pnls.reduce((s, p) => s + (p - avg) ** 2, 0) / (pnls.length - 1));
  return std === 0 ? 0 : (avg / std) * Math.sqrt(250);
}

function computeProfitFactor(trades) {
  let gp = 0, gl = 0;
  for (const t of trades) {
    if (t.netPnlPct > 0) gp += t.netPnlPct;
    else gl += Math.abs(t.netPnlPct);
  }
  return gl > 0 ? gp / gl : Infinity;
}

function safeMean(arr) {
  return arr.length === 0 ? 0 : arr.reduce((a, b) => a + b, 0) / arr.length;
}

function r2(n) { return Math.round(n * 100) / 100; }
function r4(n) { return Math.round(n * 10000) / 10000; }
```

```javascript
// ═══════════════════════════════════════════════

export function groupByDay(candles) {
  const days = new Map();
  for (const c of candles) {
    const d = c.time instanceof Date ? c.time : new Date(c.time);
    const key = d.toISOString().slice(0, 10);
    if (!days.has(key)) days.set(key, []);
    days.get(key).push(c);
  }
  return days;
}

// ═══════════════════════════════════════════════

/**
 * @param {object[]} allCandles  – flat 5-min candle array spanning multiple days
 * @param {object}   opts
 *   {
 *     capital: number,
 *     enabledSetups: string[],  // ['PULLBACK', 'BREAKOUT']
 *     usePDCFilter: boolean,
 *     maxTradesPerDay: number,
 *     prevDayCloses: { 'YYYY-MM-DD': number }  // map of date→close
 *   }
 */
export function runFullBacktest(allCandles, opts = {}) {
  const {
    capital = CONFIG.DEFAULT_CAPITAL,
    enabledSetups = [VWAP_SETUP.PULLBACK, VWAP_SETUP.BREAKOUT],
    usePDCFilter = false,
    maxTradesPerDay = 2,
    prevDayCloses = {},
  } = opts;

  const dayGroups = groupByDay(allCandles);
  const sortedDates = [...dayGroups.keys()].sort();

  const allTrades = [];
  const skippedDays = [];
  let runningPrevClose = null;

  for (let i = 0; i < sortedDates.length; i++) {
    const dateKey = sortedDates[i];
    const dayCandles = dayGroups.get(dateKey);

    if (dayCandles.length < 10) {
      skippedDays.push({ date: dateKey, reason: 'Insufficient candles' });
      if (dayCandles.length > 0) {
        runningPrevClose = dayCandles[dayCandles.length - 1].close;
      }
      continue;
    }

    // Use explicit prevDayClose if provided, else use running tracker
    const prevClose = prevDayCloses[dateKey] ?? runningPrevClose;

    const dayTrades = runVWAPForDay(dayCandles, {
      prevDayClose: prevClose,
      usePDCFilter,
      enabledSetups,
      maxTradesPerDay,
    });

    for (const t of dayTrades) {
      t.date = dateKey;
      allTrades.push(t);
    }

    if (dayTrades.length === 0) {
      skippedDays.push({ date: dateKey, reason: 'No valid signal' });
    }

    // Update prev close
    runningPrevClose = dayCandles[dayCandles.length - 1].close;
  }

  // Analysis
  const report = allTrades.length > 0 ? analyzePerformance(allTrades, capital) : null;

  return {
    trades: allTrades,
    report,
    totalDays: sortedDates.length,
    tradedDays: new Set(allTrades.map(t => t.date)).size,
    totalSignals: allTrades.length,
    skippedDays,
  };
}

  return { pullbackOnly, breakoutOnly, combined };
}

/**
 * Run with and without PDC filter to measure its impact.
 */
export function analyzePDCImpact(allCandles, opts = {}) {
  const withoutPDC = runFullBacktest(allCandles, { ...opts, usePDCFilter: false });
  const withPDC = runFullBacktest(allCandles, { ...opts, usePDCFilter: true });

  console.log('\n┌──────────────────────────────────────────────────┐');
  console.log('│        PDC FILTER IMPACT                          │');
  console.log('├──────────────────────────────────────────────────┤');

  const metrics = [
    ['Trades', withoutPDC.report?.totalTrades, withPDC.report?.totalTrades],
    ['Win Rate %', withoutPDC.report?.winRatePct, withPDC.report?.winRatePct],
    ['Expectancy %', withoutPDC.report?.expectancyPct, withPDC.report?.expectancyPct],
    ['Return %', withoutPDC.report?.totalReturnPct, withPDC.report?.totalReturnPct],
    ['Max DD %', withoutPDC.report?.maxDrawdownPct, withPDC.report?.maxDrawdownPct],
  ];

  console.log(`│ ${'Metric'.padEnd(16)} ${'No PDC'.padStart(12)} ${'With PDC'.padStart(12)} │`);
  for (const [m, a, b] of metrics) {
    console.log(`│ ${m.padEnd(16)} ${String(a ?? '-').padStart(12)} ${String(b ?? '-').padStart(12)} │`);
  }
  console.log('└──────────────────────────────────────────────────┘\n');

// MULTI-STOCK SCAN
// ═══════════════════════════════════════════════

/**
 * @param {Map<string, object[]>} stockCandles – symbol → candles[]
 * @param {object} opts
 * @returns {object[]} – ranked by expectancy
 */
export function multiStockScan(stockCandles, opts = {}) {
  const results = [];

  for (const [symbol, candles] of stockCandles) {
    const res = runFullBacktest(candles, opts);

    results.push({
      symbol,
      trades: res.report?.totalTrades ?? 0,
      winRatePct: res.report?.winRatePct ?? 0,
      expectancyPct: res.report?.expectancyPct ?? 0,
      returnPct: res.report?.totalReturnPct ?? 0,
      maxDDPct: res.report?.maxDrawdownPct ?? 0,
      sharpe: res.report?.sharpeRatio ?? 0,
      profitFactor: res.report?.profitFactor ?? 0,
    });
  }

  results.sort((a, b) => (b.expectancyPct || 0) - (a.expectancyPct || 0));
  return results;
}

export function runAndReport(allCandles, opts = {}) {
  const result = runFullBacktest(allCandles, opts);

  if (!result.report) {
    console.log('No trades generated.');
    return result;
  }

  const trades = result.trades;

  const extras = {
    setupBreakdown: setupBreakdown(trades),
    rejectionPatterns: rejectionPatternAnalysis(trades),
    timeOfDay: timeOfDayAnalysis(trades),
    directions: directionBreakdown(trades),
    exitReasons: exitReasonBreakdown(trades),
    trendDuration: trendDurationAnalysis(trades),
  };

  printReport(result.report, extras);

  console.log('\n📅 MONTHLY:');
  console.table(monthlyBreakdown(trades));

  console.log('\n📈 DIRECTION:');
  console.table(extras.directions);

  console.log(`\nDays: ${result.totalDays} | Traded: ${result.tradedDays} | Signals: ${result.totalSignals}`);

  return { ...result, extras };
}
```

Key architectural difference from ORB: the VWAP strategy uses a **state machine** (`IDLE → TRENDING → PULLING_BACK → SIGNAL`) to detect pullback setups because they require tracking how long price has trended on one side before approaching VWAP — something a simple threshold check can't capture. The breakout detector uses a **confirmation gate** (2 consecutive candles holding above VWAP after cross) to filter fakeouts, and all trailing is done via **EMA(9)** rather than previous candle low/high, matching the strategy's institutional mean-reversion character.
