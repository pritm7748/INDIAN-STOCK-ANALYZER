## `ema-indicators.js`

```javascript
/**
 * Technical Indicator Computations
 *
 * All indicator functions mutate candles in-place for performance.
 * Each sets named properties on the candle objects.
 *
 * Candle contract:
 * {
 *   time: Date | string,
 *   open: number,
 *   high: number,
 *   low: number,
 *   close: number,
 *   volume: number,
 *   // Set by this module:
 *   ema9?: number,  ema20?: number,
 *   ema5?: number,  ema13?: number, ema26?: number,
 *   macdLine?: number, macdSignal?: number, macdHist?: number,
 *   vwap?: number, vwapUpper1?: number, vwapLower1?: number,
 *   avgVolume?: number,
 * }
 */

import { CONFIG } from './ema-config.js';

// ═══════════════════════════════════════════════
// SECTION 1: EMA COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Generic EMA computation on close prices.
 *
 * @param {object[]} candles
 * @param {number}   period
 * @param {string}   fieldName – property name to set on each candle
 */
export function computeEMA(candles, period, fieldName) {
  if (candles.length === 0) return;

  const k = 2 / (period + 1);

  // Seed: SMA of first `period` candles
  let seed = 0;
  const seedEnd = Math.min(period, candles.length);
  for (let i = 0; i < seedEnd; i++) {
    seed += candles[i].close;
  }
  seed /= seedEnd;

  // Set initial EMA for first candle
  candles[0][fieldName] = candles[0].close;

  // For candles before period, use progressive SMA as approximation
  for (let i = 1; i < seedEnd; i++) {
    candles[i][fieldName] = candles[i].close * k + candles[i - 1][fieldName] * (1 - k);
  }

  // Proper EMA from seed onwards
  if (seedEnd > 1) {
    candles[seedEnd - 1][fieldName] = seed;
  }

  for (let i = seedEnd; i < candles.length; i++) {
    candles[i][fieldName] = candles[i].close * k + candles[i - 1][fieldName] * (1 - k);
  }
}

/**
 * Compute all EMAs needed for the dual 9/20 system.
 */
export function computeDualEMAs(candles) {
  computeEMA(candles, CONFIG.DUAL_FAST_PERIOD, 'ema9');
  computeEMA(candles, CONFIG.DUAL_SLOW_PERIOD, 'ema20');
}

/**
 * Compute all EMAs for the triple 5/13/26 system.
 */
export function computeTripleEMAs(candles) {
  computeEMA(candles, CONFIG.TRIPLE_FAST_PERIOD, 'ema5');
  computeEMA(candles, CONFIG.TRIPLE_MID_PERIOD, 'ema13');
  computeEMA(candles, CONFIG.TRIPLE_SLOW_PERIOD, 'ema26');
}


// ═══════════════════════════════════════════════
// SECTION 2: MACD COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Compute MACD with custom periods for intraday.
 * Sets: candle.macdLine, candle.macdSignal, candle.macdHist
 *
 * MACD Line = EMA(fast) - EMA(slow)     on close prices
 * Signal Line = EMA(signal period)       on MACD Line
 * Histogram = MACD Line - Signal Line
 */
export function computeMACD(candles) {
  const fastField = '_macd_fast_ema';
  const slowField = '_macd_slow_ema';

  // Compute fast and slow EMAs for MACD (separate from display EMAs)
  computeEMA(candles, CONFIG.MACD_FAST, fastField);
  computeEMA(candles, CONFIG.MACD_SLOW, slowField);

  // MACD Line
  for (const c of candles) {
    c.macdLine = (c[fastField] ?? c.close) - (c[slowField] ?? c.close);
  }

  // Signal Line = EMA of MACD Line
  computeEMAOnField(candles, CONFIG.MACD_SIGNAL, 'macdLine', 'macdSignal');

  // Histogram
  for (const c of candles) {
    c.macdHist = (c.macdLine ?? 0) - (c.macdSignal ?? 0);
  }

  // Clean up temp fields
  for (const c of candles) {
    delete c[fastField];
    delete c[slowField];
  }
}

/**
 * Compute EMA on an arbitrary numeric field (not necessarily close price).
 *
 * @param {object[]} candles
 * @param {number}   period
 * @param {string}   sourceField – field to read values from
 * @param {string}   targetField – field to write EMA to
 */
function computeEMAOnField(candles, period, sourceField, targetField) {
  if (candles.length === 0) return;

  const k = 2 / (period + 1);

  candles[0][targetField] = candles[0][sourceField] ?? 0;

  for (let i = 1; i < candles.length; i++) {
    const val = candles[i][sourceField] ?? 0;
    const prevEma = candles[i - 1][targetField] ?? 0;
    candles[i][targetField] = val * k + prevEma * (1 - k);
  }
}


// ═══════════════════════════════════════════════
// SECTION 3: VWAP + BANDS
// ═══════════════════════════════════════════════

/**
 * Compute VWAP with ±1 SD band.
 * Sets: candle.vwap, candle.vwapUpper1, candle.vwapLower1
 */
export function computeVWAPWithBands(candles) {
  let cumTPV = 0;
  let cumVol = 0;
  let cumTP2V = 0;

  for (const c of candles) {
    const tp = (c.high + c.low + c.close) / 3;

    cumTPV += tp * c.volume;
    cumVol += c.volume;
    cumTP2V += tp * tp * c.volume;

    if (cumVol === 0) {
      c.vwap = c.close;
      c.vwapUpper1 = c.close;
      c.vwapLower1 = c.close;
      continue;
    }

    const vwap = cumTPV / cumVol;
    const variance = (cumTP2V / cumVol) - (vwap * vwap);
    const sd = Math.sqrt(Math.max(0, variance));

    c.vwap = vwap;
    c.vwapUpper1 = vwap + CONFIG.VWAP_SD_MULTIPLIER * sd;
    c.vwapLower1 = vwap - CONFIG.VWAP_SD_MULTIPLIER * sd;
  }
}


// ═══════════════════════════════════════════════
// SECTION 4: ROLLING AVERAGE VOLUME
// ═══════════════════════════════════════════════

/**
 * Compute rolling average volume for each candle.
 * Sets: candle.avgVolume
 */
export function computeAvgVolume(candles, lookback = CONFIG.VOLUME_AVG_LOOKBACK) {
  for (let i = 0; i < candles.length; i++) {
    const start = Math.max(0, i - lookback);
    const window = candles.slice(start, i + 1);
    const sum = window.reduce((s, c) => s + c.volume, 0);
    candles[i].avgVolume = sum / window.length;
  }
}


// ═══════════════════════════════════════════════
// SECTION 5: EMA SLOPE COMPUTATION
// ═══════════════════════════════════════════════

/**
 * Compute the slope of an EMA over the last N candles.
 * Slope = (EMA[now] - EMA[now - N]) / (N × price) — normalised.
 *
 * @param {object[]} candles
 * @param {string}   emaField    – 'ema20', 'ema26', etc.
 * @param {number}   lookback    – candles to measure slope over
 * @param {number}   currentIdx  – index of current candle
 * @returns {number} slope per candle (positive = rising, negative = falling)
 */
export function computeEMASlope(candles, emaField, lookback, currentIdx) {
  const i = currentIdx;
  const startIdx = Math.max(0, i - lookback);

  const endVal = candles[i]?.[emaField];
  const startVal = candles[startIdx]?.[emaField];

  if (endVal == null || startVal == null) return 0;

  const candleSpan = i - startIdx;
  if (candleSpan === 0) return 0;

  // Normalise by price level to make slope comparable across stocks
  const refPrice = candles[i].close || 1;
  return (endVal - startVal) / (candleSpan * refPrice);
}

/**
 * Compute slope for every candle and store it.
 */
export function computeAllSlopes(candles, emaField, slopeField, lookback = CONFIG.SLOPE_LOOKBACK_CANDLES) {
  for (let i = 0; i < candles.length; i++) {
    candles[i][slopeField] = computeEMASlope(candles, emaField, lookback, i);
  }
}


// ═══════════════════════════════════════════════
// SECTION 6: EMA CROSSOVER DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect where fast EMA crosses slow EMA.
 *
 * @param {object[]} candles
 * @param {string}   fastField – 'ema9', 'ema5'
 * @param {string}   slowField – 'ema20', 'ema13'
 * @returns {object[]} array of crossover events:
 *   { index, time, type: 'BULLISH'|'BEARISH', fastVal, slowVal, price }
 */
export function detectCrossovers(candles, fastField, slowField) {
  const crossovers = [];

  for (let i = 1; i < candles.length; i++) {
    const prevFast = candles[i - 1][fastField];
    const prevSlow = candles[i - 1][slowField];
    const currFast = candles[i][fastField];
    const currSlow = candles[i][slowField];

    if (prevFast == null || prevSlow == null || currFast == null || currSlow == null) continue;

    // Bullish: fast crosses above slow
    if (prevFast <= prevSlow && currFast > currSlow) {
      crossovers.push({
        index: i,
        time: candles[i].time,
        type: 'BULLISH',
        fastVal: currFast,
        slowVal: currSlow,
        price: candles[i].close,
        spread: currFast - currSlow,
        spreadPct: (currFast - currSlow) / candles[i].close,
      });
    }

    // Bearish: fast crosses below slow
    if (prevFast >= prevSlow && currFast < currSlow) {
      crossovers.push({
        index: i,
        time: candles[i].time,
        type: 'BEARISH',
        fastVal: currFast,
        slowVal: currSlow,
        price: candles[i].close,
        spread: currFast - currSlow,
        spreadPct: (currFast - currSlow) / candles[i].close,
      });
    }
  }

  return crossovers;
}


// ═══════════════════════════════════════════════
// SECTION 7: MACD CROSSOVER DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect MACD line crossing signal line.
 *
 * @param {object[]} candles
 * @returns {object[]} crossover events
 */
export function detectMACDCrossovers(candles) {
  const crossovers = [];

  for (let i = 1; i < candles.length; i++) {
    const prevLine = candles[i - 1].macdLine;
    const prevSig = candles[i - 1].macdSignal;
    const currLine = candles[i].macdLine;
    const currSig = candles[i].macdSignal;

    if (prevLine == null || prevSig == null || currLine == null || currSig == null) continue;

    if (prevLine <= prevSig && currLine > currSig) {
      crossovers.push({
        index: i, time: candles[i].time, type: 'BULLISH',
        macdLine: currLine, signal: currSig, hist: candles[i].macdHist,
      });
    }

    if (prevLine >= prevSig && currLine < currSig) {
      crossovers.push({
        index: i, time: candles[i].time, type: 'BEARISH',
        macdLine: currLine, signal: currSig, hist: candles[i].macdHist,
      });
    }
  }

  return crossovers;
}


// ═══════════════════════════════════════════════
// SECTION 8: TRIPLE EMA ALIGNMENT
// ═══════════════════════════════════════════════

/**
 * Check if triple EMAs are aligned (stacked in order) or tangled.
 *
 * BULLISH: EMA5 > EMA13 > EMA26
 * BEARISH: EMA5 < EMA13 < EMA26
 * TANGLED: anything else
 *
 * @param {object} candle
 * @returns {string} ALIGNMENT enum
 */
export function getTripleAlignment(candle) {
  const { ema5, ema13, ema26 } = candle;
  if (ema5 == null || ema13 == null || ema26 == null) return 'TANGLED';

  if (ema5 > ema13 && ema13 > ema26) return 'BULLISH';
  if (ema5 < ema13 && ema13 < ema26) return 'BEARISH';
  return 'TANGLED';
}

/**
 * Check alignment over a window of candles.
 * All candles in the window must show the same alignment.
 *
 * @param {object[]} candles
 * @param {number}   endIdx     – end of window (inclusive)
 * @param {number}   windowSize – number of candles to check
 * @returns {{ alignment: string, consistent: boolean, duration: number }}
 */
export function checkAlignmentWindow(candles, endIdx, windowSize = 3) {
  const start = Math.max(0, endIdx - windowSize + 1);
  let alignment = null;
  let consistent = true;
  let duration = 0;

  for (let i = endIdx; i >= start; i--) {
    const a = getTripleAlignment(candles[i]);
    if (alignment === null) alignment = a;

    if (a === alignment && a !== 'TANGLED') {
      duration++;
    } else if (a !== alignment) {
      consistent = false;
      break;
    }
  }

  return { alignment: alignment || 'TANGLED', consistent, duration };
}


// ═══════════════════════════════════════════════
// SECTION 9: TANGLING DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect whether EMAs are "tangled" — intertwined with multiple
 * micro-crossovers indicating a range-bound market.
 *
 * @param {object[]} candles
 * @param {string}   fastField
 * @param {string}   slowField
 * @param {number}   currentIdx
 * @returns {{ isTangled: boolean, crossoverCount: number, avgDistance: number }}
 */
export function detectTangling(candles, fastField, slowField, currentIdx) {
  const lookback = CONFIG.TANGLE_LOOKBACK_CANDLES;
  const start = Math.max(0, currentIdx - lookback);

  let crossoverCount = 0;
  let totalDistancePct = 0;
  let dataPoints = 0;

  for (let i = start + 1; i <= currentIdx; i++) {
    const prevFast = candles[i - 1][fastField];
    const prevSlow = candles[i - 1][slowField];
    const currFast = candles[i][fastField];
    const currSlow = candles[i][slowField];

    if (prevFast == null || currFast == null) continue;

    // Count crossovers in window
    if ((prevFast <= prevSlow && currFast > currSlow) ||
        (prevFast >= prevSlow && currFast < currSlow)) {
      crossoverCount++;
    }

    // Average distance between EMAs
    const distance = Math.abs(currFast - currSlow) / candles[i].close;
    totalDistancePct += distance;
    dataPoints++;
  }

  const avgDistance = dataPoints > 0 ? totalDistancePct / dataPoints : 0;

  const isTangled =
    crossoverCount >= CONFIG.TANGLE_MIN_CROSSOVERS ||
    avgDistance < CONFIG.TANGLE_MAX_DISTANCE_PCT;

  return { isTangled, crossoverCount, avgDistance };
}

// SECTION 10: COMPUTE ALL INDICATORS
// ═══════════════════════════════════════════════

/**
 * One-call to compute everything needed for both systems.
 *
 * @param {object[]} candles
 * @param {string}   system – EMA_SYSTEM enum
 */
export function computeAllIndicators(candles, system) {
  // Always compute VWAP and MACD
  computeVWAPWithBands(candles);
  computeMACD(candles);
  computeAvgVolume(candles);

  if (system === 'DUAL_9_20') {
    computeDualEMAs(candles);

  } else if (system === 'TRIPLE_5_13_26') {

    computeAllSlopes(candles, 'ema26', 'ema26Slope');
    computeAllSlopes(candles, 'ema13', 'ema13Slope');

  if (system === 'BOTH') {

    computeTripleEMAs(candles);
    computeAllSlopes(candles, 'ema20', 'ema20Slope');
    computeAllSlopes(candles, 'ema26', 'ema26Slope');
    computeAllSlopes(candles, 'ema13', 'ema13Slope');
  }
}
```

## `ema-filters.js`

```javascript
/**
 * EMA Crossover Pre-Trade Filters
 *
 * Composable filters to reject low-quality crossover signals:
 *  - Time window
 *  - EMA slope (reject flat slow EMA)
 *  - Tangling detection (reject range-bound)
 *  - MACD confirmation
 *  - VWAP alignment
 *  - Volume confirmation
 *  - Triple EMA alignment
 */

import { CONFIG, DIRECTION, ALIGNMENT } from './ema-config.js';
import {
  computeEMASlope, detectTangling, getTripleAlignment,
  checkAlignmentWindow,
} from './ema-indicators.js';

// ════════════════════════════════════════════════
// TIME HELPERS
// ════════════════════════════════════════════════

export function toMinutes(h, m) { return h * 60 + m; }

export function candleTime(candle) {
  const d = candle.time instanceof Date ? candle.time : new Date(candle.time);
  return toMinutes(d.getHours(), d.getMinutes());
}

/**
 * Is the candle within the valid entry window (9:30 AM – 1:00 PM)?
 */
export function isInEntryWindow(candle) {
  const t = candleTime(candle);
  const start = toMinutes(CONFIG.TRADE_WINDOW_START_H, CONFIG.TRADE_WINDOW_START_M);
  const end = toMinutes(CONFIG.TRADE_WINDOW_END_H, CONFIG.TRADE_WINDOW_END_M);
  return t >= start && t <= end;
}

/**
 * Extended window: allows entry up to LAST_ENTRY time (2:00 PM).
 */
export function isBeforeLastEntry(candle) {
  const t = candleTime(candle);
  return t <= toMinutes(CONFIG.LAST_ENTRY_H, CONFIG.LAST_ENTRY_M);
}

export function isPastMISExit(candle) {
  return candleTime(candle) >= toMinutes(CONFIG.MIS_EXIT_H, CONFIG.MIS_EXIT_M);
}


// ════════════════════════════════════════════════
// FILTER 1: EMA SLOPE
// ════════════════════════════════════════════════

/**
 * Reject crossovers when the slow EMA is flat (no clear trend direction).
 *
 * @param {object[]} candles
 * @param {string}   slowEMAField  – 'ema20' or 'ema26'
 * @param {number}   currentIdx
 * @param {string}   direction     – expected direction
 */
export function filterSlope(candles, slowEMAField, currentIdx, direction) {
  if (!CONFIG.SLOPE_FILTER_ENABLED) return { pass: true };

  const slope = computeEMASlope(
    candles, slowEMAField, CONFIG.SLOPE_LOOKBACK_CANDLES, currentIdx
  );
  const absSlope = Math.abs(slope);
  const isDirectional = absSlope >= CONFIG.SLOPE_MIN_THRESHOLD;

  // Slope should be in the same direction as trade
  const slopeAligned = direction === DIRECTION.LONG ? slope > 0 : slope < 0;

  const pass = isDirectional && slopeAligned;

  return {
    pass,
    slope: Math.round(slope * 1_000_000) / 1_000_000,
    absSlope,
    isDirectional,
    slopeAligned,
    reason: pass ? null : `Slow EMA slope ${slope.toFixed(6)} not directional/aligned for ${direction}`,
  };
}


// ════════════════════════════════════════════════
// FILTER 2: TANGLING REJECTION
// ════════════════════════════════════════════════

/**
 * Reject if EMAs are intertwined (range-bound, whipsaw territory).
 */
export function filterTangling(candles, fastField, slowField, currentIdx) {
  const { isTangled, crossoverCount, avgDistance } = detectTangling(
    candles, fastField, slowField, currentIdx
  );

  return {
    pass: !isTangled,
    isTangled,
    crossoverCount,
    avgDistancePct: Math.round(avgDistance * 10000) / 10000,
    reason: isTangled
      ? `EMAs tangled: ${crossoverCount} crossovers in last ${CONFIG.TANGLE_LOOKBACK_CANDLES} candles, avg distance ${(avgDistance * 100).toFixed(3)}%`
      : null,
  };
}


// ════════════════════════════════════════════════
// FILTER 3: MACD CONFIRMATION
// ════════════════════════════════════════════════

/**
 * MACD line must be crossing or have recently crossed signal line
 * in the same direction as the EMA crossover.
 *
 * @param {object[]} candles
 * @param {number}   crossoverIdx  – index of EMA crossover candle
 * @param {string}   direction
 * @param {number}   lookback      – how many candles back/forward to check MACD
 */
export function filterMACD(candles, crossoverIdx, direction, lookback = 3) {
  // Check MACD alignment at crossover candle and a few candles around it
  const startCheck = Math.max(0, crossoverIdx - lookback);
  const endCheck = Math.min(candles.length - 1, crossoverIdx + lookback);

  let macdConfirmed = false;
  let macdCrossIdx = -1;
  let histogram = 0;

  for (let i = startCheck + 1; i <= endCheck; i++) {
    const prevLine = candles[i - 1].macdLine;
    const prevSig = candles[i - 1].macdSignal;
    const currLine = candles[i].macdLine;
    const currSig = candles[i].macdSignal;

    if (prevLine == null || currLine == null) continue;

    // Bullish MACD cross near bullish EMA cross
    if (direction === DIRECTION.LONG && prevLine <= prevSig && currLine > currSig) {
      macdConfirmed = true;
      macdCrossIdx = i;
      histogram = candles[i].macdHist;
      break;
    }

    // Bearish MACD cross near bearish EMA cross
    if (direction === DIRECTION.SHORT && prevLine >= prevSig && currLine < currSig) {
      macdConfirmed = true;
      macdCrossIdx = i;
      histogram = candles[i].macdHist;
      break;
    }
  }

  // Fallback: even without a cross, if MACD histogram is already in our direction
  if (!macdConfirmed) {
    const hist = candles[crossoverIdx].macdHist;
    if (direction === DIRECTION.LONG && hist > 0) {
      macdConfirmed = true;
      histogram = hist;
    }
    if (direction === DIRECTION.SHORT && hist < 0) {
      macdConfirmed = true;
      histogram = hist;
    }
  }

  return {
    pass: macdConfirmed,
    macdCrossIdx,
    histogram: Math.round(histogram * 10000) / 10000,
    macdLine: candles[crossoverIdx].macdLine,
    macdSignal: candles[crossoverIdx].macdSignal,
    reason: macdConfirmed ? null : `MACD not confirming ${direction} within ±${lookback} candles`,
  };
}


// ════════════════════════════════════════════════
// FILTER 4: VWAP ALIGNMENT
// ════════════════════════════════════════════════

/**
 * Price must be above VWAP for longs, below for shorts.
 */
export function filterVWAP(candle, direction) {
  if (!CONFIG.VWAP_ALIGNMENT_REQUIRED) return { pass: true };

  const { close, vwap } = candle;
  if (vwap == null) return { pass: true, reason: 'VWAP not available' };

  const aligned = direction === DIRECTION.LONG
    ? close > vwap
    : close < vwap;

  return {
    pass: aligned,
    price: close,
    vwap,
    reason: aligned ? null : `Price ${close} not ${direction === DIRECTION.LONG ? 'above' : 'below'} VWAP ${vwap.toFixed(2)}`,
  };
}


// ════════════════════════════════════════════════
// FILTER 5: VOLUME CONFIRMATION
// ════════════════════════════════════════════════

/**
 * Crossover candle volume should exceed average volume.
 */
export function filterVolume(candle) {
  const { volume, avgVolume } = candle;
  if (avgVolume == null || avgVolume === 0) return { pass: true };

  const ratio = volume / avgVolume;
  const pass = ratio >= CONFIG.VOLUME_CROSSOVER_MULTIPLE;

  return {
    pass,
    volume,
    avgVolume: Math.round(avgVolume),
    ratio: Math.round(ratio * 100) / 100,
    reason: pass ? null : `Volume ${volume} < ${CONFIG.VOLUME_CROSSOVER_MULTIPLE}× avg ${Math.round(avgVolume)}`,
  };
}


// ════════════════════════════════════════════════
// FILTER 6: TRIPLE EMA ALIGNMENT
// ════════════════════════════════════════════════

/**
 * For triple EMA system: all three must be stacked.
 * EMA5 > EMA13 > EMA26 for LONG
 * EMA5 < EMA13 < EMA26 for SHORT
 */
export function filterTripleAlignment(candles, currentIdx, direction, windowSize = 3) {
  const { alignment, consistent, duration } = checkAlignmentWindow(candles, currentIdx, windowSize);

  const expectedAlignment = direction === DIRECTION.LONG
    ? ALIGNMENT.BULLISH
    : ALIGNMENT.BEARISH;

  const pass = alignment === expectedAlignment && consistent;

  return {
    pass,
    alignment,
    consistent,
    duration,
    expected: expectedAlignment,
    reason: pass ? null : `Triple EMA alignment: ${alignment} (need ${expectedAlignment}, consistent: ${consistent})`,
  };
}
```

## `ema-core.js`

```javascript
/**
 * EMA Crossover + MACD Core Engine
 *
 * System A: 9/20 EMA Crossover
 *   - Detect EMA9 crossing EMA20
 *   - Entry at crossover close or pullback to EMA9
 *   - SL below EMA20
 *   - Target: fixed R:R, trail with EMA9, or VWAP band
 *
 * System B: Triple EMA (5/13/26)
 *   - Detect full alignment (all stacked)
 *   - Entry on pullback to EMA13
 *   - SL below EMA26
 *   - Fewer signals, higher quality
 */

import {
  CONFIG, DIRECTION, EMA_SYSTEM, ENTRY_METHOD, EXIT_METHOD, EXIT_REASON,
} from './ema-config.js';
import {
  computeAllIndicators, detectCrossovers, getTripleAlignment,
} from './ema-indicators.js';
import {
  filterDualCrossover, filterTriplePullback,
  isInEntryWindow, isBeforeLastEntry, isPastMISExit,
  candleTime, toMinutes,
} from './ema-filters.js';


// ═══════════════════════════════════════════════
// SECTION 1: SYSTEM A — 9/20 EMA CROSSOVER
// ═══════════════════════════════════════════════

/**
 * Detect all valid 9/20 EMA crossover signals in a day.
 *
 * @param {object[]} candles – day's candles with indicators computed
 * @param {object}   opts    – { entryMethod, exitMethod }
 * @returns {object[]} array of signal objects
 */
function detectDualCrossoverSignals(candles, opts = {}) {
  const {
    entryMethod = CONFIG.DEFAULT_ENTRY_METHOD,
  } = opts;

  const crossovers = detectCrossovers(candles, 'ema9', 'ema20');
  const signals = [];

  for (const xo of crossovers) {
    const direction = xo.type === 'BULLISH' ? DIRECTION.LONG : DIRECTION.SHORT;

    // Run all filters
    const filterResult = filterDualCrossover({
      candles,
      crossoverIdx: xo.index,
      direction,
    });

    if (!filterResult.pass) continue;

    // Determine entry point
    let entryIdx = xo.index;
    let entryPrice = candles[xo.index].close;

    if (entryMethod === ENTRY_METHOD.PULLBACK_TO_FAST) {
      const pullback = findPullbackToEMA(candles, xo.index, 'ema9', direction);
      if (pullback) {
        entryIdx = pullback.index;
        entryPrice = pullback.price;
      } else {
        continue; // no pullback found, skip signal
      }
    }

    // Compute SL and targets
    const { sl, targets } = computeDualSLTargets(
      candles, entryIdx, entryPrice, direction, opts
    );

    signals.push({
      system: EMA_SYSTEM.DUAL_9_20,
      direction,
      entryPrice,
      entryTime: candles[entryIdx].time,
      entryIndex: entryIdx,
      crossoverIndex: xo.index,
      crossoverTime: xo.time,
      sl,
      targets,
      ema9AtEntry: candles[entryIdx].ema9,
      ema20AtEntry: candles[entryIdx].ema20,
      macdHistAtEntry: candles[entryIdx].macdHist,
      vwapAtEntry: candles[entryIdx].vwap,
      filterResults: filterResult.filters,
      entryMethod,
    });
  }

  return signals;
}


/**
 * Find a pullback to the fast EMA after a crossover.
 * Price must return close to EMA9 within N candles.
 *
 * @param {object[]} candles
 * @param {number}   crossoverIdx
 * @param {string}   fastField    – 'ema9'
 * @param {string}   direction
 * @returns {object|null} { index, price }
 */
function findPullbackToEMA(candles, crossoverIdx, fastField, direction) {
  const maxWait = CONFIG.PULLBACK_MAX_CANDLES;
  const tolerance = CONFIG.PULLBACK_TOUCH_TOLERANCE;

  for (let i = crossoverIdx + 1; i < Math.min(candles.length, crossoverIdx + maxWait + 1); i++) {
    const c = candles[i];
    const ema = c[fastField];
    if (ema == null) continue;

    // Check if candle comes close to fast EMA
    const distance = Math.abs(c.low - ema) / ema; // for long
    const distanceHigh = Math.abs(c.high - ema) / ema; // for short

    if (direction === DIRECTION.LONG) {
      // Price dips near EMA9 (low is close to EMA)
      if (distance <= tolerance && c.close > ema) {
        return { index: i, price: c.close };
      }
      // Candle's low touches or pierces EMA but closes above
      if (c.low <= ema * (1 + tolerance) && c.close > ema) {
        return { index: i, price: c.close };
      }
    } else {
      // Price rallies near EMA9 (high is close to EMA)
      if (distanceHigh <= tolerance && c.close < ema) {
        return { index: i, price: c.close };
      }
      if (c.high >= ema * (1 - tolerance) && c.close < ema) {
        return { index: i, price: c.close };
      }
    }
  }

  return null; // no pullback within window
}


/**
 * Compute SL and targets for dual system.
 */
function computeDualSLTargets(candles, entryIdx, entryPrice, direction, opts = {}) {
  const exitMethod = opts.exitMethod || CONFIG.DEFAULT_EXIT_METHOD;
  const isLong = direction === DIRECTION.LONG;
  const candle = candles[entryIdx];

  // ── Stop-Loss ──
  let sl;
  if (CONFIG.SL_METHOD_DUAL === 'BELOW_SLOW_EMA') {
    const ema20 = candle.ema20 || entryPrice;
    const buffer = ema20 * CONFIG.SL_EMA_BUFFER_PCT;
    sl = isLong ? ema20 - buffer : ema20 + buffer;
  } else {
    sl = isLong
      ? entryPrice * (1 - CONFIG.SL_FIXED_PCT)
      : entryPrice * (1 + CONFIG.SL_FIXED_PCT);
  }

  // Enforce max SL
  const slPct = Math.abs(entryPrice - sl) / entryPrice;
  if (slPct > CONFIG.MAX_SL_PCT) {
    sl = isLong
      ? entryPrice * (1 - CONFIG.MAX_SL_PCT)
      : entryPrice * (1 + CONFIG.MAX_SL_PCT);
  }

  // ── Targets ──
  const slDistance = Math.abs(entryPrice - sl);
  const targets = [];

  if (exitMethod === EXIT_METHOD.FIXED_RR) {
    const t1Price = isLong
      ? entryPrice + slDistance * CONFIG.FIXED_RR_RATIO
      : entryPrice - slDistance * CONFIG.FIXED_RR_RATIO;

    targets.push({ price: t1Price, qtyPct: 1.0, label: 'FIXED_RR_TARGET' });
  }

  if (exitMethod === EXIT_METHOD.TRAIL_FAST_EMA) {
    // T1: 1× risk distance for 50%, then trail rest with EMA9
    const t1Price = isLong
      ? entryPrice + slDistance
      : entryPrice - slDistance;

    targets.push({ price: t1Price, qtyPct: CONFIG.TARGET_1_QTY_PCT, label: 'T1' });
    targets.push({ price: null, qtyPct: CONFIG.TRAIL_QTY_PCT, label: 'TRAIL_EMA9' });
  }

  if (exitMethod === EXIT_METHOD.VWAP_BAND) {
    const vwapTarget = isLong
      ? candle.vwapUpper1 || entryPrice * 1.01
      : candle.vwapLower1 || entryPrice * 0.99;

    targets.push({ price: vwapTarget, qtyPct: 1.0, label: 'VWAP_1SD' });
  }

  return { sl, targets, exitMethod };
}


// ═══════════════════════════════════════════════
// SECTION 2: SYSTEM B — TRIPLE EMA (5/13/26)
// ═══════════════════════════════════════════════

/**
 * Detect triple EMA pullback signals.
 *
 * Logic:
 *   1. All three EMAs must be aligned (stacked in order)
 *   2. Price pulls back to EMA13 (middle EMA)
 *   3. MACD confirms direction
 *   4. Enter on bounce from EMA13
 *   5. SL below EMA26
 *
 * @param {object[]} candles
 * @param {object}   opts
 * @returns {object[]} signals
 */
function detectTriplePullbackSignals(candles, opts = {}) {
  const signals = [];

  // State tracking
  let alignmentStart = -1;
  let currentAlignment = 'TANGLED';

  for (let i = CONFIG.TRIPLE_SLOW_PERIOD + 1; i < candles.length; i++) {
    const c = candles[i];
    const alignment = getTripleAlignment(c);

    // Track when alignment begins
    if (alignment !== currentAlignment) {
      currentAlignment = alignment;
      alignmentStart = i;
    }

    // Need at least a few candles of alignment
    if (alignment === 'TANGLED') continue;
    if (i - alignmentStart < 3) continue;

    const direction = alignment === 'BULLISH' ? DIRECTION.LONG : DIRECTION.SHORT;

    // Check for pullback to EMA13
    const ema13 = c.ema13;
    if (ema13 == null) continue;

    const isPullback = direction === DIRECTION.LONG
      ? c.low <= ema13 * (1 + CONFIG.PULLBACK_TOUCH_TOLERANCE) && c.close > ema13
      : c.high >= ema13 * (1 - CONFIG.PULLBACK_TOUCH_TOLERANCE) && c.close < ema13;

    if (!isPullback) continue;

    // Run filters
    const filterResult = filterTriplePullback({
      candles,
      signalIdx: i,
      direction,
    });

    if (!filterResult.pass) continue;

    // Entry, SL, Targets
    const entryPrice = c.close;
    const { sl, targets } = computeTripleSLTargets(candles, i, entryPrice, direction, opts);

    signals.push({
      system: EMA_SYSTEM.TRIPLE_5_13_26,
      direction,
      entryPrice,
      entryTime: c.time,
      entryIndex: i,
      sl,
      targets,
      ema5AtEntry: c.ema5,
      ema13AtEntry: c.ema13,
      ema26AtEntry: c.ema26,
      alignment,
      alignmentDuration: i - alignmentStart,
      macdHistAtEntry: c.macdHist,
      vwapAtEntry: c.vwap,
      filterResults: filterResult.filters,
    });

    // Don't look for another signal in same alignment block for a few candles
    i += 3;
  }

  return signals;
}


/**
 * Compute SL and targets for triple EMA system.
 */
function computeTripleSLTargets(candles, entryIdx, entryPrice, direction, opts = {}) {
  const exitMethod = opts.exitMethod || CONFIG.DEFAULT_EXIT_METHOD;
  const isLong = direction === DIRECTION.LONG;
  const candle = candles[entryIdx];

  // SL below EMA26 (slowest)
  const ema26 = candle.ema26 || entryPrice;
  const buffer = ema26 * CONFIG.SL_EMA_BUFFER_PCT;
  let sl = isLong ? ema26 - buffer : ema26 + buffer;

  // Enforce max SL
  const slPct = Math.abs(entryPrice - sl) / entryPrice;
  if (slPct > CONFIG.MAX_SL_PCT) {
    sl = isLong
      ? entryPrice * (1 - CONFIG.MAX_SL_PCT)
      : entryPrice * (1 + CONFIG.MAX_SL_PCT);
  }

  const slDistance = Math.abs(entryPrice - sl);
  const targets = [];

  // T1: 1× risk (book 50%)
  const t1Price = isLong
    ? entryPrice + slDistance
    : entryPrice - slDistance;

  targets.push({ price: t1Price, qtyPct: CONFIG.TARGET_1_QTY_PCT, label: 'T1' });

  // Trail remaining with EMA13 (middle EMA — more appropriate for triple system)
  targets.push({ price: null, qtyPct: CONFIG.TRAIL_QTY_PCT, label: 'TRAIL_EMA13' });

  return { sl, targets, exitMethod };
}


// ═══════════════════════════════════════════════
// SECTION 3: TRADE SIMULATION
// ═══════════════════════════════════════════════

/**
 * Walk forward through candles managing a trade.
 * Handles: SL, partial exits, EMA trailing, reverse crossover exit,
 * MACD divergence exit, time stops.
 *
 * @param {object[]} candles
 * @param {object}   signal
 * @returns {object} trade record
 */
export function simulateTrade(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets, system } = signal;
  const exitMethod = signal.exitMethod || CONFIG.DEFAULT_EXIT_METHOD;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  let currentSL = sl;
  const partialExits = [];
  let t1Hit = false;
  let trailActive = false;

  // Determine which EMA to trail with
  const trailEMAField = system === EMA_SYSTEM.TRIPLE_5_13_26 ? 'ema13' : 'ema9';
  const fastEMAField = system === EMA_SYSTEM.TRIPLE_5_13_26 ? 'ema5' : 'ema9';
  const slowEMAField = system === EMA_SYSTEM.TRIPLE_5_13_26 ? 'ema13' : 'ema20';

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // ── EOD forced exit ──
    if (isPastMISExit(c)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_EOD,
      });
      remainingQty = 0;
      break;
    }

    // ── Stop-loss ──
    const hitSL = isLong ? c.low <= currentSL : c.high >= currentSL;
    if (hitSL) {
      partialExits.push({
        time: c.time, price: currentSL,
        qtyPct: remainingQty,
        reason: trailActive ? EXIT_REASON.TRAIL_EMA_CLOSE : EXIT_REASON.STOP_LOSS,
      });
      remainingQty = 0;
      break;
    }

    // ── Reverse crossover exit ──
    // If fast EMA crosses back against our direction, exit everything
    const fastEMA = c[fastEMAField];
    const slowEMA = c[slowEMAField];
    if (fastEMA != null && slowEMA != null) {
      const prevFast = candles[i - 1]?.[fastEMAField];
      const prevSlow = candles[i - 1]?.[slowEMAField];

      if (prevFast != null && prevSlow != null) {
        const reverseCross = isLong
          ? (prevFast >= prevSlow && fastEMA < slowEMA)   // bullish → bearish
          : (prevFast <= prevSlow && fastEMA > slowEMA);  // bearish → bullish

        if (reverseCross && trailActive) {
          partialExits.push({
            time: c.time, price: c.close,
            qtyPct: remainingQty, reason: EXIT_REASON.REVERSE_CROSSOVER,
          });
          remainingQty = 0;
          break;
        }
      }
    }

    // ── MACD divergence exit ──
    // If MACD histogram flips against our direction after T1
    if (trailActive && c.macdHist != null) {
      const prevHist = candles[i - 1]?.macdHist ?? 0;
      const macdFlipped = isLong
        ? (prevHist > 0 && c.macdHist < 0)
        : (prevHist < 0 && c.macdHist > 0);

      if (macdFlipped) {
        partialExits.push({
          time: c.time, price: c.close,
          qtyPct: remainingQty, reason: EXIT_REASON.MACD_DIVERGENCE,
        });
        remainingQty = 0;
        break;
      }
    }

    // ── Fixed R:R exit ──
    if (exitMethod === EXIT_METHOD.FIXED_RR && targets[0]?.price != null) {
      const hitTarget = isLong
        ? c.high >= targets[0].price
        : c.low <= targets[0].price;

      if (hitTarget) {
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: remainingQty, reason: EXIT_REASON.TARGET_FIXED_RR,
        });
        remainingQty = 0;
        break;
      }
    }

    // ── VWAP Band exit ──
    if (exitMethod === EXIT_METHOD.VWAP_BAND && targets[0]?.price != null) {
      const hitBand = isLong
        ? c.high >= targets[0].price
        : c.low <= targets[0].price;

      if (hitBand) {
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: remainingQty, reason: EXIT_REASON.VWAP_BAND_TARGET,
        });
        remainingQty = 0;
        break;
      }
    }

    // ── Target 1 (for trail method) ──
    if (exitMethod === EXIT_METHOD.TRAIL_FAST_EMA && !t1Hit && targets[0]?.price != null) {
      const hitT1 = isLong
        ? c.high >= targets[0].price
        : c.low <= targets[0].price;

      if (hitT1) {
        t1Hit = true;
        const exitQty = Math.min(targets[0].qtyPct, remainingQty);
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: exitQty, reason: EXIT_REASON.TARGET_FIXED_RR,
        });
        remainingQty -= exitQty;
        trailActive = true;

        // Move SL to breakeven
        currentSL = entryPrice;
      }
    }

    // ── EMA Trail ──
    if (trailActive && remainingQty > 0) {
      const trailEMA = candles[i - 1]?.[trailEMAField];
      if (trailEMA != null) {
        if (isLong) {
          const newSL = trailEMA - trailEMA * CONFIG.SL_EMA_BUFFER_PCT;
          if (newSL > currentSL) currentSL = newSL;
        } else {
          const newSL = trailEMA + trailEMA * CONFIG.SL_EMA_BUFFER_PCT;
          if (newSL < currentSL) currentSL = newSL;
        }
      }

      // Also check: candle closes on wrong side of trail EMA
      const trailEMACurrent = c[trailEMAField];
      if (trailEMACurrent != null) {
        const closedWrongSide = isLong
          ? c.close < trailEMACurrent
          : c.close > trailEMACurrent;

        if (closedWrongSide) {
          partialExits.push({
            time: c.time, price: c.close,
            qtyPct: remainingQty, reason: EXIT_REASON.TRAIL_EMA_CLOSE,
          });
          remainingQty = 0;
          break;
        }
      }
    }

    // ── Time stop: afternoon ──
    if (!t1Hit) {
      const t = candleTime(c);
      if (t >= toMinutes(CONFIG.LAST_ENTRY_H, CONFIG.LAST_ENTRY_M)) {
        partialExits.push({
          time: c.time, price: c.close,
          qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_AFTERNOON,
        });
        remainingQty = 0;
        break;
      }
    }
  }

  return buildTradeRecord(signal, partialExits);
}


// ═══════════════════════════════════════════════
// SECTION 4: TRADE RECORD BUILDER
// ═══════════════════════════════════════════════

function buildTradeRecord(signal, partialExits) {
  const { direction, entryPrice, system } = signal;
  const isLong = direction === DIRECTION.LONG;

  let weightedExit = 0;
  let totalQty = 0;

  for (const e of partialExits) {
    weightedExit += e.price * e.qtyPct;
    totalQty += e.qtyPct;
  }

  const avgExitPrice = totalQty > 0 ? weightedExit / totalQty : entryPrice;
  const grossPnlPct = isLong
    ? (avgExitPrice - entryPrice) / entryPrice
    : (entryPrice - avgExitPrice) / entryPrice;
  const netPnlPct = grossPnlPct - (CONFIG.SLIPPAGE_PER_SIDE * 2);

  const lastExit = partialExits[partialExits.length - 1];

  // Primary exit reason
  const reasonMap = {};
  for (const e of partialExits) {
    reasonMap[e.reason] = (reasonMap[e.reason] || 0) + e.qtyPct;
  }
  let primaryReason = 'NONE';
  let maxQty = 0;
  for (const [r, q] of Object.entries(reasonMap)) {
    if (q > maxQty) { maxQty = q; primaryReason = r; }
  }

  return {
    system,
    direction,
    entryPrice,
    avgExitPrice,
    entryTime: signal.entryTime,
    exitTime: lastExit?.time ?? null,
    grossPnlPct,
    netPnlPct,
    isWinner: netPnlPct > 0,
    partialExits,
    primaryExitReason: primaryReason,
    entryMethod: signal.entryMethod || null,
    exitMethod: signal.exitMethod || CONFIG.DEFAULT_EXIT_METHOD,

    // EMA-specific fields
    ema9AtEntry: signal.ema9AtEntry ?? null,
    ema20AtEntry: signal.ema20AtEntry ?? null,
    ema5AtEntry: signal.ema5AtEntry ?? null,
    ema13AtEntry: signal.ema13AtEntry ?? null,
    ema26AtEntry: signal.ema26AtEntry ?? null,
    macdHistAtEntry: signal.macdHistAtEntry ?? null,
    vwapAtEntry: signal.vwapAtEntry ?? null,
    alignment: signal.alignment ?? null,
    alignmentDuration: signal.alignmentDuration ?? null,
    crossoverIndex: signal.crossoverIndex ?? null,
    sl: signal.sl,
  };
}


// ═══════════════════════════════════════════════
// SECTION 5: SINGLE-DAY ORCHESTRATOR
// ═══════════════════════════════════════════════

/**
 * Run the EMA strategy on one day.
 *
 * @param {object[]} dayCandlesRaw
 * @param {object}   opts
 *   {
 *     system: EMA_SYSTEM,
 *     entryMethod: ENTRY_METHOD,
 *     exitMethod: EXIT_METHOD,
 *     maxTradesPerDay: number,
 *   }
 * @returns {object[]} trade records for the day
 */
export function runEMAForDay(dayCandlesRaw, opts = {}) {
  const {
    system = EMA_SYSTEM.DUAL_9_20,
    entryMethod = CONFIG.DEFAULT_ENTRY_METHOD,
    exitMethod = CONFIG.DEFAULT_EXIT_METHOD,
    maxTradesPerDay,
  } = opts;

  const maxTrades = maxTradesPerDay
    ?? (system === EMA_SYSTEM.DUAL_9_20 ? CONFIG.MAX_TRADES_PER_DAY_DUAL : CONFIG.MAX_TRADES_PER_DAY_TRIPLE);

  // Deep copy + parse dates
  const candles = dayCandlesRaw.map(c => ({
    ...c,
    time: c.time instanceof Date ? c.time : new Date(c.time),
  }));

  if (candles.length < CONFIG.TRIPLE_SLOW_PERIOD + 5) return [];

  // Compute all indicators
  computeAllIndicators(candles, system === EMA_SYSTEM.DUAL_9_20 ? 'DUAL_9_20' : 'TRIPLE_5_13_26');

  // Detect signals
  let allSignals = [];

  if (system === EMA_SYSTEM.DUAL_9_20) {
    allSignals = detectDualCrossoverSignals(candles, { entryMethod, exitMethod });
  } else {
    allSignals = detectTriplePullbackSignals(candles, { exitMethod });
  }

  // Sort by entry time
  allSignals.sort((a, b) => a.entryIndex - b.entryIndex);

  // Simulate trades, prevent overlap
  const trades = [];
  let lastExitIndex = -1;

  for (const signal of allSignals) {
    if (trades.length >= maxTrades) break;
    if (signal.entryIndex <= lastExitIndex) continue;

    const trade = simulateTrade(candles, signal);
    trades.push(trade);

    // Find exit index
    const lastPartial = trade.partialExits[trade.partialExits.length - 1];
    if (lastPartial) {
      const exitTime = lastPartial.time instanceof Date
        ? lastPartial.time : new Date(lastPartial.time);
      for (let j = signal.entryIndex; j < candles.length; j++) {
        const ct = candles[j].time instanceof Date ? candles[j].time : new Date(candles[j].time);
        if (ct >= exitTime) { lastExitIndex = j; break; }
      }
    }
  }

  return trades;
}
```

```javascript
/**
 * EMA Strategy Analyzer
 *
 * System-specific analytics:
 *  - 9/20 vs Triple performance comparison
 *  - Entry method comparison (crossover close vs pullback)
 *  - Exit method comparison (fixed R:R vs EMA trail vs VWAP band)
 *  - Filter effectiveness analysis
 *  - Crossover quality metrics (spread, slope at entry)
 *  - MACD histogram strength correlation

import { CONFIG, EMA_SYSTEM, DIRECTION, EXIT_METHOD, ENTRY_METHOD, EXIT_REASON } from './ema-config.js';

// ═══════════════════════════════════════════════
// CORE PERFORMANCE
// ═══════════════════════════════════════════════

export function analyzePerformance(trades, capital = CONFIG.DEFAULT_CAPITAL) {
  if (trades.length === 0) return { error: 'No trades' };

  const winners = trades.filter(t => t.isWinner);
  const losers = trades.filter(t => !t.isWinner);
  const pnls = trades.map(t => t.netPnlPct);

  const totalTrades = trades.length;
  const winRate = winners.length / totalTrades;
  const avgWin = safeMean(winners.map(t => t.netPnlPct));
  const avgLoss = safeMean(losers.map(t => t.netPnlPct));
  const avgWLR = Math.abs(avgLoss) > 0 ? avgWin / Math.abs(avgLoss) : Infinity;
  const expectancy = (winRate * avgWin) + ((1 - winRate) * avgLoss);

  const equity = buildEquityCurve(trades, capital);
  const totalReturn = ((equity[equity.length - 1].equity - capital) / capital) * 100;
  const maxDD = computeMaxDrawdown(equity);
  const streaks = computeStreaks(trades);

  return {
    totalTrades,
    winners: winners.length,
    losers: losers.length,
    winRatePct: r2(winRate * 100),
    avgWinPct: r4(avgWin * 100),
    avgLossPct: r4(avgLoss * 100),
    avgWinLossRatio: r2(avgWLR),
    expectancyPct: r4(expectancy * 100),
    totalReturnPct: r2(totalReturn),
    totalReturnAbs: r2(equity[equity.length - 1].equity - capital),
    maxDrawdownPct: r2(maxDD.pct),
    maxDrawdownDuration: maxDD.duration,
    sharpeRatio: r2(computeSharpe(pnls)),
    profitFactor: r2(computeProfitFactor(trades)),
    maxWinStreak: streaks.maxWinStreak,
    maxLossStreak: streaks.maxLossStreak,
    equityCurve: equity,
  };
}

// SYSTEM BREAKDOWN (9/20 vs Triple)

export function systemBreakdown(trades) {
  const bySystem = {};

  for (const t of trades) {
    const key = t.system || 'UNKNOWN';
    if (!bySystem[key]) bySystem[key] = [];
    bySystem[key].push(t);
  }

  const result = {};
  for (const [sys, sTrades] of Object.entries(bySystem)) {
    if (sTrades.length === 0) { result[sys] = { count: 0 }; continue; }
    const wins = sTrades.filter(t => t.isWinner).length;
    const avgWin = safeMean(sTrades.filter(t => t.isWinner).map(t => t.netPnlPct));
    const avgLoss = safeMean(sTrades.filter(t => !t.isWinner).map(t => t.netPnlPct));

    result[sys] = {
      count: sTrades.length,
      winRatePct: r2((wins / sTrades.length) * 100),
      avgWinPct: r4(avgWin * 100),
      avgLossPct: r4(avgLoss * 100),
      avgPnlPct: r4(safeMean(sTrades.map(t => t.netPnlPct)) * 100),
      winLossRatio: Math.abs(avgLoss) > 0 ? r2(avgWin / Math.abs(avgLoss)) : 'INF',
      tradesPerDay: r2(sTrades.length / new Set(sTrades.map(t => t.date)).size),
    };
  }

  return result;
}

    const emaVal = t[emaKey];

    const slowKey = emaKey;

  if (extras.alignmentDuration) {
    console.log(`\n  📏 ALIGNMENT DURATION (Triple EMA)`);
    for (const [b, data] of Object.entries(extras.alignmentDuration)) {
      if (data.count === 0) continue;
      console.log(`  ${b.padEnd(28)} | N=${data.count} | WR: ${data.winRatePct}% | Avg: ${data.avgPnlPct}%`);
    }
  }

  if (extras.timeOfDay) {
    console.log(`\n  🕐 TIME-OF-DAY (30-min slots)`);
    for (const [slot, data] of Object.entries(extras.timeOfDay)) {
      console.log(`  ${slot.padEnd(8)} | N=${String(data.count).padStart(3)} | WR: ${data.winRatePct}% | Avg: ${data.avgPnlPct}%`);
    }
  }

  if (extras.filterEffectiveness) {
    console.log(`\n  🔧 FILTER EFFECTIVENESS (EMA spread at entry)`);
    for (const [b, data] of Object.entries(extras.filterEffectiveness)) {
      if (data.count === 0) continue;
      console.log(`  ${b.padEnd(18)} | N=${data.count} | WR: ${data.winRatePct}% | Avg: ${data.avgPnlPct}%`);
    }
  }

  console.log(`\n${sep}\n`);
}

// INTERNALS

function buildEquityCurve(trades, startCapital) {
  const curve = [{ tradeNum: 0, equity: startCapital }];
  let equity = startCapital;
  for (let i = 0; i < trades.length; i++) {
    const t = trades[i];
    const riskAmt = equity * CONFIG.DEFAULT_RISK_PER_TRADE;
    const slDist = Math.abs(t.entryPrice - t.sl) || t.entryPrice * 0.005;
    const qty = Math.floor(riskAmt / slDist);
    const posVal = qty * t.entryPrice;
    const pnl = posVal * t.netPnlPct;
    equity += pnl - CONFIG.COMMISSION_PER_SIDE * 2;
    curve.push({ tradeNum: i + 1, equity: r2(equity), date: t.entryTime, system: t.system });
  }
  return curve;
}

function computeMaxDrawdown(curve) {
  let peak = curve[0].equity, maxDD = 0, ddStart = 0, maxDur = 0;
  for (let i = 1; i < curve.length; i++) {
    if (curve[i].equity > peak) { peak = curve[i].equity; ddStart = i; }
    const dd = (peak - curve[i].equity) / peak;
    if (dd > maxDD) { maxDD = dd; maxDur = i - ddStart; }
  }
  return { pct: maxDD * 100, duration: maxDur };
}

function computeStreaks(trades) {
  let mw = 0, ml = 0, cw = 0, cl = 0;
  for (const t of trades) {
    if (t.isWinner) { cw++; cl = 0; if (cw > mw) mw = cw; }
    else { cl++; cw = 0; if (cl > ml) ml = cl; }
  }
  return { maxWinStreak: mw, maxLossStreak: ml };
}

function computeSharpe(pnls) {
  if (pnls.length < 2) return 0;
  const avg = safeMean(pnls);
  const std = Math.sqrt(pnls.reduce((s, p) => s + (p - avg) ** 2, 0) / (pnls.length - 1));
  return std === 0 ? 0 : (avg / std) * Math.sqrt(250);
}

function computeProfitFactor(trades) {
  let gp = 0, gl = 0;
  for (const t of trades) {
    if (t.netPnlPct > 0) gp += t.netPnlPct; else gl += Math.abs(t.netPnlPct);
  }
  return gl > 0 ? gp / gl : Infinity;
}

function safeMean(arr) { return arr.length === 0 ? 0 : arr.reduce((a, b) => a + b, 0) / arr.length; }
function r2(n) { return Math.round(n * 100) / 100; }
function r4(n) { return Math.round(n * 10000) / 10000; }
```

```javascript
// GROUP CANDLES BY DAY
// ═══════════════════════════════════════════════

export function groupByDay(candles) {
  const days = new Map();
  for (const c of candles) {
    const d = c.time instanceof Date ? c.time : new Date(c.time);
    const key = d.toISOString().slice(0, 10);
    if (!days.has(key)) days.set(key, []);
    days.get(key).push(c);
  }
  return days;
}

/**
 * @param {object[]} allCandles
 * @param {object}   opts
 *   { system, entryMethod, exitMethod, capital, maxTradesPerDay }
 */
export function runFullBacktest(allCandles, opts = {}) {
  const {
    system = EMA_SYSTEM.DUAL_9_20,
    entryMethod = CONFIG.DEFAULT_ENTRY_METHOD,
    exitMethod = CONFIG.DEFAULT_EXIT_METHOD,
    capital = CONFIG.DEFAULT_CAPITAL,
    maxTradesPerDay,
  } = opts;

  const dayGroups = groupByDay(allCandles);
  const sortedDates = [...dayGroups.keys()].sort();

  const allTrades = [];
  const skippedDays = [];

  for (const dateKey of sortedDates) {
    const dayCandles = dayGroups.get(dateKey);

    if (dayCandles.length < 30) {
      skippedDays.push({ date: dateKey, reason: 'Insufficient candles' });
      continue;
    }

    const dayTrades = runEMAForDay(dayCandles, {
      system, entryMethod, exitMethod, maxTradesPerDay,
    });

    for (const t of dayTrades) {
      t.date = dateKey;
      allTrades.push(t);
    }

    if (dayTrades.length === 0) {
      skippedDays.push({ date: dateKey, reason: 'No valid signal' });
    }
  }

  const report = allTrades.length > 0 ? analyzePerformance(allTrades, capital) : null;

  return {
    trades: allTrades,
    report,
    totalDays: sortedDates.length,
    tradedDays: new Set(allTrades.map(t => t.date)).size,
    totalSignals: allTrades.length,
    skippedDays,
    config: { system, entryMethod, exitMethod },
  };
}

    const res = runFullBacktest(candles, opts);

    results.push({
      symbol,
      trades: res.report?.totalTrades ?? 0,
      winRatePct: res.report?.winRatePct ?? 0,
      expectancyPct: res.report?.expectancyPct ?? 0,
      returnPct: res.report?.totalReturnPct ?? 0,
      maxDDPct: res.report?.maxDrawdownPct ?? 0,
      profitFactor: res.report?.profitFactor ?? 0,
      sharpe: res.report?.sharpeRatio ?? 0,
      tradesPerDay: res.tradedDays > 0
        ? r2(res.totalSignals / res.tradedDays)
        : 0,
    });
  }

  results.sort((a, b) => (b.expectancyPct || 0) - (a.expectancyPct || 0));
  return results;
}

  console.log(`Days: ${result.totalDays} | Traded: ${result.tradedDays} | Signals: ${result.totalSignals}`);

  return { ...result, extras };
}
```

The key architectural distinction of this strategy is the **filter cascade against false signals**. Unlike ORB (single breakout event) or Gap Trading (single classification), EMA crossovers fire constantly in range-bound markets, producing whipsaws. The tangling detector (counting micro-crossovers in a sliding window), slope filter (rejecting flat EMAs), and MACD confirmation gate work together to suppress the ~40% of crossovers that would be losers in choppy conditions. The triple EMA system takes this further by requiring full alignment before even looking for entries, which is why it shows fewer signals but higher quality in the performance table.
