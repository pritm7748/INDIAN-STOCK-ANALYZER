## `gap-classifier.js`

```javascript
/**
 * Gap Classification Engine
 *
 * Determines gap type, size, direction, and whether it's catalyst-driven.
 * This is the foundation — every sub-strategy starts by classifying the gap.
 */

import {
  GAP_TYPE, GAP_SIZE, DIRECTION, CATALYST_TYPE, CONFIG,
} from './gap-config.js';


// ═══════════════════════════════════════════════
// SECTION 1: GAP TYPE CLASSIFICATION
// ═══════════════════════════════════════════════

/**
 * Classify the gap based on today's open vs yesterday's OHLC.
 *
 * @param {number} todayOpen
 * @param {object} prevDay – { open, high, low, close }
 * @returns {object}
 *   {
 *     type: GAP_TYPE,
 *     direction: DIRECTION,
 *     gapSize: GAP_SIZE,
 *     gapPct: number,         // signed percentage (+ for up, - for down)
 *     gapAbsolute: number,    // absolute point difference
 *     gapFromClose: number,   // today open - prev close
 *     gapFromRange: number,   // today open - prev high (for full gap up)
 *     prevClose: number,
 *     prevHigh: number,
 *     prevLow: number,
 *     gapFillTarget: number,  // price to fill the gap
 *   }
 */
export function classifyGap(todayOpen, prevDay) {
  const { high: prevHigh, low: prevLow, close: prevClose } = prevDay;

  const gapFromClose = todayOpen - prevClose;
  const gapPct = gapFromClose / prevClose;
  const absGapPct = Math.abs(gapPct);

  let type = GAP_TYPE.NO_GAP;
  let direction = DIRECTION.NONE;
  let gapFillTarget = prevClose;
  let gapFromRange = 0;

  // ── Full Gap Up: open above yesterday's HIGH ──
  if (todayOpen > prevHigh) {
    type = GAP_TYPE.FULL_GAP_UP;
    direction = DIRECTION.LONG;
    gapFromRange = todayOpen - prevHigh;
    gapFillTarget = prevHigh; // for full gap, filling means reaching prev high
  }
  // ── Full Gap Down: open below yesterday's LOW ──
  else if (todayOpen < prevLow) {
    type = GAP_TYPE.FULL_GAP_DOWN;
    direction = DIRECTION.SHORT;
    gapFromRange = prevLow - todayOpen;
    gapFillTarget = prevLow;
  }
  // ── Partial Gap Up: open > prev close but ≤ prev high ──
  else if (todayOpen > prevClose && absGapPct >= CONFIG.GAP_MIN_THRESHOLD) {
    type = GAP_TYPE.PARTIAL_GAP_UP;
    direction = DIRECTION.LONG;
    gapFillTarget = prevClose;
  }
  // ── Partial Gap Down: open < prev close but ≥ prev low ──
  else if (todayOpen < prevClose && absGapPct >= CONFIG.GAP_MIN_THRESHOLD) {
    type = GAP_TYPE.PARTIAL_GAP_DOWN;
    direction = DIRECTION.SHORT;
    gapFillTarget = prevClose;
  }

  // ── Size classification ──
  const gapSize = classifyGapSize(absGapPct);

  return {
    type,
    direction,
    gapSize,
    gapPct: Math.round(gapPct * 10000) / 10000,    // 4 decimal places
    gapPctAbs: Math.round(absGapPct * 10000) / 10000,
    gapAbsolute: Math.round(Math.abs(gapFromClose) * 100) / 100,
    gapFromClose,
    gapFromRange: Math.round(gapFromRange * 100) / 100,
    prevClose,
    prevHigh,
    prevLow,
    todayOpen,
    gapFillTarget,
    isGapUp: gapFromClose > 0,
    isGapDown: gapFromClose < 0,
    isFullGap: type === GAP_TYPE.FULL_GAP_UP || type === GAP_TYPE.FULL_GAP_DOWN,
    isPartialGap: type === GAP_TYPE.PARTIAL_GAP_UP || type === GAP_TYPE.PARTIAL_GAP_DOWN,
  };
}


// ═══════════════════════════════════════════════
// SECTION 2: GAP SIZE CLASSIFICATION
// ═══════════════════════════════════════════════

/**
 * @param {number} absGapPct – absolute gap as fraction (e.g., 0.012 = 1.2%)
 * @returns {string} GAP_SIZE enum value
 */
export function classifyGapSize(absGapPct) {
  if (absGapPct < CONFIG.GAP_MIN_THRESHOLD) return GAP_SIZE.NONE;
  if (absGapPct < CONFIG.GAP_SMALL_MAX)     return GAP_SIZE.SMALL;
  if (absGapPct < CONFIG.GAP_MEDIUM_MAX)    return GAP_SIZE.MEDIUM;
  if (absGapPct < CONFIG.GAP_LARGE_MAX)     return GAP_SIZE.LARGE;
  return GAP_SIZE.HUGE;
}


// ═══════════════════════════════════════════════
// SECTION 3: CATALYST SCORING
// ═══════════════════════════════════════════════

/**
 * Score the catalyst strength for a gap.
 *
 * In live trading, catalyst data comes from news feeds.
 * In backtesting, it can be:
 *   1. Manually tagged per day (ideal)
 *   2. Inferred from volume + gap size heuristics (proxy)
 *
 * This function provides BOTH approaches.
 *
 * @param {object} meta – day-level metadata
 *   {
 *     catalyst?: string,          // CATALYST_TYPE if manually tagged
 *     gapClassification: object,  // from classifyGap()
 *     first15Volume: number,      // total volume in first 15 minutes
 *     avgFirst15Volume: number,   // average first-15-min volume (20-day)
 *     preMarketBias?: string,     // 'BULLISH' | 'BEARISH' | 'NEUTRAL'
 *   }
 * @returns {object}
 *   {
 *     catalystType: string,
 *     hasCatalyst: boolean,
 *     catalystScore: number,    // 0 (none) to 100 (strongest)
 *     inference: string,        // how we determined it
 *   }
 */
export function scoreCatalyst(meta) {
  const { catalyst, gapClassification, first15Volume, avgFirst15Volume } = meta;

  // ── If manually tagged ──
  if (catalyst && catalyst !== CATALYST_TYPE.UNKNOWN) {
    const manualScores = {
      [CATALYST_TYPE.EARNINGS_BEAT]: 90,
      [CATALYST_TYPE.EARNINGS_MISS]: 85,
      [CATALYST_TYPE.UPGRADE]: 75,
      [CATALYST_TYPE.DOWNGRADE]: 75,
      [CATALYST_TYPE.ORDER_WIN]: 70,
      [CATALYST_TYPE.SECTOR_NEWS]: 50,
      [CATALYST_TYPE.GLOBAL_MACRO]: 40,
      [CATALYST_TYPE.NO_CATALYST]: 0,
    };

    return {
      catalystType: catalyst,
      hasCatalyst: catalyst !== CATALYST_TYPE.NO_CATALYST,
      catalystScore: manualScores[catalyst] ?? 30,
      inference: 'MANUAL_TAG',
    };
  }

  // ── Infer from market data (proxy when no news feed) ──
  const gap = gapClassification;
  let score = 0;
  const signals = [];

  // Large/huge gap = likely catalyst-driven
  if (gap.gapSize === GAP_SIZE.LARGE) { score += 40; signals.push('large_gap'); }
  if (gap.gapSize === GAP_SIZE.HUGE) { score += 60; signals.push('huge_gap'); }

  // Full gap (beyond prev day range) = stronger conviction
  if (gap.isFullGap) { score += 20; signals.push('full_gap'); }

  // First-15-min volume surge = real buying/selling pressure
  if (avgFirst15Volume > 0) {
    const volRatio = first15Volume / avgFirst15Volume;
    if (volRatio >= 3.0) { score += 30; signals.push(`vol_surge_${volRatio.toFixed(1)}x`); }
    else if (volRatio >= 2.0) { score += 15; signals.push(`vol_elevated_${volRatio.toFixed(1)}x`); }
    else if (volRatio < 1.0) { score -= 10; signals.push('vol_below_avg'); }
  }

  // Cap
  score = Math.max(0, Math.min(100, score));
  const hasCatalyst = score >= 50;

  return {
    catalystType: hasCatalyst ? CATALYST_TYPE.UNKNOWN : CATALYST_TYPE.NO_CATALYST,
    hasCatalyst,
    catalystScore: score,
    inference: `HEURISTIC: ${signals.join(', ')}`,
  };
}


// ═══════════════════════════════════════════════
// SECTION 4: STRATEGY ELIGIBILITY
// ═══════════════════════════════════════════════

/**
 * Determine which gap sub-strategy (if any) applies to this gap.
 *
 * @param {object} gapClass     – from classifyGap()
 * @param {object} catalystInfo – from scoreCatalyst()
 * @returns {object}
 *   { eligible: string[], primary: string | null, reasons: string[] }
 */
export function determineEligibleStrategies(gapClass, catalystInfo) {
  const eligible = [];
  const reasons = [];

  if (gapClass.type === GAP_TYPE.NO_GAP) {
    return { eligible: [], primary: null, reasons: ['No gap detected'] };
  }

  // ── Strategy A: Gap and Go ──
  if (gapClass.gapPctAbs >= CONFIG.GAG_MIN_GAP_PCT && catalystInfo.hasCatalyst) {
    eligible.push(GAP_STRATEGY.GAP_AND_GO);
    reasons.push(`Gap ${(gapClass.gapPctAbs * 100).toFixed(2)}% >= ${CONFIG.GAG_MIN_GAP_PCT * 100}% with catalyst (score: ${catalystInfo.catalystScore})`);
  }

  // ── Strategy B: Gap Fill ──
  if (
    gapClass.gapPctAbs >= CONFIG.GF_MIN_GAP_PCT &&
    gapClass.gapPctAbs <= CONFIG.GF_MAX_GAP_PCT &&
    !catalystInfo.hasCatalyst
  ) {
    eligible.push(GAP_STRATEGY.GAP_FILL);
    reasons.push(`Gap ${(gapClass.gapPctAbs * 100).toFixed(2)}% in fill range [${CONFIG.GF_MIN_GAP_PCT * 100}%-${CONFIG.GF_MAX_GAP_PCT * 100}%] without catalyst`);
  }

  // Both can also apply if gap is in medium range with weak catalyst
  if (
    gapClass.gapPctAbs >= CONFIG.GF_MIN_GAP_PCT &&
    gapClass.gapPctAbs <= CONFIG.GF_MAX_GAP_PCT &&
    catalystInfo.catalystScore > 0 &&
    catalystInfo.catalystScore < 50
  ) {
    if (!eligible.includes(GAP_STRATEGY.GAP_FILL)) {
      eligible.push(GAP_STRATEGY.GAP_FILL);
      reasons.push('Weak catalyst — gap fill still possible');
    }
  }

  const primary = eligible.length > 0 ? eligible[0] : null;

  return { eligible, primary, reasons };
}


// ═══════════════════════════════════════════════
// SECTION 5: INDEX GAP CLASSIFICATION
// ═══════════════════════════════════════════════

/**
 * Special classifier for Nifty / Bank Nifty index gaps.
 * Works with points rather than percentages.
 *
 * @param {number} todayOpen   – index open value
 * @param {number} prevClose   – index previous close
 * @param {object} contextMeta – { hasGlobalCatalyst, vixChange, giftNiftyDirection }
 * @returns {object}
 */
export function classifyIndexGap(todayOpen, prevClose, contextMeta = {}) {
  const gapPoints = todayOpen - prevClose;
  const absGapPoints = Math.abs(gapPoints);
  const gapPct = gapPoints / prevClose;
  const isUp = gapPoints > 0;

  const meetsThreshold = absGapPoints >= CONFIG.IG_MIN_GAP_POINTS;

  const { hasGlobalCatalyst = false, vixChange = 0 } = contextMeta;

  // Determine if gap-fill or gap-and-go more likely
  let strategy = null;
  let fillProbability = 0;
  const rationale = [];

  if (meetsThreshold) {
    if (!hasGlobalCatalyst) {
      strategy = GAP_STRATEGY.GAP_FILL;
      fillProbability = CONFIG.NIFTY_GAP_FILL_PROBABILITY;
      rationale.push(`No global catalyst, base fill prob: ${fillProbability * 100}%`);

      // VIX falling during gap-down increases fill probability
      if (!isUp && vixChange < 0) {
        fillProbability += 0.05;
        rationale.push('VIX falling on gap-down → higher fill prob');
      }
    } else {
      strategy = GAP_STRATEGY.GAP_AND_GO;
      fillProbability = 1 - CONFIG.NIFTY_GAP_FILL_PROBABILITY;
      rationale.push('Global catalyst present → continuation likely');
    }
  }

  return {
    gapPoints,
    absGapPoints,
    gapPct: Math.round(gapPct * 10000) / 10000,
    isUp,
    meetsThreshold,
    strategy,
    fillProbability: Math.round(fillProbability * 100) / 100,
    rationale,
    todayOpen,
    prevClose,
    gapFillTarget: prevClose,
  };
}
```

## `gap-filters.js`

```javascript
/**
 * Gap Strategy Filters
 *
 * Pre-trade validation for each sub-strategy.
 * Works with 5-min candle data from the trading day.
 */

import { CONFIG, DIRECTION, GAP_SIZE, GAP_STRATEGY } from './gap-config.js';

// ════════════════════════════════════════════════
// TIME HELPERS
// ════════════════════════════════════════════════

export function toMinutes(h, m) { return h * 60 + m; }

export function candleTime(candle) {
  const d = candle.time instanceof Date ? candle.time : new Date(candle.time);
  return toMinutes(d.getHours(), d.getMinutes());
}

export function isInORBWindow(candle) {
  const t = candleTime(candle);
  const start = toMinutes(CONFIG.MARKET_OPEN_H, CONFIG.MARKET_OPEN_M);
  const end = toMinutes(CONFIG.ORB_END_H, CONFIG.ORB_END_M);
  return t >= start && t < end;
}

export function isPostORB(candle) {
  return candleTime(candle) >= toMinutes(CONFIG.ORB_END_H, CONFIG.ORB_END_M);
}

export function isPastTimeStop(candle, h, m) {
  return candleTime(candle) >= toMinutes(h, m);
}


// ════════════════════════════════════════════════
// ORB RANGE BUILDER (for first 15 minutes)
// ════════════════════════════════════════════════

/**
 * Build the opening range from first N minutes of candles.
 *
 * @param {object[]} candles – day's candles sorted by time
 * @returns {object|null}
 *   { orbHigh, orbLow, orbRange, orbCandles, orbVolume, openPrice }
 */
export function buildORBRange(candles) {
  const orbCandles = candles.filter(c => isInORBWindow(c));
  if (orbCandles.length === 0) return null;

  let orbHigh = -Infinity;
  let orbLow = Infinity;
  let orbVolume = 0;

  for (const c of orbCandles) {
    if (c.high > orbHigh) orbHigh = c.high;
    if (c.low < orbLow) orbLow = c.low;
    orbVolume += c.volume;
  }

  return {
    orbHigh,
    orbLow,
    orbRange: orbHigh - orbLow,
    orbCandles,
    orbVolume,
    openPrice: orbCandles[0].open,
  };
}


// ════════════════════════════════════════════════
// RSI COMPUTATION (for Gap Fill)
// ════════════════════════════════════════════════

/**
 * Compute RSI on candle closes.
 * Mutates candles (sets candle.rsi).
 *
 * @param {object[]} candles
 * @param {number}   period  – default CONFIG.GF_RSI_PERIOD (14)
 */
export function computeRSI(candles, period = CONFIG.GF_RSI_PERIOD) {
  if (candles.length < period + 1) {
    candles.forEach(c => { c.rsi = 50; }); // neutral default
    return;
  }

  let avgGain = 0;
  let avgLoss = 0;

  // Initial SMA of gains/losses
  for (let i = 1; i <= period; i++) {
    const change = candles[i].close - candles[i - 1].close;
    if (change > 0) avgGain += change;
    else avgLoss += Math.abs(change);
  }
  avgGain /= period;
  avgLoss /= period;

  // Set RSI for first period candles as unavailable
  for (let i = 0; i <= period; i++) {
    candles[i].rsi = avgLoss === 0 ? 100 : 100 - (100 / (1 + avgGain / avgLoss));
  }

  // Smoothed RSI for remaining candles
  for (let i = period + 1; i < candles.length; i++) {
    const change = candles[i].close - candles[i - 1].close;
    const gain = change > 0 ? change : 0;
    const loss = change < 0 ? Math.abs(change) : 0;

    avgGain = ((avgGain * (period - 1)) + gain) / period;
    avgLoss = ((avgLoss * (period - 1)) + loss) / period;

    candles[i].rsi = avgLoss === 0 ? 100 : 100 - (100 / (1 + avgGain / avgLoss));
  }
}


// ════════════════════════════════════════════════
// VWAP COMPUTATION
// ════════════════════════════════════════════════

export function computeVWAP(candles) {
  let cumTPV = 0;
  let cumVol = 0;

  for (const c of candles) {
    const tp = (c.high + c.low + c.close) / 3;
    cumTPV += tp * c.volume;
    cumVol += c.volume;
    c.vwap = cumVol > 0 ? cumTPV / cumVol : c.close;
  }
}


// ════════════════════════════════════════════════
// FILTER: GAP AND GO (Strategy A)
// ════════════════════════════════════════════════

/**
 * Validate Gap and Go entry conditions.
 *
 * @param {object} ctx
 *   {
 *     gapClassification,  // from classifyGap()
 *     catalystInfo,       // from scoreCatalyst()
 *     orb,                // from buildORBRange()
 *     avgFirst15Volume,   // historical average first-15-min volume
 *     breakoutCandle,     // the candle that broke ORB high/low
 *   }
 * @returns {{ pass: boolean, filters: object[] }}
 */
export function filterGapAndGo(ctx) {
  const { gapClassification: gap, catalystInfo, orb, avgFirst15Volume, breakoutCandle } = ctx;
  const results = [];

  // 1. Gap size must be ≥ 1.5%
  results.push({
    name: 'GapSize',
    pass: gap.gapPctAbs >= CONFIG.GAG_MIN_GAP_PCT,
    value: `${(gap.gapPctAbs * 100).toFixed(2)}%`,
    required: `>= ${CONFIG.GAG_MIN_GAP_PCT * 100}%`,
  });

  // 2. Must have catalyst
  results.push({
    name: 'Catalyst',
    pass: catalystInfo.hasCatalyst,
    score: catalystInfo.catalystScore,
    type: catalystInfo.catalystType,
  });

  // 3. Price must hold above opening price in first 15 min
  // (for gap-up: ORB low should be >= today's open)
  if (CONFIG.GAG_MUST_HOLD_ABOVE_OPEN && orb) {
    const holdAboveOpen = gap.isGapUp
      ? orb.orbLow >= gap.todayOpen * 0.998  // allow 0.2% dip tolerance
      : orb.orbHigh <= gap.todayOpen * 1.002;

    results.push({
      name: 'HoldsAboveOpen',
      pass: holdAboveOpen,
      orbLow: orb.orbLow,
      openPrice: gap.todayOpen,
    });
  }

  // 4. Volume in first 15 min > 3× average
  if (orb && avgFirst15Volume > 0) {
    const volRatio = orb.orbVolume / avgFirst15Volume;
    results.push({
      name: 'First15Volume',
      pass: volRatio >= CONFIG.GAG_FIRST_15_VOLUME_MULTIPLE,
      ratio: Math.round(volRatio * 100) / 100,
      required: `>= ${CONFIG.GAG_FIRST_15_VOLUME_MULTIPLE}×`,
    });
  }

  // 5. Breakout candle present
  if (breakoutCandle) {
    const breakoutValid = gap.isGapUp
      ? breakoutCandle.close > orb.orbHigh
      : breakoutCandle.close < orb.orbLow;

    results.push({
      name: 'BreakoutConfirm',
      pass: breakoutValid,
    });
  }

  const allPassed = results.every(r => r.pass);
  return { pass: allPassed, filters: results };
}


// ════════════════════════════════════════════════
// FILTER: GAP FILL (Strategy B)
// ════════════════════════════════════════════════

/**
 * Validate Gap Fill entry conditions.
 *
 * @param {object} ctx
 *   {
 *     gapClassification,
 *     catalystInfo,
 *     orb,
 *     avgFirst15Volume,
 *     weaknessCandle,   // candle showing weakness (below open / bearish)
 *     rsiAtEntry,       // RSI value at the entry candle
 *     vwapAtEntry,      // VWAP value at entry candle
 *     entryPrice,       // proposed entry price
 *   }
 */
export function filterGapFill(ctx) {
  const { gapClassification: gap, catalystInfo, orb, avgFirst15Volume, rsiAtEntry, entryPrice } = ctx;
  const results = [];

  // 1. Gap size in fill range (0.3% – 1.5%)
  results.push({
    name: 'GapSizeInRange',
    pass: gap.gapPctAbs >= CONFIG.GF_MIN_GAP_PCT && gap.gapPctAbs <= CONFIG.GF_MAX_GAP_PCT,
    value: `${(gap.gapPctAbs * 100).toFixed(2)}%`,
    range: `${CONFIG.GF_MIN_GAP_PCT * 100}% – ${CONFIG.GF_MAX_GAP_PCT * 100}%`,
  });

  // 2. No strong catalyst
  if (CONFIG.GF_REQUIRES_NO_CATALYST) {
    results.push({
      name: 'NoCatalyst',
      pass: !catalystInfo.hasCatalyst,
      catalystScore: catalystInfo.catalystScore,
    });
  }

  // 3. Price shows weakness in first 15 min
  if (CONFIG.GF_BELOW_OPEN_IN_ORB && orb) {
    const showsWeakness = gap.isGapUp
      ? orb.orbCandles.some(c => c.close < gap.todayOpen)   // drops below open
      : orb.orbCandles.some(c => c.close > gap.todayOpen);  // rises above open

    results.push({
      name: 'FirstPeriodWeakness',
      pass: showsWeakness,
    });
  }

  // 4. Volume is average or below (no conviction behind gap)
  if (CONFIG.GF_VOLUME_BELOW_AVERAGE && orb && avgFirst15Volume > 0) {
    const volRatio = orb.orbVolume / avgFirst15Volume;
    results.push({
      name: 'LowConvictionVolume',
      pass: volRatio <= 1.5, // not a volume surge
      ratio: Math.round(volRatio * 100) / 100,
    });
  }

  // 5. RSI overbought/oversold
  if (rsiAtEntry != null) {
    const rsiValid = gap.isGapUp
      ? rsiAtEntry >= CONFIG.GF_RSI_OVERBOUGHT
      : rsiAtEntry <= CONFIG.GF_RSI_OVERSOLD;

    results.push({
      name: 'RSIExtreme',
      pass: rsiValid,
      rsi: Math.round(rsiAtEntry * 100) / 100,
      threshold: gap.isGapUp ? `>= ${CONFIG.GF_RSI_OVERBOUGHT}` : `<= ${CONFIG.GF_RSI_OVERSOLD}`,
    });
  }

  // 6. Entry trigger (below VWAP or below ORB low)
  if (CONFIG.GF_ENTRY_TRIGGER === 'BELOW_VWAP' && ctx.vwapAtEntry != null) {
    const triggerMet = gap.isGapUp
      ? entryPrice < ctx.vwapAtEntry
      : entryPrice > ctx.vwapAtEntry;

    results.push({
      name: 'EntryTrigger_VWAP',
      pass: triggerMet,
      entryPrice,
      vwap: ctx.vwapAtEntry,
    });
  } else if (CONFIG.GF_ENTRY_TRIGGER === 'BELOW_ORB_LOW' && orb) {
    const triggerMet = gap.isGapUp
      ? entryPrice < orb.orbLow
      : entryPrice > orb.orbHigh;

    results.push({
      name: 'EntryTrigger_ORB',
      pass: triggerMet,
      entryPrice,
      orbLevel: gap.isGapUp ? orb.orbLow : orb.orbHigh,
    });
  }

  const allPassed = results.every(r => r.pass);
  return { pass: allPassed, filters: results };
}
```

## `gap-core.js`

```javascript
/**
 * Gap Trading Core Engine
 *
 * Three complete sub-strategies:
 *   A. Gap and Go — ride large catalyst-driven gaps
 *   B. Gap Fill — fade small/medium gaps without catalysts
 *   C. Index Gap — Nifty/BankNifty options on statistical gap fills
 *
 * Each has:
 *   1. Signal detection
 *   2. Entry computation
 *   3. Stop-loss & target calculation
 *   4. Trade simulation (walk-forward)
 */

import {
  CONFIG, DIRECTION, GAP_STRATEGY, GAP_TYPE, GAP_SIZE, EXIT_REASON,
} from './gap-config.js';
import {
  classifyGap, scoreCatalyst, determineEligibleStrategies, classifyIndexGap,
} from './gap-classifier.js';
import {
  buildORBRange, computeRSI, computeVWAP,
  filterGapAndGo, filterGapFill, filterIndexGap,
  isPostORB, isPastTimeStop, candleTime, toMinutes,
} from './gap-filters.js';


// ═══════════════════════════════════════════════
// SECTION 1: DAY HIGH/LOW TRACKER
// ═══════════════════════════════════════════════

function computeDayHighLow(candles) {
  let high = -Infinity;
  let low = Infinity;
  for (const c of candles) {
    if (c.high > high) high = c.high;
    if (c.low < low) low = c.low;
    c.dayHigh = high;
    c.dayLow = low;
  }
}


// ═══════════════════════════════════════════════
// SECTION 2: STRATEGY A — GAP AND GO
// ═══════════════════════════════════════════════

/**
 * Detect Gap and Go signal.
 *
 * Flow:
 *   1. Classify gap → must be LARGE or HUGE with catalyst
 *   2. Build ORB (first 15 min) → price must hold above open
 *   3. Wait for ORB breakout in gap direction
 *   4. Validate volume, catalyst, price holding
 *
 * @param {object[]} candles
 * @param {object}   gapInfo       – from classifyGap()
 * @param {object}   catalystInfo  – from scoreCatalyst()
 * @param {number}   avgFirst15Vol – historical average first-15-min volume
 * @returns {object|null} signal
 */
function detectGapAndGoSignal(candles, gapInfo, catalystInfo, avgFirst15Vol) {
  const orb = buildORBRange(candles);
  if (!orb) return null;

  // Determine trade direction from gap
  const direction = gapInfo.isGapUp ? DIRECTION.LONG : DIRECTION.SHORT;

  // Walk post-ORB candles for breakout
  for (let i = 0; i < candles.length; i++) {
    const c = candles[i];
    if (!isPostORB(c)) continue;

    // Time limit: don't enter after noon
    if (isPastTimeStop(c, CONFIG.GF_TIME_STOP_H, CONFIG.GF_TIME_STOP_M)) break;

    // Check breakout
    const isBreakout = direction === DIRECTION.LONG
      ? c.close > orb.orbHigh
      : c.close < orb.orbLow;

    if (!isBreakout) continue;

    // Run filters
    const filterResult = filterGapAndGo({
      gapClassification: gapInfo,
      catalystInfo,
      orb,
      avgFirst15Volume: avgFirst15Vol,
      breakoutCandle: c,
    });

    if (!filterResult.pass) continue;

    // Compute entry, SL, targets
    const entryPrice = direction === DIRECTION.LONG
      ? Math.max(c.close, orb.orbHigh + CONFIG.GAG_ENTRY_BUFFER)
      : Math.min(c.close, orb.orbLow - CONFIG.GAG_ENTRY_BUFFER);

    const { sl, targets } = computeGapAndGoSLTargets(direction, entryPrice, orb, gapInfo);

    return {
      strategy: GAP_STRATEGY.GAP_AND_GO,
      direction,
      entryPrice,
      entryTime: c.time,
      entryIndex: i,
      sl,
      targets,
      gapInfo,
      catalystInfo,
      orbData: orb,
      filterResults: filterResult.filters,
    };
  }

  return null;
}

/**
 * Compute SL and targets for Gap and Go.
 */
function computeGapAndGoSLTargets(direction, entryPrice, orb, gapInfo) {
  const isLong = direction === DIRECTION.LONG;
  let sl;

  // SL option: ORB low or previous day's high
  if (CONFIG.GAG_SL_OPTION === 'PREV_DAY_HIGH' && gapInfo.isGapUp) {
    sl = gapInfo.prevHigh;
  } else if (CONFIG.GAG_SL_OPTION === 'PREV_DAY_LOW' && gapInfo.isGapDown) {
    sl = gapInfo.prevLow;
  } else {
    sl = isLong ? orb.orbLow : orb.orbHigh;
  }

  // Enforce max SL
  const slPct = Math.abs(entryPrice - sl) / entryPrice;
  if (slPct > CONFIG.MAX_SL_PCT) {
    sl = isLong
      ? entryPrice * (1 - CONFIG.MAX_SL_PCT)
      : entryPrice * (1 + CONFIG.MAX_SL_PCT);
  }

  // Target 1: 1× gap size from breakout point
  const gapSize = gapInfo.gapAbsolute;
  const t1Price = isLong ? entryPrice + gapSize : entryPrice - gapSize;

  const targets = [
    { price: t1Price, qtyPct: CONFIG.GAG_TARGET_1_QTY_PCT, label: 'T1_GAP_SIZE' },
    { price: null, qtyPct: CONFIG.GAG_TRAIL_QTY_PCT, label: 'TRAIL' },
  ];

  return { sl, targets };
}


// ═══════════════════════════════════════════════
// SECTION 3: STRATEGY B — GAP FILL
// ═══════════════════════════════════════════════

/**
 * Detect Gap Fill signal.
 *
 * Flow:
 *   1. Classify gap → must be SMALL or MEDIUM without catalyst
 *   2. Check first 15 min for weakness (price drops below open)
 *   3. Check RSI overbought/oversold
 *   4. Wait for VWAP cross or ORB breakdown as entry trigger
 *   5. Target = previous close (gap fill)
 *
 * @param {object[]} candles
 * @param {object}   gapInfo
 * @param {object}   catalystInfo
 * @param {number}   avgFirst15Vol
 * @returns {object|null}
 */
function detectGapFillSignal(candles, gapInfo, catalystInfo, avgFirst15Vol) {
  const orb = buildORBRange(candles);
  if (!orb) return null;

  // Direction: FADE the gap
  // Gap up → go SHORT, gap down → go LONG
  const direction = gapInfo.isGapUp ? DIRECTION.SHORT : DIRECTION.LONG;

  // Walk post-ORB candles
  for (let i = 0; i < candles.length; i++) {
    const c = candles[i];
    if (!isPostORB(c)) continue;

    // Time limit for gap fill
    if (isPastTimeStop(c, CONFIG.GF_TIME_STOP_H, CONFIG.GF_TIME_STOP_M)) break;

    // Check entry trigger
    let triggerMet = false;

    if (CONFIG.GF_ENTRY_TRIGGER === 'BELOW_VWAP') {
      if (c.vwap != null) {
        triggerMet = gapInfo.isGapUp
          ? c.close < c.vwap     // for gap-up fade: price drops below VWAP
          : c.close > c.vwap;    // for gap-down fade: price rises above VWAP
      }
    } else if (CONFIG.GF_ENTRY_TRIGGER === 'BELOW_ORB_LOW') {
      triggerMet = gapInfo.isGapUp
        ? c.close < orb.orbLow
        : c.close > orb.orbHigh;
    }

    if (!triggerMet) continue;

    const entryPrice = c.close;

    // Run filters
    const filterResult = filterGapFill({
      gapClassification: gapInfo,
      catalystInfo,
      orb,
      avgFirst15Volume: avgFirst15Vol,
      rsiAtEntry: c.rsi,
      vwapAtEntry: c.vwap,
      entryPrice,
    });

    if (!filterResult.pass) continue;

    // SL & target
    const { sl, targets } = computeGapFillSLTargets(direction, entryPrice, gapInfo, candles.slice(0, i + 1));

    // Validate R:R
    const risk = Math.abs(entryPrice - sl);
    const reward = Math.abs(targets[0].price - entryPrice);
    if (risk > 0 && reward / risk < 0.8) continue; // need at least 0.8:1 R:R

    return {
      strategy: GAP_STRATEGY.GAP_FILL,
      direction,
      entryPrice,
      entryTime: c.time,
      entryIndex: i,
      sl,
      targets,
      gapInfo,
      catalystInfo,
      orbData: orb,
      filterResults: filterResult.filters,
      rsiAtEntry: c.rsi,
    };
  }

  return null;
}

/**
 * Compute SL and targets for Gap Fill.
 */
function computeGapFillSLTargets(direction, entryPrice, gapInfo, candlesUpTo) {
  const isLong = direction === DIRECTION.LONG;
  let sl;

  // SL: above day's high + 0.3% (for short), below day's low - 0.3% (for long)
  const lastCandle = candlesUpTo[candlesUpTo.length - 1];
  const dayHigh = lastCandle.dayHigh || entryPrice * 1.01;
  const dayLow = lastCandle.dayLow || entryPrice * 0.99;

  if (isLong) {
    // Fading a gap down → go long → SL below day low
    sl = dayLow * (1 - CONFIG.GF_SL_ABOVE_DAY_HIGH_PCT);
  } else {
    // Fading a gap up → go short → SL above day high
    sl = dayHigh * (1 + CONFIG.GF_SL_ABOVE_DAY_HIGH_PCT);
  }

  // Enforce max SL
  const slPct = Math.abs(entryPrice - sl) / entryPrice;
  if (slPct > CONFIG.MAX_SL_PCT) {
    sl = isLong
      ? entryPrice * (1 - CONFIG.MAX_SL_PCT)
      : entryPrice * (1 + CONFIG.MAX_SL_PCT);
  }

  // Target: previous close (gap fill)
  const gapFillPrice = gapInfo.gapFillTarget;

  const targets = [
    {
      price: gapFillPrice,
      qtyPct: 1.0, // full exit at gap fill
      label: 'GAP_FILL_TARGET',
    },
  ];

  return { sl, targets };
}

// SECTION 4: STRATEGY C — INDEX GAP
// ═══════════════════════════════════════════════

/**
 * Detect Index Gap signal.
 *
 * @param {object[]} candles
 * @param {object}   indexGap   – from classifyIndexGap()
 * @returns {object|null}
 */
function detectIndexGapSignal(candles, indexGap) {
  if (!indexGap.meetsThreshold || !indexGap.strategy) return null;

  const filterResult = filterIndexGap({ indexGap });
  if (!filterResult.pass) return null;

  // Entry timing: 9:20 AM (5 minutes after open)
  const entryTime = toMinutes(CONFIG.MARKET_OPEN_H, CONFIG.IG_GAP_FILL_ENTRY_TIME_M);
  let entryCandle = null;
  let entryIndex = -1;

  for (let i = 0; i < candles.length; i++) {
    if (candleTime(candles[i]) >= entryTime) {
      entryCandle = candles[i];
      entryIndex = i;
      break;
    }
  }

  if (!entryCandle) return null;

  const entryPrice = entryCandle.close;
  let direction;
  let optionType;

  if (indexGap.strategy === GAP_STRATEGY.GAP_FILL) {
    // Gap fill: trade against the gap
    direction = indexGap.isUp ? DIRECTION.SHORT : DIRECTION.LONG;
    optionType = indexGap.isUp ? 'PUT' : 'CALL';
  } else {
    // Gap and go: trade with the gap
    direction = indexGap.isUp ? DIRECTION.LONG : DIRECTION.SHORT;
    optionType = indexGap.isUp ? 'CALL' : 'PUT';
  }

  // For options: SL and target in premium terms
  const premiumEstimate = entryPrice; // simplified — in reality, option premium
  const sl = premiumEstimate * (1 - CONFIG.IG_OPTION_SL_PCT);   // 30% loss
  const target = premiumEstimate * (1 + CONFIG.IG_OPTION_TARGET_PCT); // 50% gain

  return {
    strategy: GAP_STRATEGY.INDEX_GAP,
    direction,
    optionType,
    entryPrice,
    entryTime: entryCandle.time,
    entryIndex,
    sl,
    targets: [
      {
        price: target,
        qtyPct: 1.0,
        label: `OPTION_TARGET_${CONFIG.IG_OPTION_TARGET_PCT * 100}%`,
      },
    ],
    gapFillTarget: indexGap.gapFillTarget,
    gapInfo: indexGap,
    fillProbability: indexGap.fillProbability,
    filterResults: filterResult.filters,
    isOptionsTrade: true,
  };
}

// SECTION 5: TRADE SIMULATION
// ═══════════════════════════════════════════════

/**
 * Universal trade simulator for all three gap strategies.
 *
 * @param {object[]} candles
 * @param {object}   signal
 * @returns {object}  trade record
 */
export function simulateTrade(candles, signal) {
  switch (signal.strategy) {
    case GAP_STRATEGY.GAP_AND_GO:
      return simulateGapAndGo(candles, signal);
    case GAP_STRATEGY.GAP_FILL:
      return simulateGapFill(candles, signal);
    case GAP_STRATEGY.INDEX_GAP:
      return simulateIndexGap(candles, signal);
    default:
      return null;
  }
}


// ── Strategy A simulation ──

function simulateGapAndGo(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets } = signal;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  let currentSL = sl;
  const partialExits = [];
  let t1Hit = false;
  let trailActive = false;

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // EOD exit
    if (isPastTimeStop(c, CONFIG.MIS_EXIT_H, CONFIG.MIS_EXIT_M)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_EOD,
      });
      remainingQty = 0;
      break;
    }

    // Stop-loss check
    const hitSL = isLong ? c.low <= currentSL : c.high >= currentSL;
    if (hitSL) {
      partialExits.push({
        time: c.time, price: currentSL,
        qtyPct: remainingQty,
        reason: trailActive ? EXIT_REASON.TRAIL_STOP : EXIT_REASON.STOP_LOSS,
      });
      remainingQty = 0;
      break;
    }

    // Target 1
    if (!t1Hit && targets[0] && targets[0].price != null) {
      const hitT1 = isLong ? c.high >= targets[0].price : c.low <= targets[0].price;
      if (hitT1) {
        t1Hit = true;
        const exitQty = Math.min(targets[0].qtyPct, remainingQty);
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: exitQty, reason: EXIT_REASON.TARGET_1,
        });
        remainingQty -= exitQty;

        // SL to breakeven, activate trail
        currentSL = entryPrice;
        trailActive = true;
      }
    }

    // Trail with 5-min candle lows/highs
    if (trailActive && remainingQty > 0) {
      const prevCandle = candles[i - 1];
      if (isLong) {
        const newTrail = prevCandle.low;
        if (newTrail > currentSL) currentSL = newTrail;
      } else {
        const newTrail = prevCandle.high;
        if (newTrail < currentSL) currentSL = newTrail;
      }
    }

    // Noon time stop if T1 not hit
    if (!t1Hit && isPastTimeStop(c, CONFIG.GF_TIME_STOP_H, CONFIG.GF_TIME_STOP_M)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_NOON,
      });
      remainingQty = 0;
      break;
    }
  }

  return buildTradeRecord(signal, partialExits);
}


// ── Strategy B simulation ──

function simulateGapFill(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets, gapInfo } = signal;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  const partialExits = [];
  const gapFillPrice = gapInfo.gapFillTarget;

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // EOD exit
    if (isPastTimeStop(c, CONFIG.MIS_EXIT_H, CONFIG.MIS_EXIT_M)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_EOD,
      });
      remainingQty = 0;
      break;
    }

    // Stop-loss
    const hitSL = isLong ? c.low <= sl : c.high >= sl;
    if (hitSL) {
      partialExits.push({
        time: c.time, price: sl,
        qtyPct: remainingQty, reason: EXIT_REASON.STOP_LOSS,
      });
      remainingQty = 0;
      break;
    }

    // Gap fill target
    const hitFill = isLong
      ? c.high >= gapFillPrice      // gap down fill = price rises to prev close
      : c.low <= gapFillPrice;      // gap up fill = price drops to prev close

    if (hitFill) {
      partialExits.push({
        time: c.time, price: gapFillPrice,
        qtyPct: remainingQty, reason: EXIT_REASON.GAP_FILL_COMPLETE,
      });
      remainingQty = 0;
      break;
    }

    // Time stop: noon
    if (isPastTimeStop(c, CONFIG.GF_TIME_STOP_H, CONFIG.GF_TIME_STOP_M)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_NOON,
      });
      remainingQty = 0;
      break;
    }
  }

  return buildTradeRecord(signal, partialExits);
}


// ── Strategy C simulation ──

function simulateIndexGap(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets, gapInfo } = signal;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  const partialExits = [];

  // Options trade: SL and target are premium-based
  const premiumEntry = entryPrice;
  const slPremium = premiumEntry * (1 - CONFIG.IG_OPTION_SL_PCT);
  const targetPremium = premiumEntry * (1 + CONFIG.IG_OPTION_TARGET_PCT);

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // Time stop: 11:00 AM
    if (isPastTimeStop(c, CONFIG.IG_TIME_STOP_H, CONFIG.IG_TIME_STOP_M)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_11AM,
      });
      remainingQty = 0;
      break;
    }

    // For options simulation, we track the underlying movement
    // and estimate premium change as a simplified proxy
    const underlyingMove = isLong
      ? (c.close - candles[entryIndex].close) / candles[entryIndex].close
      : (candles[entryIndex].close - c.close) / candles[entryIndex].close;

    // Rough delta approximation: ATM option moves ~50% of underlying
    const estimatedPremiumChange = underlyingMove * 2.0; // simplified leverage
    const currentPremium = premiumEntry * (1 + estimatedPremiumChange);

    // Option SL (30% loss)
    if (currentPremium <= slPremium) {
      partialExits.push({
        time: c.time, price: currentPremium,
        qtyPct: remainingQty, reason: EXIT_REASON.OPTION_SL,
      });
      remainingQty = 0;
      break;
    }

    // Option target (50% gain)
    if (currentPremium >= targetPremium) {
      partialExits.push({
        time: c.time, price: currentPremium,
        qtyPct: remainingQty, reason: EXIT_REASON.OPTION_TARGET,
      });
      remainingQty = 0;
      break;
    }

    // Check gap fill on underlying
    const gapFilled = gapInfo.isUp
      ? c.low <= gapInfo.gapFillTarget
      : c.high >= gapInfo.gapFillTarget;

    if (gapFilled && signal.strategy === GAP_STRATEGY.GAP_FILL) {
      partialExits.push({
        time: c.time, price: currentPremium,
        qtyPct: remainingQty, reason: EXIT_REASON.GAP_FILL_COMPLETE,
      });
      remainingQty = 0;
      break;
    }
  }

  return buildTradeRecord(signal, partialExits);
}


// ═══════════════════════════════════════════════
// SECTION 6: TRADE RECORD BUILDER
// ═══════════════════════════════════════════════

function buildTradeRecord(signal, partialExits) {
  const { direction, entryPrice, strategy, gapInfo } = signal;
  const isLong = direction === DIRECTION.LONG;

  let weightedExit = 0;
  let totalQty = 0;

  for (const e of partialExits) {
    weightedExit += e.price * e.qtyPct;
    totalQty += e.qtyPct;
  }

  const avgExitPrice = totalQty > 0 ? weightedExit / totalQty : entryPrice;

  // For options trades, PnL is on premium
  let grossPnlPct;
  if (signal.isOptionsTrade) {
    grossPnlPct = (avgExitPrice - entryPrice) / entryPrice;
  } else {
    grossPnlPct = isLong
      ? (avgExitPrice - entryPrice) / entryPrice
      : (entryPrice - avgExitPrice) / entryPrice;
  }

  const netPnlPct = grossPnlPct - (CONFIG.SLIPPAGE_PER_SIDE * 2);
  const lastExit = partialExits[partialExits.length - 1];

  // Primary exit reason
  const reasonMap = {};
  for (const e of partialExits) {
    reasonMap[e.reason] = (reasonMap[e.reason] || 0) + e.qtyPct;
  }
  let primaryReason = 'NONE';
  let maxQty = 0;
  for (const [r, q] of Object.entries(reasonMap)) {
    if (q > maxQty) { maxQty = q; primaryReason = r; }
  }

  // Did gap fill?
  const gapFilled = partialExits.some(e => e.reason === EXIT_REASON.GAP_FILL_COMPLETE);

  return {
    strategy,
    direction,
    entryPrice,
    avgExitPrice,
    entryTime: signal.entryTime,
    exitTime: lastExit?.time ?? null,
    grossPnlPct,
    netPnlPct,
    isWinner: netPnlPct > 0,
    partialExits,
    primaryExitReason: primaryReason,
    isOptionsTrade: signal.isOptionsTrade || false,

    // Gap-specific fields
    gapType: gapInfo?.type ?? null,
    gapSize: gapInfo?.gapSize ?? null,
    gapPctAbs: gapInfo?.gapPctAbs ?? null,
    gapDirection: gapInfo?.isGapUp ? 'UP' : 'DOWN',
    gapFilled,
    gapFillTarget: gapInfo?.gapFillTarget ?? null,
    catalystScore: signal.catalystInfo?.catalystScore ?? null,
    sl: signal.sl,
  };
}


// ═══════════════════════════════════════════════
// SECTION 7: SINGLE-DAY ORCHESTRATOR
// ═══════════════════════════════════════════════

/**
 * Run gap trading strategies on one day.
 *
 * @param {object[]} dayCandlesRaw
 * @param {object}   opts
 *   {
 *     prevDay: { open, high, low, close },
 *     catalyst: string,          // CATALYST_TYPE manual tag (optional)
 *     avgFirst15Volume: number,  // historical avg (optional, estimated if missing)
 *     enabledStrategies: string[],
 *     isIndex: boolean,          // true for Nifty/BankNifty
 *     hasGlobalCatalyst: boolean,
 *     vixChange: number,
 *   }
 * @returns {object[]} array of trade records
 */
export function runGapForDay(dayCandlesRaw, opts = {}) {
  const {
    prevDay,
    catalyst = null,
    avgFirst15Volume = 0,
    enabledStrategies = [GAP_STRATEGY.GAP_AND_GO, GAP_STRATEGY.GAP_FILL],
    isIndex = false,
    hasGlobalCatalyst = false,
    vixChange = 0,
  } = opts;

  if (!prevDay) return [];

  // Deep copy & parse
  const candles = dayCandlesRaw.map(c => ({
    ...c,
    time: c.time instanceof Date ? c.time : new Date(c.time),
  }));

  if (candles.length < 6) return [];

  // Compute indicators
  computeVWAP(candles);
  computeRSI(candles);
  computeDayHighLow(candles);

  const todayOpen = candles[0].open;

  // Classify the gap
  const gapInfo = classifyGap(todayOpen, prevDay);
  if (gapInfo.type === GAP_TYPE.NO_GAP) return [];

  // Score catalyst
  const orb = buildORBRange(candles);
  const first15Vol = orb ? orb.orbVolume : 0;
  const estimatedAvg15 = avgFirst15Volume > 0
    ? avgFirst15Volume
    : first15Vol;   // fallback: use today's as baseline (weaker signal)

  const catalystInfo = scoreCatalyst({
    catalyst,
    gapClassification: gapInfo,
    first15Volume: first15Vol,
    avgFirst15Volume: estimatedAvg15,
  });

  // Determine eligible strategies
  const eligibility = determineEligibleStrategies(gapInfo, catalystInfo);

  const trades = [];

  // ── Index Gap (Strategy C) ──
  if (isIndex && enabledStrategies.includes(GAP_STRATEGY.INDEX_GAP)) {
    const indexGap = classifyIndexGap(todayOpen, prevDay.close, {
      hasGlobalCatalyst, vixChange,
    });

    const indexSignal = detectIndexGapSignal(candles, indexGap);
    if (indexSignal) {
      const trade = simulateTrade(candles, indexSignal);
      if (trade) trades.push(trade);
    }
  }

  // ── Gap and Go (Strategy A) ──
  if (
    enabledStrategies.includes(GAP_STRATEGY.GAP_AND_GO) &&
    eligibility.eligible.includes(GAP_STRATEGY.GAP_AND_GO)
  ) {
    const signal = detectGapAndGoSignal(candles, gapInfo, catalystInfo, estimatedAvg15);
    if (signal) {
      const trade = simulateTrade(candles, signal);
      if (trade) trades.push(trade);
    }
  }

  // ── Gap Fill (Strategy B) ──
  if (
    enabledStrategies.includes(GAP_STRATEGY.GAP_FILL) &&
    eligibility.eligible.includes(GAP_STRATEGY.GAP_FILL)
  ) {
    const signal = detectGapFillSignal(candles, gapInfo, catalystInfo, estimatedAvg15);
    if (signal) {
      const trade = simulateTrade(candles, signal);
      if (trade) trades.push(trade);
    }
  }

  // Tag date
  for (const t of trades) {
    t.date = candles[0].time instanceof Date
      ? candles[0].time.toISOString().slice(0, 10)
      : new Date(candles[0].time).toISOString().slice(0, 10);
  }

  return trades;
}
```

```javascript
/**
 * Gap Strategy Analyzer
 *
 * Gap-specific analytics on top of standard backtest metrics:
 *  - Gap fill rate by size bucket
 *  - Gap-and-Go vs Gap-Fill comparison
 *  - Catalyst impact analysis
 *  - Full vs Partial gap performance
 *  - Time-to-fill analysis
 *  - Day-of-week edge (expiry days, etc.)

import { CONFIG, GAP_STRATEGY, GAP_SIZE, GAP_TYPE, DIRECTION, EXIT_REASON } from './gap-config.js';

// ═══════════════════════════════════════════════
// CORE PERFORMANCE
// ═══════════════════════════════════════════════

export function analyzePerformance(trades, capital = CONFIG.DEFAULT_CAPITAL) {
  if (trades.length === 0) return { error: 'No trades' };

  const winners = trades.filter(t => t.isWinner);
  const losers = trades.filter(t => !t.isWinner);
  const pnls = trades.map(t => t.netPnlPct);

  const totalTrades = trades.length;
  const winRate = winners.length / totalTrades;
  const avgWin = safeMean(winners.map(t => t.netPnlPct));
  const avgLoss = safeMean(losers.map(t => t.netPnlPct));
  const avgWLR = Math.abs(avgLoss) > 0 ? avgWin / Math.abs(avgLoss) : Infinity;
  const expectancy = (winRate * avgWin) + ((1 - winRate) * avgLoss);

  const equity = buildEquityCurve(trades, capital);
  const totalReturn = ((equity[equity.length - 1].equity - capital) / capital) * 100;
  const maxDD = computeMaxDrawdown(equity);
  const streaks = computeStreaks(trades);
  const sharpe = computeSharpe(pnls);
  const pf = computeProfitFactor(trades);

  return {
    totalTrades,
    winners: winners.length,
    losers: losers.length,
    winRatePct: r2(winRate * 100),
    avgWinPct: r4(avgWin * 100),
    avgLossPct: r4(avgLoss * 100),
    avgWinLossRatio: r2(avgWLR),
    expectancyPct: r4(expectancy * 100),
    totalReturnPct: r2(totalReturn),
    totalReturnAbs: r2(equity[equity.length - 1].equity - capital),
    maxDrawdownPct: r2(maxDD.pct),
    maxDrawdownDuration: maxDD.duration,
    sharpeRatio: r2(sharpe),
    profitFactor: r2(pf),
    maxWinStreak: streaks.maxWinStreak,
    maxLossStreak: streaks.maxLossStreak,
    equityCurve: equity,
  };
}


// ═══════════════════════════════════════════════
// GAP FILL RATE BY SIZE
// ═══════════════════════════════════════════════

/**
 * The most important gap-specific metric:
 * What percentage of gaps actually filled, broken down by size?
 *
 * @param {object[]} trades  – all gap trades
 * @returns {object}
 */
export function gapFillRateBySize(trades) {
  const buckets = {};

  for (const size of Object.values(GAP_SIZE)) {
    if (size === GAP_SIZE.NONE) continue;
    buckets[size] = { total: 0, filled: 0, trades: [] };
  }

  for (const t of trades) {
    const size = t.gapSize;
    if (!buckets[size]) continue;

    buckets[size].total++;
    buckets[size].trades.push(t);

    if (t.gapFilled) {
      buckets[size].filled++;
    }
  }

  const result = {};
  for (const [size, data] of Object.entries(buckets)) {
    result[size] = {
      totalGaps: data.total,
      gapsFilled: data.filled,
      fillRatePct: data.total > 0 ? r2((data.filled / data.total) * 100) : 0,
      avgPnlPct: r4(safeMean(data.trades.map(t => t.netPnlPct)) * 100),
      winRatePct: data.total > 0
        ? r2((data.trades.filter(t => t.isWinner).length / data.total) * 100) : 0,
    };
  }

  return result;
}

export function gapFillRateByType(trades) {

        ? r2((data.trades.filter(t => t.isWinner).length / data.total) * 100)
        : 0,
      avgPnlPct: r4(safeMean(data.trades.map(t => t.netPnlPct)) * 100),
    };
  }

  return result;
}


// ═══════════════════════════════════════════════
// STRATEGY COMPARISON (Gap&Go vs Gap Fill vs Index)
// ═══════════════════════════════════════════════

export function strategyComparison(trades) {
  const strategies = {};

  for (const strat of Object.values(GAP_STRATEGY)) {
    strategies[strat] = [];
  }
  strategies['UNKNOWN'] = [];

  for (const t of trades) {
    const key = t.strategy || 'UNKNOWN';
    if (!strategies[key]) strategies[key] = [];
    strategies[key].push(t);
  }

  const result = {};
  for (const [strat, sTrades] of Object.entries(strategies)) {
    if (sTrades.length === 0) { result[strat] = { count: 0 }; continue; }

    const wins = sTrades.filter(t => t.isWinner).length;
    const avgWin = safeMean(sTrades.filter(t => t.isWinner).map(t => t.netPnlPct));
    const avgLoss = safeMean(sTrades.filter(t => !t.isWinner).map(t => t.netPnlPct));

    result[strat] = {
      count: sTrades.length,
      winRatePct: r2((wins / sTrades.length) * 100),
      avgWinPct: r4(avgWin * 100),
      avgLossPct: r4(avgLoss * 100),
      avgPnlPct: r4(safeMean(sTrades.map(t => t.netPnlPct)) * 100),
      winLossRatio: Math.abs(avgLoss) > 0 ? r2(avgWin / Math.abs(avgLoss)) : 'INF',
      gapsFilled: sTrades.filter(t => t.gapFilled).length,
      gapFillRate: r2((sTrades.filter(t => t.gapFilled).length / sTrades.length) * 100),
    };
  }

  return result;
}

/**
 * How does catalyst presence/strength correlate with gap trade outcomes?
 */
export function catalystImpactAnalysis(trades) {
  const buckets = {
    'No Catalyst (0)': [],
    'Weak (1-30)': [],
    'Moderate (31-60)': [],
    'Strong (61-80)': [],
    'Very Strong (81-100)': [],
  };

  for (const t of trades) {
    const score = t.catalystScore ?? 0;
    if (score === 0) buckets['No Catalyst (0)'].push(t);
    else if (score <= 30) buckets['Weak (1-30)'].push(t);
    else if (score <= 60) buckets['Moderate (31-60)'].push(t);
    else if (score <= 80) buckets['Strong (61-80)'].push(t);
    else buckets['Very Strong (81-100)'].push(t);
  }

  const result = {};
  for (const [bucket, bTrades] of Object.entries(buckets)) {
    if (bTrades.length === 0) { result[bucket] = { count: 0 }; continue; }
    const wins = bTrades.filter(t => t.isWinner).length;
    result[bucket] = {
      count: bTrades.length,
      winRatePct: r2((wins / bTrades.length) * 100),
      avgPnlPct: r4(safeMean(bTrades.map(t => t.netPnlPct)) * 100),
      dominantStrategy: mode(bTrades.map(t => t.strategy)),
    };
  }

  return result;
}

// ═══════════════════════════════════════════════
// TIME TO FILL ANALYSIS
// ═══════════════════════════════════════════════

/**
 * For trades where gap filled: how long did it take?
 */
export function timeToFillAnalysis(trades) {
  const filledTrades = trades.filter(t => t.gapFilled && t.entryTime && t.exitTime);

  if (filledTrades.length === 0) return { filledCount: 0 };

  const durations = filledTrades.map(t => {
    const entry = t.entryTime instanceof Date ? t.entryTime : new Date(t.entryTime);
    const exit = t.exitTime instanceof Date ? t.exitTime : new Date(t.exitTime);
    return (exit - entry) / (1000 * 60); // minutes
  });

  const buckets = {
    '< 30 min': 0,
    '30-60 min': 0,
    '1-2 hours': 0,
    '2-3 hours': 0,
    '3+ hours': 0,
  };

  for (const d of durations) {
    if (d < 30) buckets['< 30 min']++;
    else if (d < 60) buckets['30-60 min']++;
    else if (d < 120) buckets['1-2 hours']++;
    else if (d < 180) buckets['2-3 hours']++;
    else buckets['3+ hours']++;
  }

  return {
    filledCount: filledTrades.length,
    avgMinutesToFill: r2(safeMean(durations)),
    medianMinutesToFill: r2(median(durations)),
    distribution: buckets,
  };
}

export function exitReasonBreakdown(trades) {
  const reasons = {};

  for (const t of trades) {
    const r = t.primaryExitReason || 'UNKNOWN';
    if (!reasons[r]) reasons[r] = { count: 0, totalPnl: 0, wins: 0 };
    reasons[r].count++;
    reasons[r].totalPnl += t.netPnlPct;
    if (t.isWinner) reasons[r].wins++;
  }

  for (const r of Object.keys(reasons)) {
    const d = reasons[r];
    d.winRatePct = r2((d.wins / d.count) * 100);
    d.avgPnlPct = r4((d.totalPnl / d.count) * 100);
  }

  return reasons;
}


// ═══════════════════════════════════════════════
// MONTHLY BREAKDOWN
// ═══════════════════════════════════════════════

export function monthlyBreakdown(trades) {
  const groups = {};

  for (const t of trades) {
    if (!t.entryTime) continue;
    const d = t.entryTime instanceof Date ? t.entryTime : new Date(t.entryTime);
    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
    if (!groups[key]) groups[key] = [];
    groups[key].push(t);
  }

  return Object.entries(groups).sort().map(([month, mTrades]) => {
    const wins = mTrades.filter(t => t.isWinner).length;
    return {
      month,
      trades: mTrades.length,
      winRatePct: r2((wins / mTrades.length) * 100),
      netPnlPct: r4(mTrades.reduce((s, t) => s + t.netPnlPct, 0) * 100),
      gapsFilled: mTrades.filter(t => t.gapFilled).length,
    };
  });
}

export function dayOfWeekBreakdown(trades) {

  for (const t of trades) {

    const d = t.entryTime instanceof Date ? t.entryTime : new Date(t.entryTime);
    const dow = days[d.getDay()];
    if (!groups[dow]) groups[dow] = [];
    groups[dow].push(t);
  }

  const result = {};
  for (const [dow, dTrades] of Object.entries(groups)) {
    const wins = dTrades.filter(t => t.isWinner).length;
    result[dow] = {
      count: dTrades.length,
      winRatePct: r2((wins / dTrades.length) * 100),
      avgPnlPct: r4(safeMean(dTrades.map(t => t.netPnlPct)) * 100),
      fillRate: r2((dTrades.filter(t => t.gapFilled).length / dTrades.length) * 100),
    };
  }

  return result;
}

function buildEquityCurve(trades, startCapital) {
  const curve = [{ tradeNum: 0, equity: startCapital }];
  let equity = startCapital;
  for (let i = 0; i < trades.length; i++) {
    const t = trades[i];
    const riskAmt = equity * CONFIG.DEFAULT_RISK_PER_TRADE;
    const slDist = Math.abs(t.entryPrice - t.sl) || t.entryPrice * 0.005;
    const qty = Math.floor(riskAmt / slDist);
    const posVal = qty * t.entryPrice;
    const pnl = t.isOptionsTrade
      ? riskAmt * t.netPnlPct * 3    // options leverage approximation
      : posVal * t.netPnlPct;
    equity += pnl - CONFIG.COMMISSION_PER_SIDE * 2;
    curve.push({ tradeNum: i + 1, equity: r2(equity), date: t.entryTime, strategy: t.strategy });
  }
  return curve;
}

function computeMaxDrawdown(curve) {
  let peak = curve[0].equity, maxDD = 0, ddStart = 0, maxDur = 0;
  for (let i = 1; i < curve.length; i++) {
    if (curve[i].equity > peak) { peak = curve[i].equity; ddStart = i; }
    const dd = (peak - curve[i].equity) / peak;
    if (dd > maxDD) { maxDD = dd; maxDur = i - ddStart; }
  }
  return { pct: maxDD * 100, duration: maxDur };
}

function computeStreaks(trades) {
  let mw = 0, ml = 0, cw = 0, cl = 0;
  for (const t of trades) {
    if (t.isWinner) { cw++; cl = 0; if (cw > mw) mw = cw; }
    else { cl++; cw = 0; if (cl > ml) ml = cl; }
  }
  return { maxWinStreak: mw, maxLossStreak: ml };
}

function computeSharpe(pnls) {
  if (pnls.length < 2) return 0;
  const avg = safeMean(pnls);
  const std = Math.sqrt(pnls.reduce((s, p) => s + (p - avg) ** 2, 0) / (pnls.length - 1));
  return std === 0 ? 0 : (avg / std) * Math.sqrt(250);
}

function computeProfitFactor(trades) {
  let gp = 0, gl = 0;
  for (const t of trades) {
    if (t.netPnlPct > 0) gp += t.netPnlPct; else gl += Math.abs(t.netPnlPct);
  }
  return gl > 0 ? gp / gl : Infinity;
}

function safeMean(arr) { return arr.length === 0 ? 0 : arr.reduce((a, b) => a + b, 0) / arr.length; }
function median(arr) {
  if (arr.length === 0) return 0;
  const sorted = [...arr].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}
function mode(arr) {
  const freq = {};
  for (const v of arr) freq[v] = (freq[v] || 0) + 1;
  let modeVal = arr[0], maxFreq = 0;
  for (const [v, f] of Object.entries(freq)) { if (f > maxFreq) { maxFreq = f; modeVal = v; } }
  return modeVal;
}
function r2(n) { return Math.round(n * 100) / 100; }
function r4(n) { return Math.round(n * 10000) / 10000; }
```

```javascript
// GROUP CANDLES BY DAY
// ═══════════════════════════════════════════════

export function groupByDay(candles) {
  const days = new Map();
  for (const c of candles) {
    const d = c.time instanceof Date ? c.time : new Date(c.time);
    const key = d.toISOString().slice(0, 10);
    if (!days.has(key)) days.set(key, []);
    days.get(key).push(c);
  }
  return days;
}

// COMPUTE HISTORICAL AVERAGES
// ═══════════════════════════════════════════════

/**
 * Compute average first-15-min volume over a rolling window.
 * Needed for catalyst scoring and volume filters.
 *
 * @param {Map<string, object[]>} dayGroups
 * @param {number} lookback – number of days to average
 * @returns {Map<string, number>} date → avg first-15-min volume
 */
function computeAvgFirst15Volumes(dayGroups, lookback = 20) {
  const sortedDates = [...dayGroups.keys()].sort();
  const first15Volumes = [];
  const avgMap = new Map();

  const orbEnd = (CONFIG.ORB_END_H * 60) + CONFIG.ORB_END_M;
  const mktOpen = (CONFIG.MARKET_OPEN_H * 60) + CONFIG.MARKET_OPEN_M;

  for (const dateKey of sortedDates) {
    const dayCandles = dayGroups.get(dateKey);

    // Sum volume in first 15 minutes
    let vol15 = 0;
    for (const c of dayCandles) {
      const d = c.time instanceof Date ? c.time : new Date(c.time);
      const t = d.getHours() * 60 + d.getMinutes();
      if (t >= mktOpen && t < orbEnd) {
        vol15 += c.volume;
      }
    }

    first15Volumes.push(vol15);

    // Rolling average
    const windowStart = Math.max(0, first15Volumes.length - lookback - 1);
    const windowEnd = first15Volumes.length - 1; // exclude current day
    const window = first15Volumes.slice(windowStart, windowEnd);

    avgMap.set(dateKey, window.length > 0
      ? window.reduce((a, b) => a + b, 0) / window.length
      : vol15);
  }

  return avgMap;
}

/**
 * @param {object[]} allCandles – flat 5-min candle array, multi-day
 * @param {object}   opts
 *   {
 *     capital: number,
 *     enabledStrategies: string[],
 *     isIndex: boolean,
 *     dailyMeta: {
 *       'YYYY-MM-DD': {
 *         catalyst?: string,          // CATALYST_TYPE
 *         hasGlobalCatalyst?: boolean,
 *         vixChange?: number,
 *       }
 *     }
 *   }
 */
export function runFullBacktest(allCandles, opts = {}) {
  const {
    capital = CONFIG.DEFAULT_CAPITAL,
    enabledStrategies = [GAP_STRATEGY.GAP_AND_GO, GAP_STRATEGY.GAP_FILL],
    isIndex = false,
    dailyMeta = {},
  } = opts;

  const dayGroups = groupByDay(allCandles);
  const sortedDates = [...dayGroups.keys()].sort();

  if (sortedDates.length < 2) {
    return { trades: [], report: null, error: 'Need at least 2 days of data' };
  }

  // Pre-compute rolling average first-15-min volumes
  const avgFirst15Map = computeAvgFirst15Volumes(dayGroups);

  const allTrades = [];
  const skippedDays = [];
  const gapLog = []; // log every gap classification for analysis

  // We need previous day data, so start from day 2
  for (let i = 1; i < sortedDates.length; i++) {
    const dateKey = sortedDates[i];
    const prevDateKey = sortedDates[i - 1];

    const dayCandles = dayGroups.get(dateKey);
    const prevDayCandles = dayGroups.get(prevDateKey);

    if (dayCandles.length < 6 || prevDayCandles.length < 3) {
      skippedDays.push({ date: dateKey, reason: 'Insufficient candles' });
      continue;
    }

    // Build previous day OHLC
    const prevDay = {
      open: prevDayCandles[0].open,
      high: Math.max(...prevDayCandles.map(c => c.high)),
      low: Math.min(...prevDayCandles.map(c => c.low)),
      close: prevDayCandles[prevDayCandles.length - 1].close,
    };

    const meta = dailyMeta[dateKey] || {};
    const avgFirst15Vol = avgFirst15Map.get(dateKey) || 0;

    const dayTrades = runGapForDay(dayCandles, {
      prevDay,
      catalyst: meta.catalyst || null,
      avgFirst15Volume: avgFirst15Vol,
      enabledStrategies,
      isIndex,
      hasGlobalCatalyst: meta.hasGlobalCatalyst || false,
      vixChange: meta.vixChange || 0,
    });

    // Log gap classification even if no trade taken
    const todayOpen = dayCandles[0].open;
    const { classifyGap } = await import('./gap-classifier.js');
    const gapInfo = classifyGap(todayOpen, prevDay);
    gapLog.push({ date: dateKey, ...gapInfo, traded: dayTrades.length > 0 });

    for (const t of dayTrades) {
      t.date = dateKey;
      allTrades.push(t);
    }

    if (dayTrades.length === 0) {
      skippedDays.push({ date: dateKey, reason: gapInfo.type === 'NO_GAP' ? 'No gap' : 'No valid signal' });
    }
  }

  const report = allTrades.length > 0 ? analyzePerformance(allTrades, capital) : null;

  return {
    trades: allTrades,
    report,
    gapLog,
    totalDays: sortedDates.length - 1, // exclude first day (no prev data)
    tradedDays: new Set(allTrades.map(t => t.date)).size,
    gapDays: gapLog.filter(g => g.type !== 'NO_GAP').length,
    skippedDays,
  };
}

export function multiStockScan(stockCandles, opts = {}) {
  const results = [];

  for (const [symbol, candles] of stockCandles) {
    const res = runFullBacktestSync(candles, opts);

    if (res.report) {
      // Additional gap-specific metrics
      const fillRate = res.trades.length > 0
        ? r2((res.trades.filter(t => t.gapFilled).length / res.trades.length) * 100)
        : 0;

      results.push({
        symbol,
        trades: res.report.totalTrades,
        winRatePct: res.report.winRatePct,
        expectancyPct: res.report.expectancyPct,
        returnPct: res.report.totalReturnPct,
        maxDDPct: res.report.maxDrawdownPct,
        profitFactor: res.report.profitFactor,
        gapFillRate: fillRate,
        gapDays: res.gapDays || 0,
      });
    } else {
      results.push({ symbol, skipped: true });
    }
  }

  results.sort((a, b) => {
    if (a.skipped && !b.skipped) return 1;
    if (!a.skipped && b.skipped) return -1;
    return (b.expectancyPct || 0) - (a.expectancyPct || 0);
  });

  return results;
}
```

The key architectural distinction of the gap strategy is the **classifier layer** (`gap-classifier.js`) that sits between raw data and strategy logic. Every trading day starts with gap classification (type + size + catalyst scoring) before any strategy runs. This mirrors the real-world workflow: the night-before scan identifies candidates, then at 9:15 AM the classifier determines which playbook applies. The catalyst scoring system provides both manual tagging (for live trading with news feeds) and heuristic inference (for backtesting without news data), making the system usable in both modes without code changes.
