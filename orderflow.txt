## `orderflow-depth.js`

```javascript
/**
 * Market Depth Processing Engine
 *
 * Data contracts:
 *
 * DepthSnapshot:
 * {
 *   time: Date | string | number,       // timestamp
 *   bids: [ { price, qty }, ... ],       // top 5, sorted best-first (highest price first)
 *   asks: [ { price, qty }, ... ],       // top 5, sorted best-first (lowest price first)
 *   ltp: number,                         // last traded price
 *   ltq?: number,                        // last traded quantity
 *   totalBuyQty?: number,                // total market buy qty (if available)
 *   totalSellQty?: number,               // total market sell qty
 * }
 *
 * Candle (augmented):
 * {
 *   ...standard OHLCV candle,
 *   depthSnapshots: DepthSnapshot[],     // all depth snapshots within this candle
 * }
 */

import { CONFIG, OF_BIAS } from './orderflow-config.js';

// ═══════════════════════════════════════════════
// SECTION 1: BASIC DEPTH METRICS
// ═══════════════════════════════════════════════

/**
 * Compute raw bid-ask quantities from a single depth snapshot.
 *
 * @param {object} snapshot – DepthSnapshot
 * @returns {object}
 */
export function computeRawDepthMetrics(snapshot) {
  const { bids, asks } = snapshot;

  let totalBidQty = 0;
  let totalAskQty = 0;
  let weightedBidQty = 0;
  let weightedAskQty = 0;

  const weights = CONFIG.DEPTH_LEVEL_WEIGHTS;

  for (let i = 0; i < Math.min(bids.length, CONFIG.DEPTH_LEVELS); i++) {
    const qty = bids[i]?.qty || 0;
    totalBidQty += qty;
    weightedBidQty += qty * (weights[i] || 0.1);
  }

  for (let i = 0; i < Math.min(asks.length, CONFIG.DEPTH_LEVELS); i++) {
    const qty = asks[i]?.qty || 0;
    totalAskQty += qty;
    weightedAskQty += qty * (weights[i] || 0.1);
  }

  // Bid-Ask Ratios
  const rawBAR = totalAskQty > 0 ? totalBidQty / totalAskQty : Infinity;
  const weightedBAR = weightedAskQty > 0 ? weightedBidQty / weightedAskQty : Infinity;

  // Spread
  const bestBid = bids[0]?.price || 0;
  const bestAsk = asks[0]?.price || 0;
  const spread = bestAsk - bestBid;
  const spreadPct = bestBid > 0 ? spread / bestBid : 0;
  const midPrice = (bestBid + bestAsk) / 2;

  // Imbalance: normalised –1 (all sellers) to +1 (all buyers)
  const totalQty = totalBidQty + totalAskQty;
  const imbalance = totalQty > 0 ? (totalBidQty - totalAskQty) / totalQty : 0;

  return {
    totalBidQty,
    totalAskQty,
    weightedBidQty: Math.round(weightedBidQty),
    weightedAskQty: Math.round(weightedAskQty),
    rawBAR: round4(rawBAR),
    weightedBAR: round4(weightedBAR),
    bar: CONFIG.USE_WEIGHTED_BAR ? round4(weightedBAR) : round4(rawBAR),
    spread,
    spreadPct: round6(spreadPct),
    midPrice: round2(midPrice),
    bestBid,
    bestAsk,
    imbalance: round4(imbalance),
    time: snapshot.time,
    ltp: snapshot.ltp,
  };
}


// ═══════════════════════════════════════════════
// SECTION 2: SMOOTHED / AGGREGATED METRICS
// ═══════════════════════════════════════════════

/**
 * Compute smoothed bid-ask ratio over a window of snapshots.
 *
 * @param {object[]} snapshots    – recent DepthSnapshots
 * @param {number}   windowSize  – how many to average
 * @returns {object}
 */
export function computeSmoothedBAR(snapshots, windowSize = CONFIG.BAR_SMOOTHING_SNAPSHOTS) {
  const recent = snapshots.slice(-windowSize);
  if (recent.length === 0) return { smoothedBAR: 1.0, samples: 0 };

  const metrics = recent.map(s => computeRawDepthMetrics(s));

  const avgBAR = metrics.reduce((sum, m) => sum + m.bar, 0) / metrics.length;
  const avgImbalance = metrics.reduce((sum, m) => sum + m.imbalance, 0) / metrics.length;

  // Standard deviation of BAR (for volatility of depth)
  const barValues = metrics.map(m => m.bar);
  const barMean = avgBAR;
  const barVariance = barValues.reduce((sum, v) => sum + (v - barMean) ** 2, 0) / barValues.length;
  const barStd = Math.sqrt(barVariance);

  return {
    smoothedBAR: round4(avgBAR),
    avgImbalance: round4(avgImbalance),
    barStd: round4(barStd),
    barStable: barStd < 0.3,  // low std = consistent reading
    samples: recent.length,
    latestBAR: metrics[metrics.length - 1]?.bar || 1.0,
  };
}


/**
 * Determine directional bias from depth metrics.
 *
 * @param {number} bar          – bid-ask ratio (smoothed or raw)
 * @param {number} imbalance    – -1 to +1
 * @returns {{ bias: string, strength: number, reason: string }}
 */
export function determineBias(bar, imbalance) {
  let bias = OF_BIAS.NEUTRAL;
  let strength = 0; // 0–100
  let reason = '';

  if (bar >= CONFIG.BAR_BULLISH_THRESHOLD) {
    bias = OF_BIAS.BULLISH;
    // Strength scales with how far above threshold
    strength = Math.min(100, Math.round(((bar - 1.0) / 1.0) * 100));
    reason = `BAR ${bar} ≥ ${CONFIG.BAR_BULLISH_THRESHOLD} (buyers dominant)`;
  } else if (bar <= CONFIG.BAR_BEARISH_THRESHOLD) {
    bias = OF_BIAS.BEARISH;
    strength = Math.min(100, Math.round(((1.0 - bar) / 1.0) * 100));
    reason = `BAR ${bar} ≤ ${CONFIG.BAR_BEARISH_THRESHOLD} (sellers dominant)`;
  } else if (bar >= CONFIG.BAR_NEUTRAL_LOW && bar <= CONFIG.BAR_NEUTRAL_HIGH) {
    bias = OF_BIAS.NEUTRAL;
    strength = 0;
    reason = `BAR ${bar} in neutral zone [${CONFIG.BAR_NEUTRAL_LOW}–${CONFIG.BAR_NEUTRAL_HIGH}]`;
  } else if (bar > CONFIG.BAR_NEUTRAL_HIGH) {
    bias = OF_BIAS.BULLISH;
    strength = Math.min(50, Math.round(((bar - 1.0) / 0.3) * 50));
    reason = `BAR ${bar} mildly bullish`;
  } else {
    bias = OF_BIAS.BEARISH;
    strength = Math.min(50, Math.round(((1.0 - bar) / 0.3) * 50));
    reason = `BAR ${bar} mildly bearish`;
  }

  // Boost strength if imbalance aligns
  if (imbalance > 0.3 && bias === OF_BIAS.BULLISH) strength = Math.min(100, strength + 15);
  if (imbalance < -0.3 && bias === OF_BIAS.BEARISH) strength = Math.min(100, strength + 15);

  return { bias, strength, reason };
}


// ═══════════════════════════════════════════════
// SECTION 3: CUMULATIVE DELTA
// ═══════════════════════════════════════════════

/**
 * Compute cumulative delta from trade data within depth snapshots.
 *
 * Delta = buy-initiated volume – sell-initiated volume.
 * Approximation: trades at ask = buy-initiated, trades at bid = sell-initiated.
 *
 * If trade-level data isn't available, approximate from LTP movement
 * relative to bid/ask.
 *
 * @param {object[]} snapshots – ordered DepthSnapshots
 * @returns {object[]} – array with delta metrics per snapshot
 */
export function computeCumulativeDelta(snapshots) {
  let cumDelta = 0;
  const results = [];
  const recentDeltas = [];

  for (let i = 0; i < snapshots.length; i++) {
    const snap = snapshots[i];
    const metrics = computeRawDepthMetrics(snap);

    let tickDelta = 0;

    if (snap.ltq && snap.ltp) {
      // Classify: if LTP is at or near ask → buy-initiated
      if (snap.ltp >= metrics.bestAsk) {
        tickDelta = snap.ltq;  // positive = buying pressure
      } else if (snap.ltp <= metrics.bestBid) {
        tickDelta = -snap.ltq; // negative = selling pressure
      } else {
        // Mid-range: proportional assignment
        const range = metrics.bestAsk - metrics.bestBid;
        const position = range > 0
          ? (snap.ltp - metrics.bestBid) / range
          : 0.5;
        tickDelta = snap.ltq * (2 * position - 1); // maps to [-ltq, +ltq]
      }
    }

    cumDelta += tickDelta;
    recentDeltas.push(tickDelta);

    // Keep rolling window
    if (recentDeltas.length > CONFIG.DELTA_AVG_WINDOW) {
      recentDeltas.shift();
    }

    const avgDelta = recentDeltas.length > 0
      ? recentDeltas.reduce((a, b) => a + b, 0) / recentDeltas.length
      : 0;

    const absAvg = Math.abs(avgDelta) || 1;
    const deltaRatio = Math.abs(tickDelta) / absAvg;

    results.push({
      time: snap.time,
      tickDelta: Math.round(tickDelta),
      cumDelta: Math.round(cumDelta),
      avgDelta: Math.round(avgDelta),
      deltaRatio: round2(deltaRatio),
      isSurge: deltaRatio >= CONFIG.DELTA_SURGE_THRESHOLD,
      surgeDirection: tickDelta > 0 ? 'POSITIVE' : tickDelta < 0 ? 'NEGATIVE' : 'NONE',
    });
  }

  return results;
}


// ═══════════════════════════════════════════════
// SECTION 4: PER-LEVEL DEPTH TRACKING
// ═══════════════════════════════════════════════

/**
 * Build a map of quantities at each price level over time.
 * Essential for absorption and spoofing detection.
 *
 * @param {object[]} snapshots
 * @param {string}   side – 'bids' | 'asks'
 * @returns {Map<number, object[]>}
 *   key = price level (rounded)
 *   value = [{ time, qty, snapshotIdx }]
 */
export function buildLevelHistory(snapshots, side) {
  const history = new Map();

  for (let i = 0; i < snapshots.length; i++) {
    const snap = snapshots[i];
    const levels = snap[side] || [];

    for (const level of levels) {
      const priceKey = roundPrice(level.price);

      if (!history.has(priceKey)) {
        history.set(priceKey, []);
      }

      history.get(priceKey).push({
        time: snap.time,
        qty: level.qty,
        snapshotIdx: i,
      });
    }
  }

  return history;
}


/**
 * Get the quantity at a specific price level at a specific snapshot.
 * Returns 0 if not present.
 *
 * @param {object}   snapshot
 * @param {string}   side  – 'bids' | 'asks'
 * @param {number}   price – target price (within tolerance)
 * @returns {number} qty at that level
 */
export function getQtyAtLevel(snapshot, side, price) {
  const levels = snapshot[side] || [];
  const tolerance = price * CONFIG.ABSORPTION_PRICE_TOLERANCE;

  for (const level of levels) {
    if (Math.abs(level.price - price) <= tolerance) {
      return level.qty;
    }
  }
  return 0;
}


// ═══════════════════════════════════════════════
// SECTION 5: AGGREGATE DEPTH FOR A CANDLE PERIOD
// ═══════════════════════════════════════════════

/**
 * Aggregate depth metrics for all snapshots within a candle's time window.
 * Used to attach depth summary to each 5-min candle.
 *
 * @param {object[]} snapshots – all depth snapshots for this candle period
 * @returns {object} candle-level depth summary
 */
export function aggregateDepthForCandle(snapshots) {
  if (snapshots.length === 0) {
    return {
      avgBAR: 1.0, avgImbalance: 0, bias: OF_BIAS.NEUTRAL,
      biasStrength: 0, samples: 0,
      maxBidQty: 0, maxAskQty: 0,
      cumDelta: 0, hasData: false,
    };
  }

  const metricsArr = snapshots.map(s => computeRawDepthMetrics(s));
  const deltas = computeCumulativeDelta(snapshots);

  const avgBAR = safeMean(metricsArr.map(m => m.bar));
  const avgImbalance = safeMean(metricsArr.map(m => m.imbalance));
  const maxBidQty = Math.max(...metricsArr.map(m => m.totalBidQty));
  const maxAskQty = Math.max(...metricsArr.map(m => m.totalAskQty));

  const { bias, strength } = determineBias(avgBAR, avgImbalance);

  const lastDelta = deltas[deltas.length - 1];

  return {
    avgBAR: round4(avgBAR),
    avgImbalance: round4(avgImbalance),
    bias,
    biasStrength: strength,
    samples: snapshots.length,
    maxBidQty,
    maxAskQty,
    cumDelta: lastDelta?.cumDelta || 0,
    avgDelta: lastDelta?.avgDelta || 0,
    hasDeltaSurge: deltas.some(d => d.isSurge),
    hasData: true,
  };
}


// ═══════════════════════════════════════════════
// UTILITIES
// ═══════════════════════════════════════════════

function roundPrice(p) { return Math.round(p * 20) / 20; } // round to 0.05
function round2(n) { return Math.round(n * 100) / 100; }
function round4(n) { return Math.round(n * 10000) / 10000; }
function round6(n) { return Math.round(n * 1000000) / 1000000; }
function safeMean(arr) { return arr.length === 0 ? 0 : arr.reduce((a, b) => a + b, 0) / arr.length; }

export { round2, round4, safeMean, roundPrice };
```

## `orderflow-patterns.js`

```javascript
/**
 * Order Flow Pattern Detection
 *
 * Three primary patterns:
 *   1. ABSORPTION — large order being consumed by opposing flow
 *   2. SPOOFING / ICEBERG — fake or hidden orders
 *   3. STACKING — institutional accumulation across multiple levels
 *
 * Each detector works on a time-series of depth snapshots.
 */

import { CONFIG, OF_PATTERN } from './orderflow-config.js';
import {
  computeRawDepthMetrics, buildLevelHistory, getQtyAtLevel, roundPrice,
} from './orderflow-depth.js';

// ═══════════════════════════════════════════════
// SECTION 1: ABSORPTION DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect absorption patterns where a large resting order is being
 * consumed by aggressive counter-orders.
 *
 * Bullish absorption: Large ASK (sellers) at a price is being consumed
 *   → asks shrinking without price falling → buyers winning → bullish breakout
 *
 * Bearish absorption: Large BID (buyers) at a price is being consumed
 *   → bids shrinking without price rising → sellers winning → bearish breakdown
 *
 * @param {object[]} snapshots – ordered depth snapshots
 * @param {object}   opts      – { side?: 'bids'|'asks', priceLevel?: number }
 * @returns {object[]} detected absorption events
 */
export function detectAbsorption(snapshots, opts = {}) {
  if (snapshots.length < 5) return [];

  const events = [];

  // Scan both sides unless specified
  const sides = opts.side ? [opts.side] : ['asks', 'bids'];

  for (const side of sides) {
    const history = buildLevelHistory(snapshots, side);

    for (const [priceLevel, entries] of history) {
      if (entries.length < CONFIG.ABSORPTION_MIN_HITS) continue;

      // Find initial large order
      const firstEntry = entries[0];
      if (firstEntry.qty < CONFIG.ABSORPTION_INITIAL_QTY_MIN) continue;

      const initialQty = firstEntry.qty;
      const startTime = firstEntry.snapshotIdx;

      // Track consumption over time
      let minQty = initialQty;
      let peakQty = initialQty;
      let hitCount = 0;
      let lastIdx = startTime;
      let consumed = false;

      for (const entry of entries) {
        if (entry.snapshotIdx - startTime > CONFIG.ABSORPTION_MAX_SNAPSHOTS) break;

        if (entry.qty > peakQty) peakQty = entry.qty;
        if (entry.qty < minQty) minQty = entry.qty;

        lastIdx = entry.snapshotIdx;

        // Count times price touches this level
        const snap = snapshots[entry.snapshotIdx];
        const ltp = snap?.ltp || 0;
        const distance = Math.abs(ltp - priceLevel) / priceLevel;
        if (distance < CONFIG.ABSORPTION_PRICE_TOLERANCE) {
          hitCount++;
        }

        // Check consumption level
        const consumedPct = 1 - (entry.qty / initialQty);

        if (consumedPct >= CONFIG.ABSORPTION_COMPLETE_PCT) {
          consumed = true;
          break;
        }
      }

      const consumedPct = 1 - (minQty / initialQty);

      if (consumedPct >= CONFIG.ABSORPTION_CONSUMED_PCT && hitCount >= CONFIG.ABSORPTION_MIN_HITS) {
        // Determine direction
        // Asks being absorbed = BULLISH (buyers consuming sellers)
        // Bids being absorbed = BEARISH (sellers consuming buyers)
        const pattern = side === 'asks'
          ? OF_PATTERN.ABSORPTION_BULLISH
          : OF_PATTERN.ABSORPTION_BEARISH;

        events.push({
          pattern,
          priceLevel,
          side,
          initialQty,
          finalQty: minQty,
          consumedPct: Math.round(consumedPct * 100),
          hitCount,
          startSnapshotIdx: startTime,
          endSnapshotIdx: lastIdx,
          startTime: snapshots[startTime]?.time,
          endTime: snapshots[lastIdx]?.time,
          fullyConsumed: consumed,
          duration: lastIdx - startTime,
          score: computeAbsorptionScore(consumedPct, hitCount, consumed, initialQty),
        });
      }
    }
  }

  // Sort by score descending
  events.sort((a, b) => b.score - a.score);
  return events;
}

/**
 * Score an absorption event (0–100).
 */
function computeAbsorptionScore(consumedPct, hitCount, fullyConsumed, initialQty) {
  let score = 0;

  // Consumption percentage (max 40 points)
  score += Math.min(40, Math.round(consumedPct * 40));

  // Hit count (max 20 points)
  score += Math.min(20, hitCount * 4);

  // Full consumption bonus (15 points)
  if (fullyConsumed) score += 15;

  // Size of initial order (max 15 points)
  const qtyNorm = Math.min(1, initialQty / 100_000);
  score += Math.round(qtyNorm * 15);

  // Extra for very large orders (max 10 points)
  if (initialQty > 200_000) score += 10;

  return Math.min(100, score);
}


// ═══════════════════════════════════════════════
// SECTION 2: SPOOFING / FAKE ORDER DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect spoofing: large orders that appear and quickly disappear,
 * designed to mislead other traders.
 *
 * Characteristics:
 *   - Large qty appears at a bid/ask level
 *   - Within seconds, qty drops by 80%+ or vanishes entirely
 *   - This pattern repeats
 *   - Price may briefly move in the fake order's direction before reversing
 *
 * @param {object[]} snapshots
 * @returns {object[]} detected spoof events
 */
export function detectSpoofing(snapshots) {
  if (snapshots.length < 3) return [];

  const events = [];

  for (const side of ['bids', 'asks']) {
    const history = buildLevelHistory(snapshots, side);

    for (const [priceLevel, entries] of history) {
      let spoofSequences = [];
      let currentSpoof = null;

      for (let i = 0; i < entries.length; i++) {
        const entry = entries[i];

        // Large order appears
        if (entry.qty >= CONFIG.SPOOF_MIN_QTY && !currentSpoof) {
          currentSpoof = {
            appearedIdx: entry.snapshotIdx,
            appearedQty: entry.qty,
            priceLevel,
          };
          continue;
        }

        // Track if it disappears quickly
        if (currentSpoof) {
          const elapsedSnapshots = entry.snapshotIdx - currentSpoof.appearedIdx;

          if (elapsedSnapshots <= CONFIG.SPOOF_APPEAR_DISAPPEAR_SEC) {
            const qtyDrop = 1 - (entry.qty / currentSpoof.appearedQty);

            if (qtyDrop >= CONFIG.SPOOF_QTY_DROP_PCT || entry.qty === 0) {
              spoofSequences.push({
                appearedIdx: currentSpoof.appearedIdx,
                disappearedIdx: entry.snapshotIdx,
                appearedQty: currentSpoof.appearedQty,
                remainingQty: entry.qty,
                dropPct: Math.round(qtyDrop * 100),
                duration: elapsedSnapshots,
              });
              currentSpoof = null;
            }
          } else {
            // Held too long to be a spoof
            currentSpoof = null;
          }
        }
      }

      // Need repeated spoof behaviour at same level
      if (spoofSequences.length >= CONFIG.SPOOF_REPEAT_COUNT) {
        const pattern = side === 'bids' ? OF_PATTERN.SPOOF_BID : OF_PATTERN.SPOOF_ASK;

        events.push({
          pattern,
          priceLevel,
          side,
          spoofCount: spoofSequences.length,
          sequences: spoofSequences,
          firstAppearance: snapshots[spoofSequences[0].appearedIdx]?.time,
          avgAppearedQty: Math.round(
            spoofSequences.reduce((s, sq) => s + sq.appearedQty, 0) / spoofSequences.length
          ),
          avgDuration: Math.round(
            spoofSequences.reduce((s, sq) => s + sq.duration, 0) / spoofSequences.length
          ),
          // Spoof on bids = fake buying support = actually bearish
          // Spoof on asks = fake selling pressure = actually bullish
          impliedDirection: side === 'bids' ? 'BEARISH' : 'BULLISH',
          score: Math.min(100, spoofSequences.length * 25 + 20),
        });
      }
    }
  }

  return events;
}


// ═══════════════════════════════════════════════
// SECTION 3: ICEBERG ORDER DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect iceberg orders: small displayed quantity that keeps refilling
 * at the same price level, indicating a large hidden order.
 *
 * @param {object[]} snapshots
 * @returns {object[]} detected iceberg events
 */
export function detectIcebergOrders(snapshots) {
  if (snapshots.length < 5) return [];

  const events = [];

  for (const side of ['bids', 'asks']) {
    const history = buildLevelHistory(snapshots, side);

    for (const [priceLevel, entries] of history) {
      if (entries.length < CONFIG.ICEBERG_REFILL_COUNT) continue;

      // Look for pattern: qty drops to near 0, then jumps back up
      let refills = [];
      let totalConsumed = 0;

      for (let i = 1; i < entries.length; i++) {
        const prev = entries[i - 1];
        const curr = entries[i];
        const windowElapsed = curr.snapshotIdx - entries[0].snapshotIdx;

        if (windowElapsed > CONFIG.ICEBERG_REFILL_WINDOW_SEC * (1 / CONFIG.SNAPSHOT_INTERVAL_SEC)) break;

        // Qty dropped significantly then refilled
        if (prev.qty < CONFIG.ICEBERG_REFILL_QTY_MIN && curr.qty >= CONFIG.ICEBERG_REFILL_QTY_MIN) {
          // This looks like a refill
          const consumed = prev.qty === 0 ? curr.qty : curr.qty - prev.qty;
          totalConsumed += Math.abs(consumed);

          refills.push({
            snapshotIdx: curr.snapshotIdx,
            preRefillQty: prev.qty,
            postRefillQty: curr.qty,
            consumed: Math.abs(consumed),
          });
        }

        // Also track when qty decreases (consumption between refills)
        if (curr.qty < prev.qty) {
          totalConsumed += prev.qty - curr.qty;
        }
      }

      if (refills.length >= CONFIG.ICEBERG_REFILL_COUNT &&
          totalConsumed >= CONFIG.ICEBERG_TOTAL_CONSUMED_MIN) {

        const pattern = side === 'bids' ? OF_PATTERN.ICEBERG_BID : OF_PATTERN.ICEBERG_ASK;

        events.push({
          pattern,
          priceLevel,
          side,
          refillCount: refills.length,
          totalConsumed,
          refills,
          startTime: snapshots[entries[0].snapshotIdx]?.time,
          // Iceberg bid = real buying support = bullish
          // Iceberg ask = real selling pressure = bearish
          impliedDirection: side === 'bids' ? 'BULLISH' : 'BEARISH',
          score: Math.min(100, refills.length * 15 + Math.min(40, Math.round(totalConsumed / 5000))),
        });
      }
    }
  }

  return events;
}


// ═══════════════════════════════════════════════
// SECTION 4: STACKING DETECTION
// ═══════════════════════════════════════════════

/**
 * Detect stacking: multiple large orders at consecutive price levels,
 * indicating institutional position building.
 *
 * @param {object[]} snapshots – use latest snapshot(s)
 * @returns {object[]} detected stacking events
 */
export function detectStacking(snapshots) {
  if (snapshots.length === 0) return [];

  const events = [];

  // Use the most recent snapshot
  const snap = snapshots[snapshots.length - 1];

  for (const side of ['bids', 'asks']) {
    const levels = snap[side] || [];
    if (levels.length < CONFIG.STACKING_MIN_LEVELS) continue;

    // Check consecutive levels with large quantities
    let stackStart = -1;
    let stackLevels = [];
    let stackTotalQty = 0;

    for (let i = 0; i < levels.length; i++) {
      const level = levels[i];

      if (level.qty >= CONFIG.STACKING_MIN_QTY_PER_LEVEL) {
        if (stackStart === -1) stackStart = i;
        stackLevels.push(level);
        stackTotalQty += level.qty;
      } else {
        // Streak broken — evaluate what we have
        if (stackLevels.length >= CONFIG.STACKING_MIN_LEVELS &&
            stackTotalQty >= CONFIG.STACKING_TOTAL_QTY_THRESHOLD) {

          // Verify price spacing is tight (institutional, not random)
          const prices = stackLevels.map(l => l.price);
          const maxSpacing = Math.max(
            ...prices.slice(1).map((p, idx) => Math.abs(p - prices[idx]) / prices[idx])
          );

          if (maxSpacing <= CONFIG.STACKING_LEVEL_SPACING_PCT) {
            const pattern = side === 'bids' ? OF_PATTERN.STACKING_BID : OF_PATTERN.STACKING_ASK;

            events.push({
              pattern,
              side,
              levels: [...stackLevels],
              levelCount: stackLevels.length,
              totalQty: stackTotalQty,
              avgQtyPerLevel: Math.round(stackTotalQty / stackLevels.length),
              priceRange: {
                top: Math.max(...prices),
                bottom: Math.min(...prices),
              },
              maxSpacingPct: Math.round(maxSpacing * 10000) / 10000,
              time: snap.time,
              // Stacked bids = buying wall = bullish
              // Stacked asks = selling wall = bearish (but could be absorbed)
              impliedDirection: side === 'bids' ? 'BULLISH' : 'BEARISH',
              score: computeStackingScore(stackLevels.length, stackTotalQty, maxSpacing),
            });
          }
        }

        // Reset
        stackStart = -1;
        stackLevels = [];
        stackTotalQty = 0;
      }
    }

    // Check final streak
    if (stackLevels.length >= CONFIG.STACKING_MIN_LEVELS &&
        stackTotalQty >= CONFIG.STACKING_TOTAL_QTY_THRESHOLD) {
      const prices = stackLevels.map(l => l.price);
      const maxSpacing = Math.max(
        ...prices.slice(1).map((p, idx) => Math.abs(p - prices[idx]) / prices[idx])
      );

      if (maxSpacing <= CONFIG.STACKING_LEVEL_SPACING_PCT) {
        const pattern = side === 'bids' ? OF_PATTERN.STACKING_BID : OF_PATTERN.STACKING_ASK;
        events.push({
          pattern, side,
          levels: [...stackLevels], levelCount: stackLevels.length,
          totalQty: stackTotalQty,
          avgQtyPerLevel: Math.round(stackTotalQty / stackLevels.length),
          priceRange: { top: Math.max(...prices), bottom: Math.min(...prices) },
          maxSpacingPct: Math.round(maxSpacing * 10000) / 10000,
          time: snap.time,
          impliedDirection: side === 'bids' ? 'BULLISH' : 'BEARISH',
          score: computeStackingScore(stackLevels.length, stackTotalQty, maxSpacing),
        });
      }
    }
  }

  return events;
}

function computeStackingScore(levelCount, totalQty, maxSpacing) {
  let score = 0;
  score += Math.min(30, (levelCount - 2) * 10);            // levels (max 30)
  score += Math.min(40, Math.round(totalQty / 10_000));     // qty (max 40)
  score += maxSpacing < 0.001 ? 20 : 10;                    // tight spacing bonus
  score += totalQty > 200_000 ? 10 : 0;                     // very large bonus
  return Math.min(100, score);
}


// ═══════════════════════════════════════════════
// SECTION 5: COMPOSITE PATTERN SCAN
// ═══════════════════════════════════════════════

/**
 * Run all pattern detectors and return a consolidated view.
 *
 * @param {object[]} snapshots
 * @returns {object}
 *   {
 *     absorptions: [],
 *     spoofs: [],
 *     icebergs: [],
 *     stackings: [],
 *     allPatterns: [],           // sorted by score
 *     compositeBias: string,     // net implied direction
 *     compositeScore: number,    // 0–100
 *   }
 */
export function scanAllPatterns(snapshots) {
  const absorptions = detectAbsorption(snapshots);
  const spoofs = detectSpoofing(snapshots);
  const icebergs = detectIcebergOrders(snapshots);
  const stackings = detectStacking(snapshots);

  const allPatterns = [...absorptions, ...spoofs, ...icebergs, ...stackings]
    .sort((a, b) => b.score - a.score);

  // Composite scoring
  let bullishScore = 0;
  let bearishScore = 0;

  for (const p of allPatterns) {
    const dir = p.impliedDirection;
    if (dir === 'BULLISH') bullishScore += p.score;
    else if (dir === 'BEARISH') bearishScore += p.score;
  }

  let compositeBias = 'NEUTRAL';
  let compositeScore = 0;

  if (bullishScore > bearishScore && bullishScore > 30) {
    compositeBias = 'BULLISH';
    compositeScore = Math.min(100, bullishScore);
  } else if (bearishScore > bullishScore && bearishScore > 30) {
    compositeBias = 'BEARISH';
    compositeScore = Math.min(100, bearishScore);
  }

  return {
    absorptions,
    spoofs,
    icebergs,
    stackings,
    allPatterns,
    compositeBias,
    compositeScore,
    bullishScore,
    bearishScore,
    patternCount: allPatterns.length,
  };
}
```

## `orderflow-filters.js`

```javascript
/**
 * Order Flow Confirmation Filters
 *
 * Primary use: overlay on top of other strategies (ORB, VWAP, EMA, Gap)
 * to confirm or reject a signal based on real-time depth data.
 *
 * "This CONFIRMATION approach improves win rate by ~5-8%"
 */

import { CONFIG, DIRECTION, OF_BIAS, OF_PATTERN } from './orderflow-config.js';
import {
  computeSmoothedBAR, determineBias, aggregateDepthForCandle,
} from './orderflow-depth.js';
import { scanAllPatterns } from './orderflow-patterns.js';

// ════════════════════════════════════════════════
// TIME HELPERS
// ════════════════════════════════════════════════

export function toMinutes(h, m) { return h * 60 + m; }

export function candleTime(candle) {
  const d = candle.time instanceof Date ? candle.time : new Date(candle.time);
  return toMinutes(d.getHours(), d.getMinutes());
}

export function isInTradeWindow(candle) {
  const t = candleTime(candle);
  return t >= toMinutes(CONFIG.NO_TRADE_BEFORE_H, CONFIG.NO_TRADE_BEFORE_M) &&
         t <= toMinutes(CONFIG.LAST_ENTRY_H, CONFIG.LAST_ENTRY_M);
}

export function isPastMISExit(candle) {
  return candleTime(candle) >= toMinutes(CONFIG.MIS_EXIT_H, CONFIG.MIS_EXIT_M);
}


// ════════════════════════════════════════════════
// CONFIRMATION FILTER: BID-ASK RATIO
// ════════════════════════════════════════════════

/**
 * Check if bid-ask ratio aligns with proposed trade direction.
 *
 * For LONG: BAR should be ≥ CONFIRM_MIN_BAR (1.2)
 * For SHORT: BAR should be ≤ CONFIRM_MAX_BAR (0.8)
 *
 * @param {object[]} depthSnapshots – recent depth snapshots
 * @param {string}   direction      – DIRECTION.LONG | SHORT
 * @returns {{ pass: boolean, bar: number, bias: string, strength: number, reason: string }}
 */
export function confirmBAR(depthSnapshots, direction) {
  const smoothed = computeSmoothedBAR(depthSnapshots);
  const { bias, strength, reason } = determineBias(smoothed.smoothedBAR, 0);

  let pass = false;

  if (direction === DIRECTION.LONG) {
    pass = smoothed.smoothedBAR >= CONFIG.CONFIRM_MIN_BAR;
  } else if (direction === DIRECTION.SHORT) {
    pass = smoothed.smoothedBAR <= CONFIG.CONFIRM_MAX_BAR;
  }

  return {
    pass,
    bar: smoothed.smoothedBAR,
    latestBAR: smoothed.latestBAR,
    bias,
    strength,
    barStable: smoothed.barStable,
    reason: pass
      ? `BAR ${smoothed.smoothedBAR} confirms ${direction}`
      : `BAR ${smoothed.smoothedBAR} does NOT confirm ${direction} (need ${direction === DIRECTION.LONG ? `≥${CONFIG.CONFIRM_MIN_BAR}` : `≤${CONFIG.CONFIRM_MAX_BAR}`})`,
  };
}


// ════════════════════════════════════════════════
// CONFIRMATION FILTER: PATTERN ALIGNMENT
// ════════════════════════════════════════════════

/**
 * Check if detected order flow patterns align with trade direction.
 *
 * @param {object[]} depthSnapshots
 * @param {string}   direction
 * @returns {{ pass: boolean, patterns: object[], score: number, reason: string }}
 */
export function confirmPatterns(depthSnapshots, direction) {
  if (!CONFIG.CONFIRM_PATTERN_BONUS) return { pass: true, patterns: [], score: 0 };

  const scan = scanAllPatterns(depthSnapshots);

  const expectedBias = direction === DIRECTION.LONG ? 'BULLISH' : 'BEARISH';
  const alignedPatterns = scan.allPatterns.filter(p => p.impliedDirection === expectedBias);
  const opposingPatterns = scan.allPatterns.filter(p => p.impliedDirection !== expectedBias && p.impliedDirection !== 'NEUTRAL');

  // Check for spoofing AGAINST our direction
  const spoofAgainst = scan.spoofs.filter(s => {
    // Spoof on bids implies bearish (fake buy support) → bad for longs
    if (direction === DIRECTION.LONG && s.pattern === OF_PATTERN.SPOOF_BID) return true;
    // Spoof on asks implies bullish (fake sell pressure) → bad for shorts
    if (direction === DIRECTION.SHORT && s.pattern === OF_PATTERN.SPOOF_ASK) return true;
    return false;
  });

  const hasSpoofAgainst = CONFIG.CONFIRM_SPOOF_REJECTS && spoofAgainst.length > 0;

  // Aligned absorption or stacking = strong confirmation
  const hasStrongPattern = alignedPatterns.some(p =>
    p.score >= 60 && (
      p.pattern.includes('ABSORPTION') ||
      p.pattern.includes('STACKING') ||
      p.pattern.includes('ICEBERG')
    )
  );

  const alignedScore = alignedPatterns.reduce((s, p) => s + p.score, 0);
  const opposingScore = opposingPatterns.reduce((s, p) => s + p.score, 0);

  const netScore = alignedScore - opposingScore;

  // Pass if no spoof against AND (aligned patterns exist OR no opposing patterns)
  const pass = !hasSpoofAgainst && netScore >= 0;

  return {
    pass,
    patterns: alignedPatterns,
    opposingPatterns,
    spoofAgainst,
    hasStrongPattern,
    alignedScore,
    opposingScore,
    netScore,
    compositeBias: scan.compositeBias,
    reason: pass
      ? `Patterns confirm ${direction}: ${alignedPatterns.length} aligned (score ${alignedScore})`
      : hasSpoofAgainst
        ? `Spoof detected against ${direction} — rejecting`
        : `Net pattern score ${netScore} opposing ${direction}`,
  };
}


// ════════════════════════════════════════════════
// CONFIRMATION FILTER: DELTA ALIGNMENT
// ════════════════════════════════════════════════

/**
 * Check if cumulative delta supports the trade direction.
 *
 * @param {object} depthSummary – from aggregateDepthForCandle()
 * @param {string} direction
 */
export function confirmDelta(depthSummary, direction) {
  if (!depthSummary || !depthSummary.hasData) {
    return { pass: true, reason: 'No delta data available — neutral' };
  }

  const { cumDelta, avgDelta, hasDeltaSurge } = depthSummary;

  let pass = true;
  const reasons = [];

  // Delta should be positive for longs, negative for shorts
  if (direction === DIRECTION.LONG) {
    if (cumDelta < 0) {
      pass = false;
      reasons.push(`Cum delta ${cumDelta} negative (sellers in control)`);
    }
    if (hasDeltaSurge && avgDelta < 0) {
      pass = false;
      reasons.push('Delta surge in selling direction');
    }
  } else {
    if (cumDelta > 0) {
      pass = false;
      reasons.push(`Cum delta ${cumDelta} positive (buyers in control)`);
    }
    if (hasDeltaSurge && avgDelta > 0) {
      pass = false;
      reasons.push('Delta surge in buying direction');
    }
  }

  return {
    pass,
    cumDelta,
    avgDelta,
    hasDeltaSurge,
    reason: pass ? `Delta confirms ${direction}` : reasons.join('; '),
  };
}


// ════════════════════════════════════════════════
// COMPOSITE CONFIRMATION FILTER
// ════════════════════════════════════════════════

/**
 * Run all order flow confirmation checks for a proposed trade.
 *
 * This is the primary API that other strategies call.
 *
 * @param {object} ctx
 *   {
 *     direction: string,
 *     depthSnapshots: object[],     // recent depth snapshots
 *     depthSummary?: object,        // pre-aggregated candle depth summary
 *     requireAllPass?: boolean,     // all filters must pass (default: true)
 *   }
 * @returns {object}
 *   {
 *     confirmed: boolean,
 *     overallScore: number,    // 0–100, composite confirmation strength
 *     filters: object[],
 *     recommendation: string,  // 'ENTER' | 'SKIP' | 'CAUTION'
 *   }
 */
export function confirmOrderFlow(ctx) {
  const {
    direction,
    depthSnapshots = [],
    depthSummary = null,
    requireAllPass = true,
  } = ctx;

  if (depthSnapshots.length === 0 && !depthSummary) {
    return {
      confirmed: true,   // no data = don't block
      overallScore: 50,
      filters: [],
      recommendation: 'ENTER',
      reason: 'No depth data — proceeding without confirmation',
    };
  }

  const filters = [];

  // 1. Bid-Ask Ratio
  const barResult = confirmBAR(depthSnapshots, direction);
  filters.push({ name: 'BidAskRatio', ...barResult });

  // 2. Pattern alignment
  const patternResult = confirmPatterns(depthSnapshots, direction);
  filters.push({ name: 'Patterns', ...patternResult });

  // 3. Delta alignment
  const summary = depthSummary || aggregateDepthForCandle(depthSnapshots);
  const deltaResult = confirmDelta(summary, direction);
  filters.push({ name: 'Delta', ...deltaResult });

  // Compute overall
  const allPassed = filters.every(f => f.pass);
  const passCount = filters.filter(f => f.pass).length;
  const totalFilters = filters.length;

  const confirmed = requireAllPass ? allPassed : passCount >= Math.ceil(totalFilters / 2);

  // Score: weighted average of filter results
  let overallScore = 50; // neutral base
  if (barResult.pass) overallScore += 15;
  if (barResult.barStable && barResult.pass) overallScore += 5;
  if (patternResult.hasStrongPattern) overallScore += 20;
  if (patternResult.pass) overallScore += 5;
  if (deltaResult.pass) overallScore += 5;
  if (!barResult.pass) overallScore -= 15;
  if (patternResult.spoofAgainst?.length > 0) overallScore -= 25;
  if (!deltaResult.pass) overallScore -= 10;

  overallScore = Math.max(0, Math.min(100, overallScore));

  let recommendation = 'ENTER';
  if (!confirmed) recommendation = 'SKIP';
  else if (overallScore < 60) recommendation = 'CAUTION';

  return {
    confirmed,
    overallScore,
    filters,
    recommendation,
    passCount,
    totalFilters,
    depthBias: summary.bias,
    depthBiasStrength: summary.biasStrength,
  };
}


// ════════════════════════════════════════════════
// EXIT FILTER: DEPTH REVERSAL
// ════════════════════════════════════════════════

/**
 * During a live trade, check if order flow has turned against the position.
 * Used to tighten stops or force exit.
 *
 * @param {object[]} depthSnapshots – recent snapshots (current candle)
 * @param {string}   direction      – current position direction
 * @returns {{ shouldExit: boolean, shouldTightenSL: boolean, reason: string }}
 */
export function checkDepthReversal(depthSnapshots, direction) {
  if (depthSnapshots.length < 3) {
    return { shouldExit: false, shouldTightenSL: false, reason: 'Insufficient data' };
  }

  const smoothed = computeSmoothedBAR(depthSnapshots);
  const scan = scanAllPatterns(depthSnapshots);

  const isLong = direction === DIRECTION.LONG;

  // Strong reversal: BAR completely flipped
  const barFlipped = isLong
    ? smoothed.smoothedBAR <= CONFIG.BAR_BEARISH_THRESHOLD
    : smoothed.smoothedBAR >= CONFIG.BAR_BULLISH_THRESHOLD;

  // Large opposing order appeared
  const latestSnap = depthSnapshots[depthSnapshots.length - 1];
  const opposingSide = isLong ? 'asks' : 'bids';
  const largeOpposingOrder = (latestSnap[opposingSide] || [])
    .some(level => level.qty >= CONFIG.SPOOF_MIN_QTY);

  // Opposing stacking
  const opposingStack = scan.stackings.some(s =>
    (isLong && s.side === 'asks') || (!isLong && s.side === 'bids')
  );

  const shouldExit = barFlipped && (opposingStack || largeOpposingOrder);
  const shouldTightenSL = !shouldExit && (barFlipped || largeOpposingOrder);

  return {
    shouldExit,
    shouldTightenSL,
    bar: smoothed.smoothedBAR,
    barFlipped,
    largeOpposingOrder,
    opposingStack,
    reason: shouldExit
      ? `BAR flipped + opposing flow detected — exit ${direction}`
      : shouldTightenSL
        ? `BAR shifting against ${direction} — tighten SL`
        : `Depth still supports ${direction}`,
  };
}
```

## `orderflow-core.js`

```javascript
/**
 * Order Flow Core Engine
 *
 * Two operation modes:
 *   1. STANDALONE — generate independent signals from depth data
 *   2. CONFIRMATION — attach to candle data, augment other strategies
 *
 * Data contract for augmented candles:
 * {
 *   ...standard OHLCV candle,
 *   depthSnapshots?: DepthSnapshot[],   // raw depth data for this candle period
 *   depthSummary?: object,              // pre-aggregated depth metrics
 * }
 */

import { CONFIG, DIRECTION, OF_MODE, OF_PATTERN, EXIT_REASON } from './orderflow-config.js';
import {
  computeRawDepthMetrics, computeSmoothedBAR, determineBias,
  aggregateDepthForCandle, computeCumulativeDelta,
} from './orderflow-depth.js';
import { scanAllPatterns, detectAbsorption } from './orderflow-patterns.js';
import {
  confirmOrderFlow, checkDepthReversal,
  isInTradeWindow, isPastMISExit, candleTime, toMinutes,
} from './orderflow-filters.js';


// ═══════════════════════════════════════════════
// SECTION 1: AUGMENT CANDLES WITH DEPTH DATA
// ═══════════════════════════════════════════════

/**
 * Attach depth summaries to each candle.
 * Maps depth snapshots to their corresponding candle time windows.
 *
 * @param {object[]} candles     – sorted 5-min candles
 * @param {object[]} allSnapshots – all depth snapshots for the day, sorted by time
 * @returns {object[]} candles with depthSummary attached
 */
export function augmentCandlesWithDepth(candles, allSnapshots) {
  if (allSnapshots.length === 0) {
    for (const c of candles) c.depthSummary = null;
    return candles;
  }

  const intervalMs = CONFIG.CANDLE_INTERVAL_MIN * 60 * 1000;

  for (const candle of candles) {
    const candleStart = candle.time instanceof Date ? candle.time.getTime() : new Date(candle.time).getTime();
    const candleEnd = candleStart + intervalMs;

    // Find snapshots within this candle's window
    const candleSnapshots = allSnapshots.filter(s => {
      const t = s.time instanceof Date ? s.time.getTime()
        : typeof s.time === 'number' ? s.time
        : new Date(s.time).getTime();
      return t >= candleStart && t < candleEnd;
    });

    candle.depthSnapshots = candleSnapshots;
    candle.depthSummary = aggregateDepthForCandle(candleSnapshots);
  }

  return candles;
}


// ═══════════════════════════════════════════════
// SECTION 2: STANDALONE SIGNAL DETECTION
// ═══════════════════════════════════════════════

/**
 * Generate standalone trading signals purely from order flow patterns.
 *
 * Signal logic:
 *   1. Absorption complete → breakout signal
 *   2. Stacking + strong BAR alignment → trend signal
 *   3. Spoof detected → contrarian signal (fade the spoof direction)
 *   4. Delta surge → momentum signal
 *
 * @param {object[]} candles – augmented with depthSummary
 * @param {object}   opts
 * @returns {object[]} signals
 */
export function detectStandaloneSignals(candles, opts = {}) {
  const signals = [];

  for (let i = 1; i < candles.length; i++) {
    const candle = candles[i];
    if (!isInTradeWindow(candle)) continue;
    if (!candle.depthSnapshots || candle.depthSnapshots.length < 3) continue;

    const snapshots = candle.depthSnapshots;
    const prevSnapshots = candles[i - 1]?.depthSnapshots || [];
    const combinedSnapshots = [...prevSnapshots.slice(-20), ...snapshots];

    // Run pattern scan
    const scan = scanAllPatterns(combinedSnapshots);
    const depthSummary = candle.depthSummary;

    // Compute composite signal score
    const signalResult = evaluateStandaloneSignal(scan, depthSummary, candle);

    if (signalResult.score >= CONFIG.STANDALONE_MIN_PATTERN_SCORE) {
      const direction = signalResult.direction;
      const entryPrice = candle.close;
      const { sl, targets } = computeStandaloneSLTargets(entryPrice, direction);

      signals.push({
        mode: OF_MODE.STANDALONE,
        direction,
        entryPrice,
        entryTime: candle.time,
        entryIndex: i,
        sl,
        targets,
        patternScore: signalResult.score,
        primaryPattern: signalResult.primaryPattern,
        patterns: signalResult.patterns,
        depthBias: depthSummary?.bias,
        depthBiasStrength: depthSummary?.biasStrength,
        bar: depthSummary?.avgBAR,
        cumDelta: depthSummary?.cumDelta,
      });
    }
  }

  return signals;
}

/**
 * Evaluate whether depth data generates a standalone signal.
 */
function evaluateStandaloneSignal(scan, depthSummary, candle) {
  let bullScore = 0;
  let bearScore = 0;
  const patterns = [];

  // Absorption events (highest value)
  for (const abs of scan.absorptions) {
    if (abs.pattern === OF_PATTERN.ABSORPTION_BULLISH && abs.fullyConsumed) {
      bullScore += 40;
      patterns.push(abs);
    } else if (abs.pattern === OF_PATTERN.ABSORPTION_BULLISH) {
      bullScore += 20;
      patterns.push(abs);
    }
    if (abs.pattern === OF_PATTERN.ABSORPTION_BEARISH && abs.fullyConsumed) {
      bearScore += 40;
      patterns.push(abs);
    } else if (abs.pattern === OF_PATTERN.ABSORPTION_BEARISH) {
      bearScore += 20;
      patterns.push(abs);
    }
  }

  // Stacking
  for (const stack of scan.stackings) {
    if (stack.impliedDirection === 'BULLISH') bullScore += stack.score * 0.3;
    else bearScore += stack.score * 0.3;
    patterns.push(stack);
  }

  // Iceberg orders
  for (const ice of scan.icebergs) {
    if (ice.impliedDirection === 'BULLISH') bullScore += ice.score * 0.25;
    else bearScore += ice.score * 0.25;
    patterns.push(ice);
  }

  // Spoofs (contrarian — spoof bid = actually bearish)
  for (const spoof of scan.spoofs) {
    if (spoof.impliedDirection === 'BULLISH') bullScore += 15;
    else bearScore += 15;
    patterns.push(spoof);
  }

  // BAR alignment bonus
  if (depthSummary) {
    if (depthSummary.bias === 'BULLISH') bullScore += depthSummary.biasStrength * 0.2;
    if (depthSummary.bias === 'BEARISH') bearScore += depthSummary.biasStrength * 0.2;
  }

  // Delta alignment
  if (depthSummary?.cumDelta > 0) bullScore += 10;
  if (depthSummary?.cumDelta < 0) bearScore += 10;

  const direction = bullScore > bearScore ? DIRECTION.LONG : DIRECTION.SHORT;
  const score = Math.round(Math.max(bullScore, bearScore));

  // Primary pattern: highest scoring one
  const primaryPattern = patterns.length > 0
    ? patterns.sort((a, b) => b.score - a.score)[0].pattern
    : null;

  return { direction, score, primaryPattern, patterns, bullScore, bearScore };
}


/**
 * Compute SL and targets for standalone order flow signals.
 */
function computeStandaloneSLTargets(entryPrice, direction) {
  const isLong = direction === DIRECTION.LONG;
  let sl = isLong
    ? entryPrice * (1 - CONFIG.STANDALONE_SL_PCT)
    : entryPrice * (1 + CONFIG.STANDALONE_SL_PCT);

  // Enforce max SL
  const slPct = Math.abs(entryPrice - sl) / entryPrice;
  if (slPct > CONFIG.MAX_SL_PCT) {
    sl = isLong
      ? entryPrice * (1 - CONFIG.MAX_SL_PCT)
      : entryPrice * (1 + CONFIG.MAX_SL_PCT);
  }

  const slDistance = Math.abs(entryPrice - sl);

  const targets = [
    {
      price: isLong
        ? entryPrice + slDistance * CONFIG.STANDALONE_TARGET_RR
        : entryPrice - slDistance * CONFIG.STANDALONE_TARGET_RR,
      qtyPct: CONFIG.TARGET_1_QTY_PCT,
      label: 'T1',
    },
    { price: null, qtyPct: CONFIG.TRAIL_QTY_PCT, label: 'TRAIL' },
  ];

  return { sl, targets };
}


// ═══════════════════════════════════════════════
// SECTION 3: TRADE SIMULATION
// ═══════════════════════════════════════════════

/**
 * Walk forward through candles managing a trade.
 * Uses depth reversal detection for dynamic exits.
 *
 * @param {object[]} candles  – augmented with depthSummary
 * @param {object}   signal
 * @returns {object} trade record
 */
export function simulateTrade(candles, signal) {
  const { direction, entryPrice, entryIndex, sl, targets } = signal;
  const isLong = direction === DIRECTION.LONG;

  let remainingQty = 1.0;
  let currentSL = sl;
  const partialExits = [];
  let t1Hit = false;
  let trailActive = false;

  for (let i = entryIndex + 1; i < candles.length; i++) {
    const c = candles[i];
    if (remainingQty <= 0) break;

    // ── EOD ──
    if (isPastMISExit(c)) {
      partialExits.push({
        time: c.time, price: c.close,
        qtyPct: remainingQty, reason: EXIT_REASON.TIME_STOP_EOD,
      });
      remainingQty = 0;
      break;
    }

    // ── Stop-loss ──
    const hitSL = isLong ? c.low <= currentSL : c.high >= currentSL;
    if (hitSL) {
      partialExits.push({
        time: c.time, price: currentSL,
        qtyPct: remainingQty,
        reason: trailActive ? EXIT_REASON.TRAIL_STOP : EXIT_REASON.STOP_LOSS,
      });
      remainingQty = 0;
      break;
    }

    // ── Depth reversal check ──
    if (c.depthSnapshots && c.depthSnapshots.length > 0) {
      const reversal = checkDepthReversal(c.depthSnapshots, direction);

      if (reversal.shouldExit && trailActive) {
        partialExits.push({
          time: c.time, price: c.close,
          qtyPct: remainingQty, reason: EXIT_REASON.DEPTH_REVERSAL,
        });
        remainingQty = 0;
        break;
      }

      if (reversal.shouldTightenSL) {
        // Tighten SL to previous candle's extreme
        if (isLong) {
          const tighterSL = candles[i - 1]?.low;
          if (tighterSL && tighterSL > currentSL) currentSL = tighterSL;
        } else {
          const tighterSL = candles[i - 1]?.high;
          if (tighterSL && tighterSL < currentSL) currentSL = tighterSL;
        }
      }

      // Check for large opposing order
      if (reversal.largeOpposingOrder && !t1Hit) {
        partialExits.push({
          time: c.time, price: c.close,
          qtyPct: remainingQty, reason: EXIT_REASON.LARGE_OPPOSING_ORDER,
        });
        remainingQty = 0;
        break;
      }
    }

    // ── Target 1 ──
    if (!t1Hit && targets[0]?.price != null) {
      const hitT1 = isLong ? c.high >= targets[0].price : c.low <= targets[0].price;

      if (hitT1) {
        t1Hit = true;
        const exitQty = Math.min(targets[0].qtyPct, remainingQty);
        partialExits.push({
          time: c.time, price: targets[0].price,
          qtyPct: exitQty, reason: EXIT_REASON.TARGET_1,
        });
        remainingQty -= exitQty;
        trailActive = CONFIG.STANDALONE_TRAIL_AFTER_T1;
        currentSL = entryPrice; // breakeven
      }
    }

    // ── Trail with previous candle low/high ──
    if (trailActive && remainingQty > 0) {
      const prevCandle = candles[i - 1];
      if (isLong) {
        const newTrail = prevCandle?.low;
        if (newTrail && newTrail > currentSL) currentSL = newTrail;
      } else {
        const newTrail = prevCandle?.high;
        if (newTrail && newTrail < currentSL) currentSL = newTrail;
      }
    }
  }

  return buildTradeRecord(signal, partialExits);
}

function buildTradeRecord(signal, partialExits) {
  const { direction, entryPrice } = signal;
  const isLong = direction === DIRECTION.LONG;

  let weightedExit = 0, totalQty = 0;
  for (const e of partialExits) {
    weightedExit += e.price * e.qtyPct;
    totalQty += e.qtyPct;
  }

  const avgExitPrice = totalQty > 0 ? weightedExit / totalQty : entryPrice;
  const grossPnlPct = isLong
    ? (avgExitPrice - entryPrice) / entryPrice
    : (entryPrice - avgExitPrice) / entryPrice;
  const netPnlPct = grossPnlPct - (CONFIG.SLIPPAGE_PER_SIDE * 2);

  const lastExit = partialExits[partialExits.length - 1];
  const reasonMap = {};
  for (const e of partialExits) reasonMap[e.reason] = (reasonMap[e.reason] || 0) + e.qtyPct;
  let primaryReason = 'NONE', maxQ = 0;
  for (const [r, q] of Object.entries(reasonMap)) { if (q > maxQ) { maxQ = q; primaryReason = r; } }

  return {
    mode: signal.mode || OF_MODE.STANDALONE,
    direction,
    entryPrice,
    avgExitPrice,
    entryTime: signal.entryTime,
    exitTime: lastExit?.time ?? null,
    grossPnlPct,
    netPnlPct,
    isWinner: netPnlPct > 0,
    partialExits,
    primaryExitReason: primaryReason,
    patternScore: signal.patternScore ?? null,
    primaryPattern: signal.primaryPattern ?? null,
    depthBias: signal.depthBias ?? null,
    depthBiasStrength: signal.depthBiasStrength ?? null,
    bar: signal.bar ?? null,
    cumDelta: signal.cumDelta ?? null,
    sl: signal.sl,
  };
}


// ═══════════════════════════════════════════════
// SECTION 4: SINGLE-DAY ORCHESTRATOR
// ═══════════════════════════════════════════════

/**
 * Run order flow strategy on one day.
 *
 * @param {object[]} dayCandlesRaw    – 5-min candles
 * @param {object[]} daySnapshots     – all depth snapshots for the day
 * @param {object}   opts
 *   { mode, maxTradesPerDay }
 * @returns {object[]} trade records
 */
export function runOrderFlowForDay(dayCandlesRaw, daySnapshots = [], opts = {}) {
  const { mode = OF_MODE.STANDALONE, maxTradesPerDay = 3 } = opts;

  const candles = dayCandlesRaw.map(c => ({
    ...c,
    time: c.time instanceof Date ? c.time : new Date(c.time),
  }));

  // Augment candles with depth data
  augmentCandlesWithDepth(candles, daySnapshots);

  if (mode === OF_MODE.STANDALONE) {
    const signals = detectStandaloneSignals(candles);

    // Simulate, no overlap
    const trades = [];
    let lastExitIdx = -1;

    for (const signal of signals) {
      if (trades.length >= maxTradesPerDay) break;
      if (signal.entryIndex <= lastExitIdx) continue;

      const trade = simulateTrade(candles, signal);
      trades.push(trade);

      const lastPartial = trade.partialExits[trade.partialExits.length - 1];
      if (lastPartial) {
        for (let j = signal.entryIndex; j < candles.length; j++) {
          const ct = candles[j].time instanceof Date ? candles[j].time : new Date(candles[j].time);
          const et = lastPartial.time instanceof Date ? lastPartial.time : new Date(lastPartial.time);
          if (ct >= et) { lastExitIdx = j; break; }
        }
      }
    }

    return trades;
  }

  // CONFIRMATION mode returns the augmented candles (other strategies use confirmOrderFlow)
  return candles;
}
```

```javascript
/**

 *
 * Order-flow-specific analytics:

 *  - BAR threshold optimization

export function analyzePerformance(trades, capital = CONFIG.DEFAULT_CAPITAL) {
  if (trades.length === 0) return { error: 'No trades' };

  const winners = trades.filter(t => t.isWinner);
  const losers = trades.filter(t => !t.isWinner);
  const pnls = trades.map(t => t.netPnlPct);

  const totalTrades = trades.length;
  const winRate = winners.length / totalTrades;
  const avgWin = safeMean(winners.map(t => t.netPnlPct));
  const avgLoss = safeMean(losers.map(t => t.netPnlPct));
  const avgWLR = Math.abs(avgLoss) > 0 ? avgWin / Math.abs(avgLoss) : Infinity;
  const expectancy = (winRate * avgWin) + ((1 - winRate) * avgLoss);

  const equity = buildEquityCurve(trades, capital);
  const totalReturn = ((equity[equity.length - 1].equity - capital) / capital) * 100;
  const maxDD = computeMaxDrawdown(equity);
  const streaks = computeStreaks(trades);

  return {
    totalTrades,
    winners: winners.length,
    losers: losers.length,
    winRatePct: r2(winRate * 100),
    avgWinPct: r4(avgWin * 100),
    avgLossPct: r4(avgLoss * 100),
    avgWinLossRatio: r2(avgWLR),
    expectancyPct: r4(expectancy * 100),
    totalReturnPct: r2(totalReturn),
    totalReturnAbs: r2(equity[equity.length - 1].equity - capital),
    maxDrawdownPct: r2(maxDD.pct),
    maxDrawdownDuration: maxDD.duration,
    sharpeRatio: r2(computeSharpe(pnls)),
    profitFactor: r2(computeProfitFactor(trades)),
    maxWinStreak: streaks.maxWinStreak,
    maxLossStreak: streaks.maxLossStreak,
    equityCurve: equity,
  };
}

/**

  for (const t of trades) {
    const key = t.primaryPattern || 'NO_PATTERN';
    if (!byPattern[key]) byPattern[key] = [];
    byPattern[key].push(t);
  }

  const result = {};
  for (const [pattern, pTrades] of Object.entries(byPattern)) {

    result[pattern] = {
      count: pTrades.length,
      winRatePct: r2((wins / pTrades.length) * 100),
      avgPnlPct: r4(safeMean(pTrades.map(t => t.netPnlPct)) * 100),
      avgScore: r2(safeMean(pTrades.map(t => t.patternScore || 0))),
    };
  }

  return result;
}

/**
 * Compare trades that were confirmed by order flow vs those that weren't.
 * Expects trades to have a `depthConfirmed` boolean field.
 *
 * @param {object[]} confirmedTrades   – trades where depth confirmed
 * @param {object[]} unconfirmedTrades – trades where depth was ignored/unavailable
 */
export function confirmationImpact(confirmedTrades, unconfirmedTrades) {
  const analyse = (trades, label) => {
    if (trades.length === 0) return { label, count: 0 };
    const wins = trades.filter(t => t.isWinner).length;
    return {
      label,
      count: trades.length,
      winRatePct: r2((wins / trades.length) * 100),
      avgPnlPct: r4(safeMean(trades.map(t => t.netPnlPct)) * 100),
      expectancy: r4(
        ((wins / trades.length) * safeMean(trades.filter(t => t.isWinner).map(t => t.netPnlPct)) +
         ((1 - wins / trades.length) * safeMean(trades.filter(t => !t.isWinner).map(t => t.netPnlPct)))) * 100
      ),
    };
  };

  const confirmed = analyse(confirmedTrades, 'With OF Confirmation');
  const unconfirmed = analyse(unconfirmedTrades, 'Without OF Confirmation');

  const winRateImprovement = confirmed.winRatePct - unconfirmed.winRatePct;

  return {
    confirmed,
    unconfirmed,
    winRateImprovementPct: r2(winRateImprovement),
    tradesFiltered: unconfirmedTrades.length - confirmedTrades.length,
    filterRate: unconfirmedTrades.length > 0
      ? r2(((unconfirmedTrades.length - confirmedTrades.length) / unconfirmedTrades.length) * 100)
      : 0,
  };
}

export function exitReasonBreakdown(trades) {
  const reasons = {};
  for (const t of trades) {
    const r = t.primaryExitReason || 'UNKNOWN';
    if (!reasons[r]) reasons[r] = { count: 0, totalPnl: 0, wins: 0 };
    reasons[r].count++;
    reasons[r].totalPnl += t.netPnlPct;
    if (t.isWinner) reasons[r].wins++;
  }
  for (const r of Object.keys(reasons)) {
    const d = reasons[r];
    d.winRatePct = r2((d.wins / d.count) * 100);
    d.avgPnlPct = r4((d.totalPnl / d.count) * 100);
  }
  return reasons;
}

export function monthlyBreakdown(trades) {
  const groups = {};
  for (const t of trades) {
    if (!t.entryTime) continue;
    const d = t.entryTime instanceof Date ? t.entryTime : new Date(t.entryTime);
    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
    if (!groups[key]) groups[key] = [];
    groups[key].push(t);
  }
  return Object.entries(groups).sort().map(([month, mTrades]) => {
    const wins = mTrades.filter(t => t.isWinner).length;
    return {
      month, trades: mTrades.length,
      winRatePct: r2((wins / mTrades.length) * 100),
      netPnlPct: r4(mTrades.reduce((s, t) => s + t.netPnlPct, 0) * 100),
    };
  });
}

function buildEquityCurve(trades, startCapital) {
  const curve = [{ tradeNum: 0, equity: startCapital }];
  let equity = startCapital;
  for (let i = 0; i < trades.length; i++) {
    const t = trades[i];
    const riskAmt = equity * CONFIG.DEFAULT_RISK_PER_TRADE;
    const slDist = Math.abs(t.entryPrice - t.sl) || t.entryPrice * 0.005;
    const qty = Math.floor(riskAmt / slDist);
    const pnl = qty * t.entryPrice * t.netPnlPct;
    equity += pnl - CONFIG.COMMISSION_PER_SIDE * 2;
    curve.push({ tradeNum: i + 1, equity: r2(equity), date: t.entryTime });
  }
  return curve;
}

function computeMaxDrawdown(curve) {
  let peak = curve[0].equity, maxDD = 0, ddStart = 0, maxDur = 0;
  for (let i = 1; i < curve.length; i++) {
    if (curve[i].equity > peak) { peak = curve[i].equity; ddStart = i; }
    const dd = (peak - curve[i].equity) / peak;
    if (dd > maxDD) { maxDD = dd; maxDur = i - ddStart; }
  }
  return { pct: maxDD * 100, duration: maxDur };
}

function computeStreaks(trades) {
  let mw = 0, ml = 0, cw = 0, cl = 0;
  for (const t of trades) {
    if (t.isWinner) { cw++; cl = 0; if (cw > mw) mw = cw; }
    else { cl++; cw = 0; if (cl > ml) ml = cl; }
  }
  return { maxWinStreak: mw, maxLossStreak: ml };
}

function computeSharpe(pnls) {
  if (pnls.length < 2) return 0;
  const avg = safeMean(pnls);
  const std = Math.sqrt(pnls.reduce((s, p) => s + (p - avg) ** 2, 0) / (pnls.length - 1));
  return std === 0 ? 0 : (avg / std) * Math.sqrt(250);
}

function computeProfitFactor(trades) {
  let gp = 0, gl = 0;
  for (const t of trades) { if (t.netPnlPct > 0) gp += t.netPnlPct; else gl += Math.abs(t.netPnlPct); }
  return gl > 0 ? gp / gl : Infinity;
}

function safeMean(arr) { return arr.length === 0 ? 0 : arr.reduce((a, b) => a + b, 0) / arr.length; }
function r2(n) { return Math.round(n * 100) / 100; }
function r4(n) { return Math.round(n * 10000) / 10000; }
```

```javascript
/**
 * Order Flow Runner
 *
 * Orchestrates both modes:
 *   1. STANDALONE — run order flow as the primary strategy
 *   2. CONFIRMATION — wrap around another strategy's signals
 *
 * Also provides confirmation impact measurement — the key value-add:
 *   "Did order flow confirmation actually improve win rate by 5-8%?"
 *
 * USAGE:
 *
 *   // Standalone
 *   import { runAndReport } from './orderflow-runner.js';
 *   const results = runAndReport(allCandles, allDepthSnapshots);
 *
 *   // Confirmation layer for another strategy
 *   import { wrapWithConfirmation, measureConfirmationImpact } from './orderflow-runner.js';
 *   const enhanced = wrapWithConfirmation(existingSignals, allCandles, allDepthSnapshots);
 */

import { CONFIG, OF_MODE } from './orderflow-config.js';
import { augmentCandlesWithDepth, runOrderFlowForDay, detectStandaloneSignals, simulateTrade } from './orderflow-core.js';
import { confirmOrderFlow } from './orderflow-filters.js';
import {
  analyzePerformance, patternEffectiveness, scoreVsOutcome,
  barThresholdAnalysis, confirmationImpact, exitReasonBreakdown,
  depthExitEffectiveness, monthlyBreakdown, printReport,
} from './orderflow-analyzer.js';


// ═══════════════════════════════════════════════
// GROUP BY DAY
// ═══════════════════════════════════════════════

export function groupByDay(items, timeField = 'time') {
  const days = new Map();
  for (const item of items) {
    const d = item[timeField] instanceof Date ? item[timeField] : new Date(item[timeField]);
    const key = d.toISOString().slice(0, 10);
    if (!days.has(key)) days.set(key, []);
    days.get(key).push(item);
  }
  return days;
}


// ═══════════════════════════════════════════════
// STANDALONE MULTI-DAY BACKTEST
// ═══════════════════════════════════════════════

/**
 * @param {object[]} allCandles    – flat 5-min candle array, multi-day
 * @param {object[]} allSnapshots  – flat depth snapshot array, multi-day
 * @param {object}   opts          – { capital, maxTradesPerDay }
 */
export function runFullBacktest(allCandles, allSnapshots, opts = {}) {
  const { capital = CONFIG.DEFAULT_CAPITAL, maxTradesPerDay = 3 } = opts;

  const candleDays = groupByDay(allCandles);
  const snapshotDays = groupByDay(allSnapshots);
  const sortedDates = [...candleDays.keys()].sort();

  const allTrades = [];
  const skippedDays = [];

  for (const dateKey of sortedDates) {
    const dayCandles = candleDays.get(dateKey) || [];
    const daySnapshots = snapshotDays.get(dateKey) || [];

    if (dayCandles.length < 10) {
      skippedDays.push({ date: dateKey, reason: 'Insufficient candles' });
      continue;
    }

    if (daySnapshots.length < 30) {
      skippedDays.push({ date: dateKey, reason: 'Insufficient depth data' });
      continue;
    }

    const dayTrades = runOrderFlowForDay(dayCandles, daySnapshots, {
      mode: OF_MODE.STANDALONE,
      maxTradesPerDay,
    });

    for (const t of dayTrades) {
      t.date = dateKey;
      allTrades.push(t);
    }

    if (dayTrades.length === 0) {
      skippedDays.push({ date: dateKey, reason: 'No valid signal' });
    }
  }

  const report = allTrades.length > 0 ? analyzePerformance(allTrades, capital) : null;

  return {
    trades: allTrades,
    report,
    totalDays: sortedDates.length,
    tradedDays: new Set(allTrades.map(t => t.date)).size,
    skippedDays,
  };
}


// ═══════════════════════════════════════════════
// CONFIRMATION MODE: WRAP EXISTING SIGNALS
// ═══════════════════════════════════════════════

/**
 * Take signals from another strategy (ORB, VWAP, EMA, etc.) and
 * filter them through order flow confirmation.
 *
 * Each signal must have:
 *   { direction, entryTime, entryIndex, entryPrice, ... }
 *
 * Returns:
 *   - confirmedSignals: signals that passed OF check
 *   - rejectedSignals: signals that failed OF check
 *   - Each signal gets `ofConfirmation` object attached
 *
 * @param {object[]} signals       – from another strategy
 * @param {object[]} allCandles    – flat candle array
 * @param {object[]} allSnapshots  – flat depth snapshot array
 * @param {object}   opts
 *   { requireAllPass, lookbackSnapshots }
 * @returns {{ confirmed: object[], rejected: object[] }}
 */
export function wrapWithConfirmation(signals, allCandles, allSnapshots, opts = {}) {
  const {
    requireAllPass = true,
    lookbackSnapshots = CONFIG.BAR_SMOOTHING_SNAPSHOTS * 3,
  } = opts;

  // Pre-augment candles
  const candleDays = groupByDay(allCandles);
  const snapshotDays = groupByDay(allSnapshots);

  for (const [dateKey, dayCandles] of candleDays) {
    const daySnaps = snapshotDays.get(dateKey) || [];
    augmentCandlesWithDepth(dayCandles, daySnaps);
  }

  // Build flat augmented candle array for lookup
  const augmented = [...candleDays.values()].flat();

  const confirmed = [];
  const rejected = [];

  for (const signal of signals) {
    const signalTime = signal.entryTime instanceof Date
      ? signal.entryTime : new Date(signal.entryTime);

    // Find relevant depth snapshots around signal time
    const signalTs = signalTime.getTime();
    const lookbackMs = lookbackSnapshots * CONFIG.SNAPSHOT_INTERVAL_SEC * 1000;

    const relevantSnapshots = allSnapshots.filter(s => {
      const t = s.time instanceof Date ? s.time.getTime()
        : typeof s.time === 'number' ? s.time
        : new Date(s.time).getTime();
      return t >= (signalTs - lookbackMs) && t <= signalTs;
    });

    // Find corresponding candle for depth summary
    const signalDate = signalTime.toISOString().slice(0, 10);
    const dayCandles = candleDays.get(signalDate) || [];
    const matchCandle = dayCandles.find(c => {
      const ct = c.time instanceof Date ? c.time : new Date(c.time);
      return ct.getTime() <= signalTs &&
             ct.getTime() + CONFIG.CANDLE_INTERVAL_MIN * 60 * 1000 > signalTs;
    });

    const ofResult = confirmOrderFlow({
      direction: signal.direction,
      depthSnapshots: relevantSnapshots,
      depthSummary: matchCandle?.depthSummary || null,
      requireAllPass,
    });

    const enrichedSignal = {
      ...signal,
      ofConfirmation: ofResult,
      depthConfirmed: ofResult.confirmed,
    };

    if (ofResult.confirmed) {
      confirmed.push(enrichedSignal);
    } else {
      rejected.push(enrichedSignal);
    }
  }

  return { confirmed, rejected };
}

/**
 * Run another strategy's trades with and without OF confirmation,
 * compare win rates and PnL.
 *
 * @param {object[]} allTrades         – original trades from another strategy
 * @param {object[]} confirmedTrades   – trades that passed OF confirmation
 * @returns {object} impact analysis
 */
export function measureConfirmationImpact(allTrades, confirmedTrades) {
  return confirmationImpact(confirmedTrades, allTrades);
}


// ═══════════════════════════════════════════════
// MULTI-STOCK SCAN
// ═══════════════════════════════════════════════

export function multiStockScan(stockData, opts = {}) {
  const results = [];

  // stockData: Map<symbol, { candles: [], snapshots: [] }>
  for (const [symbol, data] of stockData) {
    const res = runFullBacktest(data.candles, data.snapshots, opts);

    results.push({
      symbol,
      trades: res.report?.totalTrades ?? 0,
      winRatePct: res.report?.winRatePct ?? 0,
      expectancyPct: res.report?.expectancyPct ?? 0,
      returnPct: res.report?.totalReturnPct ?? 0,
      maxDDPct: res.report?.maxDrawdownPct ?? 0,
      profitFactor: res.report?.profitFactor ?? 0,
    });
  }

  results.sort((a, b) => (b.expectancyPct || 0) - (a.expectancyPct || 0));
  return results;
}


// ═══════════════════════════════════════════════
// FULL REPORT
// ═══════════════════════════════════════════════

export function runAndReport(allCandles, allSnapshots, opts = {}) {
  const result = runFullBacktest(allCandles, allSnapshots, opts);

  if (!result.report) {
    console.log('No trades generated.');
    return result;
  }

  const trades = result.trades;

  const extras = {
    patternEffectiveness: patternEffectiveness(trades),
    scoreVsOutcome: scoreVsOutcome(trades),
    barAnalysis: barThresholdAnalysis(trades),
    exitReasons: exitReasonBreakdown(trades),
    depthExits: depthExitEffectiveness(trades),
  };

  printReport(result.report, extras);

  console.log('\n📅 MONTHLY:');
  console.table(monthlyBreakdown(trades));

  console.log(`\nDays: ${result.totalDays} | Traded: ${result.tradedDays}`);

  return { ...result, extras };
}


/**
 * Full confirmation mode demo: run another strategy, apply OF filter, compare.
 *
 * @param {object[]} baseSignals   – signals from ORB/VWAP/EMA/Gap strategy
 * @param {object[]} baseTrades    – simulated trades from base strategy
 * @param {object[]} allCandles
 * @param {object[]} allSnapshots
 */
export function runConfirmationAnalysis(baseSignals, baseTrades, allCandles, allSnapshots, opts = {}) {
  console.log('\n' + '═'.repeat(60));
  console.log('  ORDER FLOW CONFIRMATION ANALYSIS');
  console.log('═'.repeat(60));

  // Step 1: Filter signals through OF
  const { confirmed, rejected } = wrapWithConfirmation(baseSignals, allCandles, allSnapshots, opts);

  console.log(`\n  Base strategy signals:     ${baseSignals.length}`);
  console.log(`  OF confirmed:              ${confirmed.length}`);
  console.log(`  OF rejected:               ${rejected.length}`);
  console.log(`  Filter rate:               ${r2((rejected.length / baseSignals.length) * 100)}%`);

  // Step 2: Match confirmed signals to their trades
  const confirmedTrades = baseTrades.filter(t =>
    confirmed.some(c => {
      const ct = c.entryTime instanceof Date ? c.entryTime.getTime() : new Date(c.entryTime).getTime();
      const tt = t.entryTime instanceof Date ? t.entryTime.getTime() : new Date(t.entryTime).getTime();
      return Math.abs(ct - tt) < 60000 && c.direction === t.direction;
    })
  );

  // Step 3: Compare
  const impact = measureConfirmationImpact(baseTrades, confirmedTrades);

  console.log(`\n  ── IMPACT ──`);
  console.log(`  Base WR:       ${impact.unconfirmed.winRatePct}% (${impact.unconfirmed.count} trades)`);
  console.log(`  Confirmed WR:  ${impact.confirmed.winRatePct}% (${impact.confirmed.count} trades)`);
  console.log(`  Improvement:   ${impact.winRateImprovementPct > 0 ? '+' : ''}${impact.winRateImprovementPct}%`);
  console.log(`  Base Avg PnL:  ${impact.unconfirmed.avgPnlPct}%`);
  console.log(`  Conf Avg PnL:  ${impact.confirmed.avgPnlPct}%`);
  console.log('\n' + '═'.repeat(60) + '\n');

  return { confirmed, rejected, confirmedTrades, impact };
}
```

```
│  │  monthlyBreakdown()           → seasonal patterns                         │     │
│  │  printReport()                → formatted console output                  │     │
│  └────────────────────────────────────────────────────────────────────────────┘     │
└─────────────────────────────────────────────────────────────────────────────────────┘
```

The key architectural distinction of order flow is that it operates on **two different time scales simultaneously**: depth snapshots arrive every second (or faster) while candles are 5-minute, so the `augmentCandlesWithDepth` bridge maps hundreds of snapshots into each candle's window. The pattern detectors (`detectAbsorption`, `detectSpoofing`, `detectStacking`) work on the raw snapshot stream using `buildLevelHistory` to track quantity changes at individual price levels across time — something impossible with candle data alone. The confirmation filter (`confirmOrderFlow`) is designed as a **composable function** that any other strategy can call with just a direction and snapshot array, making it trivial to add the "5-8% win rate improvement" to ORB, VWAP, EMA, or Gap strategies without modifying their core logic.
