// ============================================================================
// CORE CALCULATIONS
// ============================================================================

/**
 * Simple Moving Average
 */
function calcSMA(data, period) {
  const result = new Array(data.length).fill(null);
  if (data.length < period) return result;
  for (let i = period - 1; i < data.length; i++) {
    let sum = 0;
    for (let j = i - period + 1; j <= i; j++) sum += data[j];
    result[i] = sum / period;
  }
  return result;
}

/**
 * Exponential Moving Average
 */
function calcEMA(data, period) {
  const result = new Array(data.length).fill(null);
  if (data.length < period) return result;
  let sum = 0;
  for (let i = 0; i < period; i++) sum += data[i];
  result[period - 1] = sum / period;
  const k = 2 / (period + 1);
  for (let i = period; i < data.length; i++) {
    result[i] = (data[i] - result[i - 1]) * k + result[i - 1];
  }
  return result;
}

/**
 * Annualized realized volatility
 */
function calcVolatility(prices, window = 20) {
  if (prices.length < window + 1) return null;
  const rets = [];
  for (let i = prices.length - window; i < prices.length; i++) {
    if (prices[i - 1] === 0) return null;
    rets.push(Math.log(prices[i] / prices[i - 1]));
  }
  const mean = rets.reduce((s, r) => s + r, 0) / rets.length;
  const variance = rets.reduce((s, r) => s + (r - mean) ** 2, 0) / (rets.length - 1);
  return Math.sqrt(variance) * Math.sqrt(CONFIG.TRADING_DAYS_PER_YEAR);
}

/**
 * Simple return over N days
 */
function calcReturn(prices, lookback) {
  if (!prices || prices.length < lookback + 1) return null;
  const curr = prices[prices.length - 1];
  const prev = prices[prices.length - 1 - lookback];
  if (!prev || prev === 0) return null;
  return (curr / prev) - 1;
}

/**
 * Calculate the slope/trend of an array over recent N bars
 * Uses simple linear regression slope
 */
function calcSlope(data, period) {
  const validData = data.filter(v => v !== null);
  if (validData.length < period) return null;
  const slice = validData.slice(-period);
  const n = slice.length;

  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
  for (let i = 0; i < n; i++) {
    sumX += i;
    sumY += slice[i];
    sumXY += i * slice[i];
    sumX2 += i * i;
  }

  const denom = (n * sumX2) - (sumX * sumX);
  if (denom === 0) return 0;

  return ((n * sumXY) - (sumX * sumY)) / denom;
}

/**
 * Compute drawdown at each point from an equity series
 */
function calcRunningDrawdown(equitySeries) {
  let peak = equitySeries[0] || 0;
  return equitySeries.map(v => {
    if (v > peak) peak = v;
    return peak > 0 ? (peak - v) / peak : 0;
  });
}

/**
 * Compute the Relative Strength ratio series: Sector / Benchmark
 * @param {number[]} sectorPrices - daily closes for the sector index
 * @param {number[]} benchmarkPrices - daily closes for Nifty 50
 * @returns {number[]} RS ratio series
 */
function calcRSRatio(sectorPrices, benchmarkPrices) {
  const len = Math.min(sectorPrices.length, benchmarkPrices.length);
  const rsRatio = new Array(len).fill(null);

  for (let i = 0; i < len; i++) {
    if (benchmarkPrices[i] && benchmarkPrices[i] !== 0) {
      rsRatio[i] = sectorPrices[i] / benchmarkPrices[i];
    }
  }

  return rsRatio;
}

/**
 * Compute RS momentum: how much has the RS ratio itself changed over a lookback
 * RS_1M = RS_ratio_today / RS_ratio_22d_ago
 * If > 1.0, the sector is outperforming the benchmark at an accelerating rate
 */
function calcRSMomentum(rsRatio, lookback) {
  const len = rsRatio.length;
  if (len < lookback + 1) return null;

  const current = rsRatio[len - 1];
  const past = rsRatio[len - 1 - lookback];

  if (current === null || past === null || past === 0) return null;
  return current / past;
}

/**
 * Full relative strength analysis for one sector
 * @param {number[]} sectorPrices
 * @param {number[]} niftyPrices
 * @param {string} sectorKey
 * @returns {Object} comprehensive RS analysis
 */
function analyzeSectorRS(sectorPrices, niftyPrices, sectorKey) {
  const rsRatio = calcRSRatio(sectorPrices, niftyPrices);

  const rs1M = calcRSMomentum(rsRatio, CONFIG.RS_LOOKBACK_1M);
  const rs3M = calcRSMomentum(rsRatio, CONFIG.RS_LOOKBACK_3M);
  const rs6M = calcRSMomentum(rsRatio, CONFIG.RS_LOOKBACK_6M);

  // Composite strength score
  let compositeScore = null;
  if (rs1M !== null && rs3M !== null) {
    compositeScore = CONFIG.RS_WEIGHT_1M * rs1M + CONFIG.RS_WEIGHT_3M * rs3M;
  }

  // RS ratio slope (is relative strength improving or deteriorating?)
  const rsSlope = calcSlope(rsRatio, CONFIG.MIN_RS_RATIO_SLOPE_DAYS);

  // Absolute sector return
  const sectorReturn1M = calcReturn(sectorPrices, CONFIG.RS_LOOKBACK_1M);
  const sectorReturn3M = calcReturn(sectorPrices, CONFIG.RS_LOOKBACK_3M);

  // Sector volatility
  const sectorVol = calcVolatility(sectorPrices);

  // RS trend classification
  let rsTrend = 'NEUTRAL';
  if (rs1M !== null) {
    if (rs1M > 1.02 && rsSlope > 0) rsTrend = 'STRONG_IMPROVING';
    else if (rs1M > 1.0) rsTrend = 'IMPROVING';
    else if (rs1M > 0.98) rsTrend = 'NEUTRAL';
    else if (rs1M > 0.95) rsTrend = 'WEAKENING';
    else rsTrend = 'STRONG_WEAKENING';
  }

  // RRG (Relative Rotation Graph) quadrant classification
  // Based on RS-Ratio (relative position) and RS-Momentum (rate of change)
  const currentRSRatio = rsRatio[rsRatio.length - 1];
  const rsRatioSMA = calcSMA(rsRatio.filter(v => v !== null), 10);
  const normalizedRSRatio = currentRSRatio && rsRatioSMA.length > 0
    ? currentRSRatio / rsRatioSMA[rsRatioSMA.length - 1]
    : null;

  let rrgQuadrant = 'UNKNOWN';
  if (normalizedRSRatio !== null && rs1M !== null) {
    if (normalizedRSRatio > 1.0 && rs1M > 1.0) rrgQuadrant = 'LEADING';
    else if (normalizedRSRatio > 1.0 && rs1M <= 1.0) rrgQuadrant = 'WEAKENING';
    else if (normalizedRSRatio <= 1.0 && rs1M <= 1.0) rrgQuadrant = 'LAGGING';
    else rrgQuadrant = 'IMPROVING';
  }

  const sectorMeta = SECTORS[sectorKey] || {};

  return {
    sector: sectorKey,
    sectorName: sectorMeta.name || sectorKey,
    cycleSensitivity: sectorMeta.cycleSensitivity || 'UNKNOWN',

    // RS metrics
    rsRatio: r2(currentRSRatio),
    rs1M: r2(rs1M),
    rs3M: r2(rs3M),
    rs6M: r2(rs6M),
    compositeScore: r2(compositeScore),
    rsSlope: rsSlope !== null ? r2(rsSlope * 10000) : null, // scaled for readability
    rsTrend,
    rrgQuadrant,

    // Absolute performance
    return1M: sectorReturn1M !== null ? r2(sectorReturn1M * 100) + '%' : null,
    return3M: sectorReturn3M !== null ? r2(sectorReturn3M * 100) + '%' : null,

    // Risk
    volatility: sectorVol !== null ? r2(sectorVol * 100) + '%' : null,

    // Raw arrays for further analysis
    rsRatioArray: rsRatio,
    pricesArray: sectorPrices,
  };
}

/**
 * Determine the current economic cycle phase based on macro indicators
 *
 * In practice you'd feed real macro data. This implementation uses
 * sector RS patterns as a proxy (which sectors are leading tells you the phase).
 *
 * @param {Object} sectorAnalyses - { BANK: {...}, IT: {...}, ... }
 * @param {Object} [macroData] - optional real macro inputs
 * @returns {Object} detected phase + confidence
 */
function detectEconomicPhase(sectorAnalyses, macroData = null) {
  // If real macro data is provided, use it directly
  if (macroData && macroData.rbiStance) {
    return detectFromMacro(macroData);
  }

  // Otherwise, infer from sector RS patterns
  return detectFromSectorRS(sectorAnalyses);
}

/**
 * Detect cycle phase from real macro data
 */
function detectFromMacro(macroData) {
  const { rbiStance, gdpTrend, inflationTrend, creditGrowth } = macroData;

  const scores = {
    EARLY_RECOVERY: 0,
    MID_EXPANSION: 0,
    LATE_EXPANSION: 0,
    SLOWDOWN: 0,
  };

  // RBI stance
  if (rbiStance === 'DOVISH' || rbiStance === 'ACCOMMODATIVE') {
    scores.EARLY_RECOVERY += 3;
  } else if (rbiStance === 'NEUTRAL') {
    scores.MID_EXPANSION += 2;
    scores.EARLY_RECOVERY += 1;
  } else if (rbiStance === 'HAWKISH' || rbiStance === 'TIGHTENING') {
    scores.LATE_EXPANSION += 2;
    scores.SLOWDOWN += 2;
  }

  // GDP trend
  if (gdpTrend === 'ACCELERATING') {
    scores.MID_EXPANSION += 3;
  } else if (gdpTrend === 'STABLE_HIGH') {
    scores.LATE_EXPANSION += 2;
  } else if (gdpTrend === 'DECELERATING') {
    scores.SLOWDOWN += 3;
  } else if (gdpTrend === 'RECOVERING') {
    scores.EARLY_RECOVERY += 3;
  }

  // Inflation
  if (inflationTrend === 'RISING') {
    scores.LATE_EXPANSION += 3;
  } else if (inflationTrend === 'HIGH') {
    scores.SLOWDOWN += 2;
  } else if (inflationTrend === 'FALLING') {
    scores.EARLY_RECOVERY += 2;
  } else if (inflationTrend === 'LOW') {
    scores.EARLY_RECOVERY += 1;
    scores.MID_EXPANSION += 1;
  }

  // Credit growth
  if (creditGrowth === 'ACCELERATING') {
    scores.EARLY_RECOVERY += 2;
    scores.MID_EXPANSION += 2;
  } else if (creditGrowth === 'SLOWING') {
    scores.SLOWDOWN += 2;
  }

  // Pick highest scoring phase
  let bestPhase = 'MID_EXPANSION';
  let bestScore = 0;
  for (const [phase, score] of Object.entries(scores)) {
    if (score > bestScore) {
      bestScore = score;
      bestPhase = phase;
    }
  }

  const totalPossible = 11; // rough max
  const confidence = Math.min(bestScore / totalPossible, 1.0);

  return {
    phase: bestPhase,
    phaseInfo: ECONOMIC_CYCLE[bestPhase],
    confidence: r2(confidence * 100) + '%',
    scores,
    source: 'MACRO_DATA',
    inputs: macroData,
  };
}

/**
 * Infer cycle phase from which sectors are showing relative strength
 * (when macro data isn't available)
 */
function detectFromSectorRS(sectorAnalyses) {
  const scores = {
    EARLY_RECOVERY: 0,
    MID_EXPANSION: 0,
    LATE_EXPANSION: 0,
    SLOWDOWN: 0,
  };

  // For each sector, if it's in the "LEADING" or "IMPROVING" RRG quadrant,
  // boost the phases where that sector typically leads
  for (const [sectorKey, analysis] of Object.entries(sectorAnalyses)) {
    const meta = SECTORS[sectorKey];
    if (!meta || !meta.cyclePhases) continue;

    const strength = analysis.rrgQuadrant;
    let multiplier = 0;
    if (strength === 'LEADING') multiplier = 3;
    else if (strength === 'IMPROVING') multiplier = 2;
    else if (strength === 'WEAKENING') multiplier = -1;
    else if (strength === 'LAGGING') multiplier = -2;

    for (const phase of meta.cyclePhases) {
      scores[phase] = (scores[phase] || 0) + multiplier;
    }
  }

  let bestPhase = 'MID_EXPANSION';
  let bestScore = -Infinity;
  for (const [phase, score] of Object.entries(scores)) {
    if (score > bestScore) {
      bestScore = score;
      bestPhase = phase;
    }
  }

  const allScores = Object.values(scores);
  const maxPossible = Math.max(...allScores) - Math.min(...allScores);
  const confidence = maxPossible > 0
    ? Math.min((bestScore - Math.min(...allScores)) / maxPossible, 1.0)
    : 0.5;

  return {
    phase: bestPhase,
    phaseInfo: ECONOMIC_CYCLE[bestPhase],
    confidence: r2(confidence * 100) + '%',
    scores,
    source: 'SECTOR_RS_INFERENCE',
    note: 'Inferred from sector relative strength patterns — provide macro data for higher accuracy',
  };
}

// SECTOR RANKING ENGINE
// ============================================================================

/**
 * Rank all sectors and select the top performers
 *
 * @param {Object} sectorData - { BANK: { prices: [] }, IT: { prices: [] }, ... }
 * @param {number[]} niftyPrices - Nifty 50 daily closes
 * @param {Object} [macroData] - optional macro inputs
 * @returns {Object} ranked sectors with metadata
 */
function rankSectors(sectorData, niftyPrices, macroData = null) {
  const analyses = {};
  const ranked = [];

  // Step 1: Compute RS for each sector
  for (const [sectorKey, data] of Object.entries(sectorData)) {
    if (!data.prices || data.prices.length < CONFIG.RS_LOOKBACK_3M + 10) continue;

    const analysis = analyzeSectorRS(data.prices, niftyPrices, sectorKey);
    analyses[sectorKey] = analysis;

    if (analysis.compositeScore !== null) {
      ranked.push({
        sector: sectorKey,
        sectorName: analysis.sectorName,
        compositeScore: analysis.compositeScore,
        rs1M: analysis.rs1M,
        rs3M: analysis.rs3M,
        rsTrend: analysis.rsTrend,
        rrgQuadrant: analysis.rrgQuadrant,
        return1M: analysis.return1M,
        return3M: analysis.return3M,
        volatility: analysis.volatility,
        cycleSensitivity: analysis.cycleSensitivity,
      });
    }
  }

  // Step 2: Sort by composite score
  ranked.sort((a, b) => b.compositeScore - a.compositeScore);

  // Assign ranks
  ranked.forEach((s, i) => { s.rank = i + 1; });

  // Step 3: Detect economic phase
  const economicPhase = detectEconomicPhase(analyses, macroData);

  // Step 4: Apply cycle overlay — boost/penalize sectors based on phase
  const cycleAdjusted = ranked.map(s => {
    let cycleBonus = 0;
    const phaseInfo = ECONOMIC_CYCLE[economicPhase.phase];

    if (phaseInfo) {
      if (phaseInfo.leadingSectors.includes(s.sector)) {
        cycleBonus = 0.02; // boost leading sectors
      }
      if (phaseInfo.laggingSectors.includes(s.sector)) {
        cycleBonus = -0.02; // penalize lagging sectors
      }
    }

    return {
      ...s,
      cycleBonus: r2(cycleBonus),
      adjustedScore: r2(s.compositeScore + cycleBonus),
      cycleAlignment: cycleBonus > 0 ? 'FAVORABLE' : cycleBonus < 0 ? 'UNFAVORABLE' : 'NEUTRAL',
    };
  });

  // Re-sort by adjusted score
  cycleAdjusted.sort((a, b) => b.adjustedScore - a.adjustedScore);
  cycleAdjusted.forEach((s, i) => { s.adjustedRank = i + 1; });

  // Step 5: Select top sectors
  const topSectors = cycleAdjusted.slice(0, CONFIG.TOP_SECTORS);
  const bottomSectors = cycleAdjusted.slice(-CONFIG.TOP_SECTORS);

  return {
    allSectors: cycleAdjusted,
    topSectors,
    bottomSectors,
    economicPhase,
    sectorCount: ranked.length,
    analyses,
  };
}

/**
 * Compute individual stock momentum score (adapted from Strategy 1)
 */
function calcStockMomentum(prices) {
  const ret1m = calcReturn(prices, CONFIG.RS_LOOKBACK_1M);
  const ret3m = calcReturn(prices, CONFIG.RS_LOOKBACK_3M);
  const ret6m = calcReturn(prices, CONFIG.RS_LOOKBACK_6M);

  if (ret1m === null || ret3m === null) return null;

  // Allow missing 6M data (use available)
  const r6m = ret6m !== null ? ret6m : 0;
  const w6m = ret6m !== null ? CONFIG.STOCK_MOM_6M_WEIGHT : 0;
  const totalWeight = CONFIG.STOCK_MOM_1M_WEIGHT + CONFIG.STOCK_MOM_3M_WEIGHT + w6m;

  const composite = (
    CONFIG.STOCK_MOM_1M_WEIGHT * ret1m +
    CONFIG.STOCK_MOM_3M_WEIGHT * ret3m +
    w6m * r6m
  ) / totalWeight;

  return { composite, ret1m, ret3m, ret6m };
}

/**
 * Select top stocks within a sector
 * @param {Object[]} sectorStocks - stocks belonging to this sector
 * @param {number} count - how many to pick
 * @returns {Object[]} selected stocks
 */
function selectStocksInSector(sectorStocks, count = CONFIG.STOCKS_PER_SECTOR) {
  const scored = [];

  for (const stock of sectorStocks) {
    // Filter check
    if (stock.avgDailyTurnoverCr && stock.avgDailyTurnoverCr < CONFIG.MIN_AVG_TURNOVER_CR) {
      continue;
    }

    if (!stock.prices || stock.prices.length < CONFIG.RS_LOOKBACK_3M + 10) continue;

    const momentum = calcStockMomentum(stock.prices);
    if (!momentum) continue;

    const vol = calcVolatility(stock.prices);

    // Volume health check
    let volumeHealthy = true;
    if (stock.volumes && stock.volumes.length >= 50) {
      const avg20 = stock.volumes.slice(-20).reduce((s, v) => s + v, 0) / 20;
      const avg50 = stock.volumes.slice(-50).reduce((s, v) => s + v, 0) / 50;
      if (avg50 > 0 && avg20 / avg50 < 0.5) volumeHealthy = false;
    }

    scored.push({
      symbol: stock.symbol,
      momentumScore: momentum.composite,
      ret1m: momentum.ret1m,
      ret3m: momentum.ret3m,
      ret6m: momentum.ret6m,
      volatility: vol,
      currentPrice: stock.prices[stock.prices.length - 1],
      volumeHealthy,
      avgDailyTurnoverCr: stock.avgDailyTurnoverCr || 0,
    });
  }

  // Filter out unhealthy volume stocks
  const healthy = scored.filter(s => s.volumeHealthy);
  const pool = healthy.length >= count ? healthy : scored;

  // Sort by momentum, pick top N
  pool.sort((a, b) => b.momentumScore - a.momentumScore);
  return pool.slice(0, count);
}

// PORTFOLIO CONSTRUCTION
// ============================================================================

/**
 * Build the complete sector rotation portfolio
 *
 * @param {Object} sectorData - sector index prices: { BANK: { prices }, ... }
 * @param {Object} stockBySector - stocks grouped by sector: { BANK: [stocks], ... }
 * @param {number[]} niftyPrices - Nifty 50 closes
 * @param {number} totalCapital
 * @param {Object} [macroData]
 * @param {string} [currentDate]
 * @returns {Object} complete portfolio allocation
 */
function buildPortfolio(sectorData, stockBySector, niftyPrices, totalCapital, macroData = null, currentDate = null) {

  // Step 0: Market regime check
  const regime = checkMarketRegime(niftyPrices);
  if (!regime.tradeable) {
    return {
      date: currentDate,
      regime,
      signal: 'CASH',
      message: regime.reason,
      holdings: [],
    };
  }

  // Step 1: Rank sectors
  const sectorRanking = rankSectors(sectorData, niftyPrices, macroData);

  // Step 2: Select stocks within top sectors
  const allHoldings = [];
  const sectorAllocations = [];

  for (const topSector of sectorRanking.topSectors) {
    const sectorStocks = stockBySector[topSector.sector] || [];

    if (sectorStocks.length < CONFIG.MIN_SECTOR_STOCKS) {
      sectorAllocations.push({
        sector: topSector.sector,
        sectorName: topSector.sectorName,
        skipped: true,
        reason: `Only ${sectorStocks.length} stocks (need ${CONFIG.MIN_SECTOR_STOCKS})`,
      });
      continue;
    }

    const selectedStocks = selectStocksInSector(sectorStocks);

    sectorAllocations.push({
      sector: topSector.sector,
      sectorName: topSector.sectorName,
      rank: topSector.adjustedRank,
      compositeScore: topSector.adjustedScore,
      rsTrend: topSector.rsTrend,
      rrgQuadrant: topSector.rrgQuadrant,
      cycleAlignment: topSector.cycleAlignment,
      stocksSelected: selectedStocks.length,
    });

    for (const stock of selectedStocks) {
      allHoldings.push({
        ...stock,
        sector: topSector.sector,
        sectorName: topSector.sectorName,
        sectorRank: topSector.adjustedRank,
        sectorScore: topSector.adjustedScore,
      });
    }
  }

  // Step 3: Calculate weights
  const weighted = calcPortfolioWeights(allHoldings, sectorRanking, totalCapital);

  return {
    date: currentDate,
    regime,
    signal: 'INVEST',
    economicPhase: sectorRanking.economicPhase,
    sectorRanking: sectorRanking.allSectors,
    topSectors: sectorAllocations,
    bottomSectors: sectorRanking.bottomSectors.map(s => ({
      sector: s.sector,
      sectorName: s.sectorName,
      adjustedScore: s.adjustedScore,
      rsTrend: s.rsTrend,
    })),
    holdings: weighted,
    portfolioStats: calcPortfolioStats(weighted, totalCapital),
    nextRebalance: `${CONFIG.HOLDING_PERIOD_DAYS} trading days`,
  };
}

/**
 * Calculate portfolio weights using the configured method
 */
function calcPortfolioWeights(holdings, sectorRanking, totalCapital) {
  if (holdings.length === 0) return [];

  const investableCapital = totalCapital * (1 - CONFIG.CASH_RESERVE_PCT);

  if (CONFIG.WEIGHTING_METHOD === 'EQUAL') {
    return calcEqualWeights(holdings, investableCapital);
  } else if (CONFIG.WEIGHTING_METHOD === 'INV_VOL') {
    return calcInvVolWeights(holdings, investableCapital);
  } else {
    return calcRSWeights(holdings, sectorRanking, investableCapital);
  }
}

function calcEqualWeights(holdings, capital) {
  const weight = 1 / holdings.length;
  return holdings.map(h => ({
    ...h,
    weight: r2(weight),
    allocatedCapital: Math.round(weight * capital),
    shares: Math.floor((weight * capital) / h.currentPrice),
    stopLoss: r2(h.currentPrice * (1 - CONFIG.STOCK_STOP_LOSS_PCT)),
  }));
}

function calcInvVolWeights(holdings, capital) {
  const totalInvVol = holdings.reduce((sum, h) => {
    return sum + (h.volatility && h.volatility > 0 ? 1 / h.volatility : 1);
  }, 0);

  return holdings.map(h => {
    const invVol = h.volatility && h.volatility > 0 ? 1 / h.volatility : 1;
    let weight = invVol / totalInvVol;

    // Cap at max stock weight
    weight = Math.min(weight, CONFIG.MAX_STOCK_WEIGHT);

    return {
      ...h,
      weight: r2(weight),
      allocatedCapital: Math.round(weight * capital),
      shares: Math.floor((weight * capital) / h.currentPrice),
      stopLoss: r2(h.currentPrice * (1 - CONFIG.STOCK_STOP_LOSS_PCT)),
    };
  });
}

function calcRSWeights(holdings, sectorRanking, capital) {
  // Weight sectors proportionally to their adjusted RS score
  // Then within sector, equal-weight the stocks
  const sectorScores = {};
  for (const s of sectorRanking.topSectors) {
    sectorScores[s.sector] = Math.max(s.adjustedScore, 0.001);
  }

  const totalSectorScore = Object.values(sectorScores).reduce((s, v) => s + v, 0);

  // Group holdings by sector
  const bySector = {};
  for (const h of holdings) {
    if (!bySector[h.sector]) bySector[h.sector] = [];
    bySector[h.sector].push(h);
  }

  const result = [];
  for (const [sector, stocks] of Object.entries(bySector)) {
    let sectorWeight = (sectorScores[sector] || 0) / totalSectorScore;
    sectorWeight = Math.min(sectorWeight, CONFIG.MAX_SECTOR_WEIGHT);

    const perStockWeight = sectorWeight / stocks.length;

    for (const stock of stocks) {
      const weight = Math.min(perStockWeight, CONFIG.MAX_STOCK_WEIGHT);
      result.push({
        ...stock,
        sectorWeight: r2(sectorWeight),
        weight: r2(weight),
        allocatedCapital: Math.round(weight * capital),
        shares: Math.floor((weight * capital) / stock.currentPrice),
        stopLoss: r2(stock.currentPrice * (1 - CONFIG.STOCK_STOP_LOSS_PCT)),
      });
    }
  }

  return result;
}

/**
 * Calculate portfolio-level summary statistics
 */
function calcPortfolioStats(holdings, totalCapital) {
  if (holdings.length === 0) return {};

  const totalInvested = holdings.reduce((s, h) => s + (h.allocatedCapital || 0), 0);

  // Sector concentration
  const sectorWeights = {};
  for (const h of holdings) {
    sectorWeights[h.sectorName || h.sector] =
      (sectorWeights[h.sectorName || h.sector] || 0) + (h.weight || 0);
  }

  // Portfolio weighted volatility (simplified — assumes zero correlation)
  const weightedVol = Math.sqrt(
    holdings.reduce((sum, h) => {
      const w = h.weight || 0;
      const v = h.volatility || 0.2;
      return sum + (w * v) ** 2;
    }, 0)
  );

  // HHI (Herfindahl-Hirschman Index)
  const hhi = holdings.reduce((sum, h) => sum + ((h.weight || 0) * 100) ** 2, 0);

  return {
    totalHoldings: holdings.length,
    totalInvested: Math.round(totalInvested),
    cashReserve: Math.round(totalCapital - totalInvested),
    investedPct: r2((totalInvested / totalCapital) * 100) + '%',
    sectorConcentration: Object.entries(sectorWeights)
      .map(([s, w]) => ({ sector: s, weight: r2(w * 100) + '%' }))
      .sort((a, b) => parseFloat(b.weight) - parseFloat(a.weight)),
    portfolioVolatility: r2(weightedVol * 100) + '%',
    hhiIndex: Math.round(hhi),
    diversification: hhi > 2000 ? 'CONCENTRATED' : hhi > 1200 ? 'MODERATE' : 'DIVERSIFIED',
    maxStockWeight: r2(Math.max(...holdings.map(h => h.weight || 0)) * 100) + '%',
  };
}


// ============================================================================
// MARKET REGIME
// ============================================================================

function checkMarketRegime(niftyPrices) {
  if (!niftyPrices || niftyPrices.length < CONFIG.NIFTY_TREND_SMA + 1) {
    return { regime: 'UNKNOWN', tradeable: false, reason: 'Insufficient Nifty data' };
  }

  const sma50 = calcSMA(niftyPrices, CONFIG.NIFTY_SMA_PERIOD);
  const sma200 = calcSMA(niftyPrices, CONFIG.NIFTY_TREND_SMA);
  const currentNifty = niftyPrices[niftyPrices.length - 1];
  const curr50 = sma50[sma50.length - 1];
  const curr200 = sma200[sma200.length - 1];

  // Strong bull: Nifty > 50-SMA > 200-SMA
  // Moderate bull: Nifty > 200-SMA but below 50-SMA
  // Bear: Nifty < 200-SMA
  if (curr200 === null) {
    return { regime: 'UNKNOWN', tradeable: false, reason: '200-SMA not available' };
  }

  const above50 = curr50 !== null && currentNifty > curr50;
  const above200 = currentNifty > curr200;
  const sma50Above200 = curr50 !== null && curr50 > curr200;

  let regime, tradeable;
  if (above200 && above50 && sma50Above200) {
    regime = 'STRONG_BULL';
    tradeable = true;
  } else if (above200 && above50) {
    regime = 'BULL';
    tradeable = true;
  } else if (above200) {
    regime = 'CAUTIOUS_BULL';
    tradeable = true;
  } else {
    regime = 'BEAR';
    tradeable = false;
  }

  return {
    regime,
    tradeable,
    nifty: r2(currentNifty),
    sma50: r2(curr50),
    sma200: r2(curr200),
    reason: tradeable ? null : 'Nifty below 200-SMA — stay in cash or reduce exposure',
  };
}

// ============================================================================

/**
 * Monitor all positions in the sector rotation portfolio
 * @param {Object[]} holdings - current portfolio holdings with entry data
 * @param {Object} currentStockPrices - { symbol: currentPrice }
 * @param {number} daysSinceRebalance
 * @returns {Object} monitoring results
 */
function monitorPortfolio(holdings, currentStockPrices, daysSinceRebalance) {
  const actions = [];
  let portfolioLevelStop = false;
  let totalPnl = 0;
  let totalAllocated = 0;

  for (const h of holdings) {
    const currentPrice = currentStockPrices[h.symbol];
    if (currentPrice === undefined) continue;

    const pnlPct = (currentPrice / h.currentPrice) - 1;
    const pnl = (currentPrice - h.currentPrice) * (h.shares || 0);
    totalPnl += pnl;
    totalAllocated += h.allocatedCapital || 0;

    // Individual stock stop
    if (currentPrice <= h.stopLoss) {
      actions.push({
        symbol: h.symbol,
        sector: h.sector,
        action: 'EXIT',
        reason: 'STOCK_STOP_LOSS',
        entryPrice: h.currentPrice,
        currentPrice: r2(currentPrice),
        pnlPct: r2(pnlPct * 100) + '%',
        urgency: 'IMMEDIATE',
      });
    } else {
      actions.push({
        symbol: h.symbol,
        sector: h.sector,
        action: 'HOLD',
        entryPrice: h.currentPrice,
        currentPrice: r2(currentPrice),
        pnlPct: r2(pnlPct * 100) + '%',
        distanceToStop: r2(((currentPrice - h.stopLoss) / currentPrice) * 100) + '%',
      });
    }
  }

  // Portfolio-level stop check
  const portfolioPnlPct = totalAllocated > 0 ? totalPnl / totalAllocated : 0;
  if (portfolioPnlPct <= -CONFIG.STOP_LOSS_PCT) {
    portfolioLevelStop = true;
  }

  // Time-based rebalance signal
  const rebalanceDue = daysSinceRebalance >= CONFIG.HOLDING_PERIOD_DAYS;

  return {
    daysSinceRebalance,
    portfolioPnlPct: r2(portfolioPnlPct * 100) + '%',
    portfolioStopTriggered: portfolioLevelStop,
    rebalanceDue,
    stockActions: actions,
    exitCount: actions.filter(a => a.action === 'EXIT').length,
    recommendation: portfolioLevelStop
      ? 'EXIT ALL — portfolio stop-loss breached'
      : rebalanceDue
        ? 'REBALANCE — re-run sector ranking and rebuild portfolio'
        : 'HOLD — continue monitoring',
  };
}

/**
 * Generate a sector rotation heatmap showing RS trends over time
 * @param {Object} sectorData - historical sector index prices
 * @param {number[]} niftyPrices
 * @param {number[]} lookbackPoints - evaluation dates as array indices
 * @returns {Object} heatmap data
 */
function generateRotationHeatmap(sectorData, niftyPrices, lookbackPoints = null) {
  // Default: monthly snapshots over the last 6 months
  if (!lookbackPoints) {
    lookbackPoints = [];
    for (let offset = 0; offset <= 5; offset++) {
      const idx = niftyPrices.length - 1 - (offset * 22);
      if (idx > CONFIG.RS_LOOKBACK_3M) lookbackPoints.unshift(idx);
    }
  }

  const heatmap = {};
  const sectorKeys = Object.keys(sectorData);

  for (const point of lookbackPoints) {
    const dateIdx = point;
    const row = {};

    for (const sectorKey of sectorKeys) {
      const sectorPrices = sectorData[sectorKey].prices;
      if (!sectorPrices || sectorPrices.length <= dateIdx) continue;

      const slicedSector = sectorPrices.slice(0, dateIdx + 1);
      const slicedNifty = niftyPrices.slice(0, dateIdx + 1);

      const rsRatio = calcRSRatio(slicedSector, slicedNifty);
      const rs1M = calcRSMomentum(rsRatio, CONFIG.RS_LOOKBACK_1M);

      let status;
      if (rs1M === null) status = '—';
      else if (rs1M > 1.03) status = '▲▲'; // strongly outperforming
      else if (rs1M > 1.01) status = '▲';  // outperforming
      else if (rs1M > 0.99) status = '●';  // neutral
      else if (rs1M > 0.97) status = '▼';  // underperforming
      else status = '▼▼'; // strongly underperforming

      row[sectorKey] = {
        rs1M: r2(rs1M),
        status,
        return1M: r2(calcReturn(slicedSector, Math.min(22, slicedSector.length - 1)) * 100) + '%',
      };
    }

    heatmap[dateIdx] = row;
  }

  return { heatmap, lookbackPoints, sectors: sectorKeys };
}


// ============================================================================
// BACKTEST ENGINE
// ============================================================================

/**
 * Full backtest of sector rotation strategy
 *
 * @param {Object} data
 *   { sectorIndices: { BANK: { prices: [] }, IT: { prices: [] }, ... },
 *     stocksBySector: { BANK: [ { symbol, prices, volumes, avgDailyTurnoverCr } ], ... },
 *     niftyPrices: [],
 *     dates: [],
 *     macroHistory: [{ date, rbiStance, gdpTrend, inflationTrend, creditGrowth }] (optional) }
 *
 * @param {number} capitalBase
 * @returns {Object} backtest results
 */
function runBacktest(data, capitalBase = 10000000) {
  const { sectorIndices, stocksBySector, niftyPrices, dates, macroHistory } = data;

  let capital = capitalBase;
  const tradeLog = [];
  const equityCurve = [];
  const rotationLog = [];
  let peakEquity = capitalBase;
  let maxDrawdown = 0;

  let currentHoldings = [];
  let holdingEntryDay = 0;

  // Start after enough history
  const startDay = CONFIG.RS_LOOKBACK_6M + 10;

  for (let day = startDay; day < dates.length; day++) {
    const currentDate = dates[day];
    const daysSinceRebalance = day - holdingEntryDay;
    const isRebalanceDay = daysSinceRebalance >= CONFIG.HOLDING_PERIOD_DAYS || currentHoldings.length === 0;

    // ===== CHECK FOR INDIVIDUAL STOP-LOSSES =====
    const stoppedOut = [];
    for (let h = 0; h < currentHoldings.length; h++) {
      const holding = currentHoldings[h];
      const stockData = findStockInSector(stocksBySector, holding.symbol);
      if (!stockData || day >= stockData.prices.length) continue;

      const currentPrice = stockData.prices[day];
      if (currentPrice <= holding.stopLoss) {
        const pnl = (currentPrice - holding.entryPrice) * holding.shares;
        capital += currentPrice * holding.shares;

        tradeLog.push({
          symbol: holding.symbol,
          sector: holding.sector,
          entryDate: dates[holding.entryDay],
          exitDate: currentDate,
          entryPrice: r2(holding.entryPrice),
          exitPrice: r2(currentPrice),
          shares: holding.shares,
          pnl: Math.round(pnl),
          pnlPct: r2(((currentPrice / holding.entryPrice) - 1) * 100) + '%',
          holdingDays: day - holding.entryDay,
          exitReason: 'STOP_LOSS',
          sectorRank: holding.sectorRank,
        });

        stoppedOut.push(h);
      }
    }

    // Remove stopped out positions
    for (let i = stoppedOut.length - 1; i >= 0; i--) {
      currentHoldings.splice(stoppedOut[i], 1);
    }

    // ===== REBALANCE =====
    if (isRebalanceDay) {
      // Close all remaining positions
      for (const holding of currentHoldings) {
        const stockData = findStockInSector(stocksBySector, holding.symbol);
        if (!stockData || day >= stockData.prices.length) continue;

        const exitPrice = stockData.prices[day];
        const pnl = (exitPrice - holding.entryPrice) * holding.shares;
        capital += exitPrice * holding.shares;

        tradeLog.push({
          symbol: holding.symbol,
          sector: holding.sector,
          entryDate: dates[holding.entryDay],
          exitDate: currentDate,
          entryPrice: r2(holding.entryPrice),
          exitPrice: r2(exitPrice),
          shares: holding.shares,
          pnl: Math.round(pnl),
          pnlPct: r2(((exitPrice / holding.entryPrice) - 1) * 100) + '%',
          holdingDays: day - holding.entryDay,
          exitReason: 'REBALANCE',
          sectorRank: holding.sectorRank,
        });
      }

      currentHoldings = [];

      // Build new portfolio
      const sectorDataSliced = {};
      for (const [key, sd] of Object.entries(sectorIndices)) {
        sectorDataSliced[key] = { prices: sd.prices.slice(0, day + 1) };
      }

      const stocksBySectorSliced = {};
      for (const [sector, stocks] of Object.entries(stocksBySector)) {
        stocksBySectorSliced[sector] = stocks.map(s => ({
          ...s,
          prices: s.prices.slice(0, day + 1),
          volumes: s.volumes ? s.volumes.slice(0, day + 1) : undefined,
        }));
      }

      const niftySliced = niftyPrices.slice(0, day + 1);

      // Get macro data for this date if available
      let macroData = null;
      if (macroHistory) {
        macroData = macroHistory.find(m => m.date <= currentDate);
      }

      const portfolio = buildPortfolio(
        sectorDataSliced, stocksBySectorSliced, niftySliced,
        capital, macroData, currentDate
      );

      // Record rotation decision
      rotationLog.push({
        date: currentDate,
        regime: portfolio.regime?.regime,
        economicPhase: portfolio.economicPhase?.phase,
        topSectors: portfolio.topSectors?.map(s => s.sector) || [],
        holdingsCount: portfolio.holdings?.length || 0,
        signal: portfolio.signal,
      });

      if (portfolio.signal === 'INVEST' && portfolio.holdings.length > 0) {
        holdingEntryDay = day;

        for (const h of portfolio.holdings) {
          const stockData = findStockInSector(stocksBySector, h.symbol);
          if (!stockData || day >= stockData.prices.length) continue;

          const entryPrice = stockData.prices[day];
          const allocatedCapital = Math.min(h.allocatedCapital, capital);
          const shares = Math.floor(allocatedCapital / entryPrice);

          if (shares <= 0) continue;

          capital -= shares * entryPrice;

          currentHoldings.push({
            symbol: h.symbol,
            sector: h.sector,
            sectorRank: h.sectorRank,
            entryPrice,
            entryDay: day,
            shares,
            stopLoss: entryPrice * (1 - CONFIG.STOCK_STOP_LOSS_PCT),
            weight: h.weight,
          });
        }
      }
    }

    // ===== EQUITY CURVE =====
    let openValue = 0;
    for (const h of currentHoldings) {
      const stockData = findStockInSector(stocksBySector, h.symbol);
      if (stockData && day < stockData.prices.length) {
        openValue += stockData.prices[day] * h.shares;
      }
    }

    const totalEquity = capital + openValue;
    if (totalEquity > peakEquity) peakEquity = totalEquity;
    const dd = peakEquity > 0 ? (peakEquity - totalEquity) / peakEquity : 0;
    if (dd > maxDrawdown) maxDrawdown = dd;

    equityCurve.push({
      date: currentDate,
      totalEquity: Math.round(totalEquity),
      cash: Math.round(capital),
      invested: Math.round(openValue),
      positions: currentHoldings.length,
      drawdown: r2(dd * 100) + '%',
    });
  }

  // Force close remaining
  for (const h of currentHoldings) {
    const stockData = findStockInSector(stocksBySector, h.symbol);
    if (!stockData) continue;
    const exitPrice = stockData.prices[stockData.prices.length - 1];
    const pnl = (exitPrice - h.entryPrice) * h.shares;
    capital += exitPrice * h.shares;

    tradeLog.push({
      symbol: h.symbol,
      sector: h.sector,
      entryDate: dates[h.entryDay],
      exitDate: dates[dates.length - 1],
      entryPrice: r2(h.entryPrice),
      exitPrice: r2(exitPrice),
      shares: h.shares,
      pnl: Math.round(pnl),
      pnlPct: r2(((exitPrice / h.entryPrice) - 1) * 100) + '%',
      holdingDays: dates.length - 1 - h.entryDay,
      exitReason: 'BACKTEST_END',
      sectorRank: h.sectorRank,
    });
  }

  return buildBacktestAnalytics(tradeLog, equityCurve, rotationLog, capitalBase, capital, maxDrawdown);
}

/**
 * Helper to find a stock across all sectors
 */
function findStockInSector(stocksBySector, symbol) {
  for (const stocks of Object.values(stocksBySector)) {
    const found = stocks.find(s => s.symbol === symbol);
    if (found) return found;
  }
  return null;
}


// ============================================================================
// BACKTEST ANALYTICS
// ============================================================================

function buildBacktestAnalytics(trades, equityCurve, rotationLog, capitalBase, finalCapital, maxDrawdown) {
  const totalTrades = trades.length;
  const winners = trades.filter(t => t.pnl > 0);
  const losers = trades.filter(t => t.pnl <= 0);

  const winRate = totalTrades > 0 ? winners.length / totalTrades : 0;
  const avgWinPct = winners.length > 0
    ? winners.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / winners.length : 0;
  const avgLossPct = losers.length > 0
    ? Math.abs(losers.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / losers.length) : 0;
  const avgHoldDays = totalTrades > 0
    ? trades.reduce((s, t) => s + t.holdingDays, 0) / totalTrades : 0;

  const grossProfit = winners.reduce((s, t) => s + t.pnl, 0);
  const grossLoss = Math.abs(losers.reduce((s, t) => s + t.pnl, 0));
  const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : Infinity;

  const totalReturn = (finalCapital / capitalBase) - 1;
  const years = equityCurve.length / CONFIG.TRADING_DAYS_PER_YEAR;
  const cagr = years > 0 ? Math.pow(finalCapital / capitalBase, 1 / years) - 1 : 0;

  // Weekly returns for Sharpe/Sortino
  const weeklyReturns = [];
  for (let i = 5; i < equityCurve.length; i += 5) {
    const prev = equityCurve[i - 5].totalEquity;
    const curr = equityCurve[i].totalEquity;
    if (prev > 0) weeklyReturns.push((curr / prev) - 1);
  }

  const sharpe = calcRatio(weeklyReturns, 52, 'sharpe');
  const sortino = calcRatio(weeklyReturns, 52, 'sortino');
  const calmar = maxDrawdown > 0 ? cagr / maxDrawdown : null;

  // By sector analysis
  const bySector = analyzeBySector(trades);

  // By exit reason
  const byExitReason = analyzeByField(trades, 'exitReason');

  // Rotation frequency — how often do top sectors change?
  const rotationAnalysis = analyzeRotations(rotationLog);

  // Sector hit rate — which sectors produce the best trades?
  const sectorHitRate = {};
  for (const [sector, data] of Object.entries(bySector)) {
    sectorHitRate[sector] = {
      trades: data.count,
      winRate: data.winRate,
      avgPnlPct: data.avgPnlPct,
      contribution: data.totalPnl,
    };
  }

  // Drawdown analysis
  const drawdowns = analyzeDrawdowns(equityCurve);

  // Monthly returns
  const monthly = calcMonthlyReturns(equityCurve);

  // Streaks
  const { maxWin, maxLose } = calcStreaks(trades);

  return {
    summary: {
      startingCapital: capitalBase,
      endingCapital: Math.round(finalCapital),
      totalReturn: r2(totalReturn * 100) + '%',
      cagr: r2(cagr * 100) + '%',
      totalTrades,
      winRate: r2(winRate * 100) + '%',
      avgWinPct: r2(avgWinPct) + '%',
      avgLossPct: r2(avgLossPct) + '%',
      winLossRatio: avgLossPct > 0 ? r2(avgWinPct / avgLossPct) : 'N/A',
      profitFactor: profitFactor !== Infinity ? r2(profitFactor) : 'N/A',
      maxDrawdown: r2(maxDrawdown * 100) + '%',
      avgHoldingDays: r2(avgHoldDays),
      sharpeRatio: sharpe !== null ? r2(sharpe) : 'N/A',
      sortinoRatio: sortino !== null ? r2(sortino) : 'N/A',
      calmarRatio: calmar !== null ? r2(calmar) : 'N/A',
      maxWinStreak: maxWin,
      maxLoseStreak: maxLose,
      totalRotations: rotationLog.length,
    },
    bySector: sectorHitRate,
    byExitReason,
    rotationAnalysis,
    drawdownPeriods: drawdowns.slice(0, 5),
    monthlyReturns: monthly,
    topWinners: [...trades].sort((a, b) => b.pnl - a.pnl).slice(0, 10).map(formatTrade),
    topLosers: [...trades].sort((a, b) => a.pnl - b.pnl).slice(0, 10).map(formatTrade),
    rotationLog,
    trades,
    equityCurve,
  };
}

function analyzeBySector(trades) {
  const buckets = {};
  for (const t of trades) {
    if (!buckets[t.sector]) buckets[t.sector] = [];
    buckets[t.sector].push(t);
  }

  const result = {};
  for (const [sector, list] of Object.entries(buckets)) {
    const wins = list.filter(t => t.pnl > 0).length;
    result[sector] = {
      count: list.length,
      winRate: r2((wins / list.length) * 100) + '%',
      avgPnlPct: r2(list.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / list.length) + '%',
      totalPnl: Math.round(list.reduce((s, t) => s + t.pnl, 0)),
    };
  }
  return result;
}

function analyzeByField(trades, field) {
  const buckets = {};
  for (const t of trades) {
    const key = t[field] || 'UNKNOWN';
    if (!buckets[key]) buckets[key] = [];
    buckets[key].push(t);
  }

  const result = {};
  for (const [key, list] of Object.entries(buckets)) {
    const wins = list.filter(t => t.pnl > 0).length;
    result[key] = {
      count: list.length,
      winRate: r2((wins / list.length) * 100) + '%',
      avgPnlPct: r2(list.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / list.length) + '%',
      totalPnl: Math.round(list.reduce((s, t) => s + t.pnl, 0)),
    };
  }
  return result;
}

/**
 * Analyze how frequently sector leadership rotates
 */
function analyzeRotations(rotationLog) {
  if (rotationLog.length < 2) return { rotations: 0 };

  let leadershipChanges = 0;
  const sectorAppearances = {};

  for (let i = 1; i < rotationLog.length; i++) {
    const prev = new Set(rotationLog[i - 1].topSectors || []);
    const curr = new Set(rotationLog[i].topSectors || []);

    // Count sectors that are NEW to the top (not in previous period)
    let newSectors = 0;
    for (const s of curr) {
      if (!prev.has(s)) newSectors++;
      sectorAppearances[s] = (sectorAppearances[s] || 0) + 1;
    }

    if (newSectors > 0) leadershipChanges++;
  }

  // Most frequent leading sectors
  const sortedSectors = Object.entries(sectorAppearances)
    .sort((a, b) => b[1] - a[1])
    .map(([sector, count]) => ({
      sector,
      timesInTop: count,
      pctOfPeriods: r2((count / rotationLog.length) * 100) + '%',
    }));

  return {
    totalPeriods: rotationLog.length,
    periodsWithChange: leadershipChanges,
    turnoverRate: r2((leadershipChanges / (rotationLog.length - 1)) * 100) + '%',
    avgSectorsChanged: r2(leadershipChanges / Math.max(rotationLog.length - 1, 1)),
    sectorFrequency: sortedSectors,
    phasesObserved: [...new Set(rotationLog.map(r => r.economicPhase).filter(Boolean))],
  };
}

function analyzeDrawdowns(equityCurve) {
  const drawdowns = [];
  let peak = equityCurve[0]?.totalEquity || 0;
  let ddStart = null;
  let maxDD = 0;

  for (let i = 0; i < equityCurve.length; i++) {
    const eq = equityCurve[i].totalEquity;
    if (eq >= peak) {
      if (ddStart !== null && maxDD > 0.02) {
        drawdowns.push({
          start: equityCurve[ddStart].date,
          recovery: equityCurve[i].date,
          maxDrawdown: r2(maxDD * 100) + '%',
          duration: i - ddStart,
        });
      }
      peak = eq;
      ddStart = null;
      maxDD = 0;
    } else {
      if (ddStart === null) ddStart = i;
      const dd = (peak - eq) / peak;
      if (dd > maxDD) maxDD = dd;
    }
  }

  if (ddStart !== null && maxDD > 0.02) {
    drawdowns.push({
      start: equityCurve[ddStart].date,
      recovery: 'ONGOING',
      maxDrawdown: r2(maxDD * 100) + '%',
      duration: equityCurve.length - ddStart,
    });
  }

  return drawdowns.sort((a, b) => parseFloat(b.maxDrawdown) - parseFloat(a.maxDrawdown));
}

function calcMonthlyReturns(equityCurve) {
  if (equityCurve.length < 2) return [];
  const monthly = {};
  for (const p of equityCurve) {
    monthly[p.date.substring(0, 7)] = p.totalEquity;
  }
  const months = Object.keys(monthly).sort();
  return months.slice(1).map((m, i) => ({
    month: m,
    return: r2(((monthly[m] / monthly[months[i]]) - 1) * 100) + '%',
  }));
}

 */
function sensitivityAnalysis(data, capitalBase = 10000000) {

    { w1: 1.0, w3: 0.0, label: 'Pure 1M RS' },

  for (const combo of weightCombos) {

      return: bt.summary.totalReturn, maxDD: bt.summary.maxDrawdown,

    });

  }

  return results;
}

function r2(n) {
  return n !== null && n !== undefined && !isNaN(n) ? Math.round(n * 100) / 100 : null;
}

function formatTrade(t) {
  return {
    symbol: t.symbol,
    sector: t.sector,
    entry: `${t.entryDate} @ ₹${t.entryPrice}`,
    exit: `${t.exitDate} @ ₹${t.exitPrice}`,
    pnlPct: t.pnlPct,
    holdingDays: t.holdingDays,
    exitReason: t.exitReason,
  };
}

function calcStreaks(trades) {
  let maxWin = 0, maxLose = 0, curWin = 0, curLose = 0;
  for (const t of trades) {
    if (t.pnl > 0) { curWin++; curLose = 0; maxWin = Math.max(maxWin, curWin); }
    else { curLose++; curWin = 0; maxLose = Math.max(maxLose, curLose); }
  }
  return { maxWin, maxLose };
}

function calcRatio(returns, periodsPerYear, type) {
  if (!returns || returns.length < 2) return null;
  const mean = returns.reduce((s, r) => s + r, 0) / returns.length;
  const annReturn = mean * periodsPerYear;

  if (type === 'sharpe') {
    const std = Math.sqrt(returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (returns.length - 1));
    if (std === 0) return null;
    return (annReturn - CONFIG.RISK_FREE_RATE) / (std * Math.sqrt(periodsPerYear));
  } else {
    const down = returns.filter(r => r < 0);
    if (down.length === 0) return Infinity;
    const dsDev = Math.sqrt(down.reduce((s, r) => s + r ** 2, 0) / down.length);
    if (dsDev === 0) return Infinity;
    return (annReturn - CONFIG.RISK_FREE_RATE) / (dsDev * Math.sqrt(periodsPerYear));
  }
}

Run with `node sector-rotation-strategy.js`. Here's how every spec element maps:

---

**Relative Strength Engine (`analyzeSectorRS`):**
- Computes `RS_ratio = Sector Index / Nifty 50` at every bar
- `RS_1M` and `RS_3M` momentum of the ratio itself (is outperformance accelerating?)
- Composite score: `0.5 × RS_1M + 0.5 × RS_3M` exactly as specified
- RS slope via linear regression (direction of RS trend)
- **RRG Quadrant classification** (Leading / Improving / Weakening / Lagging) — same framework used by institutional rotation desks

**All 12 sector indices** defined with cycle metadata — Bank, IT, Pharma, FMCG, Metal, Realty, Auto, Energy, Financial Services, PSU Bank, Media, Infra.

**Economic Cycle Overlay (`detectEconomicPhase`):**
- Full 4-phase model: Early Recovery → Mid Expansion → Late Expansion → Slowdown
- Each phase maps leading/lagging sectors exactly from your India-specific table
- **Dual detection**: from real macro data (RBI stance, GDP trend, inflation, credit growth) OR inferred from sector RS patterns when macro data isn't available
- Cycle-adjusted scores boost favorable sectors by +2% and penalize unfavorable by −2%

**Stock Selection Within Sectors (`selectStocksInSector`):**
- Uses Strategy 1's momentum scoring (weighted 1M/3M/6M returns) on individual stocks within each top sector
- Picks top 3 per sector → 6-9 total holdings
- Volume health filter (20d vs 50d volume decline check)

**Three portfolio weighting methods:**
- `EQUAL` — simple equal weight
- `RS_WEIGHTED` — sectors weighted proportionally to their RS score, stocks equal within sector
- `INV_VOL` — inverse volatility (more capital to calmer stocks)
- Hard caps: 40% max per sector, 15% max per stock

**Rotation Heatmap** (`generateRotationHeatmap`) — visual sector leadership over 6 monthly snapshots with `▲▲ ▲ ● ▼ ▼▼` status indicators.

**Portfolio monitoring** tracks individual stock stop-losses (8%) and portfolio-level stop (10%), with rebalance-due signals at month-end.

**Backtest analytics unique to rotation:**
- **Sector hit rate** — which sectors produced the best/worst trades
- **Rotation turnover analysis** — how often leadership changed, most frequent leading sectors
- **Economic phases observed** during the backtest period
- Performance by exit reason (REBALANCE vs STOP_LOSS)
- Sensitivity: top-N sectors, stocks/sector, RS weights, holding period, weighting method
