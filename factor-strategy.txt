// ============================================================================
// TECHNICAL CALCULATIONS
// ============================================================================

function calcSMA(data, period) {
  const result = new Array(data.length).fill(null);
  if (data.length < period) return result;
  for (let i = period - 1; i < data.length; i++) {
    let sum = 0;
    for (let j = i - period + 1; j <= i; j++) sum += data[j];
    result[i] = sum / period;
  }
  return result;
}

function calcReturn(prices, lookback, skipRecent = 0) {
  if (!prices || prices.length < lookback + 1 + skipRecent) return null;
  const endIdx = prices.length - 1 - skipRecent;
  const startIdx = endIdx - lookback;
  if (startIdx < 0 || prices[startIdx] === 0) return null;
  return (prices[endIdx] / prices[startIdx]) - 1;
}

function calcVolatility(prices, window = 252) {
  const effectiveWindow = Math.min(window, prices.length - 1);
  if (effectiveWindow < 20) return null;
  const rets = [];
  for (let i = prices.length - effectiveWindow; i < prices.length; i++) {
    if (prices[i - 1] === 0) return null;
    rets.push(Math.log(prices[i] / prices[i - 1]));
  }
  const mean = rets.reduce((s, r) => s + r, 0) / rets.length;
  const variance = rets.reduce((s, r) => s + (r - mean) ** 2, 0) / (rets.length - 1);
  return Math.sqrt(variance) * Math.sqrt(CONFIG.TRADING_DAYS_PER_YEAR);
}

function calcDrawdown(prices) {
  let peak = prices[0];
  let maxDD = 0;
  for (const p of prices) {
    if (p > peak) peak = p;
    const dd = (peak - p) / peak;
    if (dd > maxDD) maxDD = dd;
  }
  return maxDD;
}

function calcBeta(stockReturns, benchmarkReturns) {
  if (stockReturns.length !== benchmarkReturns.length || stockReturns.length < 20) return null;
  const n = stockReturns.length;
  const meanS = stockReturns.reduce((s, r) => s + r, 0) / n;
  const meanB = benchmarkReturns.reduce((s, r) => s + r, 0) / n;

  let cov = 0, varB = 0;
  for (let i = 0; i < n; i++) {
    cov += (stockReturns[i] - meanS) * (benchmarkReturns[i] - meanB);
    varB += (benchmarkReturns[i] - meanB) ** 2;
  }

  return varB !== 0 ? cov / varB : null;
}

// FACTOR SCORING ENGINE
// ============================================================================

/**
 * Percentile rank within a sorted array
 * Returns 0–100 (100 = best)
 */
function percentileRank(value, allValues, higherIsBetter = true) {
  if (allValues.length === 0 || value === null || value === undefined) return null;

  const sorted = [...allValues].filter(v => v !== null && v !== undefined && !isNaN(v));
  if (sorted.length === 0) return null;

  sorted.sort((a, b) => a - b);
  let rank = sorted.filter(v => v < value).length;
  const percentile = (rank / sorted.length) * 100;

  return higherIsBetter ? percentile : (100 - percentile);
}


// ── MOMENTUM FACTOR ──

/**
 * Calculate momentum factor score for a stock
 * @param {number[]} prices - daily close prices
 * @returns {Object} momentum sub-scores and composite
 */
function calcMomentumFactor(prices) {
  if (!prices || prices.length < CONFIG.MOMENTUM.LOOKBACK_6M + CONFIG.MOMENTUM.SKIP_RECENT_DAYS + 1) {
    return null;
  }

  const skip = CONFIG.MOMENTUM.SKIP_RECENT_DAYS;

  const ret1m = calcReturn(prices, CONFIG.MOMENTUM.LOOKBACK_1M, skip);
  const ret3m = calcReturn(prices, CONFIG.MOMENTUM.LOOKBACK_3M, skip);
  const ret6m = calcReturn(prices, CONFIG.MOMENTUM.LOOKBACK_6M, skip);
  const ret12m = prices.length >= CONFIG.MOMENTUM.LOOKBACK_12M + skip + 1
    ? calcReturn(prices, CONFIG.MOMENTUM.LOOKBACK_12M, skip)
    : null;

  if (ret3m === null) return null;

  // Raw composite (before percentile ranking)
  let weights = CONFIG.MOMENTUM.WEIGHT_1M + CONFIG.MOMENTUM.WEIGHT_3M + CONFIG.MOMENTUM.WEIGHT_6M;
  let rawScore =
    (ret1m !== null ? CONFIG.MOMENTUM.WEIGHT_1M * ret1m : 0) +
    CONFIG.MOMENTUM.WEIGHT_3M * ret3m +
    (ret6m !== null ? CONFIG.MOMENTUM.WEIGHT_6M * ret6m : 0);

  if (ret12m !== null) {
    rawScore += CONFIG.MOMENTUM.WEIGHT_12M * ret12m;
    weights += CONFIG.MOMENTUM.WEIGHT_12M;
  }

  rawScore /= weights;

  return {
    rawScore,
    ret1m,
    ret3m,
    ret6m,
    ret12m,
  };
}


// ── VALUE FACTOR ──

/**
 * Calculate value factor score from fundamental data
 * @param {Object} fundamentals - { evEbitda, peRatio, pbRatio, earningsYield, eps }
 * @returns {Object} value sub-scores and raw composite
 */
function calcValueFactor(fundamentals) {
  const { evEbitda, peRatio, pbRatio, earningsYield, eps } = fundamentals;

  // Exclude negative earnings companies
  if (CONFIG.VALUE.EXCLUDE_NEGATIVE_EARNINGS && eps !== undefined && eps <= 0) {
    return null;
  }

  const metrics = {};
  const metricConfigs = CONFIG.VALUE.METRICS;

  // EV/EBITDA
  if (evEbitda !== null && evEbitda !== undefined) {
    let val = evEbitda;
    if (metricConfigs.EV_EBITDA.cap) val = Math.min(val, metricConfigs.EV_EBITDA.cap);
    if (metricConfigs.EV_EBITDA.floor !== undefined) val = Math.max(val, metricConfigs.EV_EBITDA.floor);
    metrics.evEbitda = val;
  }

  // P/E
  if (peRatio !== null && peRatio !== undefined && peRatio > 0) {
    let val = peRatio;
    if (metricConfigs.PE_RATIO.cap) val = Math.min(val, metricConfigs.PE_RATIO.cap);
    metrics.peRatio = val;
  }

  // P/B
  if (pbRatio !== null && pbRatio !== undefined && pbRatio > 0) {
    let val = pbRatio;
    if (metricConfigs.PB_RATIO.cap) val = Math.min(val, metricConfigs.PB_RATIO.cap);
    metrics.pbRatio = val;
  }

  // Earnings yield (1/PE or EBIT/EV)
  if (earningsYield !== null && earningsYield !== undefined) {
    metrics.earningsYield = earningsYield;
  } else if (peRatio && peRatio > 0) {
    metrics.earningsYield = 1 / peRatio;
  }

  return {
    metrics,
    rawValues: { evEbitda, peRatio, pbRatio, earningsYield },
  };
}


// ── QUALITY FACTOR ──

/**
 * Calculate quality factor score from fundamental data
 * @param {Object} fundamentals
 *   { roe, roic, debtToEquity, earningsStability (0-1 score),
 *     earningsGrowth3Y (CAGR), yearsPositiveEarnings, interestCoverage }
 * @returns {Object|null} quality sub-scores
 */
function calcQualityFactor(fundamentals) {
  const {
    roe, roic, debtToEquity, earningsStability,
    earningsGrowth3Y, yearsPositiveEarnings,
  } = fundamentals;

  // Hard gates
  if (debtToEquity !== undefined && debtToEquity > CONFIG.QUALITY.MAX_DEBT_TO_EQUITY) {
    return null; // exclude extremely leveraged
  }

  if (yearsPositiveEarnings !== undefined &&
    yearsPositiveEarnings < CONFIG.QUALITY.MIN_YEARS_POSITIVE_EARNINGS) {
    return null;
  }

  const metrics = {};

  if (roe !== null && roe !== undefined) metrics.roe = roe;
  if (roic !== null && roic !== undefined) metrics.roic = roic;
  if (debtToEquity !== null && debtToEquity !== undefined) {
    metrics.debtToEquity = Math.min(debtToEquity, CONFIG.QUALITY.METRICS.DEBT_TO_EQUITY.cap || 5);
  }
  if (earningsStability !== null && earningsStability !== undefined) {
    metrics.earningsStability = earningsStability;
  }
  if (earningsGrowth3Y !== null && earningsGrowth3Y !== undefined) {
    metrics.earningsGrowth3Y = earningsGrowth3Y;
  }

  return { metrics };
}

/**
 * Calculate earnings stability as coefficient of variation of EPS
 * Lower CV = more stable = better
 * @param {number[]} epsHistory - array of EPS for last 5+ years
 * @returns {number} stability score (0-1, 1 = perfectly stable)
 */
function calcEarningsStability(epsHistory) {
  if (!epsHistory || epsHistory.length < 3) return null;

  const positiveEps = epsHistory.filter(e => e > 0);
  if (positiveEps.length < 3) return 0;

  const mean = positiveEps.reduce((s, e) => s + e, 0) / positiveEps.length;
  if (mean === 0) return 0;

  const stdDev = Math.sqrt(
    positiveEps.reduce((s, e) => s + (e - mean) ** 2, 0) / positiveEps.length
  );

  const cv = stdDev / Math.abs(mean);

  // Convert to 0-1 score (lower CV = higher score)
  // CV of 0 = perfect stability (score 1)
  // CV of 1+ = very unstable (score ~0)
  return Math.max(0, Math.min(1, 1 - cv));
}


// ── EARNINGS REVISION FACTOR ──

/**
 * Calculate earnings revision factor
 * @param {Object} revisionData
 *   { currentFYEPS, previousFYEPS (1 month ago),
 *     currentFYRevenue, previousFYRevenue,
 *     numUpgrades, numDowngrades, numTotal }
 * @returns {Object|null} revision scores
 */
function calcEarningsRevisionFactor(revisionData) {
  if (!revisionData) return null;

  const {
    currentFYEPS, previousFYEPS,
    currentFYRevenue, previousFYRevenue,
    numUpgrades, numDowngrades, numTotal,
  } = revisionData;

  const metrics = {};

  // EPS revision %
  if (currentFYEPS !== null && previousFYEPS !== null && previousFYEPS !== 0) {
    metrics.epsRevision1M = (currentFYEPS - previousFYEPS) / Math.abs(previousFYEPS);
  }

  // Revenue revision %
  if (currentFYRevenue !== null && previousFYRevenue !== null && previousFYRevenue !== 0) {
    metrics.revenueRevision1M = (currentFYRevenue - previousFYRevenue) / Math.abs(previousFYRevenue);
  }

  // Net upgrades ratio: (upgrades - downgrades) / total
  if (numTotal !== undefined && numTotal > 0) {
    metrics.netUpgradeRatio = ((numUpgrades || 0) - (numDowngrades || 0)) / numTotal;
  }

  if (Object.keys(metrics).length === 0) return null;

  return { metrics };
}


// ============================================================================
// CROSS-SECTIONAL RANKING ENGINE
// ============================================================================

/**
 * Rank all stocks on each factor and compute composite scores
 *
 * @param {Object[]} universe - array of stock objects:
 *   { symbol, sector, marketCapCr, prices[],
 *     fundamentals: { evEbitda, peRatio, pbRatio, roe, roic, debtToEquity,
 *       earningsGrowth3Y, earningsStability, eps, earningsYield,
 *       yearsPositiveEarnings },
 *     revisionData: { currentFYEPS, previousFYEPS, ... },
 *     avgDailyTurnoverCr, freeFloatPct }
 *
 * @returns {Object} ranked universe with composite scores
 */
function rankUniverse(universe) {
  // ── Step 0: Apply universe filters ──
  const filtered = [];
  const qualifying = [];

  for (const stock of universe) {
    const filterResult = applyUniverseFilters(stock);
    if (!filterResult.pass) {
      filtered.push({ symbol: stock.symbol, reason: filterResult.reason });
      continue;
    }
    qualifying.push(stock);
  }

  // ── Step 1: Compute raw factor scores for each stock ──
  const factorData = [];

  for (const stock of qualifying) {
    const momentum = calcMomentumFactor(stock.prices);
    const value = calcValueFactor(stock.fundamentals || {});
    const quality = calcQualityFactor(stock.fundamentals || {});
    const revision = calcEarningsRevisionFactor(stock.revisionData);

    if (!momentum) continue; // minimum: must have price momentum

    factorData.push({
      symbol: stock.symbol,
      sector: stock.sector,
      marketCapCr: stock.marketCapCr,
      currentPrice: stock.prices[stock.prices.length - 1],
      momentum,
      value,
      quality,
      revision,
      volatility: calcVolatility(stock.prices),
      avgDailyTurnoverCr: stock.avgDailyTurnoverCr,
      rawStock: stock,
    });
  }

  if (factorData.length === 0) {
    return { ranked: [], filtered, qualifying: 0, message: 'No stocks passed filters' };
  }

  // ── Step 2: Percentile rank each factor ──

  // Momentum percentile
  const allMomScores = factorData.map(d => d.momentum.rawScore);
  for (const d of factorData) {
    d.momentumPercentile = percentileRank(d.momentum.rawScore, allMomScores, true);
  }

  // Value percentile (for each sub-metric, then combine)
  const valueMetricNames = Object.keys(CONFIG.VALUE.METRICS);
  for (const d of factorData) {
    if (!d.value) {
      d.valuePercentile = null;
      continue;
    }

    let weightedSum = 0;
    let totalWeight = 0;

    for (const metricName of valueMetricNames) {
      const metricConfig = CONFIG.VALUE.METRICS[metricName];
      const keyMap = {
        EV_EBITDA: 'evEbitda',
        PE_RATIO: 'peRatio',
        PB_RATIO: 'pbRatio',
        EARNINGS_YIELD: 'earningsYield',
      };
      const key = keyMap[metricName];
      const val = d.value.metrics[key];

      if (val === undefined || val === null) continue;

      const allVals = factorData
        .filter(dd => dd.value && dd.value.metrics[key] !== undefined)
        .map(dd => dd.value.metrics[key]);

      const higherBetter = metricConfig.direction === 'HIGHER_BETTER';
      const pctile = percentileRank(val, allVals, higherBetter);

      if (pctile !== null) {
        weightedSum += metricConfig.weight * pctile;
        totalWeight += metricConfig.weight;
      }
    }

    d.valuePercentile = totalWeight > 0 ? weightedSum / totalWeight : null;
  }

  // Quality percentile
  const qualityMetricNames = Object.keys(CONFIG.QUALITY.METRICS);
  for (const d of factorData) {
    if (!d.quality) {
      d.qualityPercentile = null;
      continue;
    }

    let weightedSum = 0;
    let totalWeight = 0;

    for (const metricName of qualityMetricNames) {
      const metricConfig = CONFIG.QUALITY.METRICS[metricName];
      const keyMap = {
        ROE: 'roe',
        ROIC: 'roic',
        DEBT_TO_EQUITY: 'debtToEquity',
        EARNINGS_STABILITY: 'earningsStability',
        EARNINGS_GROWTH_3Y: 'earningsGrowth3Y',
      };
      const key = keyMap[metricName];
      const val = d.quality.metrics[key];

      if (val === undefined || val === null) continue;

      const allVals = factorData
        .filter(dd => dd.quality && dd.quality.metrics[key] !== undefined)
        .map(dd => dd.quality.metrics[key]);

      const higherBetter = metricConfig.direction === 'HIGHER_BETTER';
      const pctile = percentileRank(val, allVals, higherBetter);

      if (pctile !== null) {
        weightedSum += metricConfig.weight * pctile;
        totalWeight += metricConfig.weight;
      }
    }

    d.qualityPercentile = totalWeight > 0 ? weightedSum / totalWeight : null;
  }

  // Earnings revision percentile
  const revMetricNames = Object.keys(CONFIG.EARNINGS_REVISION.METRICS);
  for (const d of factorData) {
    if (!d.revision) {
      d.revisionPercentile = null;
      continue;
    }

    let weightedSum = 0;
    let totalWeight = 0;

    const keyMap = {
      EPS_REVISION_1M: 'epsRevision1M',
      REVENUE_REVISION_1M: 'revenueRevision1M',
      NUM_UPGRADES_VS_DOWNGRADES: 'netUpgradeRatio',
    };

    for (const metricName of revMetricNames) {
      const metricConfig = CONFIG.EARNINGS_REVISION.METRICS[metricName];
      const key = keyMap[metricName];
      const val = d.revision.metrics[key];

      if (val === undefined || val === null) continue;

      const allVals = factorData
        .filter(dd => dd.revision && dd.revision.metrics[key] !== undefined)
        .map(dd => dd.revision.metrics[key]);

      const pctile = percentileRank(val, allVals, true);

      if (pctile !== null) {
        weightedSum += metricConfig.weight * pctile;
        totalWeight += metricConfig.weight;
      }
    }

    d.revisionPercentile = totalWeight > 0 ? weightedSum / totalWeight : null;
  }

  // ── Step 3: Compute composite score ──
  for (const d of factorData) {
    let composite = 0;
    let activeWeight = 0;

    if (d.momentumPercentile !== null) {
      composite += CONFIG.WEIGHTS.MOMENTUM * d.momentumPercentile;
      activeWeight += CONFIG.WEIGHTS.MOMENTUM;
    }
    if (d.valuePercentile !== null) {
      composite += CONFIG.WEIGHTS.VALUE * d.valuePercentile;
      activeWeight += CONFIG.WEIGHTS.VALUE;
    }
    if (d.qualityPercentile !== null) {
      composite += CONFIG.WEIGHTS.QUALITY * d.qualityPercentile;
      activeWeight += CONFIG.WEIGHTS.QUALITY;
    }
    if (d.revisionPercentile !== null) {
      composite += CONFIG.WEIGHTS.EARNINGS_REVISION * d.revisionPercentile;
      activeWeight += CONFIG.WEIGHTS.EARNINGS_REVISION;
    }

    // Normalize by active weight (in case some factors are missing)
    d.compositeScore = activeWeight > 0 ? composite / activeWeight : 0;
    d.factorsAvailable = {
      momentum: d.momentumPercentile !== null,
      value: d.valuePercentile !== null,
      quality: d.qualityPercentile !== null,
      revision: d.revisionPercentile !== null,
    };
    d.factorCount = Object.values(d.factorsAvailable).filter(Boolean).length;
  }

  // ── Step 4: Sort by composite score ──
  factorData.sort((a, b) => b.compositeScore - a.compositeScore);
  factorData.forEach((d, i) => { d.rank = i + 1; });

  return {
    ranked: factorData,
    totalUniverse: universe.length,
    filtered: filtered.length,
    qualifying: factorData.length,
    filteredDetails: filtered.slice(0, 20),
  };
}


// ============================================================================
// UNIVERSE FILTERS
// ============================================================================

function applyUniverseFilters(stock) {
  if (stock.avgDailyTurnoverCr && stock.avgDailyTurnoverCr < CONFIG.MIN_AVG_TURNOVER_CR) {
    return { pass: false, reason: `Turnover ₹${stock.avgDailyTurnoverCr}cr < ₹${CONFIG.MIN_AVG_TURNOVER_CR}cr` };
  }

  if (stock.marketCapCr && stock.marketCapCr < CONFIG.MIN_MARKET_CAP_CR) {
    return { pass: false, reason: `Market cap ₹${stock.marketCapCr}cr < ₹${CONFIG.MIN_MARKET_CAP_CR}cr` };
  }

  if (stock.prices && stock.prices.length > 0) {
    const price = stock.prices[stock.prices.length - 1];
    if (price < CONFIG.MIN_PRICE) {
      return { pass: false, reason: `Price ₹${r2(price)} < ₹${CONFIG.MIN_PRICE}` };
    }
  }

  if (stock.prices && stock.prices.length < CONFIG.MIN_HISTORY_DAYS) {
    return { pass: false, reason: `Only ${stock.prices.length} days history (need ${CONFIG.MIN_HISTORY_DAYS})` };
  }

  if (stock.freeFloatPct !== undefined && stock.freeFloatPct < CONFIG.MIN_FREE_FLOAT_PCT) {
    return { pass: false, reason: `Free float ${stock.freeFloatPct}% < ${CONFIG.MIN_FREE_FLOAT_PCT}%` };
  }

  if (CONFIG.EXCLUDE_SECTORS.length > 0 && CONFIG.EXCLUDE_SECTORS.includes(stock.sector)) {
    return { pass: false, reason: `Sector ${stock.sector} excluded` };
  }

  return { pass: true };
}


// ============================================================================
// PORTFOLIO CONSTRUCTION
// ============================================================================

/**
 * Build the factor-based portfolio
 *
 * @param {Object[]} universe - full stock universe with all data
 * @param {number[]} niftyPrices - Nifty 50 daily closes
 * @param {number} totalCapital
 * @param {string} [currentDate]
 * @returns {Object} portfolio
 */
function buildPortfolio(universe, niftyPrices, totalCapital, currentDate = null) {
  // Market regime check
  const regime = checkMarketRegime(niftyPrices);

  if (!regime.tradeable && CONFIG.REDUCE_EXPOSURE_IN_BEAR) {
    return {
      date: currentDate,
      regime,
      signal: 'CASH',
      message: 'Bear market — reduce or eliminate equity exposure',
      holdings: [],
    };
  }

  // Rank the universe
  const ranking = rankUniverse(universe);

  if (ranking.qualifying === 0) {
    return {
      date: currentDate,
      regime,
      signal: 'NO_STOCKS',
      message: 'No stocks passed filters',
      holdings: [],
    };
  }

  // Select top N
  const topStocks = ranking.ranked.slice(0, CONFIG.TOP_N_STOCKS);

  // Apply sector concentration limits
  const sectorCounts = {};
  const selected = [];

  for (const stock of topStocks) {
    const sector = stock.sector || 'Unknown';
    sectorCounts[sector] = (sectorCounts[sector] || 0) + 1;

    // Simple sector cap: max weight check done in weighting step
    selected.push(stock);
  }

  // Calculate weights
  const holdings = calcWeights(selected, totalCapital, regime);

  // Factor exposure analysis
  const factorExposure = analyzeFactorExposure(holdings);

  return {
    date: currentDate,
    regime,
    signal: 'INVEST',
    holdings,
    portfolioStats: calcPortfolioStats(holdings, totalCapital),
    factorExposure,
    rankingSummary: {
      totalUniverse: ranking.totalUniverse,
      filtered: ranking.filtered,
      qualifying: ranking.qualifying,
      selected: holdings.length,
    },
    topDecile: ranking.ranked.slice(0, Math.ceil(ranking.qualifying * 0.1)).map(d => ({
      symbol: d.symbol,
      composite: r2(d.compositeScore),
      momentum: r2(d.momentumPercentile),
      value: r2(d.valuePercentile),
      quality: r2(d.qualityPercentile),
      revision: r2(d.revisionPercentile),
    })),
    bottomDecile: ranking.ranked.slice(-Math.ceil(ranking.qualifying * 0.1)).map(d => ({
      symbol: d.symbol,
      composite: r2(d.compositeScore),
    })),
    holdingPeriod: `${CONFIG.HOLDING_PERIOD_DAYS} trading days`,
    nextRebalance: `After ${CONFIG.HOLDING_PERIOD_DAYS} trading days`,
  };
}

/**
 * Calculate portfolio weights
 */
function calcWeights(stocks, totalCapital, regime) {
  if (stocks.length === 0) return [];

  // Reduce allocation in cautious regime
  const regimeMultiplier = regime.regime === 'CAUTIOUS_BULL' ? 0.7 : 1.0;
  const investable = totalCapital * regimeMultiplier;

  let holdings;

  if (CONFIG.WEIGHTING === 'EQUAL') {
    const weight = 1 / stocks.length;
    holdings = stocks.map(s => ({
      ...formatHolding(s),
      weight: Math.min(weight, CONFIG.MAX_STOCK_WEIGHT),
    }));
  } else if (CONFIG.WEIGHTING === 'COMPOSITE_WEIGHTED') {
    const totalScore = stocks.reduce((sum, s) => sum + s.compositeScore, 0);
    holdings = stocks.map(s => ({
      ...formatHolding(s),
      weight: Math.min(
        totalScore > 0 ? s.compositeScore / totalScore : 1 / stocks.length,
        CONFIG.MAX_STOCK_WEIGHT
      ),
    }));
  } else if (CONFIG.WEIGHTING === 'INV_VOL') {
    const totalInvVol = stocks.reduce((sum, s) => {
      return sum + (s.volatility && s.volatility > 0 ? 1 / s.volatility : 1);
    }, 0);
    holdings = stocks.map(s => {
      const invVol = s.volatility && s.volatility > 0 ? 1 / s.volatility : 1;
      return {
        ...formatHolding(s),
        weight: Math.min(invVol / totalInvVol, CONFIG.MAX_STOCK_WEIGHT),
      };
    });
  } else {
    // Default: equal
    const weight = 1 / stocks.length;
    holdings = stocks.map(s => ({
      ...formatHolding(s),
      weight: Math.min(weight, CONFIG.MAX_STOCK_WEIGHT),
    }));
  }

  // Normalize weights to sum to ~1
  const totalWeight = holdings.reduce((sum, h) => sum + h.weight, 0);
  if (totalWeight > 0) {
    holdings.forEach(h => { h.weight = h.weight / totalWeight; });
  }

  // Apply sector cap
  const sectorWeights = {};
  for (const h of holdings) {
    const sector = h.sector || 'Unknown';
    sectorWeights[sector] = (sectorWeights[sector] || 0) + h.weight;
  }

  // If any sector exceeds cap, redistribute
  for (const [sector, totalSW] of Object.entries(sectorWeights)) {
    if (totalSW > CONFIG.MAX_SECTOR_WEIGHT) {
      const sectorHoldings = holdings.filter(h => (h.sector || 'Unknown') === sector);
      const scaleFactor = CONFIG.MAX_SECTOR_WEIGHT / totalSW;
      sectorHoldings.forEach(h => { h.weight *= scaleFactor; });
    }
  }

  // Re-normalize
  const finalTotal = holdings.reduce((sum, h) => sum + h.weight, 0);
  if (finalTotal > 0 && finalTotal < 0.99) {
    holdings.forEach(h => { h.weight = h.weight / finalTotal; });
  }

  // Calculate allocations
  for (const h of holdings) {
    h.allocatedCapital = Math.round(h.weight * investable);
    h.shares = h.currentPrice > 0 ? Math.floor(h.allocatedCapital / h.currentPrice) : 0;
    h.stopLoss = r2(h.currentPrice * (1 - CONFIG.STOP_LOSS_PCT));
    h.weight = r2(h.weight);
  }

  return holdings;
}

function formatHolding(stock) {
  return {
    symbol: stock.symbol,
    sector: stock.sector,
    rank: stock.rank,
    currentPrice: r2(stock.currentPrice),
    compositeScore: r2(stock.compositeScore),
    factorCount: stock.factorCount,
    momentumPctile: r2(stock.momentumPercentile),
    valuePctile: r2(stock.valuePercentile),
    qualityPctile: r2(stock.qualityPercentile),
    revisionPctile: r2(stock.revisionPercentile),
    // Raw factor highlights
    ret3m: stock.momentum ? r2(stock.momentum.ret3m * 100) + '%' : null,
    roe: stock.quality ? r2(stock.quality.metrics.roe) + '%' : null,
    evEbitda: stock.value ? r2(stock.value.rawValues.evEbitda) : null,
    volatility: stock.volatility ? r2(stock.volatility * 100) + '%' : null,
    marketCapCr: stock.marketCapCr,
  };
}

// ============================================================================

/**
 * Analyze the portfolio's factor tilts vs. the universe
 */
function analyzeFactorExposure(holdings) {
  if (holdings.length === 0) return {};

  const avgMom = avg(holdings.map(h => h.momentumPctile).filter(v => v !== null));
  const avgVal = avg(holdings.map(h => h.valuePctile).filter(v => v !== null));
  const avgQual = avg(holdings.map(h => h.qualityPctile).filter(v => v !== null));
  const avgRev = avg(holdings.map(h => h.revisionPctile).filter(v => v !== null));

  // Factor tilt: how far above 50th percentile (universe median) are we?
  return {
    momentum: {
      avgPercentile: r2(avgMom),
      tilt: avgMom !== null ? r2(avgMom - 50) : null,
      interpretation: avgMom > 70 ? 'Strong momentum tilt' :
        avgMom > 55 ? 'Moderate momentum tilt' : 'Weak/neutral momentum',
    },
    value: {
      avgPercentile: r2(avgVal),
      tilt: avgVal !== null ? r2(avgVal - 50) : null,
      interpretation: avgVal > 70 ? 'Strong value tilt' :
        avgVal > 55 ? 'Moderate value tilt' : 'Weak/neutral value',
    },
    quality: {
      avgPercentile: r2(avgQual),
      tilt: avgQual !== null ? r2(avgQual - 50) : null,
      interpretation: avgQual > 70 ? 'Strong quality tilt' :
        avgQual > 55 ? 'Moderate quality tilt' : 'Weak/neutral quality',
    },
    earningsRevision: {
      avgPercentile: r2(avgRev),
      tilt: avgRev !== null ? r2(avgRev - 50) : null,
      interpretation: avgRev > 70 ? 'Strong revision tilt' :
        avgRev > 55 ? 'Moderate revision tilt' : 'Weak/neutral revision',
    },
    compositeAvg: r2(avg(holdings.map(h => h.compositeScore).filter(v => v !== null))),
    dominantFactor: determineDominantFactor(avgMom, avgVal, avgQual, avgRev),
  };
}

function determineDominantFactor(mom, val, qual, rev) {
  const factors = [
    { name: 'Momentum', score: mom || 0 },
    { name: 'Value', score: val || 0 },
    { name: 'Quality', score: qual || 0 },
    { name: 'Earnings Revision', score: rev || 0 },
  ];
  factors.sort((a, b) => b.score - a.score);
  return factors[0].name;
}


// ============================================================================
// PORTFOLIO STATISTICS
// ============================================================================

function calcPortfolioStats(holdings, totalCapital) {
  if (holdings.length === 0) return {};

  const totalInvested = holdings.reduce((s, h) => s + (h.allocatedCapital || 0), 0);

  // Sector breakdown
  const sectors = {};
  for (const h of holdings) {
    const s = h.sector || 'Unknown';
    sectors[s] = (sectors[s] || 0) + (h.weight || 0);
  }

  // HHI
  const hhi = holdings.reduce((sum, h) => sum + ((h.weight || 0) * 100) ** 2, 0);

  // Weighted volatility (simplified)
  const vols = holdings.filter(h => h.volatility);
  const wVol = vols.length > 0
    ? Math.sqrt(vols.reduce((sum, h) => {
      const w = parseFloat(h.weight) || 0;
      const v = parseFloat(h.volatility) / 100 || 0.2;
      return sum + (w * v) ** 2;
    }, 0))
    : null;

  return {
    holdingsCount: holdings.length,
    totalInvested: Math.round(totalInvested),
    cashReserve: Math.round(totalCapital - totalInvested),
    investedPct: r2((totalInvested / totalCapital) * 100) + '%',
    avgCompositeScore: r2(avg(holdings.map(h => h.compositeScore))),
    avgFactorCount: r2(avg(holdings.map(h => h.factorCount))),
    sectorBreakdown: Object.entries(sectors)
      .map(([s, w]) => ({ sector: s, weight: r2(w * 100) + '%' }))
      .sort((a, b) => parseFloat(b.weight) - parseFloat(a.weight)),
    maxStockWeight: r2(Math.max(...holdings.map(h => parseFloat(h.weight) || 0)) * 100) + '%',
    hhiIndex: Math.round(hhi),
    diversification: hhi > 1500 ? 'CONCENTRATED' : hhi > 800 ? 'MODERATE' : 'DIVERSIFIED',
    portfolioVolatility: wVol ? r2(wVol * 100) + '%' : null,
  };
}


// ============================================================================
// MARKET REGIME
// ============================================================================

function checkMarketRegime(niftyPrices) {
  if (!niftyPrices || niftyPrices.length < CONFIG.NIFTY_TREND_SMA + 1) {
    return { regime: 'UNKNOWN', tradeable: true, reason: 'Insufficient Nifty data' };
  }

  const sma50 = calcSMA(niftyPrices, CONFIG.NIFTY_SMA_PERIOD);
  const sma200 = calcSMA(niftyPrices, CONFIG.NIFTY_TREND_SMA);
  const curr = niftyPrices[niftyPrices.length - 1];
  const s50 = sma50[sma50.length - 1];
  const s200 = sma200[sma200.length - 1];

  if (s200 === null) return { regime: 'UNKNOWN', tradeable: true };

  const above50 = s50 !== null && curr > s50;
  const above200 = curr > s200;

  if (above200 && above50) {
    return { regime: 'BULL', tradeable: true, nifty: r2(curr), sma50: r2(s50), sma200: r2(s200) };
  } else if (above200) {
    return { regime: 'CAUTIOUS_BULL', tradeable: true, nifty: r2(curr), sma50: r2(s50), sma200: r2(s200) };
  }
  return {
    regime: 'BEAR', tradeable: false,
    nifty: r2(curr), sma50: r2(s50), sma200: r2(s200),
    reason: 'Nifty below 200-SMA',
  };
}


// ============================================================================
// POSITION MONITORING
// ============================================================================

function monitorPortfolio(holdings, currentPrices, daysSinceRebalance) {
  const actions = [];
  let portfolioPnl = 0;
  let totalAllocated = 0;

  for (const h of holdings) {
    const currentPrice = currentPrices[h.symbol];
    if (currentPrice === undefined) continue;

    const entryPrice = parseFloat(h.currentPrice);
    const pnlPct = (currentPrice / entryPrice) - 1;
    const pnl = (currentPrice - entryPrice) * (h.shares || 0);
    portfolioPnl += pnl;
    totalAllocated += h.allocatedCapital || 0;

    const stopLoss = parseFloat(h.stopLoss);

    if (currentPrice <= stopLoss) {
      actions.push({
        symbol: h.symbol, action: 'EXIT', reason: 'STOP_LOSS',
        pnlPct: r2(pnlPct * 100) + '%', urgency: 'IMMEDIATE',
      });
    } else {
      actions.push({
        symbol: h.symbol, action: 'HOLD',
        pnlPct: r2(pnlPct * 100) + '%',
        distanceToStop: r2(((currentPrice - stopLoss) / currentPrice) * 100) + '%',
      });
    }
  }

  const portfolioPnlPct = totalAllocated > 0 ? portfolioPnl / totalAllocated : 0;
  const portfolioStop = portfolioPnlPct <= -CONFIG.PORTFOLIO_STOP_PCT;
  const rebalanceDue = daysSinceRebalance >= CONFIG.HOLDING_PERIOD_DAYS;

  return {
    daysSinceRebalance,
    portfolioPnlPct: r2(portfolioPnlPct * 100) + '%',
    portfolioStopTriggered: portfolioStop,
    rebalanceDue,
    stockActions: actions,
    recommendation: portfolioStop
      ? 'EXIT ALL — portfolio stop triggered'
      : rebalanceDue
        ? 'REBALANCE — re-rank and rebuild'
        : 'HOLD',
  };
}

// BACKTEST ENGINE
// ============================================================================

/**
 * Backtest the multi-factor strategy
 *
 * @param {Object[]} universeSnapshots - monthly snapshots:
 *   [{ date, stocks: [{ symbol, sector, prices[], fundamentals, revisionData, ... }],
 *      niftyPrices }]
 *
 * @param {number} capitalBase
 * @returns {Object} backtest results
 */
function runBacktest(universeSnapshots, capitalBase = 10000000) {
  let capital = capitalBase;
  const closedTrades = [];
  const equityCurve = [];
  let peakEquity = capitalBase;
  let maxDrawdown = 0;
  const decileReturns = { top: [], bottom: [] };

  for (let period = 0; period < universeSnapshots.length; period++) {
    const { date, stocks, niftyPrices } = universeSnapshots[period];

    // Build portfolio
    const portfolio = buildPortfolio(stocks, niftyPrices, capital, date);

    if (portfolio.signal !== 'INVEST' || portfolio.holdings.length === 0) {
      equityCurve.push({
        date, totalEquity: Math.round(capital), positions: 0, drawdown: '0%',
      });
      continue;
    }

    // Simulate holding period
    // Look at next period for exit prices
    const nextPeriod = period + 1 < universeSnapshots.length
      ? universeSnapshots[period + 1] : null;

    for (const h of portfolio.holdings) {
      const entryPrice = parseFloat(h.currentPrice);
      let exitPrice = entryPrice;
      let exitReason = 'REBALANCE';

      if (nextPeriod) {
        const nextStock = nextPeriod.stocks.find(s => s.symbol === h.symbol);
        if (nextStock && nextStock.prices && nextStock.prices.length > 0) {
          exitPrice = nextStock.prices[nextStock.prices.length - 1];
        }
      }

      // Check stop-loss
      const stopLoss = parseFloat(h.stopLoss);
      if (exitPrice < stopLoss) {
        exitPrice = stopLoss;
        exitReason = 'STOP_LOSS';
      }

      const pnl = (exitPrice - entryPrice) * (h.shares || 0);
      capital += pnl;

      closedTrades.push({
        symbol: h.symbol,
        sector: h.sector,
        entryDate: date,
        exitDate: nextPeriod ? nextPeriod.date : 'END',
        entryPrice: r2(entryPrice),
        exitPrice: r2(exitPrice),
        pnl: Math.round(pnl),
        pnlPct: r2(((exitPrice / entryPrice) - 1) * 100) + '%',
        exitReason,
        rank: h.rank,
        compositeScore: h.compositeScore,
        momentumPctile: h.momentumPctile,
        valuePctile: h.valuePctile,
        qualityPctile: h.qualityPctile,
        revisionPctile: h.revisionPctile,
        factorCount: h.factorCount,
      });
    }

    // Track equity
    if (capital > peakEquity) peakEquity = capital;
    const dd = (peakEquity - capital) / peakEquity;
    if (dd > maxDrawdown) maxDrawdown = dd;

    equityCurve.push({
      date, totalEquity: Math.round(capital),
      positions: portfolio.holdings.length,
      drawdown: r2(dd * 100) + '%',
    });

    // Track top vs bottom decile returns (for factor premium analysis)
    if (portfolio.topDecile && portfolio.topDecile.length > 0) {
      // Approximate: top decile average composite → did stocks with high composite outperform?
      const topAvg = avg(portfolio.holdings.slice(0, 5).map(h => parseFloat(h.compositeScore) || 0));
      const bottomAvg = avg(portfolio.holdings.slice(-5).map(h => parseFloat(h.compositeScore) || 0));
      decileReturns.top.push(topAvg);
      decileReturns.bottom.push(bottomAvg);
    }
  }

  return buildBacktestAnalytics(closedTrades, equityCurve, capitalBase, capital, maxDrawdown);
}

function buildBacktestAnalytics(trades, equityCurve, capitalBase, finalCapital, maxDrawdown) {
  const total = trades.length;
  const winners = trades.filter(t => t.pnl > 0);
  const losers = trades.filter(t => t.pnl <= 0);
  const winRate = total > 0 ? winners.length / total : 0;

  const avgWin = winners.length > 0
    ? winners.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / winners.length : 0;
  const avgLoss = losers.length > 0
    ? Math.abs(losers.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / losers.length) : 0;

  const grossProfit = winners.reduce((s, t) => s + t.pnl, 0);
  const grossLoss = Math.abs(losers.reduce((s, t) => s + t.pnl, 0));
  const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : Infinity;
  const expectancy = (winRate * avgWin) - ((1 - winRate) * avgLoss);

  const totalReturn = (finalCapital / capitalBase) - 1;
  const years = equityCurve.length > 0
    ? equityCurve.length * CONFIG.HOLDING_PERIOD_DAYS / CONFIG.TRADING_DAYS_PER_YEAR : 1;
  const cagr = years > 0 ? Math.pow(finalCapital / capitalBase, 1 / years) - 1 : 0;

  // Monthly returns for Sharpe
  const monthlyReturns = [];
  for (let i = 1; i < equityCurve.length; i++) {
    const prev = equityCurve[i - 1].totalEquity;
    const curr = equityCurve[i].totalEquity;
    if (prev > 0) monthlyReturns.push((curr / prev) - 1);
  }

  const periodsPerYear = CONFIG.TRADING_DAYS_PER_YEAR / CONFIG.HOLDING_PERIOD_DAYS;
  const sharpe = calcSharpeRatio(monthlyReturns, periodsPerYear);
  const sortino = calcSortinoRatio(monthlyReturns, periodsPerYear);
  const calmar = maxDrawdown > 0 ? cagr / maxDrawdown : null;

  // ── Factor Contribution Analysis ──
  // Which factor drove returns most?
  const factorContribution = analyzeFactorContribution(trades);

  // ── By sector ──
  const bySector = groupAnalyze(trades, 'sector');

  // ── By factor count (how many factors available) ──
  const byFactorCount = groupAnalyze(trades, 'factorCount');

  // ── By exit reason ──
  const byExitReason = groupAnalyze(trades, 'exitReason');

  // ── Factor quintile returns ──
  const quintileAnalysis = analyzeQuintileReturns(trades);

  // Streaks
  let maxWin = 0, maxLose = 0, curWin = 0, curLose = 0;
  for (const t of trades) {
    if (t.pnl > 0) { curWin++; curLose = 0; maxWin = Math.max(maxWin, curWin); }
    else { curLose++; curWin = 0; maxLose = Math.max(maxLose, curLose); }
  }

  return {
    summary: {
      startingCapital: capitalBase,
      endingCapital: Math.round(finalCapital),
      totalReturn: r2(totalReturn * 100) + '%',
      cagr: r2(cagr * 100) + '%',
      totalTrades: total,
      winRate: r2(winRate * 100) + '%',
      avgWinPct: r2(avgWin) + '%',
      avgLossPct: r2(avgLoss) + '%',
      winLossRatio: avgLoss > 0 ? r2(avgWin / avgLoss) : 'N/A',
      profitFactor: profitFactor !== Infinity ? r2(profitFactor) : 'N/A',
      expectancy: r2(expectancy) + '%',
      maxDrawdown: r2(maxDrawdown * 100) + '%',
      sharpeRatio: sharpe !== null ? r2(sharpe) : 'N/A',
      sortinoRatio: sortino !== null ? r2(sortino) : 'N/A',
      calmarRatio: calmar !== null ? r2(calmar) : 'N/A',
      maxWinStreak: maxWin,
      maxLoseStreak: maxLose,
    },
    factorContribution,
    quintileAnalysis,
    bySector,
    byFactorCount,
    byExitReason,
    topWinners: [...trades].sort((a, b) => b.pnl - a.pnl).slice(0, 10).map(formatTrade),
    topLosers: [...trades].sort((a, b) => a.pnl - b.pnl).slice(0, 10).map(formatTrade),
    monthlyReturns: equityCurve.map((e, i) => ({
      date: e.date,
      return: i > 0
        ? r2(((e.totalEquity / equityCurve[i - 1].totalEquity) - 1) * 100) + '%'
        : '0%',
    })),
    trades,
    equityCurve,
  };
}

/**
 * Analyze which factor percentile correlates most with trade P&L
 * (Information Coefficient analog)
 */
function analyzeFactorContribution(trades) {
  const factors = ['momentumPctile', 'valuePctile', 'qualityPctile', 'revisionPctile'];
  const result = {};

  for (const factor of factors) {
    const pairs = trades
      .filter(t => t[factor] !== null && t[factor] !== undefined)
      .map(t => ({ score: t[factor], pnl: parseFloat(t.pnlPct) }));

    if (pairs.length < 10) {
      result[factor] = { ic: null, hitRate: null, count: pairs.length };
      continue;
    }

    // Rank correlation (simplified Spearman IC)
    const sortedByScore = [...pairs].sort((a, b) => b.score - a.score);
    const topHalf = sortedByScore.slice(0, Math.floor(pairs.length / 2));
    const bottomHalf = sortedByScore.slice(Math.floor(pairs.length / 2));

    const topAvgPnl = avg(topHalf.map(p => p.pnl));
    const bottomAvgPnl = avg(bottomHalf.map(p => p.pnl));

    // Hit rate: % of times high-factor-score stocks outperformed
    const topWins = topHalf.filter(p => p.pnl > 0).length;
    const hitRate = topHalf.length > 0 ? topWins / topHalf.length : 0;

    // Spread: top half return minus bottom half return
    const spread = topAvgPnl - bottomAvgPnl;

    // Pearson correlation as IC proxy
    const allScores = pairs.map(p => p.score);
    const allPnls = pairs.map(p => p.pnl);
    const ic = pearsonCorrelation(allScores, allPnls);

    result[factor.replace('Pctile', '')] = {
      informationCoefficient: r2(ic),
      topHalfAvgReturn: r2(topAvgPnl) + '%',
      bottomHalfAvgReturn: r2(bottomAvgPnl) + '%',
      spread: r2(spread) + '%',
      hitRate: r2(hitRate * 100) + '%',
      count: pairs.length,
      verdict: ic > 0.05 ? 'PREDICTIVE' : ic > 0 ? 'WEAK' : 'NOT_PREDICTIVE',
    };
  }

  return result;
}

/**
 * Analyze returns by factor quintile
 * (Does higher factor score → higher returns?)
 */
function analyzeQuintileReturns(trades) {
  const factors = ['compositeScore', 'momentumPctile', 'valuePctile', 'qualityPctile'];
  const result = {};

  for (const factor of factors) {
    const valid = trades.filter(t => t[factor] !== null && t[factor] !== undefined);
    if (valid.length < 10) continue;

    const sorted = [...valid].sort((a, b) => a[factor] - b[factor]);
    const quintileSize = Math.floor(sorted.length / 5);

    const quintiles = {};
    for (let q = 1; q <= 5; q++) {
      const start = (q - 1) * quintileSize;
      const end = q === 5 ? sorted.length : q * quintileSize;
      const bucket = sorted.slice(start, end);

      const avgReturn = avg(bucket.map(t => parseFloat(t.pnlPct)));
      const wins = bucket.filter(t => t.pnl > 0).length;

      quintiles[`Q${q}`] = {
        avgReturn: r2(avgReturn) + '%',
        winRate: r2((wins / bucket.length) * 100) + '%',
        count: bucket.length,
        avgFactorScore: r2(avg(bucket.map(t => t[factor]))),
      };
    }

    // Monotonicity check: does return increase from Q1 to Q5?
    const qReturns = [1, 2, 3, 4, 5].map(q =>
      parseFloat(quintiles[`Q${q}`].avgReturn)
    );
    let monotonic = true;
    for (let i = 1; i < qReturns.length; i++) {
      if (qReturns[i] < qReturns[i - 1]) { monotonic = false; break; }
    }

    const factorName = factor.replace('Pctile', '').replace('Score', '');
    result[factorName] = {
      quintiles,
      monotonic,
      q5MinusQ1Spread: r2(qReturns[4] - qReturns[0]) + '%',
      verdict: monotonic ? 'STRONG_FACTOR' : 'WEAK/NONLINEAR',
    };
  }

  return result;
}


// ============================================================================
// SENSITIVITY ANALYSIS
// ============================================================================

function sensitivityAnalysis(universeSnapshots, capitalBase = 10000000) {
  const results = [];

  // Factor weight variations
  const weightSets = [
    { label: 'Default (35/25/20/20)', m: 0.35, v: 0.25, q: 0.20, e: 0.20 },

    CONFIG.WEIGHTS.QUALITY = ws.q;
    CONFIG.WEIGHTS.EARNINGS_REVISION = ws.e;

    const bt = runBacktest(universeSnapshots, capitalBase);
    results.push({

      maxDD: bt.summary.maxDrawdown,
      sharpe: bt.summary.sharpeRatio,
      winRate: bt.summary.winRate,
      trades: bt.summary.totalTrades,
    });

    Object.assign(CONFIG.WEIGHTS, origWeights);
  }

  // Portfolio size
  for (const n of [10, 15, 20, 25, 30, 50]) {
    const orig = CONFIG.TOP_N_STOCKS;
    CONFIG.TOP_N_STOCKS = n;
    const bt = runBacktest(universeSnapshots, capitalBase);
    results.push({
      param: 'Portfolio Size', value: `Top ${n}`,
      return: bt.summary.totalReturn, maxDD: bt.summary.maxDrawdown,
      sharpe: bt.summary.sharpeRatio, winRate: bt.summary.winRate,
      trades: bt.summary.totalTrades,
    });
    CONFIG.TOP_N_STOCKS = orig;
  }

  // Weighting method
  for (const method of ['EQUAL', 'COMPOSITE_WEIGHTED', 'INV_VOL']) {
    const orig = CONFIG.WEIGHTING;
    CONFIG.WEIGHTING = method;
    const bt = runBacktest(universeSnapshots, capitalBase);
    results.push({
      param: 'Weighting', value: method,
      return: bt.summary.totalReturn, maxDD: bt.summary.maxDrawdown,
      sharpe: bt.summary.sharpeRatio, winRate: bt.summary.winRate,
      trades: bt.summary.totalTrades,
    });
    CONFIG.WEIGHTING = orig;
  }

  return results;
}

function r2(n) {
  return n !== null && n !== undefined && !isNaN(n) ? Math.round(n * 100) / 100 : null;
}

function avg(arr) {
  const valid = arr.filter(v => v !== null && v !== undefined && !isNaN(v));
  return valid.length > 0 ? valid.reduce((s, v) => s + v, 0) / valid.length : null;
}

function pearsonCorrelation(x, y) {
  if (x.length !== y.length || x.length < 3) return null;
  const n = x.length;
  const mx = x.reduce((s, v) => s + v, 0) / n;
  const my = y.reduce((s, v) => s + v, 0) / n;
  let cov = 0, vx = 0, vy = 0;
  for (let i = 0; i < n; i++) {
    cov += (x[i] - mx) * (y[i] - my);
    vx += (x[i] - mx) ** 2;
    vy += (y[i] - my) ** 2;
  }
  const denom = Math.sqrt(vx * vy);
  return denom !== 0 ? cov / denom : 0;
}

function groupAnalyze(trades, field) {
  const groups = {};
  for (const t of trades) {
    const key = String(t[field] || 'Unknown');
    if (!groups[key]) groups[key] = [];
    groups[key].push(t);
  }
  const result = {};
  for (const [key, list] of Object.entries(groups)) {
    const wins = list.filter(t => t.pnl > 0);
    result[key] = {
      count: list.length,
      winRate: r2((wins.length / list.length) * 100) + '%',
      avgPnlPct: r2(list.reduce((s, t) => s + parseFloat(t.pnlPct), 0) / list.length) + '%',
      totalPnl: Math.round(list.reduce((s, t) => s + t.pnl, 0)),
    };
  }
  return result;
}

function formatTrade(t) {
  return {
    symbol: t.symbol, sector: t.sector, pnlPct: t.pnlPct,
    rank: t.rank, composite: t.compositeScore,
    mom: t.momentumPctile, val: t.valuePctile,
    qual: t.qualityPctile, rev: t.revisionPctile,
    exitReason: t.exitReason,
  };
}

function calcSharpeRatio(returns, periodsPerYear) {
  if (!returns || returns.length < 2) return null;
  const mean = returns.reduce((s, r) => s + r, 0) / returns.length;
  const std = Math.sqrt(returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (returns.length - 1));
  if (std === 0) return null;
  return (mean * periodsPerYear - CONFIG.RISK_FREE_RATE) / (std * Math.sqrt(periodsPerYear));
}

function calcSortinoRatio(returns, periodsPerYear) {
  if (!returns || returns.length < 2) return null;
  const mean = returns.reduce((s, r) => s + r, 0) / returns.length;
  const down = returns.filter(r => r < 0);
  if (down.length === 0) return Infinity;
  const dsDev = Math.sqrt(down.reduce((s, r) => s + r ** 2, 0) / down.length);
  if (dsDev === 0) return Infinity;
  return (mean * periodsPerYear - CONFIG.RISK_FREE_RATE) / (dsDev * Math.sqrt(periodsPerYear));
}
